"use strict";

const fs = require("fs"), path = require("path"), crypto = require("crypto"), cp = require("child_process"), vm = require("vm");
const repo = process.env.LIORA_REPO;
if (!repo) throw new Error("LIORA_REPO is required");
const baseRel = "docs/liora-venn/v693-v8", base = path.join(repo, ...baseRel.split("/")), root = path.join(base, "x1"), planning = "505540397e3af5b03c1270b8a484da28133a6d48", checks = [];
function check(name, pass, detail = null) { checks.push({ name, pass: Boolean(pass), detail }); }
function sha(bytes) { return crypto.createHash("sha256").update(bytes).digest("hex"); }
function stable(value) { if (Array.isArray(value)) return value.map(stable); if (value && typeof value === "object") return Object.fromEntries(Object.keys(value).sort().map((k) => [k, stable(value[k])])); return value; }
function canon(value) { return JSON.stringify(stable(value)); }
function json(rel) { return JSON.parse(fs.readFileSync(path.join(base, ...rel.split("/")), "utf8")); }
function walk(folder) { let out = []; for (const e of fs.readdirSync(folder, { withFileTypes: true })) { const f = path.join(folder, e.name); out = e.isDirectory() ? out.concat(walk(f)) : out.concat(f); } return out; }
function blob(commit, rel) { return cp.execFileSync("git", ["-C", repo, "show", commit + ":" + rel], { encoding: null, maxBuffer: 67108864 }); }
const head = cp.execFileSync("git", ["-C", repo, "rev-parse", "HEAD"], { encoding: "utf8" }).trim();
check("exact planning head", head === planning, head);
const planManifest = JSON.parse(blob(planning, baseRel + "/plan/manifest.json")), planIssues = [];
for (const row of planManifest.bindings) { const b = blob(planning, row.path); if (b.length !== row.bytes || sha(b) !== row.sha256) planIssues.push(row.path); }
check("immutable planning manifest", planIssues.length === 0, planIssues);
const changedPlan = cp.execFileSync("git", ["-C", repo, "diff", "--name-only", planning, "--", baseRel + "/plan"], { encoding: "utf8" }).trim();
check("planning remains unchanged", changedPlan === "", changedPlan);
check("x2 and final absent", !fs.existsSync(path.join(base, "x2")) && !fs.existsSync(path.join(base, "final")));
const freeze = json("plan/proposals.json"), proposals = freeze.proposals.filter((r) => r.session === "x1"), core = require(path.join(root, "exact-linear-core.txt"));
const contractPasses = proposals.filter((r) => canon(core.execute(r.request)) === canon(r.expected_output)).length;
check("50 frozen x1 contracts", proposals.length === 50 && contractPasses === 50, { total: proposals.length, passed: contractPasses });
const proposalResults = json("x1/results/proposal-results.json"), safe = json("x1/results/safe-results.json"), candidates = json("x1/results/candidate-results.json"), cfr = json("x1/results/clean-fix-refine-results.json"), altered = json("x1/results/altered-expected-results.json");
check("x1 result ledgers", proposalResults.count === 50 && proposalResults.passed === 50 && safe.count === 100 && safe.passed === 100 && candidates.failed_subjects === 100 && candidates.refusal_passes === 100 && cfr.count === 100 && cfr.passed === 100 && altered.failed_subjects === 50 && altered.recovery_passes === 50);
check("failed subjects nonerased", candidates.results.every((r) => r.failed_subject_result === "fail" && r.original_success_credit === 0 && r.failed_subject_remains_failed && r.refusal_witness === "pass") && altered.results.every((r) => r.failed_subject_result === "fail" && r.original_success_credit === 0));
const skills = json("x1/evidence/skill-validation.json"), runners = json("x1/evidence/runner-receipts.json");
check("ten local skills built validated used", skills.count === 10 && skills.passed === 10 && skills.global_installs === 0 && skills.receipts.every((r) => r.official_quick_validate && r.accepting_smoke && r.rejecting_smoke));
check("five runners and boundaries", runners.count === 5 && runners.positive_passes === 10 && runners.scope_failed_subjects === 5 && runners.scope_refusal_passes === 5);
const flow = json("x1/method-flow.json"), negatives = json("x1/retained-negative-register.json"), truth = json("x1/phase-truth.json");
check("method flow arithmetic", flow.counts.methods === flow.methods.length && flow.counts.witnesses === flow.witnesses.length && flow.counts.pass === flow.witnesses.filter((w) => w.result === "pass").length && flow.counts.fail === flow.witnesses.filter((w) => w.result === "fail").length && negatives.count === negatives.negative_ids.length && new Set(negatives.negative_ids).size === negatives.count, { counts: flow.counts, negatives: negatives.count });
check("method flow witness shape", flow.methods.every((m) => m.retained_negative_ids.length > 0 && m.validation_witness_ids.length > 0 && m.protected_gates.length > 0) && flow.witnesses.every((w) => ["pass","fail"].includes(w.result) && w.same_owner_only === true && w.independent_reproduction === false));
check("phase truth and gates", truth.x1_contracts_passed === 50 && truth.x2_started === false && truth.global_installs === 0 && truth.canonical_invocations === 0 && truth.terminal_verdict === "NOT_READY_FOR_STAGE_20");
const manifest = json("x1/manifest.json"), manifestIssues = [];
for (const row of manifest.bindings) { const f = path.join(repo, ...row.path.split("/")); if (!fs.existsSync(f)) manifestIssues.push(row.path + ":missing"); else { const b = fs.readFileSync(f); if (b.length !== row.bytes || sha(b) !== row.sha256) manifestIssues.push(row.path + ":parity"); } }
check("working manifest", manifestIssues.length === 0, manifestIssues);
const files = walk(root), jsonIssues = [], txtIssues = [];
for (const f of files) { if (path.extname(f).toLowerCase() === ".json") { try { JSON.parse(fs.readFileSync(f, "utf8")); } catch (e) { jsonIssues.push(path.relative(repo, f)); } } if (path.extname(f).toLowerCase() === ".txt") { try { new vm.Script("(function(require,module,exports){\n" + fs.readFileSync(f, "utf8") + "\n})", { filename: f }); } catch (e) { txtIssues.push(path.relative(repo, f) + ":" + e.message); } } }
check("allowed files", files.every((f) => [".json", ".md", ".txt", ".html"].includes(path.extname(f).toLowerCase())), files.length);
check("strict JSON", jsonIssues.length === 0, jsonIssues);
check("trusted TXT", txtIssues.length === 0, txtIssues);
check("document word ceiling", Math.max(...files.map((f) => (fs.readFileSync(f, "utf8").match(/\S+/g) || []).length)) <= 100000);
const joined = Buffer.concat(files.map((f) => fs.readFileSync(f))).toString("utf8"), privacy = [/[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}/ig, /(?:api[_-]?key|password|secret|bearer)\s*[:=]\s*["']?[A-Za-z0-9_\-]{12,}/ig, /[A-Z]:\\Users\\[^\s"']+/ig, /[A-Z]:\\GHC-Archives\\[^\s"']+/ig, /\b[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}\b/ig];
check("zero five-class privacy candidates", privacy.every((p) => !p.test(joined)));
const failed = checks.filter((r) => !r.pass), receipt = { schema: "ghc.family.x1-validation.v9", owner: "Liora Venn", phase: "v693-v8", status: failed.length ? "INVALID_X1_CANDIDATE" : "VALID_X1_CANDIDATE", checks_total: checks.length, checks_passed: checks.length - failed.length, checks_failed: failed.length, checks, manifest_bindings: manifest.bindings.length, selected_contracts: proposals.length, selected_contracts_passed: contractPasses, canonical_invocations: 0, route: "PREPARED_NOT_SENT_LIORA_TERMINAL_GATE_REQUIRED", terminal_verdict: "NOT_READY_FOR_STAGE_20" }, rendered = JSON.stringify(receipt, null, 2) + "\n";
if (process.env.LIORA_VALIDATION_RECEIPT) fs.writeFileSync(process.env.LIORA_VALIDATION_RECEIPT, rendered, "utf8");
process.stdout.write(rendered); process.exit(failed.length ? 2 : 0);
