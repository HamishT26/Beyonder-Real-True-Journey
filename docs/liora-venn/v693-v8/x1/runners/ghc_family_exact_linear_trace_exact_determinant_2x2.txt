"use strict";
const path=require("path");
const core=require(path.join(__dirname,"..","exact-linear-core.txt"));
const allowed=new Set(["trace_exact","determinant_2x2"]);
function run(request){if(!request||!allowed.has(request.operation))return {ok:false,error:"E_SCOPE",authority:false};return core.execute(request);}
module.exports={run,operations:[...allowed]};
if(require.main===module){let data="";process.stdin.setEncoding("utf8");process.stdin.on("data",c=>data+=c);process.stdin.on("end",()=>{const out=run(JSON.parse(data));process.stdout.write(JSON.stringify(out)+"\n");});}
