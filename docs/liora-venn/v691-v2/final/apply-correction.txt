function replaceExact(file, replacements) {
  let text = fs.readFileSync(file, 'utf8');
  for (const [from, to] of replacements) { C.ensure(text.includes(from), 'replacement_context_missing'); text = text.replace(from, to); }
  fs.writeFileSync(file, text, 'utf8');
}
