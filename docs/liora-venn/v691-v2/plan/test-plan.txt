'use strict';
const fs=require('node:fs'),path=require('node:path'),C=require('./tools/common.txt');
const base=path.resolve('docs/liora-venn/v691-v2');
const freeze=C.read(path.join(base,'plan/proposal-freeze.json')),source=C.read(path.join(base,'plan/source-reviews.json')),novelty=C.read(path.join(base,'plan/novelty-audit.json'));
const checks=[];const check=(name,pass,detail=null)=>checks.push({name,pass:Boolean(pass),detail});
check('planning only',freeze.status==='FROZEN_PLANNING_ONLY'&&!freeze.execution_results_present);
check('200 proposals',freeze.new_proposals.length===200);
check('100 x1',freeze.new_proposals.filter(p=>p.session==='x1').length===100);
check('100 x2',freeze.new_proposals.filter(p=>p.session==='x2').length===100);
check('four outcome labels',freeze.new_proposals.every(p=>['completed','represented','open_gap','exact_gate'].includes(p.expected_disposition)));
check('outcome ceiling',JSON.stringify(freeze.outcome_ceiling_counts)===JSON.stringify({completed:160,represented:20,open_gap:10,exact_gate:10}));
check('unique proposals',new Set(freeze.new_proposals.map(p=>p.proposal_id)).size===200);
check('unique requests',new Set(freeze.new_proposals.map(p=>C.hash(Buffer.from(C.stable(p.request))))).size===200);
check('required proposal fields',freeze.new_proposals.every(p=>p.hypothesis&&p.null_or_failure_condition&&p.approval_class&&p.execution_lane&&p.official_or_primary_source_needs.length&&p.concrete_artifacts.length&&p.falsifier_or_acceptance_gate&&p.rollback_or_recovery&&p.protected_gates.length&&p.expected_disposition));
check('200 source reviews',source.rows.length===200&&source.rows.every(r=>r.novelty_credit===0&&r.completion_credit===0&&!r.execution_replayed));
check('no exact novelty collision',novelty.exact_collisions.length===0&&novelty.unique_request_digests===200);
for(const [file,count] of [['safe-now-portfolio.json',200],['candidate-portfolio.json',200],['clean-fix-refine-portfolio.json',200]])check(file,C.read(path.join(base,'plan',file)).rows.length===count);
check('exact held',C.read(path.join(base,'plan/exact-approval-packets.json')).rows.length===50&&C.read(path.join(base,'plan/exact-approval-packets.json')).executed===0);
check('blocked held',C.read(path.join(base,'plan/blocked-packets.json')).rows.length===30&&C.read(path.join(base,'plan/blocked-packets.json')).executed===0);
check('skills floor',C.read(path.join(base,'plan/local-skill-plan.json')).rows.length===20);
check('runners floor',C.read(path.join(base,'plan/local-runner-plan.json')).rows.length===10);
check('merged tools floor',C.read(path.join(base,'plan/tool-promotion-design.json')).groups.length===5);
const flow=C.read(path.join(base,'plan/method-flow.json'));check('startup failures retained',flow.counts.witness_results.fail===12&&flow.counts.witness_results.pass===12);
check('terminal verdict',C.BOUNDARY.includes('NOT_READY_FOR_STAGE_20'));
const failed=checks.filter(c=>!c.pass);process.stdout.write(JSON.stringify({schema:'ghc.family.plan-test.v5',owner:'Liora Venn',phase:'v691-v2',valid:failed.length===0,checks_total:checks.length,checks_passed:checks.length-failed.length,checks_failed:failed.length,checks,boundary:C.BOUNDARY}));if(failed.length)process.exitCode=1;
