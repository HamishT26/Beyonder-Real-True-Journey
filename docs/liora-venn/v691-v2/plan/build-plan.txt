function makeNet(index, conflict = false) {
  const placeCount = 3 + (index % 4);
  const transitionCount = 2 + (index % 5);
  const amount = 1 + Math.floor(index / 5);
  const places = Array.from({length: placeCount}, (_, i) => 'p' + i);
  const transitions = Array.from({length: transitionCount}, (_, t) => {
    const consume = Array(placeCount).fill(0);
    const produce = Array(placeCount).fill(0);
    const source = conflict && t < 2 ? 0 : t % placeCount;
    const destination = (t + 1 + (index % 2)) % placeCount;
    consume[source] = amount;
    produce[destination] = amount;
    return {id: 't' + t, consume, produce};
  });
  const marking = Array.from({length: placeCount}, (_, p) => ((p + index) % 3) + (p === 0 ? amount : 0));
  return {places, transitions, marking};
}

function enabled(net, marking, transition) {
  return transition.consume.every((value, index) => marking[index] >= value);
}

function fire(net, marking, transitionIndex) {
  const transition = net.transitions[transitionIndex];
  if (!enabled(net, marking, transition)) return null;
  return marking.map((value, index) => value - transition.consume[index] + transition.produce[index]);
}

function reachable(net, initial, cap) {
  const queue = [initial.slice()];
  const rows = [initial.slice()];
  const seen = new Set([C.stable(initial)]);
  let cursor = 0;
  while (cursor < queue.length && rows.length < cap) {
    const marking = queue[cursor++];
    for (let t = 0; t < net.transitions.length && rows.length < cap; t++) {
      const next = fire(net, marking, t);
      if (!next) continue;
      const key = C.stable(next);
      if (!seen.has(key)) { seen.add(key); queue.push(next); rows.push(next); }
    }
  }
  const frontierEnabled = cursor < queue.length || queue.some(marking => net.transitions.some((_, t) => {
    const next = fire(net, marking, t);
    return next && !seen.has(C.stable(next));
  }));
  return {markings: rows, truncated: rows.length >= cap && frontierEnabled};
}

function support(transition) {
  const out = [];
  transition.consume.forEach((v, i) => { if (v || transition.produce[i]) out.push(i); });
  return out;
}

  const net = makeNet(index, operation === 'petri_conflict_pairs');

  if (operation === 'petri_record_shape' || operation === 'petri_incidence_matrix') return {op: operation, places: net.places, transitions: net.transitions};
  if (operation === 'petri_enabled_transitions' || operation === 'petri_deadlock_witness' || operation === 'petri_conflict_pairs') return {op: operation, places: net.places, transitions: net.transitions, marking};
  if (operation === 'petri_fire_transition') return {op: operation, places: net.places, transitions: net.transitions, marking, transition_index: transitionIndex};
  if (operation === 'petri_fire_sequence') return {op: operation, places: net.places, transitions: net.transitions, marking, sequence};
  if (operation === 'petri_place_invariant') return {op: operation, places: net.places, transitions: net.transitions, marking, transition_index: transitionIndex, weights: net.places.map((_, p) => 1 + ((p + index) % 3))};
  if (operation === 'petri_reachability_bfs') return {op: operation, places: net.places, transitions: net.transitions, marking, state_cap: 12 + index};
  if (operation === 'petri_boundedness_probe') return {op: operation, places: net.places, transitions: net.transitions, marking, state_cap: 12 + index, token_limit: 3 + index};
  if (operation === 'petri_trace_prefix') return {op: operation, places: net.places, transitions: net.transitions, marking, sequence, prefix_length: index % (sequence.length + 1)};
  if (operation === 'petri_independent_pairs') return {op: operation, places: net.places, transitions: net.transitions};

  throw new Error('unknown_operation');
}

function oracle(request) {
  const net = request.places ? {places: request.places, transitions: request.transitions} : null;
  if (request.op === 'petri_record_shape') return {place_count: net.places.length, transition_count: net.transitions.length, nonzero_pre_arcs: net.transitions.reduce((n, t) => n + t.consume.filter(Boolean).length, 0), nonzero_post_arcs: net.transitions.reduce((n, t) => n + t.produce.filter(Boolean).length, 0), shape_consistent: net.transitions.every(t => t.consume.length === net.places.length && t.produce.length === net.places.length)};
  if (request.op === 'petri_incidence_matrix') return {matrix: net.places.map((_, p) => net.transitions.map(t => t.produce[p] - t.consume[p])), orientation: 'places_by_transitions'};
  if (request.op === 'petri_enabled_transitions') return {enabled: net.transitions.filter(t => enabled(net, request.marking, t)).map(t => t.id)};
  if (request.op === 'petri_fire_transition') { const next = fire(net, request.marking, request.transition_index); return {enabled: next !== null, next_marking: next}; }
  if (request.op === 'petri_fire_sequence') { let marking = request.marking.slice(), fired = 0, blockedAt = null; for (let i = 0; i < request.sequence.length; i++) { const next = fire(net, marking, request.sequence[i]); if (!next) { blockedAt = i; break; } marking = next; fired++; } return {fired_count: fired, blocked_at: blockedAt, marking}; }
  if (request.op === 'petri_place_invariant') { const next = fire(net, request.marking, request.transition_index); const dot = marking => marking.reduce((n, v, i) => n + v * request.weights[i], 0); return {enabled: next !== null, before: dot(request.marking), after: next ? dot(next) : null, conserved: next ? dot(request.marking) === dot(next) : null, local_witness_only: true}; }
  if (request.op === 'petri_reachability_bfs') return {...reachable(net, request.marking, request.state_cap), cap: request.state_cap};
  if (request.op === 'petri_boundedness_probe') { const result = reachable(net, request.marking, request.state_cap); const maxima = net.places.map((_, p) => Math.max(...result.markings.map(m => m[p]))); return {observed_maxima: maxima, within_declared_limit: maxima.every(v => v <= request.token_limit), truncated: result.truncated, universal_boundedness_proved: false}; }
  if (request.op === 'petri_deadlock_witness') { const ids = net.transitions.filter(t => enabled(net, request.marking, t)).map(t => t.id); return {enabled: ids, deadlocked: ids.length === 0, marking_local_only: true}; }
  if (request.op === 'petri_trace_prefix') { let marking = request.marking.slice(), fired = 0, blockedAt = null; for (let i = 0; i < request.prefix_length; i++) { const next = fire(net, marking, request.sequence[i]); if (!next) { blockedAt = i; break; } marking = next; fired++; } return {requested_prefix: request.prefix_length, fired_count: fired, blocked_at: blockedAt, marking}; }
  if (request.op === 'petri_conflict_pairs') { const ids = net.transitions.map((t, i) => ({t, i})).filter(x => enabled(net, request.marking, x.t)); const pairs = []; for (let a = 0; a < ids.length; a++) for (let b = a + 1; b < ids.length; b++) { const shared = net.places.map((_, p) => p).filter(p => ids[a].t.consume[p] && ids[b].t.consume[p]); if (shared.length) pairs.push({transitions: [ids[a].t.id, ids[b].t.id], shared_places: shared.map(p => net.places[p])}); } return {pairs}; }
  if (request.op === 'petri_independent_pairs') { const pairs = []; for (let a = 0; a < net.transitions.length; a++) for (let b = a + 1; b < net.transitions.length; b++) if (support(net.transitions[a]).every(p => !support(net.transitions[b]).includes(p))) pairs.push([net.transitions[a].id, net.transitions[b].id]); return {pairs, criterion: 'disjoint_declared_support'}; }
  if (request.op === 'petri_siphon_candidate') { const set = new Set(request.subset), violations = []; for (const t of net.transitions) { const produces = request.subset.some(p => t.produce[p] > 0); const consumes = request.subset.some(p => t.consume[p] > 0); if (produces && !consumes) violations.push(t.id); } return {candidate: violations.length === 0, violations, subset: request.subset.map(p => net.places[p])}; }
  if (request.op === 'petri_trap_candidate') { const set = new Set(request.subset), violations = []; for (const t of net.transitions) { const consumes = request.subset.some(p => t.consume[p] > 0); const produces = request.subset.some(p => t.produce[p] > 0); if (consumes && !produces) violations.push(t.id); } return {candidate: violations.length === 0, violations, subset: request.subset.map(p => net.places[p])}; }
  if (request.op === 'petri_workflow_source_sink') { const entry = request.entry_place, exit = request.exit_place; return {entry: net.places[entry], exit: net.places[exit], leaving_entry: net.transitions.filter(t => t.consume[entry] > 0).map(t => t.id), entering_exit: net.transitions.filter(t => t.produce[exit] > 0).map(t => t.id), real_workflow_certified: false}; }
  if (request.op === 'petri_capacity_envelope') { const next = fire(net, request.marking, request.transition_index); return {enabled: next !== null, next_marking: next, within_declared_capacity: next ? next.every((v, i) => v <= request.capacities[i]) : null, represented_model_only: true}; }
  if (request.op === 'petri_correction_prefix') { let shared = 0; while (shared < request.original.length && shared < request.corrected.length && request.original[shared] === request.corrected[shared]) shared++; return {shared_prefix_length: shared, replay_original_suffix: request.original.slice(shared), apply_corrected_suffix: request.corrected.slice(shared), operational_replay_performed: false}; }
  if (request.op === 'petri_accessible_marking_summary') { const rows = request.labels.map((label, i) => ({position: i + 1, label, tokens: request.marking[i]})); return {rows, text: rows.map(r => r.position + '. ' + r.label + ': ' + r.tokens + ' token' + (r.tokens === 1 ? '' : 's')).join('\n'), manual_accessibility_review_required: true}; }
  if (request.op === 'petri_operator_review_vacancy') { const missing = Object.keys(request.notes).filter(k => !request.notes[k].trim()).sort(); return {missing, evaluation_count: 0, real_operator_review: false, affected_user_review: false, safety_review: false, open_gap: true}; }
  if (request.op === 'petri_release_authority_hold') { const missing = Object.keys(request.declarations).filter(k => !request.declarations[k]).sort(); return {record: request.record, missing, external_action: false, authority_verified: false, exact_gate: true}; }
  throw new Error('unknown_operation');
}
