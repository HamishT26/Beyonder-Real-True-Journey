'use strict';
const fs=require('node:fs'),path=require('node:path'),C=require('./tools/common.txt');
const repo=process.cwd(),base=path.join(repo,'docs/liora-venn/v691-v2'),plan=path.join(base,'plan');
const receipt=JSON.parse(process.argv[2]);C.ensure(receipt.valid,'plan_tests_failed');const receiptPath=path.join(plan,'test-receipt.json');if(fs.existsSync(receiptPath))C.ensure(C.equal(C.read(receiptPath),receipt),'test_receipt_drift');else C.put(receiptPath,receipt);
const exclusions=['manifest.json','staged-review.json'];const entries=C.manifest(repo,'docs/liora-venn/v691-v2/plan',exclusions);
