def stable(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'), ensure_ascii=False)

def digest(value):
    return sha256(stable(value).encode()).hexdigest()

def closure(request):
    elements = request['elements']
    index = {element: position for position, element in enumerate(elements)}
    size = len(elements)
    matrix = [[row == column for column in range(size)] for row in range(size)]
    for lower, upper in request['covers']:
        matrix[index[lower]][index[upper]] = True
    for middle in range(size):
        for lower in range(size):
            for upper in range(size):
                matrix[lower][upper] = matrix[lower][upper] or (
                    matrix[lower][middle] and matrix[middle][upper])
    return matrix

def reduced_covers(request, relation):
    elements = request['elements']
    answer = []
    for lower in range(len(elements)):
        for upper in range(len(elements)):
            if lower == upper or not relation[lower][upper]:
                continue
            mediated = any(
                middle not in (lower, upper)
                and relation[lower][middle]
                and relation[middle][upper]
                for middle in range(len(elements)))
            if not mediated:
                answer.append([elements[lower], elements[upper]])
    return answer

def mobius(relation):
    size = len(relation)
    augmented = [
        [Fraction(int(relation[row][column])) for column in range(size)] +
        [Fraction(int(row == column)) for column in range(size)]
        for row in range(size)]
    for column in range(size):
        pivot = next(row for row in range(column, size) if augmented[row][column])
        augmented[column], augmented[pivot] = augmented[pivot], augmented[column]
        scale = augmented[column][column]
        augmented[column] = [value / scale for value in augmented[column]]
        for row in range(size):
            if row == column:
                continue
            factor = augmented[row][column]
            augmented[row] = [
                value - factor * pivot_value
                for value, pivot_value in zip(augmented[row], augmented[column])]
    inverse = [row[size:] for row in augmented]
    assert all(value.denominator == 1 for row in inverse for value in row)
    return [[int(value) for value in row] for row in inverse]

def order_ideals(request, relation):
    elements = request['elements']
    answer = []
    for mask in range(1 << len(elements)):
        valid = True
        for upper in range(len(elements)):
            if not mask & (1 << upper):
                continue
            for lower in range(len(elements)):
                if relation[lower][upper] and not mask & (1 << lower):
                    valid = False
        if valid:
            answer.append([element for index, element in enumerate(elements) if mask & (1 << index)])
    return answer

def linear_extensions(request, relation):
    elements = request['elements']
    answer = []
    def visit(prefix, remaining):
        if not remaining:
            answer.append(prefix)
            return
        for candidate in remaining:
            upper = elements.index(candidate)
            blocked = any(
                other != candidate and relation[elements.index(other)][upper]
                for other in remaining)
            if not blocked:
                visit(prefix + [candidate], [other for other in remaining if other != candidate])
    visit([], elements[:])
    assert len(answer) <= request['limit']
    return answer

def coordinates(request, relation):
    answer = []
    for column, element in enumerate(request['elements']):
        rank = sum(
            other != column and relation[other][column]
            for other in range(len(request['elements'])))
        answer.append({'element': element, 'x': column * 2, 'y': rank * 2})
    return answer

def evaluate(request):
    operation = request['operation']
    elements = request['elements']
    relation = closure(request)
    index = {element: position for position, element in enumerate(elements)}
    zeta = [[int(value) for value in row] for row in relation]
    mu = mobius(relation)
    weights = [request['weights'][element] for element in elements]
    transformed = [
        sum(weights[lower] for lower in range(len(elements)) if relation[lower][upper])
        for upper in range(len(elements))]
    if operation == 'carrier_inventory':
        result = {'elements': elements, 'count': len(elements)}
    elif operation == 'relation_normalization':
        result = {'covers': sorted(request['covers'])}
    elif operation == 'reflexive_closure':
        result = {'relation': [[elements[lower], elements[upper]]
            for lower in range(len(elements)) for upper in range(len(elements))
            if relation[lower][upper]]}
    elif operation == 'transitive_closure':
        result = {'relation': [[elements[lower], elements[upper]]
            for lower in range(len(elements)) for upper in range(len(elements))
            if lower != upper and relation[lower][upper]]}
    elif operation == 'antisymmetry_audit':
        violations = [[elements[lower], elements[upper]]
            for lower in range(len(elements)) for upper in range(lower + 1, len(elements))
            if relation[lower][upper] and relation[upper][lower]]
        result = {'antisymmetric': not violations, 'violations': violations}
    elif operation == 'comparability_matrix':
        result = {'elements': elements, 'matrix': [[
            relation[lower][upper] or relation[upper][lower]
            for upper in range(len(elements))] for lower in range(len(elements))]}
    elif operation == 'cover_relation_reduction':
        result = {'covers': reduced_covers(request, relation)}
    elif operation == 'minimal_elements':
        result = {'elements': [elements[upper] for upper in range(len(elements))
            if not any(lower != upper and relation[lower][upper] for lower in range(len(elements)))]}
    elif operation == 'maximal_elements':
        result = {'elements': [elements[lower] for lower in range(len(elements))
            if not any(lower != upper and relation[lower][upper] for upper in range(len(elements)))]}
    elif operation == 'principal_ideal':
        focus = index[request['focus']]
        result = {'focus': request['focus'], 'elements': [
            elements[lower] for lower in range(len(elements)) if relation[lower][focus]]}
    elif operation == 'principal_filter':
        focus = index[request['focus']]
        result = {'focus': request['focus'], 'elements': [
            elements[upper] for upper in range(len(elements)) if relation[focus][upper]]}
    elif operation == 'zeta_matrix':
        result = {'elements': elements, 'matrix': zeta}
    elif operation == 'mobius_matrix':
        result = {'elements': elements, 'matrix': mu}
    elif operation == 'zeta_transform':
        result = {'elements': elements, 'values': dict(zip(elements, transformed))}
    elif operation == 'mobius_inversion':
        recovered = [
            sum(transformed[lower] * mu[lower][upper] for lower in range(len(elements)))
            for upper in range(len(elements))]
        result = {'elements': elements, 'values': dict(zip(elements, recovered))}
    elif operation == 'order_ideal_enumeration':
        ideals = order_ideals(request, relation)
        result = {'count': len(ideals), 'ideals': ideals}
    elif operation == 'linear_extension_enumeration':
        extensions = linear_extensions(request, relation)
        result = {'count': len(extensions), 'extensions': extensions}
    elif operation == 'hasse_scene':
        result = {'nodes': request['coordinates'], 'covers': reduced_covers(request, relation),
                  'text_equivalent': 'Finite synthetic cover graph for ' + request['profile']}
    elif operation == 'external_observation_gap':
        result = {'observation_rows': 0, 'likelihood_present': False,
                  'parameter_constraint_present': False,
                  'reason': 'No governed external observation is in scope.'}
    elif operation == 'governance_authority_gate':
        result = {'authority_observed': False, 'reservations': RESERVATIONS}
    else:
        raise ValueError(operation)
    return {'ok': True, 'operation': operation, 'outcome': OUTCOMES.get(operation, 'completed'),
            'result': result, 'scope': SCOPE, 'same_owner_only': True,
            'independent_reproduction': False, 'planning_oracle_only': True}

def request_for(operation, profile, elements, covers, focus):
    provisional = {'operation': operation, 'profile': profile, 'elements': elements[:],
        'covers': [edge[:] for edge in covers], 'focus': focus,
        'weights': {element: index + 1 for index, element in enumerate(elements)},
        'limit': 50000, 'seed': 'liora-v699-v5'}
    provisional['coordinates'] = coordinates(provisional, closure(provisional))
    return provisional
