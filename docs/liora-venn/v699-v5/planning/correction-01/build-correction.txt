def sha(raw): return hashlib.sha256(raw).hexdigest()
def stable(value): return json.dumps(value, sort_keys=True, separators=(',', ':'), ensure_ascii=False)
def write(name, value):
    target = HERE / name
    assert not target.exists(), name
    target.write_text((stable(value) if not isinstance(value, str) else value.rstrip()) + '\n',
                      encoding='utf-8', newline='\n')
def git(*args):
    return subprocess.check_output(['git', '-C', str(REPO), *args], stderr=subprocess.PIPE).decode().strip()

def corrected_portfolio(stage):
    contracts = [row for row in plan['contracts'] if row['stage'] == stage]
    assert len(contracts) == 150
    safe, candidates, cfr = [], [], []
    for row in contracts:
        safe.extend([
            {'id': row['id'] + '-SAFE-A', 'contract_id': row['id'],
             'task': 'Execute the exact positive finite subject and preserve its complete envelope.',
             'state': 'planned', 'completion_credit': 0},
            {'id': row['id'] + '-SAFE-B', 'contract_id': row['id'],
             'task': 'Verify frozen-request immutability and exact full-envelope equality within the same attributable aggregate.',
             'state': 'planned', 'completion_credit': 0},
        ])
        candidates.extend([
            {'id': row['id'] + '-CAND-N1', 'contract_id': row['id'],
             'task': 'Execute and retain the undeclared-field failed subject plus its separate refusal witness.',
             'state': 'planned', 'completion_credit': 0},
            {'id': row['id'] + '-CAND-N2', 'contract_id': row['id'],
             'task': 'Execute and retain the self-cover failed subject plus its separate refusal witness.',
             'state': 'planned', 'completion_credit': 0},
        ])
        cfr.extend([
            {'id': row['id'] + '-CFR-A', 'contract_id': row['id'],
             'task': 'Execute the altered-expected comparison as a retained zero-credit failure.',
             'state': 'planned', 'completion_credit': 0},
            {'id': row['id'] + '-CFR-B', 'contract_id': row['id'],
             'task': 'Bind the unchanged exact output as the smallest recovery without rewriting the altered-expected failure.',
             'state': 'planned', 'completion_credit': 0},
        ])
    assert len(safe) == len(candidates) == len(cfr) == 300
    return {'schema': 'liora.corrected-session-portfolio.v1', 'stage': stage,
        'safe_now': safe, 'bounded_candidates': candidates, 'clean_fix_refine': cfr,
        'original_portfolio_retained': 'planning/%s-portfolios.json' % stage,
        'original_count_per_category': 150, 'corrected_count_per_category': 300,
        'execution_credit_at_correction': 0, 'boundary': BOUNDARY}
