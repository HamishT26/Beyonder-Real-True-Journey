def stable(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'), ensure_ascii=False)

def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def read(name):
    return json.loads((ROOT / name).read_text(encoding='utf-8'))

def write(name, value):
    target = ROOT / name
    target.parent.mkdir(parents=True, exist_ok=True)
    assert not target.exists(), name
    target.write_text(
        (stable(value) if not isinstance(value, str) else value.rstrip()) + '\n',
        encoding='utf-8', newline='\n')

def git(*args):
    return subprocess.check_output(
        ['git', '-C', str(REPO), *args], stderr=subprocess.PIPE
    ).decode().strip()

def source_json(path):
    return json.loads(git('show', SOURCE + ':' + path))

def load_witnesses(stage):
    execution = read(stage + '/execution.json')
    answer = []
    for shard in execution['witnesses_shards']:
        data = read(shard['path'])
        assert data['count'] == shard['count']
        assert sha((ROOT / shard['path']).read_bytes()) == shard['sha256']
        answer.extend(data['witnesses'])
    assert len(answer) == execution['witnesses_count']
    return answer
