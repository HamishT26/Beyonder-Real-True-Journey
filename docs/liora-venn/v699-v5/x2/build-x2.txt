def stable(value): return json.dumps(value, sort_keys=True, separators=(',', ':'), ensure_ascii=False)
def h(value): return hashlib.sha256(stable(value).encode()).hexdigest()
def sha(raw): return hashlib.sha256(raw).hexdigest()
def read(name): return json.loads((ROOT / name).read_text(encoding='utf-8'))
def write(name, value):
    target = ROOT / name
    target.parent.mkdir(parents=True, exist_ok=True)
    assert not target.exists(), name
    target.write_text((stable(value) if not isinstance(value, str) else value.rstrip()) + '\n',
                      encoding='utf-8', newline='\n')
def git(*args):
    return subprocess.check_output(['git', '-C', str(REPO), *args], stderr=subprocess.PIPE).decode().strip()
def batch(requests):
    body = stable({'stage': 'x2', 'requests': requests})
    process = subprocess.run(['node', str(ROOT / 'x2/session.txt')], input=body, text=True,
                             capture_output=True, check=False)
    result = json.loads(process.stdout)
    assert process.returncode == 0 and result['ok'] and result['count'] == len(requests), (
        process.returncode, result, process.stderr)
    return result['outputs']
def chunks(rows, size):
    return [rows[index:index + size] for index in range(0, len(rows), size)]
def matrix_product(left, right):
    return [[sum(left[row][middle] * right[middle][column] for middle in range(len(right)))
             for column in range(len(right))]
            for row in range(len(left))]
