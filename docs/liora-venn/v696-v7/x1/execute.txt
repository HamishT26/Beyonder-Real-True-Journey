 const safe=[{procedure:'oracle_parity',expected:r.expected_result,observed:actual},{procedure:'input_nonmutation',expected:before,observed:JSON.stringify(r.request)}];pp.push(...safe);const cfr=[{procedure:'canonical_result_review',expected:'Reduced rational output and typed result',observed:'Exact frozen result matched'},{procedure:'protected_claim_review',expected:'No wider evidence',observed:p.BOUNDARY}];pp.push(...cfr);const cleanup=[{procedure:'retained_negative_links',expected:2,observed:negatives.length},{procedure:'fixture_scope_review',expected:'zero real rows and owner-local input',observed:'synthetic rational strings only'}];pp.push(...cleanup);results.push({proposal:r.proposal_id,actual,outcome:actual.outcome,safe_checks:safe,candidates:negatives,cfr,cleanup,boundary:p.BOUNDARY});}p.write(`x1/results/${op}.json`,{results:results.filter(x=>rs.some(r=>r.proposal_id===x.proposal)),boundary:p.BOUNDARY});p.write(`x1/method-flow/${op}.json`,p.mf(op,ff,pp));failures.push(...ff);passes.push(...pp);}
