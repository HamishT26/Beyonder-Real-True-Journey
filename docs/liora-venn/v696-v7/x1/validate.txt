"use strict";
const p=require('../plan/build-plan.txt'),cp=require('child_process'),vm=require('vm'),fs=require('fs'),path=require('path');
const patterns={private_local_path:/(?:^|[^A-Za-z])[A-Za-z]:[\\/]/,raw_uuid:/[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}/i,delegation_markup:/<codex_delegation>/,private_uri:/codex:\/\/threads\//,credential_assignment:/(?:api_key|password|secret_token)\s*[:=]\s*["'][^"']+/i};
function validate(stage){const m=p.read(stage+'/manifest.json');let issues=[],parsed=0,compiled=0,mf=0,max=0,candidates=0;for(const e of m.entries){const b=cp.execFileSync('git',['-C',p.REPO,'show',':'+p.BASE+'/'+e.path],{maxBuffer:64*1024*1024});if(b.length!==e.bytes||p.hash(b)!==e.sha256)issues.push('manifest:'+e.path);const s=b.toString('utf8'),words=s.split(/\s+/).filter(Boolean).length;max=Math.max(max,words);if(words>100000)issues.push('wordcap:'+e.path);if(!['.json','.md','.txt','.html'].includes(path.extname(e.path)))issues.push('extension');if(e.path.endsWith('.json')){JSON.parse(s);parsed++;}if(e.path.endsWith('.txt')){new vm.Script(s,{filename:e.path});compiled++;}if(e.path.includes('/method-flow/')){const r=cp.spawnSync(process.env.LIORA_PYTHON||'python',['-B','-X','utf8',path.join(process.env.LIORA_SKILLS_ROOT,'ghc-family-method-flow-state/scripts/ghc_family_method_flow_state.py'),'validate','--ledger',path.join(p.ROOT,e.path)],{encoding:'utf8'});if(r.status!==0)issues.push('method_flow:'+e.path+':'+r.stdout);mf++;}for(const [label,re]of Object.entries(patterns))if(re.test(s)){if(e.path.endsWith('/validate.txt'))candidates++;else issues.push('privacy:'+label+':'+e.path);}}
const staged=p.git(p.REPO,'diff','--cached','--name-only').split('\n');if(staged.length!==m.entries.length+1||staged.some(n=>!n.startsWith(p.BASE+'/'+stage+'/')))issues.push('exact_stage_allowlist');const report={status:issues.length?'FAILED':'VALID_OWNER_STAGE_STRUCTURE',stage,manifest_entries:m.entries.length,staged:staged.length,json_parses:parsed,txt_compiles:compiled,method_flow_ledgers:mf,max_words:max,scanner_definition_candidates:candidates,confirmed_payload_hits:issues.filter(x=>x.startsWith('privacy')).length,issues,boundary:p.BOUNDARY};console.log(JSON.stringify(report));if(issues.length)process.exitCode=1;return report;}
if(require.main===module)validate(process.argv[2]||'x1');module.exports={validate,patterns};
