"use strict";
const p=require('./build-plan.txt'),fs=require('fs'),cp=require('child_process');
const patterns={private_local_path:/(?:^|[^A-Za-z])[A-Za-z]:[\\/]/,raw_uuid:/[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}/i,delegation_markup:/<codex_delegation>/,private_uri:/codex:\/\/threads\//,credential_assignment:/(?:api_key|password|secret_token)\s*[:=]\s*["'][^"']+/i};
const m=p.read('plan/manifest.json');let issues=[],max=0,maxFile='',parsed=0;
for(const e of m.entries){const b=cp.execFileSync('git',['-C',p.REPO,'show',':'+p.BASE+'/'+e.path],{maxBuffer:64*1024*1024});if(b.length!==e.bytes||p.hash(b)!==e.sha256)issues.push('manifest:'+e.path);const s=b.toString('utf8'),w=s.split(/\s+/).filter(Boolean).length;if(w>max){max=w;maxFile=e.path;}if(w>100000)issues.push('wordcap:'+e.path);if(!['.md','.txt','.json','.html'].includes(require('path').extname(e.path)))issues.push('extension:'+e.path);if(e.path.endsWith('.json')){JSON.parse(s);parsed++;}for(const [label,re]of Object.entries(patterns))if(re.test(s)&&e.path!=='plan/validate.txt')issues.push('privacy:'+label+':'+e.path);}
const stage=p.git(p.REPO,'diff','--cached','--name-only').split('\n');if(stage.length!==m.entries.length+1||stage.some(n=>!n.startsWith(p.BASE+'/plan/')))issues.push('allowlist');
console.log(JSON.stringify({status:issues.length?'FAILED':'VALID_EXACT_STAGED_PLANNING',manifest_entries:m.entries.length,staged:stage.length,json_parses:parsed,max_words:max,max_file:maxFile,scanner_definition_candidates:1,confirmed_privacy_hits:issues.filter(x=>x.startsWith('privacy')).length,issues}));if(issues.length)process.exit(1);
