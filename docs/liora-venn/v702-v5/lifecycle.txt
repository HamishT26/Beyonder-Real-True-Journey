'use strict';
const fs=require('fs'),path=require('path'),cp=require('child_process'),Q=require('./q.txt');
const root=__dirname,repo=path.resolve(root,'../../..'),prefix='docs/liora-venn/v702-v5/',bank=process.env.GHC_PHASE_BANK,branch='codex/GHC-Family/liora-venn-v702-v5-full-tools',source='a073741b08d9042cafe793a7c0af77414ab230f5';
if(!bank)throw Error('GHC_PHASE_BANK_REQUIRED');
const git=(a,o={})=>cp.execFileSync('git',a,{cwd:repo,encoding:'utf8',maxBuffer:64*1024*1024,...o}).trim();
function walk(d=root){return fs.readdirSync(d,{withFileTypes:true}).flatMap(e=>e.isDirectory()?walk(path.join(d,e.name)):[path.join(d,e.name)]).sort();}
function gate(){const h=git(['rev-parse','HEAD']),u=git(['rev-parse','@{u}']),t=git(['rev-parse','refs/remotes/origin/'+branch]),line=git(['ls-remote','--exit-code','origin','refs/heads/'+branch]),r=line.split(/\s/)[0],div=git(['rev-list','--left-right','--count','HEAD...@{u}']);if([u,t,r].some(x=>x!==h)||git(['status','--porcelain'])||div!=='0\t0')throw Error('FOUR_WAY_GATE');return{local:h,upstream:u,tracking:t,fresh_remote:r,clean:true,divergence:[0,0]};}
function seal(stage){const stages=['planning','x1','x2','final'],index=stages.indexOf(stage);if(index<0)throw Error('STAGE');fs.mkdirSync(bank,{recursive:true});const receipt=path.join(bank,stage+'-receipt.json');if(fs.existsSync(receipt))throw Error('SUCCESSFUL_STAGE_ALREADY_SEALED');const expectedParent=index===0?source:JSON.parse(fs.readFileSync(path.join(bank,stages[index-1]+'-receipt.json'))).commit;if(git(['rev-parse','HEAD'])!==expectedParent)throw Error('PARENT');const mp='validation/'+stage+'-manifest.json';if(fs.existsSync(path.join(root,mp)))throw Error('MANIFEST_EXISTS_INSPECT_BEFORE_RETRY');const files=walk();const entries=files.map(f=>{const b=fs.readFileSync(f),rel=path.relative(repo,f).replaceAll('\\','/');if(!/\.(json|md|txt|html)$/.test(rel))throw Error('FILE_TYPE '+rel);const words=b.toString().trim()?b.toString().trim().split(/\s+/).length:0;if(words>100000)throw Error('WORD_BUDGET '+rel);if(f.endsWith('.json'))JSON.parse(b);return{path:rel,bytes:b.length,sha256:Q.sha(b)};});fs.mkdirSync(path.join(root,'validation'),{recursive:true});fs.writeFileSync(path.join(root,mp),JSON.stringify({schema:'ghc.liora.v702-v5.manifest.v1',stage,parent:expectedParent,byte_domain:'raw Git blob; generated files LF; staged with core.autocrlf=false',self_excluded:prefix+mp,entries},null,2)+'\n');git(['-c','core.autocrlf=false','add','--sparse','--',prefix]);const staged=git(['diff','--cached','--name-only']).split('\n').filter(Boolean);if(!staged.length||staged.some(p=>!p.startsWith(prefix)))throw Error('OWNER_SCOPE');const tracked=git(['ls-files']).split('\n').filter(Boolean).length;if(tracked>2000)throw Error('TRACKED_FILE_CEILING '+tracked);git(['commit','-m','Liora Venn v702-v5 '+stage+' exact dyadic sampling certificates']);const commit=git(['rev-parse','HEAD']);const raw=cp.execFileSync('git',['cat-file','--batch'],{cwd:repo,input:entries.map(e=>commit+':'+e.path).join('\n')+'\n',maxBuffer:64*1024*1024});let at=0;for(const e of entries){const end=raw.indexOf(10,at),header=raw.subarray(at,end).toString(),size=Number(header.split(' ')[2]);at=end+1;const b=raw.subarray(at,at+size);at+=size+1;if(size!==e.bytes||Q.sha(b)!==e.sha256)throw Error('COMMIT_BYTE_MISMATCH '+e.path);}git(['push','-u','origin',branch]);const equality=gate();const result={schema:'ghc.liora.lifecycle-receipt.v1',stage,commit,parent:expectedParent,source,push_exit:0,manifest:prefix+mp,manifest_entries:entries.length,owner_files:walk().length,tracked_files:tracked,commit_byte_bindings:entries.length,equality,created_utc:new Date().toISOString()};fs.writeFileSync(receipt,JSON.stringify(result,null,2)+'\n');console.log(JSON.stringify(result));return result;}
