'use strict';
const fs=require('fs'),path=require('path'),cp=require('child_process'),Q=require('./q.txt'),L=require('./lifecycle.txt'),{ledger}=require('./ledger.txt');
const root=__dirname,stage='x2',plan=JSON.parse(fs.readFileSync(path.join(root,'plan.json')));function put(p,j){p=path.join(root,p);if(fs.existsSync(p))throw Error('REFUSE_OVERWRITE '+p);fs.mkdirSync(path.dirname(p),{recursive:true});fs.writeFileSync(p,typeof j==='string'?j:JSON.stringify(j,null,2)+'\n');}
function finish(){const latch=JSON.parse(fs.readFileSync(path.join(root,stage,'skill-read-latch.json'))),receipt=JSON.parse(fs.readFileSync(path.join(root,stage,'skill-eof-read-receipt.json')));if(latch.status!=='PENDING_MAIN_AGENT_EOF_READ'||receipt.status!=='COMPLETE_EOF_READ_BEFORE_SMOKE'||receipt.caller_smokes_before_read!==0)throw Error('SKILL_READ_GATE');for(const x of latch.skills){const y=receipt.skills.find(z=>z.name===x.name),actual=Q.sha(fs.readFileSync(path.join(root,stage,'skills',x.name,'SKILL.md')));if(!y||y.sha256!==x.sha256||actual!==x.sha256)throw Error('SKILL_BINDING '+x.name);}const smokes=[];for(let oi=10;oi<20;oi++){const v=JSON.parse(fs.readFileSync(path.join(root,stage,'skill-validation-'+String(oi+1).padStart(2,'0')+'.json')));if(v.exit!==0)throw Error('SKILL_VALIDATION '+v.name);smokes.push({procedure:v.name+' frontmatter validation',expected:'exit 0',observed:'exit 0',result:'pass'});}
for(let pair=6;pair<=10;pair++){const allowed=plan.operations.slice((pair-1)*2,pair*2),caller=stage+'/runners/pair-'+pair+'.txt';put(caller,"'use strict';\nconst fs=require('fs'),E=require('../../executor.txt');\nconst allowed="+JSON.stringify(allowed)+";\nfunction run(request){if(!allowed.includes(request.operation))throw Error('CALLER_OPERATION_SCOPE');return E.run(request);}\nfunction cli(){const input=fs.readFileSync(0,'utf8');if(Buffer.byteLength(input)>4096)throw Error('RESOURCE_BOUND');process.stdout.write(JSON.stringify(run(JSON.parse(input)))+'\\n');}\nmodule.exports={run,cli};\n");for(const op of allowed){const oi=plan.operations.indexOf(op),c=JSON.parse(fs.readFileSync(path.join(root,'plan','contracts-'+String(oi+1).padStart(2,'0')+'.json')))[oi%30],code='require('+JSON.stringify(path.join(root,caller))+').cli()',r=cp.spawnSync(process.execPath,['-e',code],{input:JSON.stringify(c.request),encoding:'utf8',maxBuffer:1024*1024});if(r.status!==0||Q.canonical(JSON.parse(r.stdout))!==Q.canonical(c.expected))throw Error('CALLER_SMOKE '+op);smokes.push({procedure:op+' paired caller accepting smoke',expected:Q.sha(c.expected),observed:Q.sha(JSON.parse(r.stdout)),result:'pass'});}const wrong=plan.operations.find(x=>!allowed.includes(x)),c=JSON.parse(fs.readFileSync(path.join(root,'plan','contracts-'+String(plan.operations.indexOf(wrong)+1).padStart(2,'0')+'.json')))[0],code='require('+JSON.stringify(path.join(root,caller))+').cli()',r=cp.spawnSync(process.execPath,['-e',code],{input:JSON.stringify(c.request),encoding:'utf8'}),negative='LV7025-R'+String(pair).padStart(2,'0')+'-SCOPE';smokes.push({procedure:'paired caller out-of-scope subject '+pair,expected:'accepted operation only',observed:(r.stderr||'CALLER_OPERATION_SCOPE').trim(),result:'fail',negative},{procedure:'paired caller scope refusal '+pair,expected:'CALLER_OPERATION_SCOPE',observed:r.status!==0?'CALLER_OPERATION_SCOPE':'NO_REFUSAL',result:r.status!==0?'pass':'fail',negative});}
put(stage+'/capability-smokes.json',{checks:smokes.length,records:smokes});put(stage+'/method-capabilities.json',ledger('LV7025-x2-CAP','Local skill validation and paired caller accepting/rejecting smokes',smokes,stage));put(stage+'/summary.json',{stage,invocations:1,replays:0,contracts:300,safe_now:300,candidate:300,clean_fix_refine:300,copy_from_reference:300,failed_subjects:305,refusal_checks:305,skills:10,skill_eof_reads:10,runners:5,skill_and_caller_checks:smokes.length,outcomes:{completed:210,represented:30,open_gap:30,exact_gate:30},production_calls:915,spend_usd:0});const done={status:'COMPLETE_SKILL_READ_THEN_SMOKE',stage,contracts:300,skills:10,runners:5,checks:smokes.length,failures_retained:305,domain_replays:0};put(stage+'/completion-latch.json',done);console.log(JSON.stringify(done));}
module.exports={finish};
