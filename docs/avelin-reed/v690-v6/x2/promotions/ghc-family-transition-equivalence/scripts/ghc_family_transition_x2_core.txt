'use strict';
const p=require('./ghc_family_transition_protocol.txt'),first=require('./ghc_family_transition_x1_core.txt');
const FIELDS={lts_simulation:['peer'],lts_bisimulation:[],lts_trace_difference:['peer','depth'],lts_quotient:[],lts_product:['peer'],lts_game:['target','controller'],lts_narration:['trace'],lts_observer:['trace','hidden'],lts_evaluation_gap:['required','supplied'],lts_authority_hold:['required','supplied']};
function relation(a,b,both=false){const key=(x,y)=>JSON.stringify([x,y]),pairs=a.states.flatMap(x=>b.states.map(y=>[x,y])),keep=new Set(pairs.map(z=>key(...z)));let changed;do{changed=false;for(const [x,y] of pairs){const k=key(x,y);if(!keep.has(k))continue;const left=first.outgoing(a,x),right=first.outgoing(b,y),forward=left.every(e=>right.some(f=>e[1]===f[1]&&keep.has(key(e[2],f[2])))),reverse=!both||right.every(f=>left.some(e=>e[1]===f[1]&&keep.has(key(e[2],f[2]))));if(!forward||!reverse){keep.delete(k);changed=true;}}}while(changed);return pairs.filter(z=>keep.has(key(...z)));}
function classes(m){const R=relation(m,m,true),seen=new Set(),groups=[];for(const s of m.states)if(!seen.has(s)){const g=m.states.filter(t=>R.some(([x,y])=>x===s&&y===t));g.forEach(t=>seen.add(t));groups.push(g);}return groups;}
function traceDifference(a,b,depth){const next=(m,set,action)=>m.states.filter(v=>set.some(s=>first.outgoing(m,s).some(e=>e[1]===action&&e[2]===v)));const queue=[{left:[a.start],right:[b.start],word:[]}],visited=new Set([JSON.stringify([[a.start],[b.start]])]);for(let i=0;i<queue.length;i++){if(i>100000)p.fail('state_budget');const q=queue[i];if(!q.right.length)return{word:q.word,depth,bounded:true};if(q.word.length===depth)continue;const labels=[...new Set(q.left.flatMap(s=>first.outgoing(a,s).map(e=>e[1])))].sort();for(const action of labels){const left=next(a,q.left,action),right=next(b,q.right,action),k=JSON.stringify([left,right]);if(!visited.has(k)){visited.add(k);queue.push({left,right,word:[...q.word,action]});}}}return{word:null,depth,bounded:true};}
function product(a,b){const queue=[[a.start,b.start]],seen=new Set([JSON.stringify(queue[0])]);for(let i=0;i<queue.length;i++){const [x,y]=queue[i];for(const e of first.outgoing(a,x))for(const f of first.outgoing(b,y))if(e[1]===f[1]){const z=[e[2],f[2]],k=JSON.stringify(z);if(!seen.has(k)){seen.add(k);queue.push(z);}}}return a.states.flatMap(x=>b.states.map(y=>[x,y])).filter(z=>seen.has(JSON.stringify(z)));}
function validTrace(trace,m){if(!Array.isArray(trace)||trace.length>64)p.fail('trace_shape');let at=m.start;for(const e of trace){if(!Array.isArray(e)||e.length!==3||e[0]!==at||!m.edges.some(f=>JSON.stringify(e)===JSON.stringify(f)))p.fail('trace_edge');at=e[2];}}
function value(q){if(!q||!Object.hasOwn(FIELDS,q.op))p.fail('unknown_operation');p.object(q,['op','model',...FIELDS[q.op]]);const m=p.model(q.model);if(FIELDS[q.op].includes('peer'))p.model(q.peer);if(FIELDS[q.op].includes('target'))p.stateSet(q.target,m);if(FIELDS[q.op].includes('controller'))p.stateSet(q.controller,m);if(FIELDS[q.op].includes('trace'))validTrace(q.trace,m);if(FIELDS[q.op].includes('hidden'))p.tokens(q.hidden);if(FIELDS[q.op].includes('required')){p.tokens(q.required);p.tokens(q.supplied);if(q.supplied.some(v=>!q.required.includes(v)))p.fail('unexpected_evidence_label');}if(FIELDS[q.op].includes('depth')&&(!Number.isSafeInteger(q.depth)||q.depth<0||q.depth>8))p.fail('depth_range');
switch(q.op){case'lts_simulation':return relation(m,q.peer);case'lts_bisimulation':return classes(m);case'lts_trace_difference':return traceDifference(m,q.peer,q.depth);case'lts_quotient':{const groups=classes(m),which=s=>groups.findIndex(g=>g.includes(s)),es=m.edges.map(e=>[which(e[0]),e[1],which(e[2])]);return{classes:groups,edges:[...new Map(es.map(e=>[JSON.stringify(e),e])).values()].sort(p.compare)};}case'lts_product':return product(m,q.peer);case'lts_game':return first.least(m,q.target,(s,w)=>q.controller.includes(s)?first.succ(m,s).some(t=>w.has(t)):first.succ(m,s).every(t=>w.has(t)));case'lts_narration':return{steps:q.trace.map((e,i)=>({step:i+1,from:e[0],action:e[1],to:e[2],text:e[0]+' --'+e[1]+'--> '+e[2]})),human_evaluation:false};case'lts_observer':return{visible:q.trace.filter(e=>!q.hidden.includes(e[1])).map(e=>e[1]),hidden_count:q.trace.filter(e=>q.hidden.includes(e[1])).length,equivalence_claim:false};case'lts_evaluation_gap':return{missing:q.required.filter(r=>!q.supplied.includes(r)),real_participants:0,outcome:'open_gap'};case'lts_authority_hold':return{vacant:q.required.filter(r=>!q.supplied.includes(r)),actual_authority:false,external_action:false,outcome:'exact_gate'};}}
const run=q=>p.envelope(value,q);module.exports={run,value,FIELDS,relation,classes,traceDifference,product,validTrace};if(require.main===module)p.cli(run);
