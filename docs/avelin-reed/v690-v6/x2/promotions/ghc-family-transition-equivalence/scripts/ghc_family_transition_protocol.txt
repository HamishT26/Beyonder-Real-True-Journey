'use strict';
const fs=require('node:fs');
const token=x=>typeof x==='string'&&/^[A-Za-z][A-Za-z0-9_-]{0,31}$/.test(x);
function fail(code){throw Error(code);}
function object(x,keys){if(x===null||typeof x!=='object'||Array.isArray(x))fail('object_type');if(Object.keys(x).some(k=>!keys.includes(k)))fail('unknown_field');if(keys.some(k=>!Object.hasOwn(x,k)))fail('missing_field');}
function model(m){object(m,['states','start','edges']);if(!Array.isArray(m.states)||m.states.length<1||m.states.length>16)fail('state_count');if(m.states.some(x=>!token(x)))fail('state_type');if(new Set(m.states).size!==m.states.length)fail('duplicate_state');if(!m.states.includes(m.start))fail('unknown_state');if(!Array.isArray(m.edges)||m.edges.length>64)fail('edge_count');const seen=new Set();for(const e of m.edges){if(!Array.isArray(e)||e.length!==3)fail('edge_shape');if(!token(e[1]))fail('action_type');if(!m.states.includes(e[0])||!m.states.includes(e[2]))fail('unknown_state');const k=JSON.stringify(e);if(seen.has(k))fail('duplicate_edge');seen.add(k);}return m;}
function stateSet(xs,m){if(!Array.isArray(xs)||xs.some(x=>!m.states.includes(x))||new Set(xs).size!==xs.length)fail('state_set');}
function tokens(xs){if(!Array.isArray(xs)||xs.length>32||xs.some(x=>!token(x))||new Set(xs).size!==xs.length)fail('token_set');}
function parse(text){if(Buffer.byteLength(text)>1048576)fail('input_bytes');let i=0;const ws=()=>{while(/\s/.test(text[i]||'')&&i<text.length)i++;};function string(){const a=i++;while(i<text.length){if(text[i]==='\\'){i+=2;continue;}if(text[i++]==='"')return JSON.parse(text.slice(a,i));}fail('json_syntax');}function value(depth){if(depth>40)fail('json_depth');ws();const c=text[i];if(c==='"'){string();return;}if(c==='{'){i++;ws();const keys=new Set();if(text[i]==='}'){i++;return;}while(i<text.length){ws();if(text[i]!=='"')fail('json_syntax');const k=string();if(keys.has(k))fail('duplicate_key');keys.add(k);ws();if(text[i++]!==':')fail('json_syntax');value(depth+1);ws();const d=text[i++];if(d==='}')return;if(d!==',')fail('json_syntax');}fail('json_syntax');}if(c==='['){i++;ws();if(text[i]===']'){i++;return;}while(i<text.length){value(depth+1);ws();const d=text[i++];if(d===']')return;if(d!==',')fail('json_syntax');}fail('json_syntax');}const match=/^(?:true|false|null|-?(?:0|[1-9]\d*)(?:\.\d+)?(?:[eE][+-]?\d+)?)/.exec(text.slice(i));if(!match)fail('json_syntax');i+=match[0].length;}
try{value(0);ws();if(i!==text.length)fail('json_syntax');const x=JSON.parse(text);const finite=v=>v===null||typeof v!=='object'?(typeof v!=='number'||Number.isFinite(v)):Object.values(v).every(finite);if(!finite(x))fail('nonfinite_number');return x;}catch(e){if(['duplicate_key','json_depth','nonfinite_number'].includes(e.message))throw e;fail('json_syntax');}}
function envelope(fn,q){try{return{ok:true,value:fn(q),error:null};}catch(e){return{ok:false,value:null,error:e.message};}}
function cli(run){let text='';try{text=fs.readFileSync(0,'utf8');const result=run(parse(text));process.stdout.write(JSON.stringify(result)+'\n');if(!result.ok)process.exitCode=2;}catch(e){process.stdout.write(JSON.stringify({ok:false,value:null,error:e.message})+'\n');process.exitCode=2;}}
const compare=(a,b)=>JSON.stringify(a).localeCompare(JSON.stringify(b),'en');
module.exports={fail,object,model,stateSet,tokens,parse,envelope,cli,compare,token};
