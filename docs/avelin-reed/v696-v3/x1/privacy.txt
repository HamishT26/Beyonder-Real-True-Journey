'use strict';
const u=require('../plan/lib.txt');
const classes={private_identifiers:/\b[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}\b/gi,private_paths:/\b[A-Za-z]:[\\/](?:Users|GHC-Archives|Program Files)\b/g,credentials:/\b(?:sk-|ghp_)[A-Za-z0-9_-]{20,}|-----BEGIN [A-Z ]*PRIVATE KEY-----/g,private_route_markup:/<(?:codex_delegation|source_thread_id)>|(?:app|plugin):\/\//g,private_contact:/\b[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}\b/gi};
function scan(files){const candidates=[];for(const file of files){const text=u.fs.readFileSync(file,'utf8');for(const [kind,rx]of Object.entries(classes)){rx.lastIndex=0;for(const m of text.matchAll(rx))candidates.push({path:file,kind,line:text.slice(0,m.index).split('\n').length,classification:'unreviewed'});}}return candidates;}
module.exports={scan,classes};if(require.main===module){const stage=process.argv[2];if(!stage)throw Error('stage required');const files=u.ownedPaths(),candidates=scan(files);u.write(stage+'/privacy-scan.json',{stage,classes:Object.keys(classes),files:files.length,candidates,candidate_count:candidates.length,confirmed_hits:candidates.length===0?0:null,scope:'owner-authored files only',complete_privacy:false,boundary:u.boundary});console.log(JSON.stringify({files:files.length,candidates:candidates.length}));if(candidates.length)process.exitCode=1;}
