"use strict";
const fs = require("fs");
const path = require("path");
const Module = require("module");
function loadTextModule(p) { const m = new Module(p, module); m.filename = p; m.paths = module.paths; m._compile(fs.readFileSync(p, "utf8"), p); return m.exports; }
const corePath = path.join(__dirname, "ghc_family_stable_matching_core.txt");
loadTextModule(corePath).main(["proposer-optimal-matching", "receiver-optimal-matching"]);
