  if(!payload || !Array.isArray(payload.rows)) return {ok:false,error:"ROWS_REQUIRED",subject_success_credit:0};
  const allowed=["reduced_laplacian","configuration_degree"];

  if(payload.rows.some(row=>row.matches_expected!==true)) return {ok:false,error:"EVIDENCE_MISMATCH",subject_success_credit:0};

};
