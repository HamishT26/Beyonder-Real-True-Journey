#!/usr/bin/env python3
import runpy,sys
from pathlib import Path
sys.dont_write_bytecode=True
ns=runpy.run_path(str(Path(__file__).with_name('ghc_family_binary_code_extension.txt')))
ALLOWED=['channel-observation-gap', 'governance-authority-gate']
def handle(q):
    if not isinstance(q,dict) or q.get('op') not in ALLOWED:
        return ns['rejected']('operation')
    return ns['handle'](q)
if __name__=='__main__':
    raise SystemExit(ns['cli'](handle))
