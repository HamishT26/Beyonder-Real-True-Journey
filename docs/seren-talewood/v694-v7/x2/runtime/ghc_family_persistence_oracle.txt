'use strict';
// Separate exhaustive oracle for tiny fixtures: no reduction, pivot, rank or core import.
function faces(v){return v.length===1?[]:v.map((_,i)=>v.filter((_,j)=>i!==j).join(','));}
function span(vectors){let values=new Set([0]);for(const v of vectors)values=new Set([...values,...[...values].map(w=>w^v)]);return values;}
function model(simplices,dimension,time){const universe=simplices.filter(s=>s.vertices.length===dimension+1).map(s=>[...s.vertices].sort((a,b)=>a-b).join(',')),low=simplices.filter(s=>s.vertices.length===dimension).map(s=>[...s.vertices].sort((a,b)=>a-b).join(',')),active=simplices.filter(s=>s.vertices.length===dimension+1&&s.birth<=time);if(universe.length>20)throw Error('ORACLE_BUDGET');const columns=active.map(s=>faces([...s.vertices].sort((a,b)=>a-b)).reduce((b,k)=>b^(1<<low.indexOf(k)),0));const cycles=[];for(let mask=0;mask<(1<<active.length);mask++){let boundary=0,embedded=0;for(let j=0;j<active.length;j++)if(mask&(1<<j)){boundary^=columns[j];embedded^=1<<universe.indexOf([...active[j].vertices].sort((a,b)=>a-b).join(','));}if(!boundary)cycles.push(embedded);}return {universe,cycles};}
function persistentRank(simplices,p,a,b){const m=model(simplices,p,a),boundaryVectors=simplices.filter(s=>s.vertices.length===p+2&&s.birth<=b).map(s=>faces([...s.vertices].sort((a,b)=>a-b)).reduce((v,k)=>v^(1<<m.universe.indexOf(k)),0)),boundaries=[...span(boundaryVectors)];const cosets=new Set(m.cycles.map(z=>Math.min(...boundaries.map(b=>z^b))));const rank=Math.log2(cosets.size);if(!Number.isInteger(rank))throw Error('NON_VECTOR_QUOTIENT');return rank;}
function allRanks(simplices){const levels=[...new Set(simplices.map(s=>s.birth))].sort((a,b)=>a-b),times=[levels[0]-1,...levels,levels.at(-1)+1],dimension=Math.max(...simplices.map(s=>s.vertices.length-1));return times.flatMap((from,i)=>times.slice(i).map(to=>({from,to,ranks:Array.from({length:dimension+1},(_,p)=>persistentRank(simplices,p,from,to))})));}
module.exports={span,model,persistentRank,allRanks};
