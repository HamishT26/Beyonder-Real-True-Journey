#!/usr/bin/env python3
"""Pure metadata fixture contract for exact owner manifest admission."""
from pathlib import PurePosixPath
def validate(rows,inventory):
 if not isinstance(rows,list) or len(rows)!=len(inventory):return False
 names=[]
 for row in rows:
  if not isinstance(row,dict) or set(row)!={'path','bytes','sha256'}:return False
  p=row['path']
  if not isinstance(p,str) or ':' in p or '\\' in p or PurePosixPath(p).is_absolute() or '..' in PurePosixPath(p).parts or p not in inventory:return False
  if type(row['bytes']) is not int or row['bytes']<0:return False
  if row['bytes']!=inventory[p]['bytes'] or row['sha256']!=inventory[p]['sha256']:return False
  names.append(p)
 return len(names)==len(set(names)) and set(names)==set(inventory)
