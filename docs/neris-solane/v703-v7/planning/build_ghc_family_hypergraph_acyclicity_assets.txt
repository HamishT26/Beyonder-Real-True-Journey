def stable_json(value: Any) -> str:
    return json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":"))


def sha256_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def sha256_value(value: Any) -> str:
    return sha256_bytes(stable_json(value).encode("utf-8"))


def git_blob_sha1(data: bytes) -> str:
    prefix = f"blob {len(data)}\0".encode("ascii")
    return hashlib.sha1(prefix + data).hexdigest()


def write_text(path: Path, text: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(text.replace("\r\n", "\n"), encoding="utf-8", newline="\n")


def write_json(path: Path, value: Any) -> None:
    write_text(path, json.dumps(value, ensure_ascii=False, indent=2, sort_keys=False) + "\n")


def read_json(path: Path) -> Any:
    return json.loads(path.read_text(encoding="utf-8"))


def git(*args: str, check: bool = True) -> str:
    cp = subprocess.run(["git", *args], cwd=REPO, text=True, encoding="utf-8", capture_output=True)
    if check and cp.returncode != 0:
        raise RuntimeError(f"git {' '.join(args)} failed: {cp.stderr.strip()}")
    return cp.stdout.strip()


def git_show(spec: str) -> bytes:
    cp = subprocess.run(["git", "show", spec], cwd=REPO, capture_output=True)
    if cp.returncode != 0:
        raise RuntimeError(cp.stderr.decode("utf-8", errors="replace"))
    return cp.stdout


def normalize_hypergraph(fixture: dict[str, Any]) -> dict[str, Any]:
    vertices = tuple(str(v) for v in fixture["vertices"])
    if not vertices or len(set(vertices)) != len(vertices):
        raise ValueError("vertices_must_be_nonempty_and_unique")
    if any(not v or any(ch.isspace() for ch in v) for v in vertices):
        raise ValueError("vertex_labels_must_be_nonempty_tokens")
    vset = set(vertices)
    edges: list[tuple[str, ...]] = []
    for raw in fixture["edges"]:
        if not raw:
            raise ValueError("edges_must_be_nonempty")
        edge = tuple(sorted(str(v) for v in raw))
        if len(edge) != len(set(edge)):
            raise ValueError("edge_vertices_must_be_unique")
        if not set(edge) <= vset:
            raise ValueError("edge_vertex_outside_declared_vertex_set")
        edges.append(edge)
    edges = sorted(set(edges), key=lambda e: (len(e), e))
    if not edges:
        raise ValueError("at_least_one_edge_required")
    return {"id": fixture.get("id", "derived"), "vertices": list(vertices), "edges": [list(e) for e in edges]}


def powerset(items: Iterable[Any], include_empty: bool = False) -> Iterable[tuple[Any, ...]]:
    values = tuple(items)
    start = 0 if include_empty else 1
    for size in range(start, len(values) + 1):
        yield from itertools.combinations(values, size)


def primal_adjacency(h: dict[str, Any]) -> dict[str, set[str]]:
    adj = {v: set() for v in h["vertices"]}
    for edge in h["edges"]:
        for a, b in itertools.combinations(edge, 2):
            adj[a].add(b)
            adj[b].add(a)
    return adj


def is_clique(vertices: Iterable[str], adj: dict[str, set[str]]) -> bool:
    values = tuple(vertices)
    return all(b in adj[a] for a, b in itertools.combinations(values, 2))


def maximal_cliques_bruteforce(h: dict[str, Any]) -> list[list[str]]:
    adj = primal_adjacency(h)
    candidates = [set(s) for s in powerset(h["vertices"]) if is_clique(s, adj)]
    maximal = [c for c in candidates if not any(c < other for other in candidates)]
    return [sorted(c) for c in sorted(maximal, key=lambda s: (len(s), tuple(sorted(s))))]


def conformal_engine(h: dict[str, Any]) -> tuple[bool, list[list[str]]]:
    edges = [set(e) for e in h["edges"]]
    uncovered = [c for c in maximal_cliques_bruteforce(h) if not any(set(c) <= edge for edge in edges)]
    return not uncovered, uncovered


def conformal_oracle(h: dict[str, Any]) -> bool:
    adj = primal_adjacency(h)
    edges = [set(e) for e in h["edges"]]
    for subset in powerset(h["vertices"]):
        if is_clique(subset, adj) and not any(set(subset) <= edge for edge in edges):
            return False
    return True


def peo_valid(order: Iterable[str], adj: dict[str, set[str]]) -> bool:
    order = tuple(order)
    pos = {v: i for i, v in enumerate(order)}
    for v in order:
        later = [u for u in adj[v] if pos[u] > pos[v]]
        if not is_clique(later, adj):
            return False
    return True


def peo_oracle(h: dict[str, Any]) -> list[str] | None:
    adj = primal_adjacency(h)
    for order in itertools.permutations(h["vertices"]):
        if peo_valid(order, adj):
            return list(order)
    return None


def peo_mcs(h: dict[str, Any]) -> list[str] | None:
    adj = primal_adjacency(h)
    weights = {v: 0 for v in h["vertices"]}
    unused = set(h["vertices"])
    selected: list[str] = []
    while unused:
        v = min(unused, key=lambda x: (-weights[x], x))
        unused.remove(v)
        selected.append(v)
        for u in adj[v] & unused:
            weights[u] += 1
    order = list(reversed(selected))
    return order if peo_valid(order, adj) else None


def canonical_cycle(cycle: Iterable[str]) -> tuple[str, ...]:
    values = tuple(cycle)
    rotations = []
    for seq in (values, tuple(reversed(values))):
        rotations.extend(seq[i:] + seq[:i] for i in range(len(seq)))
    return min(rotations)


def chordless_cycles(h: dict[str, Any]) -> list[list[str]]:
    adj = primal_adjacency(h)
    found: set[tuple[str, ...]] = set()
    vertices = tuple(h["vertices"])
    for size in range(4, len(vertices) + 1):
        for subset in itertools.combinations(vertices, size):
            for perm in itertools.permutations(subset[1:]):
                cycle = (subset[0],) + perm
                if all(cycle[(i + 1) % size] in adj[cycle[i]] for i in range(size)):
                    if all(cycle[j] not in adj[cycle[i]] for i in range(size) for j in range(i + 1, size) if (j - i) not in (1, size - 1)):
                        found.add(canonical_cycle(cycle))
    return [list(c) for c in sorted(found)]


def canonical_edge_state(edges: Iterable[Iterable[str]]) -> tuple[tuple[str, ...], ...]:
    normalized = {tuple(sorted(edge)) for edge in edges if edge}
    return tuple(sorted(normalized, key=lambda e: (len(e), e)))


def gyo_deterministic(h: dict[str, Any]) -> dict[str, Any]:
    state = canonical_edge_state(h["edges"])
    trace: list[dict[str, Any]] = []
    while state:
        changed = False
        counts: dict[str, int] = {}
        for edge in state:
            for v in edge:
                counts[v] = counts.get(v, 0) + 1
        for v in sorted(counts):
            if counts[v] <= 1:
                next_state = canonical_edge_state(tuple(x for x in edge if x != v) for edge in state)
                trace.append({"rule": "remove_vertex_degree_at_most_one", "subject": v, "before": [list(e) for e in state], "after": [list(e) for e in next_state]})
                state = next_state
                changed = True
                break
        if changed:
            continue
        for i, edge in enumerate(state):
            for j, other in enumerate(state):
                if i != j and set(edge) < set(other):
                    next_state = canonical_edge_state(e for k, e in enumerate(state) if k != i)
                    trace.append({"rule": "remove_contained_edge", "subject": list(edge), "container": list(other), "before": [list(e) for e in state], "after": [list(e) for e in next_state]})
                    state = next_state
                    changed = True
                    break
            if changed:
                break
        if not changed:
            break
    return {"reduced_empty": not state, "residual": [list(e) for e in state], "trace": trace}


def gyo_oracle(h: dict[str, Any]) -> bool:
    start = canonical_edge_state(h["edges"])
    queue = [start]
    seen = {start}
    while queue:
        state = queue.pop(0)
        if not state:
            return True
        counts: dict[str, int] = {}
        for edge in state:
            for v in edge:
                counts[v] = counts.get(v, 0) + 1
        next_states: list[tuple[tuple[str, ...], ...]] = []
        for v, count in counts.items():
            if count <= 1:
                next_states.append(canonical_edge_state(tuple(x for x in edge if x != v) for edge in state))
        for i, edge in enumerate(state):
            for j, other in enumerate(state):
                if i != j and set(edge) < set(other):
                    next_states.append(canonical_edge_state(e for k, e in enumerate(state) if k != i))
        for nxt in next_states:
            if nxt not in seen:
                seen.add(nxt)
                queue.append(nxt)
    return False


def prufer_tree(sequence: tuple[int, ...], n: int) -> list[tuple[int, int]]:
    degree = [1] * n
    for x in sequence:
        degree[x] += 1
    edges: list[tuple[int, int]] = []
    for x in sequence:
        leaf = min(i for i, d in enumerate(degree) if d == 1)
        edges.append(tuple(sorted((leaf, x))))
        degree[leaf] -= 1
        degree[x] -= 1
    remaining = [i for i, d in enumerate(degree) if d == 1]
    if len(remaining) == 2:
        edges.append(tuple(sorted(remaining)))
    return sorted(edges)


def all_labeled_trees(n: int) -> Iterable[list[tuple[int, int]]]:
    if n <= 1:
        yield []
    elif n == 2:
        yield [(0, 1)]
    else:
        for seq in itertools.product(range(n), repeat=n - 2):
            yield prufer_tree(seq, n)


def running_intersection(h: dict[str, Any], tree: Iterable[tuple[int, int]]) -> tuple[bool, dict[str, Any] | None]:
    edges = [set(e) for e in h["edges"]]
    adj = {i: set() for i in range(len(edges))}
    for a, b in tree:
        adj[a].add(b)
        adj[b].add(a)
    for v in h["vertices"]:
        nodes = {i for i, edge in enumerate(edges) if v in edge}
        if len(nodes) <= 1:
            continue
        reached = {min(nodes)}
        queue = [min(nodes)]
        while queue:
            cur = queue.pop(0)
            for nxt in adj[cur] & nodes:
                if nxt not in reached:
                    reached.add(nxt)
                    queue.append(nxt)
        if reached != nodes:
            return False, {"vertex": v, "required_edge_nodes": sorted(nodes), "connected_edge_nodes": sorted(reached)}
    return True, None


def join_tree_oracle(h: dict[str, Any]) -> list[list[int]] | None:
    for tree in all_labeled_trees(len(h["edges"])):
        ok, _ = running_intersection(h, tree)
        if ok:
            return [list(e) for e in tree]
    return None


def join_tree_engine(h: dict[str, Any]) -> list[list[int]] | None:
    count = len(h["edges"])
    if count <= 1:
        return []
    weighted = []
    sets = [set(e) for e in h["edges"]]
    for a, b in itertools.combinations(range(count), 2):
        weighted.append((-len(sets[a] & sets[b]), a, b))
    weighted.sort()
    parent = list(range(count))

    def find(x: int) -> int:
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    tree: list[tuple[int, int]] = []
    for _, a, b in weighted:
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[rb] = ra
            tree.append((a, b))
            if len(tree) == count - 1:
                break
    ok, _ = running_intersection(h, tree)
    return [list(e) for e in sorted(tree)] if ok else None


def alpha_facts(fixture: dict[str, Any]) -> dict[str, Any]:
    h = normalize_hypergraph(fixture)
    conformal, uncovered = conformal_engine(h)
    conformal_o = conformal_oracle(h)

    peo_o = peo_oracle(h)
    cycles = chordless_cycles(h)
    gyo_e = gyo_deterministic(h)
    gyo_o = gyo_oracle(h)
    join_e = join_tree_engine(h)
    join_o = join_tree_oracle(h)
    booleans = {
        "engine_conformal_and_chordal": bool(conformal and peo_e is not None),
        "oracle_conformal_and_chordal": bool(conformal_o and peo_o is not None),
        "deterministic_gyo_empty": bool(gyo_e["reduced_empty"]),
        "oracle_gyo_empty": bool(gyo_o),
        "engine_join_tree_exists": join_e is not None,
        "oracle_join_tree_exists": join_o is not None,
        "no_chordless_cycle_and_conformal": bool(not cycles and conformal),
    }
    if len(set(booleans.values())) != 1:
        raise AssertionError(f"alpha characterization disagreement for {h['id']}: {booleans}")
    alpha = next(iter(booleans.values()))
    if "expected_alpha" in fixture and alpha != bool(fixture["expected_alpha"]):
        raise AssertionError(f"frozen expected alpha mismatch for {h['id']}")
    return {
        "hypergraph": h,
        "alpha": alpha,
        "characterizations": booleans,
        "conformal": conformal,
        "uncovered_maximal_cliques": uncovered,
        "peo_engine": peo_e,
        "peo_oracle": peo_o,
        "chordless_cycles": cycles,
        "gyo": gyo_e,
        "join_tree_engine": join_e,
        "join_tree_oracle": join_o,
    }


def restrict_vertices(h: dict[str, Any], subset: Iterable[str]) -> dict[str, Any] | None:
    keep = set(subset)
    edges = canonical_edge_state(keep & set(edge) for edge in h["edges"])
    if not edges:
        return None
    return {"id": "vertex-restriction", "vertices": sorted(keep), "edges": [list(e) for e in edges]}


def edge_subset_h(h: dict[str, Any], indices: Iterable[int]) -> dict[str, Any]:
    return {"id": "edge-subset", "vertices": list(h["vertices"]), "edges": [h["edges"][i] for i in indices]}


def component_join_forest(h: dict[str, Any]) -> list[dict[str, Any]]:
    edge_sets = [set(e) for e in h["edges"]]
    adjacency = {i: set() for i in range(len(edge_sets))}
    for a, b in itertools.combinations(range(len(edge_sets)), 2):
        if edge_sets[a] & edge_sets[b]:
            adjacency[a].add(b)
            adjacency[b].add(a)
    unseen = set(adjacency)
    components = []
    while unseen:
        start = min(unseen)
        seen = {start}
        queue = [start]
        while queue:
            cur = queue.pop(0)
            for nxt in adjacency[cur]:
                if nxt not in seen:
                    seen.add(nxt)
                    queue.append(nxt)
        unseen -= seen
        local_edges = [h["edges"][i] for i in sorted(seen)]
        local_vertices = sorted(set().union(*(set(e) for e in local_edges)))
        local = {"id": "component", "vertices": local_vertices, "edges": local_edges}
        tree = join_tree_oracle(normalize_hypergraph(local))
        components.append({"edge_indices": sorted(seen), "join_tree_exists": tree is not None, "join_tree": tree})
    return components


def leaf_pruning_order(h: dict[str, Any], tree_rows: list[list[int]] | None) -> list[int] | None:
    if tree_rows is None:
        return None
    nodes = set(range(len(h["edges"])))
    adjacency = {i: set() for i in nodes}
    for a, b in tree_rows:
        adjacency[a].add(b)
        adjacency[b].add(a)
    order = []
    while len(nodes) > 1:
        leaf = min(i for i in nodes if len(adjacency[i] & nodes) <= 1)
        order.append(leaf)
        nodes.remove(leaf)
    order.extend(sorted(nodes))
    return order

def evaluate_mechanism(mechanism: str, fixture: dict[str, Any]) -> dict[str, Any]:
    facts = alpha_facts(fixture)
    h = facts["hypergraph"]
    outcome = outcome_for(mechanism)
    if mechanism == "chordless_cycle_obstruction":
        value = {"exists": bool(facts["chordless_cycles"]), "lexical_witness": facts["chordless_cycles"][0] if facts["chordless_cycles"] else None}
    elif mechanism == "perfect_elimination_order":
        value = {"exists": facts["peo_engine"] is not None, "engine_order": facts["peo_engine"], "oracle_exists": facts["peo_oracle"] is not None}
    elif mechanism == "maximal_clique_conformality":
        value = {"conformal": facts["conformal"], "uncovered_maximal_cliques": facts["uncovered_maximal_cliques"], "oracle_agreement": facts["conformal"] == conformal_oracle(h)}
    elif mechanism == "gyo_reduction_trace":
        value = facts["gyo"]
    elif mechanism == "join_tree_certificate":
        value = {"exists": facts["join_tree_engine"] is not None, "engine_tree": facts["join_tree_engine"], "oracle_exists": facts["join_tree_oracle"] is not None}
    elif mechanism == "running_intersection_audit":
        candidate = facts["join_tree_engine"]
        if candidate is None:
            value = {"candidate_exists": False, "audit": "not_applicable_noncertificate"}
        else:
            ok, obstruction = running_intersection(h, [tuple(e) for e in candidate])
            value = {"candidate_exists": True, "valid": ok, "obstruction": obstruction}
    elif mechanism == "alpha_acyclicity_equivalence":
        value = {"alpha": facts["alpha"], "characterizations": facts["characterizations"], "all_equal": len(set(facts["characterizations"].values())) == 1}
    elif mechanism == "edge_induced_alpha_profile":
        rows = []
        for indices in powerset(range(len(h["edges"]))):
            derived = edge_subset_h(h, indices)
            rows.append({"edge_indices": list(indices), "alpha": alpha_facts(derived)["alpha"]})
        value = {"rows": rows, "all_alpha": all(row["alpha"] for row in rows)}
    elif mechanism == "vertex_induced_alpha_profile":
        rows = []
        for subset in powerset(h["vertices"]):
            derived = restrict_vertices(h, subset)
            rows.append({"vertices": list(subset), "alpha": True if derived is None else alpha_facts(derived)["alpha"], "vacuous_no_edge": derived is None})
        value = {"rows": rows, "all_alpha": all(row["alpha"] for row in rows)}
    elif mechanism == "single_edge_deletion_profile":
        rows = []
        for i in range(len(h["edges"])):
            remaining = [j for j in range(len(h["edges"])) if j != i]
            alpha = True if not remaining else alpha_facts(edge_subset_h(h, remaining))["alpha"]
            rows.append({"deleted_edge_index": i, "alpha": alpha, "vacuous_no_edge": not remaining})
        value = {"rows": rows}
    elif mechanism == "single_vertex_deletion_profile":
        rows = []
        for v in h["vertices"]:
            derived = restrict_vertices(h, [x for x in h["vertices"] if x != v])
            rows.append({"deleted_vertex": v, "alpha": True if derived is None else alpha_facts(derived)["alpha"], "vacuous_no_edge": derived is None})
        value = {"rows": rows}
    elif mechanism == "vertex_relabel_invariance":
        replacement = {v: f"r{index:02d}" for index, v in enumerate(reversed(h["vertices"]), 1)}
        relabeled = {"id": "relabeled", "vertices": [replacement[v] for v in h["vertices"]], "edges": [[replacement[v] for v in edge] for edge in h["edges"]]}
        after = alpha_facts(relabeled)["alpha"]
        value = {"before": facts["alpha"], "after": after, "invariant": facts["alpha"] == after, "bijection": replacement}
    elif mechanism == "edge_order_invariance":
        reordered = {"id": "reordered", "vertices": h["vertices"], "edges": list(reversed(h["edges"]))}
        after = alpha_facts(reordered)["alpha"]
        value = {"before": facts["alpha"], "after": after, "invariant": facts["alpha"] == after}
    elif mechanism == "component_join_forest":
        value = {"components": component_join_forest(h)}
    elif mechanism == "join_tree_leaf_pruning":
        value = {"join_tree_exists": facts["join_tree_engine"] is not None, "leaf_pruning_order": leaf_pruning_order(h, facts["join_tree_engine"])}
    elif mechanism == "cycle_obstruction_witness":
        if facts["alpha"]:
            obstruction = None
        elif facts["uncovered_maximal_cliques"]:
            obstruction = {"kind": "uncovered_maximal_clique", "witness": facts["uncovered_maximal_cliques"][0]}
        elif facts["chordless_cycles"]:
            obstruction = {"kind": "chordless_cycle", "witness": facts["chordless_cycles"][0]}
        else:
            obstruction = {"kind": "join_tree_absence", "witness": None}
        value = {"alpha": facts["alpha"], "obstruction": obstruction}
    elif mechanism == "contained_edge_invariance":
        contained = [h["edges"][0][0]]
        if contained in h["edges"]:
            contained = h["edges"][0][:-1]
        augmented = {"id": "contained-edge", "vertices": h["vertices"], "edges": h["edges"] + [contained]}
        after = alpha_facts(augmented)["alpha"]
        value = {"before": facts["alpha"], "after": after, "proper_contained_edge": contained, "invariant": facts["alpha"] == after}
    elif mechanism == "acyclicity_trinity_nonconversion_map":
        value = {"finite_result_represented": facts["alpha"], "gmuth_empirical_claim": False, "thos_deployment_claim": False, "freed_id_authority_claim": False}
    elif mechanism == "external_join_tree_benchmark_gap":
        value = {"external_benchmark_present": False, "independent_reproduction_present": False, "gap_preserved": True}
    elif mechanism == "hypergraph_authority_nonpromotion_gate":
        value = {"professional_authority": False, "legal_authority": False, "cultural_authority": False, "maori_authority": False, "production_authority": False, "gate_preserved": True}
    else:
        raise KeyError(mechanism)
    passed = True
    if outcome == "completed":
        if mechanism in {"perfect_elimination_order", "join_tree_certificate"}:
            passed = value.get("exists") == value.get("oracle_exists")
        elif "invariant" in value:
            passed = bool(value["invariant"])
        elif mechanism == "alpha_acyclicity_equivalence":
            passed = bool(value["all_equal"])
        elif mechanism == "maximal_clique_conformality":
            passed = bool(value["oracle_agreement"])
    if not passed:
        raise AssertionError(f"mechanism check failed: {mechanism}/{h['id']}")
    return {
        "fixture_id": h["id"],
        "mechanism": mechanism,
        "session": session_for(mechanism),
        "outcome": outcome,
        "passed": passed,
        "alpha_acyclic": facts["alpha"],
        "value": value,
        "input_sha256": sha256_value(h),
        "value_sha256": sha256_value(value),
        "boundary": "Finite synthetic same-owner evidence only; no external, empirical, production, professional, legal, cultural, Maori-authority, AGI/ASI, consciousness/personhood, Theory-of-Everything, canon, or Stage 20 promotion.",
    }

    }

    write_json(directory / "manifest.json", manifest)

def validate_mutated_subject(subject: dict[str, Any], proposal: dict[str, Any], fixture: dict[str, Any]) -> tuple[bool, str]:
    if subject.get("proposal_id") != proposal["proposal_id"]:
        return False, "proposal_id_mismatch"
    if subject.get("expected_disposition") != proposal["expected_disposition"]:
        return False, "expected_disposition_mismatch"
    try:
        normalized = normalize_hypergraph(subject["fixture"])
    except (KeyError, TypeError, ValueError) as exc:
        return False, str(exc)
    if sha256_value(normalized) != sha256_value(normalize_hypergraph(fixture)):
        return False, "fixture_digest_mismatch"
    return True, "accepted"

def portfolio_rows(stage: str, kind: str, phase_results: list[dict[str, Any]], proposal_by_id: dict[str, dict[str, Any]]) -> list[dict[str, Any]]:
    rows = []
    for i in range(300):
        result = phase_results[i % len(phase_results)]
        proposal = proposal_by_id[result["proposal_id"]]
        if kind == "safe_now":
            observed = result["passed"] and result["outcome"] in CORE_OUTCOMES
            check = "bounded_result_and_outcome_allowlist"
        elif kind == "candidate":
            observed = result["value_sha256"] == sha256_value(result["value"])
            check = "result_digest_recomputation"
        elif kind == "clean_fix_refine":
            observed = result["outcome"] == proposal["expected_disposition"] and result["fixture_id"] == proposal["fixture_id"]
            check = "proposal_result_binding"
        else:
            raise KeyError(kind)
        if not observed:
            raise AssertionError(f"portfolio task failed: {stage}/{kind}/{i+1}")
        rows.append({
            "task_id": f"NS7037-{stage.upper()}-{kind.upper().replace('_','-')}-{i+1:03d}",
            "proposal_id": result["proposal_id"],
            "check": check,
            "variant": "primary" if i < 150 else "repeat_without_domain_reclassification",
            "executed": True,
            "passed": True,
            "evidence_value_sha256": result["value_sha256"],
            "external_action": False,
            "spend_usd": 0,
        })
    return rows
