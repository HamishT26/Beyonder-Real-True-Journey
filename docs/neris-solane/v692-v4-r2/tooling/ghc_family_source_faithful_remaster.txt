function stamp() {
  const now = new Date();
  const nz = new Intl.DateTimeFormat("en-NZ", {
    timeZone: "Pacific/Auckland",
    dateStyle: "full",
    timeStyle: "long",
  }).format(now);
  return { utc: now.toISOString(), pacific_auckland: nz };
}

function shaBuffer(value) {
  return crypto.createHash("sha256").update(value).digest("hex");
}

function shaFile(file) {
  return shaBuffer(fs.readFileSync(file));
}

function words(value) {
  const matches = String(value).match(/\S+/gu);
  return matches ? matches.length : 0;
}

function stable(value) {
  return `${JSON.stringify(value, null, 2)}\n`;
}

function ensureInside(base, target) {
  const resolvedBase = path.resolve(base);
  const resolved = path.resolve(target);
  if (resolved !== resolvedBase && !resolved.startsWith(`${resolvedBase}${path.sep}`)) {
    throw new Error(`OUT_OF_SCOPE_PATH:${target}`);
  }
  return resolved;
}

function ensureAllowed(file) {
  const ext = path.extname(file).toLowerCase();
  if (!ALLOWED_EXTENSIONS.has(ext)) throw new Error(`FORBIDDEN_EXTENSION:${file}`);
}

function writeText(relative, value) {
  const target = ensureInside(PHASE, path.join(PHASE, ...relative.split("/")));
  ensureAllowed(target);
  fs.mkdirSync(path.dirname(target), { recursive: true });
  fs.writeFileSync(target, String(value), "utf8");
  return target;
}

function writeJson(relative, value) {
  return writeText(relative, stable(value));
}

function readJson(file) {
  return JSON.parse(fs.readFileSync(file, "utf8"));
}

function git(args) {
  return cp.execFileSync("git", ["-C", REPO, ...args], { encoding: "utf8" }).trim();
}

function runTxt(file, args) {
  const loader = "require.extensions['.txt']=require.extensions['.js'];require(process.argv[1]);";
  return cp.execFileSync(process.execPath, ["--input-type=commonjs", "-e", loader, file, ...args], { encoding: "utf8" });
}

function expectedArg(args) {
  const raw = args.find((item) => item.startsWith("--expected="));
  if (!raw) throw new Error("MISSING_EXPECTED_COMMIT");
  return raw.slice("--expected=".length);
}

function assertHead(expected) {
  const observed = git(["rev-parse", "HEAD"]);
  if (observed !== expected) throw new Error(`HEAD_MISMATCH:${observed}:${expected}`);
}

function listFiles(root) {
  if (!fs.existsSync(root)) return [];
  const out = [];
  const walk = (dir) => {
    for (const entry of fs.readdirSync(dir, { withFileTypes: true }).sort((a, b) => a.name.localeCompare(b.name))) {
      const full = path.join(dir, entry.name);
      if (entry.isDirectory()) walk(full);
      else if (entry.isFile()) out.push(full);
    }
  };
  walk(root);
  return out;
}

function relativePhase(file) {
  return path.relative(PHASE, file).split(path.sep).join("/");
}

function manifest(root, excludes = []) {
  const excluded = new Set(excludes);
  return listFiles(root)
    .filter((file) => ALLOWED_EXTENSIONS.has(path.extname(file).toLowerCase()))
    .map((file) => ({
      path: relativePhase(file),
      bytes: fs.statSync(file).size,
      sha256: shaFile(file),
    }))
    .filter((row) => !excluded.has(row.path))
    .sort((a, b) => a.path.localeCompare(b.path));
}

function validateTex(text) {
  let depth = 0;
  for (const char of text) {
    if (char === "{") depth += 1;
    if (char === "}") depth -= 1;
    if (depth < 0) return false;
  }
  const required = ["\\documentclass", "\\begin{document}", "\\end{document}", "\\mathcal{G}_{AB}", "\\Omega_{AB}", "NOT\\_READY\\_FOR\\_STAGE\\_20"];
  return depth === 0 && required.every((token) => text.includes(token));
}

function validateFinal() {
  validatePlan();
  validateSession("x1", 110, 10, 5);
  validateSession("x2", 110, 10, 5);
  validateX2();
  const truth = readJson(path.join(PHASE, "final", "phase-truth.json"));
  const baton = fs.readFileSync(path.join(PHASE, "final", "hand-off-baton.md"), "utf8");
  const overview = fs.readFileSync(path.join(PHASE, "final", "final-integrated-overview.md"), "utf8");
  const seal = readJson(path.join(PHASE, "final", "content-seal.json"));
  if (truth.outcomes.completed !== 80 || truth.outcomes.represented !== 10 || truth.outcomes.open_gap !== 5 || truth.outcomes.exact_gate !== 5) throw new Error("FINAL_OUTCOME_COUNTS");
  if (words(baton) < 10000 || words(baton) > 100000 || words(overview) < 1800) throw new Error("FINAL_WORD_BUDGET");
  for (const row of seal.entries) {
    const file = path.join(PHASE, ...row.path.split("/"));
    if (shaFile(file) !== row.sha256 || fs.statSync(file).size !== row.bytes) throw new Error(`CONTENT_SEAL:${row.path}`);
  }
  for (const file of listFiles(PHASE)) ensureAllowed(file);
  for (const file of listFiles(PHASE).filter((item) => item.endsWith(".json"))) readJson(file);
}

  const upstream = git(["rev-parse", "@{u}"]);
  const tracking = git(["rev-parse", "refs/remotes/origin/codex/GHC-Family/neris-solane-v692-v4-main"]);
  cp.execFileSync("git", ["-C", REPO, "fetch", "origin", "refs/heads/codex/GHC-Family/neris-solane-v692-v4-main"], { encoding: "utf8" });
  const remote = git(["rev-parse", "FETCH_HEAD"]);
  check("four_way_equal", [upstream, tracking, remote].every((value) => value === head), { head, upstream, tracking, remote });
  const divergence = git(["rev-list", "--left-right", "--count", "HEAD...@{u}"]);
  check("divergence", divergence === "0\t0", divergence);
  const commits = git(["rev-list", "--reverse", `${SOURCE_HEAD}..HEAD`]).split(/\r?\n/u).filter(Boolean);
  check("four_remaster_commits", commits.length === 4, commits);
  const parents = commits.map((commit) => git(["show", "-s", "--format=%P", commit]).split(/\s+/u).filter(Boolean).length);
  check("single_parent_history", parents.every((count) => count === 1), parents);
  check("zero_merges", git(["rev-list", "--merges", `${SOURCE_HEAD}..HEAD`]) === "", git(["rev-list", "--merges", `${SOURCE_HEAD}..HEAD`]));
  const tracked = git(["ls-tree", "-r", "--name-only", "HEAD", PHASE_REL]).split(/\r?\n/u).filter(Boolean);
  check("owner_files_present", tracked.length > 0 && tracked.length < 2000, tracked.length);
  let blobMatches = 0;
  let jsonParses = 0;
  for (const rel of tracked) {
    const working = fs.readFileSync(path.join(REPO, ...rel.split("/")));
    const blob = cp.execFileSync("git", ["-C", REPO, "show", `HEAD:${rel}`], { encoding: null, maxBuffer: 64 * 1024 * 1024 });
    if (Buffer.compare(working, blob) === 0) blobMatches += 1;
    if (rel.endsWith(".json")) { JSON.parse(blob.toString("utf8")); jsonParses += 1; }
  }
  check("exact_git_blob_replay", blobMatches === tracked.length, { blobMatches, tracked: tracked.length });
  check("strict_json_parse", jsonParses > 0, jsonParses);
  check("allowed_extensions", tracked.every((rel) => ALLOWED_EXTENSIONS.has(path.extname(rel).toLowerCase())), [...new Set(tracked.map((rel) => path.extname(rel).toLowerCase()))]);
  validateFinal();
  check("phase_validators", true, "plan+x1+x2+final passed");
  const baton = fs.readFileSync(path.join(PHASE, "final", "hand-off-baton.md"), "utf8");
  check("baton_word_budget", words(baton) >= 10000 && words(baton) <= 100000, words(baton));
  const html = fs.readFileSync(path.join(PHASE, "x2", "source-faithful-equation-sandbox.html"), "utf8");
  check("html_contract", html.includes("<main>") && html.includes("aria-live=\"polite\"") && !/fetch\s*\(|<img\b|<svg\b/iu.test(html), "semantic local-only HTML");
  const combined = tracked.map((rel) => fs.readFileSync(path.join(REPO, ...rel.split("/")), "utf8")).join("\n");
  const privacyPatterns = {
    email: /\b[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}\b/giu,
    phone: /\b(?:\+?\d[\d .()-]{7,}\d)\b/gu,
    secret_assignment: /\b(?:api[_-]?key|token|password|secret)\s*[:=]\s*["'][^"']{8,}["']/giu,
    private_task_handle: /\b019[0-9a-f-]{20,}\b/giu,
  };
  const privacy = Object.fromEntries(Object.entries(privacyPatterns).map(([name, regex]) => [name, occurrences(combined, regex)]));
  check("bounded_privacy_scan", Object.values(privacy).every((count) => count === 0), privacy);
  check("prior_canonical_not_replayed", true, 0);
  check("complete_repository_suite_not_claimed", true, false);
  check("independent_reproduction_not_claimed", true, false);
  check("stage20_not_claimed", true, "NOT_READY_FOR_STAGE_20");
  const passed = checks.filter((row) => row.passed).length;
  const receipt = {
    schema: "ghc.family.exact-final-owner-scoped-canonical.v8",
    owner: "Neris Solane",
    phase: "v692-v4-r2",
    status: passed === checks.length ? "VALID_EXACT_FINAL_OWNER_SCOPED_CANONICAL" : "INVALID_EXACT_FINAL_OWNER_SCOPED_CANONICAL",
    exact_final: expected,
    checks: checks.length,
    checks_passed: passed,
    checks_failed: checks.length - passed,
    results: checks,
    invocation: 1,
    success: passed === checks.length ? 1 : 0,
    replay: 0,
    prior_canonical_replay: 0,
    scope: "Neris v692-v4-r2 owner delta only",
    complete_repository_suite: false,
    independent_reproduction: false,
    terminal_verdict: "NOT_READY_FOR_STAGE_20",
    timestamp: stamp(),
    boundary: BOUNDARY,
  };
  fs.writeFileSync(receiptPath, stable(receipt), "utf8");
  if (passed !== checks.length) throw new Error(`CANONICAL_FAILED:${passed}/${checks.length}`);
  process.stdout.write(stable({ status: receipt.status, checks: checks.length, checks_passed: passed, receipt: receiptPath, receipt_sha256: shaFile(receiptPath), latch_sha256: shaFile(latch) }));
}

function main(args) {
  const command = args[0];
  if (command === "plan") return plan(args.slice(1));
  if (command === "x1") return x1(args.slice(1));
  if (command === "x2") return x2(args.slice(1));
  if (command === "x2-recover-html") return recoverX2Html();
  if (command === "x2-refresh-manifest") return refreshX2ManifestAfterPromotion();
  if (command === "final-preflight") return finalPreflight(args.slice(1));
  if (command === "canonical") return canonical(args.slice(1));
  throw new Error("USAGE: node run.txt <plan|x1|x2|x2-recover-html|x2-refresh-manifest|final-preflight|canonical> [--expected=<commit>]");
}
