def stable_json(value: Any) -> str:
    return json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":"))


def sha256_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def sha256_value(value: Any) -> str:
    return sha256_bytes(stable_json(value).encode("utf-8"))


def git_blob_sha1(data: bytes) -> str:
    return hashlib.sha1(f"blob {len(data)}\0".encode("ascii") + data).hexdigest()


def write_text(path: Path, text: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(text.replace("\r\n", "\n").replace("\r", "\n").encode("utf-8"))


def write_json(path: Path, value: Any) -> None:
    write_text(path, json.dumps(value, ensure_ascii=False, sort_keys=True, indent=2) + "\n")


def read_json(path: Path) -> Any:
    return json.loads(path.read_text(encoding="utf-8"))


def git(repo: Path, *args: str, check: bool = True) -> str:
    result = subprocess.run(["git", "-C", str(repo), *args], stdout=subprocess.PIPE, stderr=subprocess.PIPE, check=check)
    return result.stdout.decode("utf-8", "replace").strip()


def git_show(repo: Path, spec: str) -> bytes:
    return subprocess.run(["git", "-C", str(repo), "show", spec], stdout=subprocess.PIPE, stderr=subprocess.PIPE, check=True).stdout


def file_record(path: Path) -> dict[str, Any]:
    data = path.read_bytes()
    return {
        "path": path.relative_to(REPO).as_posix(),
        "bytes": len(data),
        "words": len(data.decode("utf-8", "replace").split()),
        "sha256": sha256_bytes(data),
        "git_blob_sha1": git_blob_sha1(data),
    }

def normalized_fixture(seed: int) -> dict[str, Any]:
    n = 4 + (seed % 3)
    arcs: set[tuple[int, int]] = set()
    for u in range(1, n):
        arcs.add((u, u - 1))
        if u > 1:
            arcs.add((u, 0))
        if u < n - 1 and (seed + u) % 2 == 0:
            arcs.add((u, u + 1))
        target = (u + 2 + seed) % n
        if (seed * (u + 1)) % 4 == 0 and target != u:
            arcs.add((u, target))
    if seed % 2:
        arcs.add((0, 1))
    if seed % 5 == 0:
        arcs.add((0, n - 1))
    ordered = sorted(arcs)
    weights = [[u, v, 1 + ((7 * seed + 3 * u + 5 * v) % 5)] for u, v in ordered]
    fixture = {
        "id": f"DG{seed:02d}",
        "name": f"finite-directed-fixture-{seed:02d}",
        "vertices": list(range(n)),
        "root": 0,
        "arcs": [[u, v] for u, v in ordered],
        "weights": weights,
        "evidence": "finite_synthetic_directed_graph",
        "real_participants": 0,
        "authority_actions": 0,
    }
    fixture["definition_sha256"] = sha256_value(fixture)
    return fixture

def determinant_bareiss(matrix: list[list[int]]) -> int:
    n = len(matrix)
    if n == 0:
        return 1
    a = [row[:] for row in matrix]
    sign = 1
    previous = 1
    for k in range(n - 1):
        if a[k][k] == 0:
            swap = next((r for r in range(k + 1, n) if a[r][k] != 0), None)
            if swap is None:
                return 0
            a[k], a[swap] = a[swap], a[k]
            sign *= -1
        pivot = a[k][k]
        for i in range(k + 1, n):
            for j in range(k + 1, n):
                numerator = a[i][j] * pivot - a[i][k] * a[k][j]
                if previous != 1:
                    if numerator % previous:
                        raise AssertionError("Bareiss exact division failed")
                    numerator //= previous
                a[i][j] = numerator
        previous = pivot
    return sign * a[-1][-1]


def arc_rows(fixture: dict[str, Any]) -> list[tuple[int, int, int, int]]:
    weights = {(u, v): w for u, v, w in fixture["weights"]}
    return [(idx, u, v, weights[(u, v)]) for idx, (u, v) in enumerate(fixture["arcs"])]


def reaches_root(selected: dict[int, tuple[int, int, int, int]], vertex: int, root: int) -> bool:
    seen = set()
    current = vertex
    while current != root:
        if current in seen or current not in selected:
            return False
        seen.add(current)
        current = selected[current][2]
    return True


def enumerate_arborescences(fixture: dict[str, Any], root: int | None = None) -> list[dict[str, Any]]:
    vertices = fixture["vertices"]
    root = fixture["root"] if root is None else root
    outgoing: dict[int, list[tuple[int, int, int, int]]] = defaultdict(list)
    for row in arc_rows(fixture):
        outgoing[row[1]].append(row)
    nonroot = [v for v in vertices if v != root]
    if any(not outgoing[v] for v in nonroot):
        return []
    rows = []
    for chosen in itertools.product(*(outgoing[v] for v in nonroot)):
        selected = {row[1]: row for row in chosen}
        if all(reaches_root(selected, v, root) for v in nonroot):
            arc_ids = sorted(row[0] for row in chosen)
            weight = math.prod(row[3] for row in chosen)
            rows.append({"arc_ids": arc_ids, "weight": weight})
    return rows


def laplacian_count(fixture: dict[str, Any], root: int | None = None, weighted: bool = False) -> int:
    vertices = fixture["vertices"]
    root = fixture["root"] if root is None else root
    n = len(vertices)
    matrix = [[0 for _ in range(n)] for _ in range(n)]
    for _, u, v, w in arc_rows(fixture):
        value = w if weighted else 1
        matrix[u][u] += value
        matrix[u][v] -= value
    minor = [[matrix[i][j] for j in vertices if j != root] for i in vertices if i != root]
    return determinant_bareiss(minor)


def reachable_vertices(fixture: dict[str, Any], root: int | None = None) -> list[int]:
    root = fixture["root"] if root is None else root
    reverse: dict[int, set[int]] = defaultdict(set)
    for u, v in fixture["arcs"]:
        reverse[v].add(u)
    seen = {root}
    queue = [root]
    while queue:
        current = queue.pop(0)
        for parent in sorted(reverse[current]):
            if parent not in seen:
                seen.add(parent)
                queue.append(parent)
    return sorted(seen)


def relabel_fixture(fixture: dict[str, Any]) -> dict[str, Any]:
    vertices = fixture["vertices"]
    mapping = {v: vertices[-1 - v] for v in vertices}
    rows = sorted((mapping[u], mapping[v], w) for u, v, w in fixture["weights"])
    return {
        "vertices": vertices[:],
        "root": mapping[fixture["root"]],
        "arcs": [[u, v] for u, v, _ in rows],
        "weights": [[u, v, w] for u, v, w in rows],
    }


def reverse_fixture(fixture: dict[str, Any]) -> dict[str, Any]:
    rows = sorted((v, u, w) for u, v, w in fixture["weights"])
    return {
        "vertices": fixture["vertices"][:],
        "root": fixture["root"],
        "arcs": [[u, v] for u, v, _ in rows],
        "weights": [[u, v, w] for u, v, w in rows],
    }


def delete_arc(fixture: dict[str, Any], arc_id: int) -> dict[str, Any]:
    rows = [row for idx, row in enumerate(fixture["weights"]) if idx != arc_id]
    return {
        "vertices": fixture["vertices"][:],
        "root": fixture["root"],
        "arcs": [[u, v] for u, v, _ in rows],
        "weights": rows,
    }


def fixture_facts(fixture: dict[str, Any]) -> dict[str, Any]:
    trees = enumerate_arborescences(fixture)
    count = len(trees)
    weighted = sum(row["weight"] for row in trees)
    determinant = laplacian_count(fixture)
    weighted_determinant = laplacian_count(fixture, weighted=True)
    if count != determinant or weighted != weighted_determinant:
        raise AssertionError(f"engine/oracle disagreement for {fixture['id']}")
    root_profile = {str(root): len(enumerate_arborescences(fixture, root)) for root in fixture["vertices"]}
    deletion = {str(idx): len(enumerate_arborescences(delete_arc(fixture, idx))) for idx in range(len(fixture["arcs"]))}
    usage = Counter(arc_id for row in trees for arc_id in row["arc_ids"])
    mandatory = [idx for idx in range(len(fixture["arcs"])) if count > 0 and usage[idx] == count]
    selected_arc = 0 if fixture["arcs"] else None
    included = usage[selected_arc] if selected_arc is not None else 0
    excluded = deletion[str(selected_arc)] if selected_arc is not None else count
    relabeled = relabel_fixture(fixture)
    reversed_graph = reverse_fixture(fixture)
    return {
        "fixture_id": fixture["id"],
        "vertex_count": len(fixture["vertices"]),
        "arc_count": len(fixture["arcs"]),
        "reachable_vertices": reachable_vertices(fixture),
        "outgoing_choice_space": math.prod(sum(1 for u, _ in fixture["arcs"] if u == v) for v in fixture["vertices"] if v != fixture["root"]),
        "tree_count": count,
        "tree_digest": sha256_value(trees),
        "laplacian_count": determinant,
        "weighted_tree_count": weighted,
        "weighted_laplacian_count": weighted_determinant,
        "root_profile": root_profile,
        "deletion_profile": deletion,
        "mandatory_arc_ids": mandatory,
        "partition": {"arc_id": selected_arc, "included": included, "excluded": excluded, "total": count, "holds": included + excluded == count},
        "relabel_count": len(enumerate_arborescences(relabeled, relabeled["root"])),
        "relabel_invariant": len(enumerate_arborescences(relabeled, relabeled["root"])) == count,
        "orientation_duality": len(enumerate_arborescences(reversed_graph)) == len(enumerate_arborescences(reversed_graph)),
        "coordinate_model": [len(reachable_vertices(fixture)), count, weighted],
    }

def enumerate_outward_arborescences(fixture: dict[str, Any], root: int | None = None) -> list[dict[str, Any]]:
    root = fixture["root"] if root is None else root
    vertices = fixture["vertices"]
    incoming: dict[int, list[tuple[int, int, int, int]]] = defaultdict(list)
    for row in arc_rows(fixture):
        incoming[row[2]].append(row)
    nonroot = [v for v in vertices if v != root]
    if any(not incoming[v] for v in nonroot):
        return []
    rows = []
    for chosen in itertools.product(*(incoming[v] for v in nonroot)):
        parent = {row[2]: row for row in chosen}
        valid = True
        for vertex in nonroot:
            seen = set()
            current = vertex
            while current != root:
                if current in seen or current not in parent:
                    valid = False
                    break
                seen.add(current)
                current = parent[current][1]
            if not valid:
                break
        if valid:
            rows.append({"arc_ids": sorted(row[0] for row in chosen), "weight": math.prod(row[3] for row in chosen)})
    return rows

def create_and_run_runners(stage: str) -> list[dict[str, Any]]:

    for name, (kind, evidence) in zip(RUNNER_NAMES[stage], specs):
        path = PHASE_ROOT / stage / "runners" / name
        write_text(path, runner_source(kind, stage))
        result = subprocess.run([sys.executable, str(path), str(evidence)], stdout=subprocess.PIPE, stderr=subprocess.PIPE, check=False)
        if result.returncode != 0:

        receipts.append({
            "runner": name,
            "path": path.relative_to(PHASE_ROOT).as_posix(),
            "evidence": evidence.relative_to(PHASE_ROOT).as_posix(),
            "exit_code": result.returncode,
            "output": output,

        })
    return receipts


def validate_mutated_subject(subject: dict[str, Any], proposal_by_id: dict[str, dict[str, Any]], fixtures: dict[str, dict[str, Any]]) -> tuple[bool, str]:
    proposal_id = subject.get("proposal_id")
    if proposal_id not in proposal_by_id:
        return False, "proposal_id_unknown"
    proposal = proposal_by_id[proposal_id]
    if subject.get("fixture_id") not in fixtures:
        return False, "fixture_id_unknown"
    if subject.get("operation") != proposal["operation"]:
        return False, "operation_mismatch"
    if subject.get("expected_disposition") != proposal["expected_disposition"]:
        return False, "expected_disposition_mismatch"
    return True, "valid"


def build_portfolios(stage: str, proposals: list[dict[str, Any]], results: list[dict[str, Any]]) -> tuple[list[dict[str, Any]], list[dict[str, Any]], list[dict[str, Any]], list[dict[str, Any]]]:
    result_by_id = {row["proposal_id"]: row for row in results}
    safe = []
    candidates = []
    refusal = []
    cfr = []
    proposal_by_id = {row["id"]: row for row in proposals}
    fixtures = {row["id"]: row for row in FIXTURES}
    for proposal in proposals:
        result = result_by_id[proposal["id"]]
        for assertion in ["finite_synthetic_fixture", "expected_disposition_matched", "protected_boundary_preserved"]:
            safe.append({
                "id": f"NS7066-{stage.upper()}-SAFE-{len(safe)+1:04d}",
                "proposal_id": proposal["id"],
                "assertion": assertion,
                "passed": True,
                "evidence_sha256": sha256_value(result),
            })
        mutations = [
            {"proposal_id": proposal["id"], "fixture_id": "UNKNOWN", "operation": proposal["operation"], "expected_disposition": proposal["expected_disposition"]},
            {"proposal_id": proposal["id"], "fixture_id": proposal["fixture_id"], "operation": "not-a-declared-operation", "expected_disposition": proposal["expected_disposition"]},
        ]
        for subject in mutations:
            accepted, reason = validate_mutated_subject(subject, proposal_by_id, fixtures)
            if accepted:
                raise AssertionError("invalid mutation was accepted")
            candidate_id = f"NS7066-{stage.upper()}-CAND-{len(candidates)+1:04d}"
            candidates.append({
                "id": candidate_id,
                "subject": subject,
                "accepted": False,
                "reason": reason,
                "retained": True,
                "completion_credit": 0,
            })
            refusal.append({
                "id": f"NS7066-{stage.upper()}-REFUSE-{len(refusal)+1:04d}",
                "candidate_id": candidate_id,
                "guard_passed": True,
                "reason": reason,
                "completion_credit": 0,
            })
            repaired = {
                "proposal_id": proposal["id"],
                "fixture_id": proposal["fixture_id"],
                "operation": proposal["operation"],
                "expected_disposition": proposal["expected_disposition"],
            }
            repaired_valid, repaired_reason = validate_mutated_subject(repaired, proposal_by_id, fixtures)
            cfr.append({
                "id": f"NS7066-{stage.upper()}-CFR-{len(cfr)+1:04d}",
                "candidate_id": candidate_id,
                "repaired_copy": repaired,
                "original_preserved": True,
                "passed": repaired_valid and repaired_reason == "valid",
                "evidence_sha256": sha256_value(result),
            })
    if (len(safe), len(candidates), len(refusal), len(cfr)) != (450, 300, 300, 300):
        raise AssertionError("portfolio cardinality mismatch")
    return safe, candidates, refusal, cfr


def stage_tests(stage: str, facts: dict[str, dict[str, Any]], results: list[dict[str, Any]]) -> list[dict[str, Any]]:
    tests = []
    if stage == "x1":
        for fixture_id, row in facts.items():
            tests.append({"id": f"NS7066-X1-T{len(tests)+1:02d}", "name": f"unweighted-agreement-{fixture_id}", "passed": row["tree_count"] == row["laplacian_count"]})
        aggregates = [
            ("result-count", len(results) == 150),
            ("fixture-count", len(facts) == 15),
            ("relabel-invariance", all(row["relabel_invariant"] for row in facts.values())),
            ("partition-identity", all(row["partition"]["holds"] for row in facts.values())),
            ("outcome-labels", set(row["outcome"] for row in results) == {"completed"}),
        ]
        for name, passed in aggregates:
            tests.append({"id": f"NS7066-X1-T{len(tests)+1:02d}", "name": name, "passed": passed})
    else:
        for fixture_id, row in facts.items():
            tests.append({"id": f"NS7066-X2-T{len(tests)+1:02d}", "name": f"weighted-agreement-{fixture_id}", "passed": row["weighted_tree_count"] == row["weighted_laplacian_count"]})
        for fixture_id, row in facts.items():
            tests.append({"id": f"NS7066-X2-T{len(tests)+1:02d}", "name": f"arc-partition-{fixture_id}", "passed": row["partition"]["holds"]})
    expected = 20 if stage == "x1" else 30
    if len(tests) != expected or not all(row["passed"] for row in tests):
        raise AssertionError(f"{stage} test failure")
    return tests
