const gcd=(a,b)=>b?gcd(b,a%b):a<0n?-a:a;
const Q=(n,d=1n)=>{n=BigInt(n);d=BigInt(d);if(!d)throw Error('zero denominator');if(d<0n){n=-n;d=-d;}let g=gcd(n,d);return [n/g,d/g];};
const add=(a,b)=>Q(a[0]*b[1]+b[0]*a[1],a[1]*b[1]),mul=(a,b)=>Q(a[0]*b[0],a[1]*b[1]),str=a=>a[0]+'/'+a[1],sum=a=>a.reduce(add,Q(0)),pc=n=>n.toString(2).replaceAll('0','').length;
const choose=(n,k)=>{let a=1;for(let j=1;j<=k;j++)a=a*(n-j+1)/j;return a;};

}
