'use strict';
const fs=require('fs'),path=require('path'),crypto=require('crypto'),cp=require('child_process'),assert=require('assert/strict');
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const json=x=>JSON.stringify(x,null,2)+'\n';
function strictJSON(text){
 const parsed=JSON.parse(text);let i=0,depth=0;const ws=()=>{while(/\s/.test(text[i]||'!'))i++;};
 function string(){const start=i++;while(i<text.length){if(text[i]==='\\'){i+=2;continue;}if(text[i++]==='"')return JSON.parse(text.slice(start,i));}throw Error('UNTERMINATED_STRING');}
 function value(){ws();assert(++depth<=128,'JSON_DEPTH');if(text[i]==='{'){i++;ws();const keys=new Set();if(text[i]!=='}')while(true){const k=string();assert(!keys.has(k),'DUPLICATE_JSON_KEY');keys.add(k);ws();assert.equal(text[i++],':');value();ws();if(text[i]!==',')break;i++;ws();}assert.equal(text[i++],'}');}
 else if(text[i]==='['){i++;ws();if(text[i]!==']')while(true){value();ws();if(text[i]!==',')break;i++;}assert.equal(text[i++],']');}
 else if(text[i]==='"')string();else while(i<text.length&&!/[\s,}\]]/.test(text[i]))i++;depth--;}
 value();ws();assert.equal(i,text.length);function finite(x){if(typeof x==='number')assert(Number.isFinite(x),'NON_FINITE_NUMBER');if(x&&typeof x==='object')Object.values(x).forEach(finite);}finite(parsed);return parsed;
}
function readJSON(p){const b=fs.readFileSync(p),s=b.toString('utf8');assert(Buffer.from(s,'utf8').equals(b),'INVALID_UTF8');return strictJSON(s);}
function write(p,value,exclusive=true){fs.mkdirSync(path.dirname(p),{recursive:true});const s=typeof value==='string'?value:json(value);assert(s.split(/\s+/).length<=100000,'DOCUMENT_WORD_CAP');fs.writeFileSync(p,s,{encoding:'utf8',flag:exclusive?'wx':'w'});}
function files(root){if(!fs.existsSync(root))return[];const result=[];for(const item of fs.readdirSync(root,{withFileTypes:true})){const p=path.join(root,item.name);assert(!item.isSymbolicLink(),'SYMLINK_OUTSIDE_PROFILE');if(item.isDirectory())result.push(...files(p));else if(item.isFile())result.push(p);}return result.sort();}
function git(root,args,opts={}){return cp.execFileSync('git',args,{cwd:root,encoding:'utf8',maxBuffer:67108864,...opts}).trim();}
function blobs(root,specs){if(!specs.length)return[];const r=cp.spawnSync('git',['cat-file','--batch'],{cwd:root,input:specs.join('\n')+'\n',maxBuffer:67108864});assert.equal(r.status,0,'GIT_BATCH');const data=r.stdout;let pos=0;return specs.map(()=>{const end=data.indexOf(10,pos),head=data.subarray(pos,end).toString('ascii').split(' ');assert.equal(head[1],'blob','MISSING_BLOB');const size=Number(head[2]),b=data.subarray(end+1,end+1+size);pos=end+size+2;return b;});}
const privacyPatterns={private_identifier:/\b[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}\b/ig,private_path:/(?<![A-Za-z0-9+.-])(?:[A-Za-z]:[\\/]|\/Users\/|\/home\/)/g,credential:/\b(?:api[_-]?key|access[_-]?token|password|secret)\s*[:=]\s*[^\s,}]+/ig,private_route:/\b(?:codex|app|session|thread):\/\//ig,application_state:/\b(?:providerTabId|clientThreadId)\b/g};
function audit(root,paths){let parsed=0;const findings=[],codeFindings=[];for(const rel of paths){assert(['.json','.md','.txt','.html'].includes(path.extname(rel)),'EXTENSION');const b=fs.readFileSync(path.join(root,rel)),s=b.toString('utf8');assert(Buffer.from(s,'utf8').equals(b),'UTF8');assert(!s.includes('\r'),'LF_DOMAIN');assert(s.split(/\s+/).length<=100000,'WORD_CAP');if(rel.endsWith('.json')){strictJSON(s);parsed++;}for(const [kind,re]of Object.entries(privacyPatterns)){re.lastIndex=0;for(const m of s.matchAll(re))findings.push({path:rel,line:s.slice(0,m.index).split('\n').length,kind});}if(rel.endsWith('.txt')||rel.endsWith('.html')){if(/\beval\s*\(|\bnew\s+Function\s*\(|shell\s*:\s*true/.test(s))codeFindings.push({path:rel,kind:'dynamic_execution_or_shell_true'});}}return{files:paths.length,parsed,privacy:{classes:Object.keys(privacyPatterns),findings,complete_privacy:false},security:{findings:codeFindings,exhaustive_security:false}};}
function stage(root,base,stageName,bank){
 const folder=path.join(root,base,stageName),manifestRel=base+'/'+stageName+'/manifest.json',reviewRel=base+'/'+stageName+'/staged-review.json';
 let paths=files(folder).map(p=>path.relative(root,p).replaceAll('\\','/')).filter(p=>![manifestRel,reviewRel,base+'/'+stageName+'/preflight.json'].includes(p));assert(paths.length<=2000);const audited=audit(root,paths);assert.equal(audited.privacy.findings.length,0,'PRIVACY');assert.equal(audited.security.findings.length,0,'SECURITY');
 write(path.join(folder,'preflight.json'),{...audited,source_executors_invoked:0,canonical_invocations:0},false);paths=files(folder).map(p=>path.relative(root,p).replaceAll('\\','/')).filter(p=>![manifestRel,reviewRel].includes(p));const allow=[...paths,manifestRel,reviewRel].sort();write(path.join(root,reviewRel),{stage:stageName,literal_allowlist:allow,additive:true,reviewed:true},false);
 const existing=[...paths,reviewRel].sort();assert(existing.every(p=>/^[A-Za-z0-9_./-]+$/.test(p)));const list=path.join(bank,stageName+'-stage-paths.txt');write(list,existing.join('\n')+'\n',false);git(root,['add','--pathspec-from-file='+list]);
 const bindings=blobs(root,existing.map(p=>':'+p)).map((b,i)=>({path:existing[i],bytes:b.length,sha256:sha(b)}));write(path.join(root,manifestRel),{schema:'ghc.family.teryn-v692-v2.manifest.v1',stage:stageName,byte_domain:'UTF8_LF_GIT_BLOB',entries:bindings,self_exclusions:[manifestRel]});git(root,['add','--',manifestRel]);
 const actual=git(root,['diff','--cached','--name-only']).split('\n').filter(Boolean).sort();assert.deepEqual(actual,allow);assert.equal(git(root,['diff','--name-only']),'');git(root,['diff','--cached','--check']);return{stage:stageName,staged_files:allow.length,bindings:bindings.length,audited_files:audited.files,parsed:audited.parsed,privacy:audited.privacy,security:audited.security};
}
module.exports={sha,json,strictJSON,readJSON,write,files,git,blobs,audit,stage};
