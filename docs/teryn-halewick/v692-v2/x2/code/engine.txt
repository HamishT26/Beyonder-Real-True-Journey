'use strict';
// Finite replica and evidence profiles. Input metadata never supplies real authority.
const crypto=require('crypto');
class InvalidInput extends Error{constructor(){super('INVALID_INPUT');}}
const need=x=>{if(!x)throw new InvalidInput();};
const object=(x,keys)=>{need(x!==null&&typeof x==='object'&&!Array.isArray(x)&&Object.keys(x).length<=1000);if(keys)need(Object.keys(x).sort().join('|')===[...keys].sort().join('|'));return x;};
const list=x=>{need(Array.isArray(x)&&x.length<=1000);return x;};
const integer=x=>{need(Number.isSafeInteger(x)&&x>=0&&x<=1000000000);return x;};
const label=x=>{need(typeof x==='string'&&/^[A-Za-z0-9_-]{1,256}$/.test(x));return x;};
const text=x=>{need(typeof x==='string'&&x.length>0&&x.length<=256);return x;};
const labels=x=>list(x).map(label),bool=x=>{need(typeof x==='boolean');return x;};
const unique=xs=>need(new Set(xs).size===xs.length),cmp=(a,b)=>a<b?-1:a>b?1:0;
const ordered=map=>Object.fromEntries([...map].sort(([a],[b])=>cmp(a,b)));
function canonical(x){if(Array.isArray(x))return '['+x.map(canonical).join(',')+']';if(x&&typeof x==='object')return '{'+Object.keys(x).sort().map(k=>JSON.stringify(k)+':'+canonical(x[k])).join(',')+'}';return JSON.stringify(x);}
function vector(v){object(v);for(const [k,n]of Object.entries(v)){label(k);integer(n);}return v;}
function joinMaps(maps){const out=new Map();for(const map of maps)for(const [key,value]of Object.entries(vector(map)))out.set(key,Math.max(out.get(key)||0,value));return ordered(out);}
function version_vector_join(s){object(s,['vectors']);list(s.vectors);return{joined:joinMaps(s.vectors)};}
function observed_remove_set(s){object(s,['adds','removes']);list(s.adds);labels(s.removes);const byTag=new Map();for(const a of s.adds){object(a,['element','tag']);label(a.element);label(a.tag);if(byTag.has(a.tag))need(byTag.get(a.tag)===a.element);byTag.set(a.tag,a.element);}const removed=new Set(s.removes),active=[...byTag].filter(([tag])=>!removed.has(tag));return{elements:[...new Set(active.map(([,element])=>element))].sort(),active_tags:active.map(([tag])=>tag).sort()};}
function bounded_pn_counter(s){object(s,['states']);list(s.states);for(const state of s.states){object(state,['p','n']);vector(state.p);vector(state.n);}const positive=joinMaps(s.states.map(x=>x.p)),negative=joinMaps(s.states.map(x=>x.n));const total=Object.values(positive).reduce((a,b)=>a+b,0)-Object.values(negative).reduce((a,b)=>a+b,0);need(Number.isSafeInteger(total));return{positive,negative,total};}
function timestamp_register_review(s){object(s,['records']);list(s.records);for(const r of s.records){object(r,['at','value','author']);integer(r.at);label(r.author);}if(!s.records.length)return{state:'MISSING',at:null,values:[]};const at=Math.max(...s.records.map(r=>r.at)),byValue=new Map();for(const r of s.records.filter(r=>r.at===at))byValue.set(canonical(r.value),r.value);const values=[...byValue].sort(([a],[b])=>cmp(a,b)).map(([,v])=>v);return{state:values.length===1?'CURRENT':'CONFLICT',at,values};}
function event_correction_asof(s){object(s,['records','at']);list(s.records);integer(s.at);const ids=[];for(const r of s.records){object(r,['id','at','value','replaces']);ids.push(label(r.id));integer(r.at);labels(r.replaces);need(!r.replaces.includes(r.id));}unique(ids);const visible=s.records.filter(r=>r.at<=s.at),present=new Set(visible.map(r=>r.id)),replaced=new Set(visible.flatMap(r=>r.replaces)),missing=[...replaced].filter(x=>!present.has(x)).sort(),current=visible.filter(r=>!replaced.has(r.id)).sort((a,b)=>cmp(a.id,b.id));return{heads:current.map(r=>r.id),values:current.map(r=>r.value),missing,conflict:current.length>1};}
function prefixed_merkle_path(s){object(s,['leaf','path','root']);need(typeof s.leaf==='string'&&s.leaf.length<=4096);list(s.path);need(s.path.length<=64);const digest=h=>need(typeof h==='string'&&/^[0-9a-f]{64}$/.test(h));digest(s.root);const hash=b=>crypto.createHash('sha256').update(b).digest();let value=hash(Buffer.concat([Buffer.from([0]),Buffer.from(s.leaf,'utf8')]));for(const p of s.path){object(p,['side','hash']);need(['left','right'].includes(p.side));digest(p.hash);const sibling=Buffer.from(p.hash,'hex');value=hash(Buffer.concat(p.side==='left'?[Buffer.from([1]),sibling,value]:[Buffer.from([1]),value,sibling]));}return{verified:value.toString('hex')===s.root,origin_authenticated:false};}
function dimensional_exponent_balance(s){object(s,['terms']);list(s.terms);const names=[];for(const t of s.terms){object(t,['name','dimension']);names.push(label(t.name));list(t.dimension);need(t.dimension.length===7&&t.dimension.every(x=>Number.isSafeInteger(x)&&Math.abs(x)<=32));}unique(names);if(!s.terms.length)return{homogeneous:false,dimension:null,mismatches:[],empirical:false};const dimension=s.terms[0].dimension,mismatches=s.terms.filter(t=>!t.dimension.every((v,i)=>v===dimension[i])).map(t=>t.name).sort();return{homogeneous:!mismatches.length,dimension,mismatches,empirical:false};}
function evidence_layer_accounting(s){object(s,['records']);list(s.records);const byId=new Map(),conflicts=new Map();let duplicates=0;for(const r of s.records){object(r,['id','result','method','negative_ids','owner','phase']);text(r.id);text(r.method);text(r.owner);text(r.phase);labels(r.negative_ids);need(['pass','fail'].includes(r.result));if(r.result==='fail')need(r.negative_ids.length>0);const key=JSON.stringify([r.owner,r.phase,r.id]);if(byId.has(key)){if(canonical(byId.get(key))===canonical(r))duplicates++;else conflicts.set(key,[r.owner,r.phase,r.id]);}else byId.set(key,r);}if(conflicts.size)return{state:'CONFLICT',counts:null,duplicates,conflicts:[...conflicts].sort(([a],[b])=>cmp(a,b)).map(([,v])=>v)};const records=[...byId.values()],methods=new Set(records.map(r=>JSON.stringify([r.owner,r.phase,r.method]))),negatives=new Set(records.flatMap(r=>r.negative_ids.map(n=>JSON.stringify([r.owner,r.phase,n])))),failed=records.filter(r=>r.result==='fail').length;return{state:'COUNTED',counts:{methods:methods.size,witnesses:records.length,failed,passed:records.length-failed,negative_refs:negatives.size},duplicates,conflicts:[]};}
function evidence_review_gap(s){object(s,['required','provided']);labels(s.required);labels(s.provided);const required=[...new Set(s.required)].sort(),provided=new Set(s.provided);return{missing:required.filter(x=>!provided.has(x)),declared:required.filter(x=>provided.has(x)),assurance_established:false};}
function authority_scope_intersection(s){object(s,['grants','requested','now']);list(s.grants);labels(s.requested);integer(s.now);const reasons=[];s.grants.forEach((g,i)=>{object(g,['scope','expires','revoked']);labels(g.scope);integer(g.expires);bool(g.revoked);if(g.expires<=s.now)reasons.push('EXPIRED:'+i);if(g.revoked)reasons.push('REVOKED:'+i);});if(!s.grants.length)reasons.push('NO_GRANTS');const requested=[...new Set(s.requested)].sort(),candidates=reasons.length?[]:requested.filter(x=>s.grants.every(g=>g.scope.includes(x)));return{candidates,outside:requested.filter(x=>!candidates.includes(x)),reasons:reasons.sort(),executed:false,authority_established:false};}
const operations={version_vector_join,observed_remove_set,bounded_pn_counter,timestamp_register_review,event_correction_asof,prefixed_merkle_path,dimensional_exponent_balance,evidence_layer_accounting,evidence_review_gap,authority_scope_intersection};
function run(operation,subject){try{need(Object.hasOwn(operations,operation));const before=JSON.stringify(subject);need(typeof before==='string'&&before.length<=262144);const value=operations[operation](subject);if(before!==JSON.stringify(subject))throw Error('INPUT_MUTATED');const disposition=operation==='dimensional_exponent_balance'?'represented':operation==='evidence_review_gap'?'open_gap':operation==='authority_scope_intersection'?'exact_gate':'completed';return{ok:true,disposition,value:structuredClone(value),error:null};}catch(e){if(e instanceof InvalidInput)return{ok:false,disposition:'open_gap',value:null,error:'INVALID_INPUT'};throw e;}}
module.exports={run,operations:Object.keys(operations)};
