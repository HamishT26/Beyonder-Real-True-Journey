'use strict';
// Finite in-memory transaction profiles. No database, network, or host operation occurs.
class InvalidInput extends Error{constructor(){super('INVALID_INPUT');}}
const need=x=>{if(!x)throw new InvalidInput();};
const object=(x,keys)=>{need(x!==null&&typeof x==='object'&&!Array.isArray(x));if(keys)need(Object.keys(x).sort().join('|')===[...keys].sort().join('|'));return x;};
const list=x=>{need(Array.isArray(x)&&x.length<=1000);return x;};
const integer=x=>{need(Number.isSafeInteger(x)&&x>=0&&x<=1000000000);return x;};
const label=x=>{need(typeof x==='string'&&/^[A-Za-z0-9_-]{1,256}$/.test(x));return x;};
const labels=x=>list(x).map(label);
const bool=x=>{need(typeof x==='boolean');return x;};
const unique=xs=>need(new Set(xs).size===xs.length);
function topo(nodes,edges){const remaining=new Set(nodes),order=[];while(remaining.size){const ready=[...remaining].filter(n=>!edges.some(([a,b])=>b===n&&remaining.has(a))).sort();if(!ready.length)return null;order.push(ready[0]);remaining.delete(ready[0]);}return order;}
function snapshot_visible_versions(s){object(s,['versions','at']);integer(s.at);const versions=list(s.versions),seen=new Set(),latest=new Map();for(const v of versions){object(v,['key','rev','value','deleted']);label(v.key);integer(v.rev);bool(v.deleted);const key=JSON.stringify([v.key,v.rev]);need(!seen.has(key));seen.add(key);if(v.rev<=s.at&&(!latest.has(v.key)||v.rev>latest.get(v.key).rev))latest.set(v.key,v);}return{view:Object.fromEntries([...latest].filter(([,v])=>!v.deleted).sort(([a],[b])=>a.localeCompare(b)).map(([k,v])=>[k,v.value]))};}
function optimistic_read_validation(s){object(s,['reads','current']);const reads=list(s.reads);object(s.current);Object.entries(s.current).forEach(([k,v])=>{label(k);integer(v);});const keys=[];for(const r of reads){object(r,['key','version']);keys.push(label(r.key));if(r.version!==null)integer(r.version);}unique(keys);const conflicts=reads.filter(r=>r.version===null?Object.hasOwn(s.current,r.key):!Object.hasOwn(s.current,r.key)||s.current[r.key]!==r.version).map(r=>r.key).sort();return{valid:conflicts.length===0,conflicts};}
function serial_conflict_witness(s){object(s,['events']);const events=list(s.events);for(const e of events){object(e,['tx','kind','key']);label(e.tx);label(e.key);need(['read','write'].includes(e.kind));}const edges=new Set();for(let i=0;i<events.length;i++)for(let j=i+1;j<events.length;j++){const a=events[i],b=events[j];if(a.tx!==b.tx&&a.key===b.key&&(a.kind==='write'||b.kind==='write'))edges.add(a.tx+'>'+b.tx);}const sorted=[...edges].sort();return{edges:sorted,acyclic:topo([...new Set(events.map(e=>e.tx))],sorted.map(e=>e.split('>')))!==null};}
function write_ahead_redo_projection(s){object(s,['records','base','committed']);list(s.records);object(s.base);const committed=new Set(labels(s.committed)),state=new Map(Object.entries(s.base));for(const [k,p]of state){label(k);object(p,['value','lsn']);integer(p.lsn);}const positions=[];for(const r of s.records){object(r,['lsn','tx','key','value']);positions.push(integer(r.lsn));label(r.tx);label(r.key);}unique(positions);const applied=[];for(const r of [...s.records].sort((a,b)=>a.lsn-b.lsn)){if(committed.has(r.tx)&&(!state.has(r.key)||state.get(r.key).lsn<r.lsn)){state.set(r.key,{value:r.value,lsn:r.lsn});applied.push(r.lsn);}}return{state:Object.fromEntries([...state].sort(([a],[b])=>a.localeCompare(b))),applied};}
function undo_compensation_projection(s){object(s,['records','applied','compensated','target']);list(s.records);list(s.applied).forEach(integer);list(s.compensated).forEach(integer);label(s.target);const positions=[];for(const r of s.records){object(r,['lsn','tx','key','before','after']);positions.push(integer(r.lsn));label(r.tx);label(r.key);}unique(positions);const applied=new Set(s.applied),compensated=new Set(s.compensated);return{instructions:s.records.filter(r=>r.tx===s.target&&applied.has(r.lsn)&&!compensated.has(r.lsn)).sort((a,b)=>b.lsn-a.lsn).map(r=>({lsn:r.lsn,key:r.key,restore:r.before})),executed:false};}
function checkpoint_log_retention(s){object(s,['segments','checkpoint','undo_floor']);list(s.segments);integer(s.checkpoint);if(s.undo_floor!==null)integer(s.undo_floor);const ids=[];for(const r of s.segments){object(r,['id','start','end','archived']);ids.push(label(r.id));integer(r.start);integer(r.end);need(r.start<=r.end);bool(r.archived);}unique(ids);const threshold=Math.min(s.checkpoint,s.undo_floor===null?s.checkpoint:s.undo_floor),review=[],retained=[];for(const r of s.segments)(r.archived&&r.end<threshold?review:retained).push(r.id);return{review:review.sort(),retained:retained.sort(),action_allowed:false};}
function idempotency_fingerprint(s){object(s,['request','entries','now']);object(s.request,['key','digest']);label(s.request.key);label(s.request.digest);list(s.entries);integer(s.now);const ids=[];for(const r of s.entries){object(r,['key','digest','expires','response']);ids.push(label(r.key));label(r.digest);integer(r.expires);}unique(ids);const found=s.entries.find(e=>e.key===s.request.key);let status='MISS',response=null;if(found){if(found.expires<=s.now)status='EXPIRED';else if(found.digest!==s.request.digest)status='CONFLICT';else{status='RECORDED_MATCH';response=found.response;}}return{status,response,executed:false};}
function lease_fencing_sequence(s){object(s,['events']);list(s.events);const ids=[];for(const e of s.events){object(e,['id','kind','epoch','holder']);ids.push(label(e.id));label(e.holder);integer(e.epoch);need(['grant','write'].includes(e.kind));}unique(ids);let epoch=0,holder=null;const grants=[],writes=[],denied=[];for(const e of s.events){if(e.kind==='grant'){if(e.epoch>epoch){epoch=e.epoch;holder=e.holder;grants.push(e.id);}else denied.push(e.id);}else if(holder!==null&&e.epoch===epoch&&e.holder===holder)writes.push(e.id);else denied.push(e.id);}return{epoch,holder,grants,writes,denied};}
function saga_compensation_order(s){object(s,['steps']);list(s.steps);const ids=[];for(const step of s.steps){object(step,['id','depends','done','reversible']);ids.push(label(step.id));labels(step.depends);unique(step.depends);bool(step.done);bool(step.reversible);}unique(ids);if(s.steps.some(step=>step.depends.some(d=>!ids.includes(d))))return{order:[],blocked:['MISSING_DEPENDENCY'],executed:false};const edges=s.steps.flatMap(step=>step.depends.map(d=>[d,step.id])),ordered=topo(ids,edges);if(ordered===null)return{order:[],blocked:['CYCLE'],executed:false};const by=new Map(s.steps.map(x=>[x.id,x]));return{order:ordered.reverse().filter(id=>by.get(id).done&&by.get(id).reversible),blocked:s.steps.filter(x=>x.done&&!x.reversible).map(x=>x.id).sort(),executed:false};}
function two_phase_commit_projection(s){object(s,['participants','decision']);list(s.participants);need(['commit','abort','none'].includes(s.decision));const ids=[];for(const p of s.participants){object(p,['id','vote']);ids.push(label(p.id));need(['prepared','committed','abort','unknown'].includes(p.vote));}unique(ids);const votes=s.participants.map(p=>p.vote);let state='WAIT';if(votes.length){if(s.decision==='abort')state=votes.includes('committed')?'CONFLICT':'ABORT_SNAPSHOT';else if(s.decision==='commit')state=votes.includes('abort')?'CONFLICT':votes.every(x=>['prepared','committed'].includes(x))?'COMMIT_SNAPSHOT':'WAIT';else if(votes.includes('committed'))state='CONFLICT';else if(votes.includes('abort'))state='ABORT_REQUIRED';}return{state,executed:false};}
const operations={snapshot_visible_versions,optimistic_read_validation,serial_conflict_witness,write_ahead_redo_projection,undo_compensation_projection,checkpoint_log_retention,idempotency_fingerprint,lease_fencing_sequence,saga_compensation_order,two_phase_commit_projection};
function run(operation,subject){try{need(Object.hasOwn(operations,operation));const before=JSON.stringify(subject);const value=operations[operation](subject);if(before!==JSON.stringify(subject))throw Error('INPUT_MUTATED');return{ok:true,disposition:'completed',value:structuredClone(value),error:null};}catch(e){if(e instanceof InvalidInput)return{ok:false,disposition:'open_gap',value:null,error:'INVALID_INPUT'};throw e;}}
module.exports={run,operations:Object.keys(operations)};
