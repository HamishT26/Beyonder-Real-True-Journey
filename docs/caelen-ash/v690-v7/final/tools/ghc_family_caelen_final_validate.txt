function verifyWorkingManifest(manifest, excludedPaths) {
  const failures = [];
  for (const entry of manifest.entries || manifest.targets) {
    const file = path.join(repo, ...entry.path.split("/"));
    if (!fs.existsSync(file)) { failures.push(entry.path); continue; }
    const bytes = fs.readFileSync(file);
    if (bytes.length !== entry.bytes || digest(bytes) !== entry.sha256) failures.push(entry.path);
  }
  return failures;
}
