function pass(name, condition, detail = null) {
  checks.push({ name, pass: Boolean(condition), detail });
  if (!condition) errors.push(detail ? `${name}: ${detail}` : name);
}

function readJson(relative) {
  return JSON.parse(fs.readFileSync(path.join(root, relative), "utf8"));
}

function normalize(value) {
  if (Array.isArray(value)) return value.map(normalize);
  if (value && typeof value === "object") {
    const out = {};
    for (const key of Object.keys(value).sort()) out[key] = normalize(value[key]);
    return out;
  }
  return value;
}

function hashObject(value) {
  return crypto.createHash("sha256").update(JSON.stringify(normalize(value)), "utf8").digest("hex");
}

function walk(dir) {
  const out = [];
  for (const entry of fs.readdirSync(dir, { withFileTypes: true }).sort((a, b) => a.name.localeCompare(b.name))) {
    const full = path.join(dir, entry.name);
    if (entry.isDirectory()) out.push(...walk(full));
    else out.push(full);
  }
  return out;
}

let crFiles = [];
let uuidCandidates = [];
let privatePathCandidates = [];
let credentialCandidates = [];

const routeFragments = [
  ["source", "thread", "id"].join("_"),
  ["provider", "Tab", "Id"].join(""),
  ["client", "Thread", "Id"].join(""),
  ["thread", "Id"].join("") + "\\s*[=:]"
];
const rawRoutePattern = new RegExp(`(?:${routeFragments.join("|")})`);
for (const file of allFiles) {
  const bytes = fs.readFileSync(file);
  const rel = path.relative(root, file).replaceAll("\\", "/");
  if (bytes.includes(13)) crFiles.push(rel);
  const text = bytes.toString("utf8");
  if (/\b[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}\b/i.test(text)) uuidCandidates.push(rel);
  if (/(?:[A-Z]:\\Users\\|[A-Z]:\\GHC-Archives\\|\/Users\/|\/home\/)/.test(text)) privatePathCandidates.push(rel);
  if (/(?:BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY|ghp_[A-Za-z0-9]{20,}|sk-[A-Za-z0-9]{20,})/.test(text)) credentialCandidates.push(rel);
  if (rawRoutePattern.test(text)) rawRouteCandidates.push(rel);
}
