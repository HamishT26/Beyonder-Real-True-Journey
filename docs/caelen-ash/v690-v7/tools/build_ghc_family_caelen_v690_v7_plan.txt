function mkdir(dir) {
  fs.mkdirSync(dir, { recursive: true });
}

function normalize(value) {
  if (Array.isArray(value)) return value.map(normalize);
  if (value && typeof value === "object") {
    const out = {};
    for (const key of Object.keys(value).sort()) out[key] = normalize(value[key]);
    return out;
  }
  return value;
}

function canonical(value) {
  return JSON.stringify(normalize(value));
}

function hashText(value) {
  return crypto.createHash("sha256").update(value, "utf8").digest("hex");
}

function hashObject(value) {
  return hashText(canonical(value));
}

function writeJson(relative, value) {
  const target = path.join(root, relative);
  mkdir(path.dirname(target));
  fs.writeFileSync(target, JSON.stringify(value, null, 2) + "\n", { encoding: "utf8" });
}

function writeText(relative, value) {
  const target = path.join(root, relative);
  mkdir(path.dirname(target));
  const clean = String(value).replace(/\r\n/g, "\n").replace(/\r/g, "\n");
  fs.writeFileSync(target, clean.endsWith("\n") ? clean : clean + "\n", { encoding: "utf8" });
}

function uniqueSorted(values) {
  return [...new Set(values)].sort((a, b) => String(a).localeCompare(String(b)));
}

function isoHour(hour) {
  return new Date(Date.UTC(2026, 8, 11, hour, 0, 0)).toISOString();
}

function scenario(index) {
  const n = index + 1;
  const concreteStates = Array.from({ length: 3 + (index % 3) }, (_, j) => `c${j}`);
  const blocks = [];
  for (let j = 0; j < concreteStates.length; j += 2) {
    blocks.push({ abstract: `a${Math.floor(j / 2)}`, members: concreteStates.slice(j, j + 2) });
  }
  const abstraction = {};
  for (const block of blocks) for (const member of block.members) abstraction[member] = block.abstract;
  const actions = ["request", "review", "correct", "handover"];
  const edges = concreteStates.slice(0, -1).map((state, j) => [state, actions[(j + index) % actions.length], concreteStates[j + 1]]);
  if (index % 2 === 1) edges.push([concreteStates.at(-1), "readback", concreteStates.at(-1)]);
  return { n, concreteStates, blocks, abstraction, actions, edges };
}

function buildRequest(operation, index) {
  const s = scenario(index);
  const hour = index;
  const even = index % 2 === 0;
  switch (operation) {
    case "abstraction_partition_check":
      return { op: operation, concrete_states: s.concreteStates, blocks: s.blocks };
    case "abstract_edge_projection":
      return { op: operation, concrete_edges: s.edges, abstraction: s.abstraction };
    case "spurious_trace_check": {
      const trace = s.edges.slice(0, Math.min(2, s.edges.length)).map((edge) => edge[1]);
      const alternatives = [trace, ["request", "missing"], ["readback"]];
      return { op: operation, abstract_actions: even ? trace : ["not-realized"], concrete_action_traces: alternatives };
    }
    case "concrete_trace_lift": {
      const pathEdges = s.edges.slice(0, Math.min(2, s.edges.length));
      const abstractTrace = pathEdges.length === 0 ? [s.abstraction[s.concreteStates[0]]] : [s.abstraction[pathEdges[0][0]], ...pathEdges.map((edge) => s.abstraction[edge[2]])];
      return { op: operation, abstract_state_trace: abstractTrace, abstraction: s.abstraction, concrete_edges: s.edges };
    }
    case "refinement_split":
      return {
        op: operation,
        members: s.concreteStates,
        attributes: Object.fromEntries(s.concreteStates.map((state, j) => [state, (j + 1) * (index + 2)])),
        threshold: (index + 2) * 2

    case "purpose_scope_intersection":

  }

function oracle(request) {
  switch (request.op) {
    case "abstraction_partition_check": {
      const seen = new Map();
      const empties = [];
      for (const block of request.blocks) {
        if (block.members.length === 0) empties.push(block.abstract);
        for (const member of block.members) seen.set(member, (seen.get(member) || 0) + 1);
      }
      const missing = request.concrete_states.filter((state) => !seen.has(state));
      const duplicateMembers = [...seen.entries()].filter(([, count]) => count > 1).map(([state]) => state).sort();
      const covered = request.concrete_states.filter((state) => seen.has(state)).sort();
      return { covered, missing, duplicate_members: duplicateMembers, empty_blocks: empties.sort(), valid: missing.length === 0 && duplicateMembers.length === 0 && empties.length === 0 };
    }
    case "abstract_edge_projection": {
      const projected = request.concrete_edges.map(([from, label, to]) => [request.abstraction[from], label, request.abstraction[to]]);
      const keyed = new Map(projected.map((edge) => [canonical(edge), edge]));
      return { abstract_edges: [...keyed.values()].sort((a, b) => canonical(a).localeCompare(canonical(b))) };
    }
    case "spurious_trace_check": {
      const target = canonical(request.abstract_actions);
      const indexes = request.concrete_action_traces.map((trace, i) => canonical(trace) === target ? i : -1).filter((i) => i >= 0);
      return { spurious: indexes.length === 0, matching_trace_indexes: indexes };
    }
    case "concrete_trace_lift": {
      const candidates = [];
      for (const [from, label, to] of request.concrete_edges) {
        const pair = [request.abstraction[from], request.abstraction[to]];
        for (let i = 0; i < request.abstract_state_trace.length - 1; i++) {
          if (pair[0] === request.abstract_state_trace[i] && pair[1] === request.abstract_state_trace[i + 1]) candidates.push({ step: i, from, label, to });
        }
      }
      return { candidates, complete_step_coverage: uniqueSorted(candidates.map((x) => x.step)).length === Math.max(0, request.abstract_state_trace.length - 1) };
    }
    case "refinement_split": {
      const lower = request.members.filter((member) => request.attributes[member] <= request.threshold).sort();
      const higher = request.members.filter((member) => request.attributes[member] > request.threshold).sort();
      return { lower_or_equal: lower, higher, disjoint_complete: lower.length + higher.length === request.members.length };
    }
    case "purpose_scope_intersection": {
      const granted = new Set(request.granted_scopes);
      return {
        permitted: uniqueSorted(request.requested_scopes.filter((scope) => granted.has(scope))),
        withheld: uniqueSorted(request.requested_scopes.filter((scope) => !granted.has(scope))),
        purpose: request.purpose
      };
    }
    case "expiry_window_decision": {
      const observed = Date.parse(request.observed_at);
      const notBefore = Date.parse(request.not_before);
      const expires = Date.parse(request.expires_at);
      const state = observed < notBefore ? "premature" : observed >= expires ? "expired" : "active";
      return { state, duration_to_expiry_seconds: Math.trunc((expires - observed) / 1000) };
    }
    case "consent_lineage": {
      const events = request.events.filter((event) => event.sequence <= request.as_of_sequence).sort((a, b) => a.sequence - b.sequence);
      const active = new Set();
      for (const event of events) {
        if (event.action === "grant") for (const scope of event.scopes) active.add(scope);
        if (event.action === "revoke") for (const scope of event.scopes) active.delete(scope);
        if (event.action === "correct") {
          active.clear();
          for (const scope of event.scopes) active.add(scope);
        }
      }
      return { active_scopes: [...active].sort(), lineage_ids: events.map((event) => event.id), last_sequence: events.length ? events.at(-1).sequence : null };
    }
    case "remedy_route": {
      const resolved = [];
      const unresolved = [];
      for (const code of request.issue_codes) {
        if (Object.hasOwn(request.declared_routes, code)) resolved.push([code, request.declared_routes[code]]);
        else unresolved.push(code);
      }
      return { declared_routes: resolved, unresolved_codes: unresolved.sort(), authority_reserved: true };
    }
    case "appeal_window": {
      let state = "missing";
      if (request.submitted_at !== null) state = Date.parse(request.submitted_at) <= Date.parse(request.appeal_by) ? "timely" : "late";
      return { state, window_seconds: Math.trunc((Date.parse(request.appeal_by) - Date.parse(request.decision_at)) / 1000), remedy_authority_reserved: true };
    }
    case "disclosure_minimization": {
      const allowed = new Set(request.allowed_fields);
      const released = {};
      const withheld = [];
      for (const field of request.requested_fields) {
        if (allowed.has(field) && Object.hasOwn(request.record, field)) released[field] = request.record[field];
        else withheld.push(field);
      }
      return { released, withheld_fields: withheld.sort(), minimum_disclosure_only: true };
    }
    case "explanation_coverage": {
      const provided = new Set(request.provided_reasons);
      const missing = request.required_reasons.filter((reason) => !provided.has(reason)).sort();
      return { covered: missing.length === 0, missing_reasons: missing, provided_reasons: uniqueSorted(request.provided_reasons) };
    }
    case "alternative_action_set": {
      const roles = new Set(request.available_roles);
      const enabled = [];
      const withheld = [];
      for (const action of request.actions) {
        const missing = action.required_roles.filter((role) => !roles.has(role));
        (missing.length === 0 ? enabled : withheld).push(missing.length === 0 ? action.id : { id: action.id, missing_roles: missing.sort() });
      }
      return { enabled_actions: enabled.sort(), withheld_actions: withheld };
    }
    case "review_quorum_gap": {
      const present = new Set(request.present_roles);
      const missing = request.required_roles.filter((role) => !present.has(role)).sort();
      return { quorum_met: missing.length === 0, missing_roles: missing, authority_reserved: true };
    }
    case "accessibility_mode_gap": {
      const provided = new Set(request.provided_modes);
      const missing = request.required_modes.filter((mode) => !provided.has(mode)).sort();
      return { declared_modes_complete: missing.length === 0, missing_modes: missing, manual_evaluation_reserved: true };
    }
    case "observation_model_separation": {
      const groups = { observation: [], model: [], assumption: [], unknown: [] };
      for (const record of request.records) groups[record.kind].push(record.id);
      return { groups, mixed_as_evidence: false, empirical_conversion_forbidden: true };
    }
    case "unit_dimension_guard": {
      const compatible = request.left.dimension === request.right.dimension;
      return { compatible, result: compatible ? request.left.value + request.right.value : null, dimension: compatible ? request.left.dimension : null, error: compatible ? null : "dimension_mismatch" };
    }
    case "workload_budget_trace": {
      let remaining = request.budget;
      const accepted = [];
      const held = [];
      for (const task of request.tasks) {
        if (task.cost <= remaining) {
          accepted.push(task.id);
          remaining -= task.cost;
        } else held.push(task.id);
      }
      return { accepted, held, remaining_budget: remaining, scheduling_authority_reserved: true };
    }
    case "rollback_checkpoint": {
      const reversible = request.events.filter((event) => event.reversible).sort((a, b) => a.sequence - b.sequence);
      return { latest_reversible_event: reversible.length ? reversible.at(-1).id : null, irreversible_events: request.events.filter((event) => !event.reversible).map((event) => event.id), real_rollback_unexecuted: true };
    }
    case "delegation_scope_gap": {
      const delegated = new Set(request.delegated_scopes);
      const missing = request.required_scopes.filter((scope) => !delegated.has(scope)).sort();
      return { delegation_complete: request.delegating_authority_verified && missing.length === 0, missing_scopes: missing, authority_verified: request.delegating_authority_verified, exact_authority_gate: true };
    }
    default:
      throw new Error(`no oracle for ${request.op}`);
  }
}

function tokens(text) {
  return new Set(String(text).toLowerCase().replace(/[^a-z0-9]+/g, " ").trim().split(/\s+/).filter(Boolean));
}

function jaccard(a, b) {
  const aa = tokens(a);
  const bb = tokens(b);
  const intersection = [...aa].filter((x) => bb.has(x)).length;
  const union = new Set([...aa, ...bb]).size;
  return union === 0 ? 1 : intersection / union;
}

function walk(dir) {
  const out = [];
  for (const entry of fs.readdirSync(dir, { withFileTypes: true }).sort((a, b) => a.name.localeCompare(b.name))) {
    const full = path.join(dir, entry.name);
    if (entry.isDirectory()) out.push(...walk(full));
    else out.push(full);
  }
  return out;
}

const manifestPath = path.join(planDir, "manifest.json");
const stagedReviewPath = path.join(planDir, "staged-review.json");
const manifestEntries = walk(root)
  .filter((file) => file !== manifestPath && file !== stagedReviewPath)
  .map((file) => {
    const bytes = fs.readFileSync(file);
    const relative = path.relative(path.dirname(path.dirname(path.dirname(root))), file).replaceAll("\\", "/");
    if (bytes.includes(13)) throw new Error(`CR byte forbidden in ${relative}`);
    return { path: relative, bytes: bytes.length, sha256: crypto.createHash("sha256").update(bytes).digest("hex") };
  });
