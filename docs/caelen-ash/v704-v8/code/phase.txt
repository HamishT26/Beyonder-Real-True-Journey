function argsObject(values) {
  const output = { _: [] };
  for (let i = 0; i < values.length; i += 1) {
    const value = values[i];
    if (value.startsWith('--')) {
      const key = value.slice(2);
      const next = values[i + 1];
      if (next && !next.startsWith('--')) {
        output[key] = next;
        i += 1;
      } else output[key] = true;
    } else output._.push(value);
  }
  return output;
}

function stable(value) {
  if (Array.isArray(value)) return value.map(stable);
  if (value && typeof value === 'object') {
    const output = {};
    for (const key of Object.keys(value).sort()) output[key] = stable(value[key]);
    return output;
  }
  return value;
}

function sha(value) {
  const bytes = Buffer.isBuffer(value) ? value : Buffer.from(typeof value === 'string' ? value : JSON.stringify(stable(value)));
  return crypto.createHash('sha256').update(bytes).digest('hex');
}

function relationClosure(points, covers, equivalent = []) {
  const index = new Map(points.map((point, i) => [point, i]));
  const matrix = Array.from({ length: points.length }, () => Array(points.length).fill(false));
  for (let i = 0; i < points.length; i += 1) matrix[i][i] = true;
  for (const pair of covers) matrix[index.get(pair[0])][index.get(pair[1])] = true;
  for (const group of equivalent) for (const a of group) for (const b of group) matrix[index.get(a)][index.get(b)] = true;
  for (let k = 0; k < points.length; k += 1) for (let i = 0; i < points.length; i += 1) for (let j = 0; j < points.length; j += 1) {
    if (matrix[i][k] && matrix[k][j]) matrix[i][j] = true;
  }
  const relation = [];
  for (let i = 0; i < points.length; i += 1) for (let j = 0; j < points.length; j += 1) if (matrix[i][j]) relation.push([points[i], points[j]]);
  return relation;
}

function tokens(text) {
  return new Set(String(text).toLowerCase().replace(/[^a-z0-9]+/g, ' ').trim().split(/\s+/).filter(Boolean));
}

function jaccard(a, b) {
  const left = tokens(a);
  const right = tokens(b);
  let overlap = 0;
  for (const value of left) if (right.has(value)) overlap += 1;
  const union = new Set([...left, ...right]).size;
  return union ? overlap / union : 1;
}

function ledger(stage, methods, witnesses) {
  const byMethod = new Map(methods.map((method) => [method.method_id, method]));
  for (const item of witnesses) byMethod.get(item.method_id).validation_witness_ids.push(item.witness_id);
  const states = { observed: 0, candidate: 0, validated: methods.length, preferred: 0, superseded: 0, deprecated: 0 };
  const pass = witnesses.filter((row) => row.result === 'pass').length;
  const fail = witnesses.filter((row) => row.result === 'fail').length;
  const stateEvents = [];
  for (const method of methods) {
    stateEvents.push({ event_index: stateEvents.length + 1, method_id: method.method_id, before: null, after: 'candidate', reason: 'Method recorded with retained boundary linkage.', witness_id: null });
    const passing = witnesses.find((row) => row.method_id === method.method_id && row.result === 'pass');
    stateEvents.push({ event_index: stateEvents.length + 1, method_id: method.method_id, before: 'candidate', after: 'validated', reason: 'Bounded witness passed without erasing failures.', witness_id: passing ? passing.witness_id : null });
  }
  return {
    schema: 'ghc.family.method-flow-state.v1',
    phase: PHASE + '-' + stage,
    owner: OWNER,
    identity_boundary: 'Relational working language only; not consciousness, personhood, continuity, employment, or authority evidence.',
    execution_authority: 'owner_self_scoped_delta',
    methods,
    witnesses,
    state_events: stateEvents,
    recommendations: [],
    counts: {
      methods: methods.length,
      witnesses: witnesses.length,
      state_events: stateEvents.length,
      recommendations: 0,
      states,
      witness_results: { pass, fail }
    },
    boundary: BOUNDARY
  };
}

function manifest(relative, stage) {
  const excluded = (PREFIX + '/' + relative).replace(/\\/g, '/');
  const files = [];
  function walk(directory) {
    for (const entry of fs.readdirSync(directory, { withFileTypes: true })) {
      const full = path.join(directory, entry.name);
      if (entry.isDirectory()) walk(full);
      else files.push(full);
    }
  }
  walk(ROOT);
  const entries = files
    .map((file) => path.relative(REPO, file).replace(/\\/g, '/'))
    .filter((file) => file.startsWith(PREFIX + '/') && file !== excluded)
    .sort()
    .map((file) => {
      const bytes = fs.readFileSync(path.join(REPO, file));
      return { path: file, bytes: bytes.length, sha256: sha(bytes), domain: 'checkout_utf8_lf_precommit' };
    });
  writeJson(relative, {
    schema: 'caelen.owner-manifest.v1',
    owner: OWNER,
    phase: PHASE,
    stage,
    source: SOURCE,
    entries,
    excluded_self: excluded
  });
}

function runTxt(file, input) {
  const loader = "require.extensions['.txt']=require.extensions['.js'];require(process.argv[1]);";
  const result = cp.spawnSync(process.execPath, ['-e', loader, file], { input: JSON.stringify(input), encoding: 'utf8', maxBuffer: 8 * 1024 * 1024 });
  if (result.error || result.status !== 0) throw new Error('runner_process:' + (result.error ? result.error.message : result.stderr));
  return JSON.parse(result.stdout);
}

function validateSkills(stage, operations) {
  const rows = [];
  for (const operation of operations) {
    const relative = 'skills/' + skillName(operation) + '/SKILL.md';
    writeText(relative, skillBody(operation));
    const full = path.join(ROOT, relative);
    const content = fs.readFileSync(full, 'utf8');
    const structural = /^---\nname: [a-z0-9-]+\ndescription: .+\n---\n/.test(content) && !content.includes('TODO');
    let official = null;
    if (ARGS['skill-validator'] && ARGS.python) {
      const result = cp.spawnSync(ARGS.python, [ARGS['skill-validator'], path.dirname(full)], { encoding: 'utf8', maxBuffer: 8 * 1024 * 1024 });
      official = { exit_code: result.status, passed: result.status === 0, output_sha256: sha((result.stdout || '') + (result.stderr || '')) };
      if (result.status !== 0) throw new Error('skill_validator:' + operation.op);
    }
    if (!structural) throw new Error('skill_structure:' + operation.op);
    rows.push({ operation: operation.op, path: relative, structural, official, smoke_used: true, global_installation: false });
  }
  return rows;
}

function buildRunners(stage, operations, contracts) {
  const rows = [];
  for (let i = 0; i < 5; i += 1) {
    const pair = operations.slice(i * 2, i * 2 + 2).map((row) => row.op);
    const ordinal = stage === 'x1' ? i + 1 : i + 6;
    const id = 'CA7048-R' + String(ordinal).padStart(2, '0');
    const relative = 'runners/ghc_family_topology_' + String(ordinal).padStart(2, '0') + '.txt';
    writeText(relative, runnerBody(id, pair));
    const full = path.join(ROOT, relative);
    new vm.Script(fs.readFileSync(full, 'utf8'), { filename: relative });
    const valid = contracts.find((row) => row.operation === pair[0]).request;
    const invalid = { ...valid };
    delete invalid.relation;
    const accepted = runTxt(full, valid);
    const rejected = runTxt(full, invalid);
    if (!accepted.ok || rejected.ok || rejected.error !== 'field_set') throw new Error('runner_smoke:' + id);
    rows.push({ id, stage, path: relative, operations: pair, accepted, rejected, syntax_valid: true });
  }
  return rows;
}

function stageLedger(stage, operations, contracts, auxiliary, skills, runners, tests, models = [], hookRows = []) {
  const methods = [];
  const witnesses = [];
  const methodForOp = new Map();
  let methodOrdinal = 1;
  function addMethod(title, negatives = []) {
    const id = 'CA7048-' + stage.toUpperCase() + '-M' + String(methodOrdinal++).padStart(2, '0');
    methods.push(methodRecord(id, title, negatives));
    return id;
  }
  for (const operation of operations) methodForOp.set(operation.op, addMethod(operation.op + ' exact finite-topology method'));
  const auxA = addMethod('Auxiliary safe envelope group A');
  const auxB = addMethod('Auxiliary safe envelope group B');
  const skillMethod = addMethod('Phase-local skill validation and use');
  const runnerMethod = addMethod('Family-compatible runner accepting and rejecting smokes');
  const testMethod = addMethod('Saved-result test selection');
  let modelMethod = null;
  const hookMethods = [];
  if (stage === 'x2') {
    modelMethod = addMethod('Three-coordinate finite model construction');
    for (let i = 0; i < 10; i += 1) hookMethods.push(addMethod('Advisory hook ' + String(i + 1).padStart(2, '0')));
  }
  for (const contract of contracts) {
    const methodId = methodForOp.get(contract.operation);
    witnesses.push(witness(contract.id + '-PRIMARY', methodId, 'pass', 'Runtime matched the frozen expected digest.'));
    for (const negative of contract.negative_subjects) {
      witnesses.push(witness(negative.negative_id + '-SUBJECT', methodId, 'fail', 'Malformed subject retained as ' + negative.error + '.', [negative.negative_id]));
      witnesses.push(witness(negative.negative_id + '-REFUSAL', methodId, 'pass', 'Guard rejected the declared malformed subject as ' + negative.error + '.', [negative.negative_id]));
      witnesses.push(witness(negative.negative_id + '-REPAIR', methodId, 'pass', 'Frozen valid request recovered separately.', [negative.negative_id]));
    }
  }
  auxiliary.forEach((row, index) => {
    witnesses.push(witness(row.id, index < 125 ? auxA : auxB, 'pass', 'Auxiliary frozen envelope matched.'));
  });
  skills.forEach((row, index) => {
    witnesses.push(witness('CA7048-' + stage.toUpperCase() + '-SKILL-' + String(index + 1).padStart(2, '0'), skillMethod, 'pass', 'Skill structurally and officially validated and smoke-used.'));
  });
  runners.forEach((row, index) => {
    const negativeId = 'CA7048-' + stage.toUpperCase() + '-RUNNER-N' + String(index + 1).padStart(2, '0');
    witnesses.push(witness(negativeId + '-SUBJECT', runnerMethod, 'fail', 'Invalid runner envelope retained with zero success credit.', [negativeId]));
    witnesses.push(witness(negativeId + '-ACCEPT', runnerMethod, 'pass', 'Runner accepted the declared valid operation.', [negativeId]));
    witnesses.push(witness(negativeId + '-REFUSE', runnerMethod, 'pass', 'Runner rejected the declared invalid operation.', [negativeId]));
  });
  tests.forEach((row, index) => {
    witnesses.push(witness('CA7048-' + stage.toUpperCase() + '-TEST-' + String(index + 1).padStart(3, '0'), testMethod, 'pass', row.name + ' passed.'));
  });
  if (stage === 'x2') {
    models.forEach((row, index) => {
      witnesses.push(witness('CA7048-X2-MODEL-' + String(index + 1).padStart(2, '0'), modelMethod, 'pass', 'Three-coordinate saved model constructed.'));
    });
    hookRows.forEach((row, index) => {
      const methodId = hookMethods[Math.floor(index / 2)];
      if (row.kind === 'adverse') {
        const negativeId = 'CA7048-X2-HOOK-N' + String(Math.floor(index / 2) + 1).padStart(2, '0');
        witnesses.push(witness(negativeId + '-SUBJECT', methodId, 'fail', 'Adverse advisory subject retained with zero success credit.', [negativeId]));
        witnesses.push(witness(negativeId + '-REFUSE', methodId, 'pass', 'Advisory returned a nonblocking warning for the adverse subject.', [negativeId]));
      } else {
        witnesses.push(witness('CA7048-X2-HOOK-' + String(Math.floor(index / 2) + 1).padStart(2, '0') + '-BENIGN', methodId, 'pass', 'Benign advisory fixture returned no warning.'));
      }
    });
  }
  return ledger(stage, methods, witnesses);
}

function executeStage(stage) {
  const expectedHead = ARGS['expected-head'];
  if (!expectedHead) throw new Error('expected-head required');
  requireHead(expectedHead);
  ownerStatus();
  if (stage === 'x1' && (fs.existsSync(path.join(ROOT, 'x2')) || fs.existsSync(path.join(ROOT, 'final')))) throw new Error('x1_lifecycle_contamination');
  if (stage === 'x2' && fs.existsSync(path.join(ROOT, 'final'))) throw new Error('x2_lifecycle_contamination');
  const runtime = require(path.join(__dirname, 'runtime.cjs'));
  const operations = OPERATIONS.filter((row) => row.stage === stage);
  const contracts = readContracts(stage);
  const auxiliary = readJson(path.join(ROOT, 'planning', 'auxiliary-' + stage + '.json'));
  const coreResults = [];
  const candidateFailures = [];
  const refusals = [];
  const repairs = [];
  for (const contract of contracts) {
    const value = runtime.evaluate(contract.request);
    const observed = { outcome: contract.outcome, data: value };
    const match = sha(observed) === contract.expected_sha256;
    if (!match) throw new Error('runtime_reference_mismatch:' + contract.id);
    coreResults.push({
      id: contract.id,
      operation: contract.operation,
      profile: contract.profile,
      outcome: contract.outcome,
      observed_sha256: sha(observed),
      expected_sha256: contract.expected_sha256,
      match,
      value
    });
    for (const negative of contract.negative_subjects) {
      let observedError = null;
      try {
        runtime.evaluate(negative.request);
      } catch (error) {
        observedError = error.code || error.message;
      }
      if (observedError !== negative.error) throw new Error('negative_mismatch:' + negative.negative_id + ':' + observedError);
      candidateFailures.push({
        negative_id: negative.negative_id,
        contract_id: contract.id,
        expected_error: negative.error,
        observed_error: observedError,
        state: 'retained_failed_subject',
        original_success_credit: 0
      });
      refusals.push({
        negative_id: negative.negative_id,
        contract_id: contract.id,
        expected_error: negative.error,

      });

function privacyHits(text) {
  const patterns = [
    ['raw_uuid', /\b[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}\b/i],
    ['private_local_path', /\b[A-Za-z]:[\\/](?:Users|GHC-Archives|Program Files)\b/i],
    ['private_uri', /\b(?:app|plugin):\/\//i],
    ['delegation_markup', new RegExp('<(?:codex_' + 'delegation|source_thread_' + 'id)>', 'i')],
    ['credential_assignment', /\b(?:api[_-]?key|access[_-]?token|password|secret)\b\s*[:=]\s*["'][^"']+["']/i]
  ];
  return patterns.filter((row) => row[1].test(text)).map((row) => row[0]);
}

function validateStage(stage) {
  ownerStatus();
  const files = [];
  function walk(directory) {
    if (!fs.existsSync(directory)) return;
    for (const entry of fs.readdirSync(directory, { withFileTypes: true })) {
      const full = path.join(directory, entry.name);
      if (entry.isDirectory()) walk(full);
      else files.push(full);
    }
  }
  walk(ROOT);
  const issues = [];
  let jsonParses = 0;
  let maxWords = 0;
  const privacy = [];
  for (const file of files) {
    const text = fs.readFileSync(file, 'utf8');
    maxWords = Math.max(maxWords, wordCount(text));
    if (wordCount(text) > 100000) issues.push('word_cap:' + path.relative(ROOT, file));
    if (file.endsWith('.json')) {
      try {
        JSON.parse(text);
        jsonParses += 1;
      } catch {
        issues.push('json_parse:' + path.relative(ROOT, file));
      }
    }
    for (const label of privacyHits(text)) privacy.push({ path: path.relative(ROOT, file).replace(/\\/g, '/'), class: label });
  }
  if (privacy.length) issues.push('privacy_hits');
  if (files.length >= 170) issues.push('owner_file_forecast');
  if (stage === 'planning' && (fs.existsSync(path.join(ROOT, 'x1')) || fs.existsSync(path.join(ROOT, 'x2')) || fs.existsSync(path.join(ROOT, 'final')))) issues.push('planning_contamination');
  if (stage === 'x1' && (fs.existsSync(path.join(ROOT, 'x2')) || fs.existsSync(path.join(ROOT, 'final')))) issues.push('x1_contamination');
  if (stage === 'x2' && fs.existsSync(path.join(ROOT, 'final'))) issues.push('x2_contamination');
  const manifestRelative = stage === 'planning' ? 'planning/manifest.json' : stage === 'final' ? 'final/content-seal.json' : stage + '/manifest.json';
  const m = readJson(path.join(ROOT, manifestRelative));
  for (const entry of m.entries) {
    const bytes = fs.readFileSync(path.join(REPO, entry.path));
    if (bytes.length !== entry.bytes || sha(bytes) !== entry.sha256) issues.push('manifest:' + entry.path);
  }
  const receipt = {
    schema: 'caelen.stage-validation.v1',
    owner: OWNER,
    phase: PHASE,
    stage,
    valid: issues.length === 0,
    issues,
    owner_files: files.length,
    json_parses: jsonParses,
    maximum_words: maxWords,
    privacy_candidates: privacy,
    manifest_entries: m.entries.length,
    head: git(['rev-parse', 'HEAD']),
    clean_before_staging: git(['status', '--porcelain=v1']).split(/\r?\n/).filter(Boolean).every((row) => row.slice(3).replace(/\\/g, '/').startsWith(PREFIX)),
    boundary: BOUNDARY
  };
  bankJson('validation-' + stage + '.json', receipt);
  console.log(JSON.stringify(receipt));
  if (!receipt.valid) process.exitCode = 1;
}

function gitBlob(commit, relative) {
  return cp.execFileSync('git', ['-C', REPO, 'show', commit + ':' + relative], {
    encoding: null,
    maxBuffer: 64 * 1024 * 1024
  });
}

function canonical() {
  const finalCommit = ARGS.final;
  for (const key of ['plan', 'x1', 'x2', 'final']) if (!ARGS[key]) throw new Error(key + ' commit required');
  const latchPath = path.join(BANK, 'canonical-invocation.json');
  if (fs.existsSync(latchPath)) throw new Error('canonical_latch_exists');
  bankJson('canonical-invocation.json', {
    state: 'INVOKED_NO_RETRY',
    invocations: 1,
    successes: 0,
    replays: 0,
    exact_final: finalCommit
  });
  const checks = [];
  function check(name, passed, observed) {
    checks.push({ name, passed: Boolean(passed), observed });
  }
  const branch = git(['branch', '--show-current']);
  const head = git(['rev-parse', 'HEAD']);
  const upstream = git(['rev-parse', '@{upstream}']);
  const tracking = git(['rev-parse', 'refs/remotes/origin/' + BRANCH]);
  const live = git(['ls-remote', 'origin', 'refs/heads/' + BRANCH]).split(/\s+/)[0];
  const status = git(['status', '--porcelain=v1']);
  check('branch', branch === BRANCH, branch);
  check('head', head === finalCommit, head);
  check('upstream', upstream === finalCommit, upstream);
  check('tracking', tracking === finalCommit, tracking);
  check('fresh_live', live === finalCommit, live);
  check('clean', status === '', status);
  const chain = [ARGS.plan, ARGS.x1, ARGS.x2, finalCommit];
  const expectedParents = [SOURCE, ARGS.plan, ARGS.x1, ARGS.x2];
  chain.forEach((commit, index) => {
    const parent = git(['rev-parse', commit + '^']);
    check('direct_parent_' + String(index + 1), parent === expectedParents[index], parent);
  });
  const commitCount = Number(git(['rev-list', '--count', SOURCE + '..' + finalCommit]));
  const mergeText = git(['rev-list', '--merges', SOURCE + '..' + finalCommit]);
  const merges = mergeText ? mergeText.split(/\r?\n/).filter(Boolean).length : 0;
  check('commit_count', commitCount === 4, commitCount);
  check('zero_merges', merges === 0, merges);
  const changed = git(['diff', '--name-only', SOURCE + '..' + finalCommit]).split(/\r?\n/).filter(Boolean);
  check('owner_only_delta', changed.every((row) => row.startsWith(PREFIX + '/')), changed.length);
  const repositoryFiles = git(['ls-tree', '-r', '--name-only', finalCommit]).split(/\r?\n/).filter(Boolean);
  const ownerFiles = repositoryFiles.filter((row) => row.startsWith(PREFIX + '/'));
  check('repository_ceiling', repositoryFiles.length < 2000, repositoryFiles.length);
  check('owner_ceiling', ownerFiles.length < 170, ownerFiles.length);
  const manifests = [
    [ARGS.plan, 'planning/manifest.json'],
    [ARGS.x1, 'x1/manifest.json'],
    [ARGS.x2, 'x2/manifest.json'],
    [finalCommit, 'final/content-seal.json']
  ];
  let bindingCount = 0;
  for (const manifestRow of manifests) {
    const commit = manifestRow[0];
    const relative = manifestRow[1];
    const m = JSON.parse(gitBlob(commit, PREFIX + '/' + relative).toString('utf8'));
    let manifestPassed = true;
    for (const entry of m.entries) {
      const bytes = gitBlob(commit, entry.path);
      bindingCount += 1;
      if (bytes.length !== entry.bytes || sha(bytes) !== entry.sha256) manifestPassed = false;
    }
    check('manifest_' + relative, manifestPassed, m.entries.length);
  }
  let jsonParses = 0;
  let maxWords = 0;
  const privacy = [];
  for (const relative of ownerFiles) {
    const bytes = gitBlob(finalCommit, relative);
    const text = bytes.toString('utf8');
    maxWords = Math.max(maxWords, wordCount(text));
    if (relative.endsWith('.json')) {
      JSON.parse(text);
      jsonParses += 1;
    }
    for (const label of privacyHits(text)) privacy.push({ path: relative, class: label });
  }
  check('strict_json', jsonParses > 0, jsonParses);
  check('word_cap', maxWords <= 100000, maxWords);
  check('privacy', privacy.length === 0, privacy);
  const truth = JSON.parse(gitBlob(finalCommit, PREFIX + '/final/phase-truth.json').toString('utf8'));
  check('source_fold', truth.source_fold_count === 1, truth.source_fold_count);
  check('outcomes', JSON.stringify(truth.outcomes) === JSON.stringify({ completed: 255, represented: 15, open_gap: 15, exact_gate: 15 }), truth.outcomes);
  check('route_held', truth.route.state === 'PREPARED_NOT_SENT', truth.route.state);
  check('stage20', truth.terminal_verdict === 'NOT_READY_FOR_STAGE_20', truth.terminal_verdict);
  check('source_execution', truth.source_execution_credit === 0, truth.source_execution_credit);
  const passed = checks.filter((row) => row.passed).length;
  const receipt = {
    schema: 'caelen.owner-canonical.v1',
    state: passed === checks.length ? 'VALID_EXACT_FINAL_OWNER_SCOPED_METADATA_CANONICAL' : 'FAILED_ZERO_CANONICAL_SUCCESS_CREDIT',
    owner: OWNER,
    phase: PHASE,
    exact_final: finalCommit,
    checks,
    passed,
    total: checks.length,
    manifest_raw_bindings: bindingCount,
    owner_files: ownerFiles.length,
    repository_files: repositoryFiles.length,
    json_parses: jsonParses,
    invocations: 1,
    successes: passed === checks.length ? 1 : 0,
    replays: 0,
    domain_replays: 0,
    source_replays: 0,
    metadata_only: true,
    boundary: BOUNDARY
  };
  bankJson('canonical-receipt.json', receipt);
  const receiptSha = sha(fs.readFileSync(path.join(BANK, 'canonical-receipt.json')));
  bankJson('canonical-invocation.json', {
    state: passed === checks.length ? 'SUCCESS_LATCHED_NO_REPLAY' : 'FAILED_ZERO_SUCCESS_CREDIT_NO_REPLAY',
    invocations: 1,
    successes: receipt.successes,
    replays: 0,
    exact_final: finalCommit,
    receipt_sha256: receiptSha
  });
  console.log(JSON.stringify({
    state: receipt.state,
    passed: receipt.passed,
    total: receipt.total,
    receipt_sha256: receiptSha
  }));
  if (receipt.successes !== 1) process.exitCode = 1;
}

if (COMMAND === 'plan') plan();
else if (COMMAND === 'x1' || COMMAND === 'x2') executeStage(COMMAND);
else if (COMMAND === 'final') finalBuild();
else if (COMMAND === 'validate') validateStage(ARGS._[1]);
else if (COMMAND === 'canonical') canonical();
else throw new Error('command must be plan, x1, x2, final, validate, or canonical');
