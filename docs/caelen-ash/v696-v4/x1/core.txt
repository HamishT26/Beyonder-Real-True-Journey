function gcd(a,b){a=a<0n?-a:a;b=b<0n?-b:b;while(b){const r=a%b;a=b;b=r;}return a||1n;}
function q(n,d=1n){n=BigInt(n);d=BigInt(d);if(d===0n)throw new Error('zero denominator');if(d<0n){n=-n;d=-d;}const g=gcd(n,d);return [n/g,d/g];}
const plus=(a,b)=>q(a[0]*b[1]+b[0]*a[1],a[1]*b[1]);

function reduceRows(input){const m=input.map(row=>row.map(value=>Array.isArray(value)?q(value[0],value[1]):q(value))),rowCount=m.length,columnCount=m[0].length,pivots=[];let target=0;for(let column=0;column<columnCount&&target<rowCount;column++){let chosen=-1;for(let row=target;row<rowCount;row++)if(!isZero(m[row][column])){chosen=row;break;}if(chosen<0)continue;if(chosen!==target)[m[target],m[chosen]]=[m[chosen],m[target]];const pivot=m[target][column];for(let j=0;j<columnCount;j++)m[target][j]=quotient(m[target][j],pivot);for(let row=0;row<rowCount;row++){if(row===target||isZero(m[row][column]))continue;const factor=m[row][column];for(let j=0;j<columnCount;j++)m[row][j]=minus(m[row][j],times(factor,m[target][j]));}pivots.push(column);target++;}return {matrix:m,pivots};}

const matrixAdd=(A,B)=>A.map((row,i)=>row.map((value,j)=>value+B[i][j]));
function matrixMultiply(A,B){const result=[];for(let i=0;i<A.length;i++){const row=[];for(let j=0;j<B[0].length;j++){let sum=0;for(let k=0;k<B.length;k++)sum+=A[i][k]*B[k][j];row.push(sum);}result.push(row);}return result;}
function determinant(A){const permutations=[[0,1,2,1],[0,2,1,-1],[1,0,2,-1],[1,2,0,1],[2,0,1,1],[2,1,0,-1]];return permutations.reduce((sum,[a,b,c,sign])=>sum+sign*A[0][a]*A[1][b]*A[2][c],0);}
function nullspace(A){const rr=reduceRows(A),free=[];for(let column=0;column<3;column++)if(!rr.pivots.includes(column))free.push(column);const basis=free.map(column=>{const vector=[q(0),q(0),q(0)];vector[column]=q(1);rr.pivots.forEach((pivot,row)=>{vector[pivot]=negative(rr.matrix[row][column]);});return vector.map(publicQ);});return {basis,dimension:basis.length};}
function solve(A,b){const rr=reduceRows(A.map((row,i)=>[...row,b[i]]));for(const row of rr.matrix)if(row.slice(0,3).every(isZero)&&!isZero(row[3]))return {status:'inconsistent',particular:null,nullspace:[],conflict_row:row.map(publicQ)};const pivots=rr.pivots.filter(x=>x<3),free=[0,1,2].filter(x=>!pivots.includes(x)),particular=[q(0),q(0),q(0)];pivots.forEach((column,row)=>{particular[column]=rr.matrix[row][3];});const basis=free.map(column=>{const vector=[q(0),q(0),q(0)];vector[column]=q(1);pivots.forEach((pivot,row)=>{vector[pivot]=negative(rr.matrix[row][column]);});return vector.map(publicQ);});return {status:free.length?'infinite':'unique',particular:particular.map(publicQ),nullspace:basis,conflict_row:null};}

function validate(request){if(!request||typeof request!=='object'||Array.isArray(request))return'invalid_request';const keys=Object.keys(request);if(!keys.includes('operation'))return'missing_operation';if(!keys.includes('fixture'))return'missing_fixture';if(keys.some(key=>!['operation','fixture'].includes(key)))return'unknown_field';if(!allowed.has(request.operation))return'unsupported_operation';const f=request.fixture;if(!f||typeof f!=='object'||Array.isArray(f))return'invalid_fixture';const fkeys=Object.keys(f);const expected=['id','matrix','other','vector','rhs','scalar'];if(expected.some(key=>!fkeys.includes(key)))return'missing_fixture_field';if(fkeys.some(key=>!expected.includes(key)))return'unknown_fixture_field';const matrix=v=>Array.isArray(v)&&v.length===3&&v.every(row=>Array.isArray(row)&&row.length===3&&row.every(x=>Number.isSafeInteger(x)&&Math.abs(x)<=12));if(!matrix(f.matrix)||!matrix(f.other))return'invalid_matrix';if(!Array.isArray(f.vector)||f.vector.length!==3||!f.vector.every(Number.isSafeInteger))return'invalid_vector';if(!Array.isArray(f.rhs)||f.rhs.length!==3||!f.rhs.every(Number.isSafeInteger))return'invalid_rhs';if(!Number.isSafeInteger(f.scalar))return'invalid_scalar';if(typeof f.id!=='string'||!f.id)return'invalid_fixture_id';return null;}
function value(operation,f){const A=f.matrix,B=f.other,v=f.vector;switch(operation){case'la_matrix_shape':return {rows:3,columns:3,square:true,bounded:true};case'la_transpose':return transpose(A);case'la_matrix_add':return matrixAdd(A,B);case'la_matrix_product':return matrixMultiply(A,B);case'la_determinant':return {determinant:determinant(A)};case'la_rref':{const rr=reduceRows(A);return {matrix:rr.matrix.map(row=>row.map(publicQ)),pivot_columns:rr.pivots};}case'la_rank':return {rank:reduceRows(A).pivots.length};case'la_nullspace':return nullspace(A);case'la_solve':return solve(A,f.rhs);case'la_linear_map':return {image:vectorImage(A,v)};default:throw new Error('unreachable');}}
function execute(request){const before=canonical(request),beforeHash=sha(Buffer.from(before)),error=validate(request);let result;if(error)result={ok:false,value:null,error,disposition:null,external_credit:false};else result={ok:true,value:value(request.operation,request.fixture),error:null,disposition:'completed',external_credit:false};const after=canonical(request);return {result,input_unchanged:before===after,input_sha256_before:beforeHash,input_sha256_after:sha(Buffer.from(after))};}
