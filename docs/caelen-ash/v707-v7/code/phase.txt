function canonical(value) {
  if (Array.isArray(value)) return '[' + value.map(canonical).join(',') + ']';
  if (value && typeof value === 'object') return '{' + Object.keys(value).sort().map(key => JSON.stringify(key) + ':' + canonical(value[key])).join(',') + '}';
  return JSON.stringify(value);
}

function sha(value) {
  return crypto.createHash('sha256').update(Buffer.isBuffer(value) ? value : typeof value === 'string' ? value : canonical(value)).digest('hex');
}

function git(args, encoding = 'utf8') {
  return cp.execFileSync('git', args, { cwd: REPO, encoding, windowsHide: true, maxBuffer: 128 * 1024 * 1024 }).toString().trim();
}

function gitBlob(commit, relative) {
  return cp.execFileSync('git', ['show', commit + ':' + relative], { cwd: REPO, encoding: null, windowsHide: true, maxBuffer: 128 * 1024 * 1024 });
}

function requireHead(expected) {
  const head = git(['rev-parse', 'HEAD']);
  if (head !== expected) throw new Error('head_mismatch:' + head);
  const branch = git(['branch', '--show-current']);
  if (branch !== BRANCH) throw new Error('branch_mismatch:' + branch);
}

function ownerStatus(requireClean = false) {
  const output = git(['status', '--porcelain=v1', '-uall']);
  const rows = output ? output.split(/\r?\n/).filter(Boolean) : [];
  const outside = rows.filter(row => {
    const item = row.slice(3).replace(/\\/g, '/');
    const target = item.includes(' -> ') ? item.split(' -> ').at(-1) : item;
    return !target.startsWith(PREFIX + '/');
  });
  if (outside.length) throw new Error('outside_owner_scope:' + outside.join(','));
  if (requireClean && rows.length) throw new Error('owner_lane_not_clean');
  return rows;
}

function walk(directory) {
  const rows = [];
  if (!fs.existsSync(directory)) return rows;
  for (const entry of fs.readdirSync(directory, { withFileTypes: true })) {
    const full = path.join(directory, entry.name);
    if (entry.isDirectory()) rows.push(...walk(full));
    else rows.push(full);
  }
  return rows;
}

function manifest(relative) {
  const target = path.join(ROOT, relative);
  const files = walk(ROOT).filter(file => path.resolve(file) !== path.resolve(target)).sort();
  const entries = files.map(file => {
    const bytes = fs.readFileSync(file);
    return {
      path: PREFIX + '/' + path.relative(ROOT, file).replace(/\\/g, '/'),
      bytes: bytes.length,
      sha256: sha(bytes),
      byte_domain: 'checkout bytes expected to equal raw Git blob after LF-only authoring'
    };
  });
  writeJson(relative, {
    schema: 'caelen.owner-manifest.v1',
    owner: OWNER,
    phase: PHASE,
    self_exclusions: [PREFIX + '/' + relative.replace(/\\/g, '/')],
    entries,
    boundary: BOUNDARY
  });
  return entries;
}

function fixtureEvents(stage, index) {
  const label = String(index + 1).padStart(2, '0');
  const entity = 'synthetic-record-' + label;
  const expiry = (stage === 'x1' ? 9 : 11) + (index % 4);
  const createId = stage + '-create-' + label;
  const correctionId = stage + '-correct-' + label;
  const events = [{ id: createId, kind: 'create', entity, recorded_at: 1, valid_from: 1, parent: null, consent_expires: expiry, payload_hash: sha(stage + '-payload-' + label) }];
  if (index % 2 === 0) events.push({ id: correctionId, kind: 'correct', entity, recorded_at: 2, valid_from: 2, parent: createId, consent_expires: expiry, payload_hash: sha(stage + '-corrected-' + label) });
  if (index % 3 === 0) events.push({ id: stage + '-revoke-' + label, kind: 'revoke', entity, recorded_at: 3, valid_from: 6 + (index % 3), parent: null, consent_expires: expiry, payload_hash: sha(stage + '-revoke-' + label) });
  if (index % 5 === 0) events.push({ id: stage + '-retain-' + label, kind: 'retain', entity, recorded_at: 4, valid_from: 4, parent: null, consent_expires: expiry, payload_hash: runtime.sha('discard') });
  const publishTime = 5 + (index % 7);
  events.push({ id: stage + '-publish-' + label, kind: 'publish', entity, recorded_at: 5, valid_from: publishTime, parent: null, consent_expires: expiry + 10, payload_hash: sha(stage + '-publish-' + label) });
  events.push({ id: stage + '-appeal-' + label, kind: 'appeal', entity, recorded_at: 6, valid_from: 6, parent: null, consent_expires: expiry, payload_hash: sha(stage + '-appeal-' + label) });
  return stage === 'x2' ? events.slice().reverse() : events;
}

function runProcess(executable, args, options = {}) {
  return cp.spawnSync(executable, args, { cwd: options.cwd || REPO, input: options.input, encoding: 'utf8', windowsHide: true, maxBuffer: 128 * 1024 * 1024 });
}

function invokeRunner(labRelease, operation, request, outputFile) {
  const runner = path.join(labRelease, 'runners', operation + '.txt');

  ensureDir(requestFile);
  fs.writeFileSync(requestFile, JSON.stringify(request, null, 2) + '\n', 'utf8');

}

function runHooks(labRelease, stage) {
  const script = path.join(labRelease, 'plugins', 'ghc-family-laboratory-v20', 'scripts', 'advisory_hook.py');
  const rows = [];
  for (const name of HOOK_NAMES) {
    const accepted = runProcess('python', ['-I', script, name], { input: JSON.stringify({ hook_event_name: 'SessionStart', cwd: 'ghc-family-laboratory/' + stage }) });
    const rejected = runProcess('python', ['-I', script, name], { input: JSON.stringify([]) });
    rows.push({
      hook: name,
      accepted_status: accepted.status,
      accepted: JSON.parse(accepted.stdout || '{}'),
      malformed_status: rejected.status,
      malformed_refusal: JSON.parse(rejected.stdout || '{}'),
      live_host_observation: false
    });
  }
  return rows;
}

function aggregateFlow() {
  const ledgers = ['planning', 'x1', 'x2'].map(stage => readJson(path.join(ROOT, stage, 'method-flow.json')));
  const own = {
    methods: ledgers.reduce((sum, row) => sum + row.counts.methods, 0),
    witnesses: ledgers.reduce((sum, row) => sum + row.counts.witnesses, 0),
    passing_witnesses: ledgers.reduce((sum, row) => sum + row.counts.witness_results.pass, 0),
    failed_witnesses: ledgers.reduce((sum, row) => sum + row.counts.witness_results.fail, 0),
    negatives: ledgers.reduce((sum, row) => sum + row.counts.witness_results.fail, 0),
    open_gaps: 2,
    exact_gates: 1
  };
  const effective = {};
  for (const key of Object.keys(SOURCE_BASELINE)) effective[key] = SOURCE_BASELINE[key] + own[key];
  return { ledgers, own, effective };
}

  writeJson('final/source-faithful-ledger.json', { source_commit: SOURCE, source_bindings: readJson(path.join(ROOT, 'planning', 'source-bindings.json')).entries, baseline: SOURCE_BASELINE, fold_count: 1, source_execution_credit: 0, boundary: BOUNDARY });
  const baton = handoffText(truth);
  if (wordCount(baton) < 2000 || wordCount(baton) > 100000) throw new Error('baton_word_bounds:' + wordCount(baton));
  writeText('final/handoff.md', baton);
  writeJson('final/handoff-metadata.json', { words: wordCount(baton), bytes: Buffer.byteLength(baton), sha256: sha(baton), literal_eof: 'LITERAL_EOF_CAELEN_V707_V7' });
  manifest('final/manifest.json');
  console.log(JSON.stringify({ ok: true, stage: 'final', outcomes, accounting: truth.accounting, baton_words: wordCount(baton) }));
}

function privacyHits(text) {
  const classes = [
    ['raw_uuid', /\b[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}\b/i],
    ['private_local_path', new RegExp('\\b[A-Za-z]:[\\\\/](?:' + ['Users', 'GHC-' + 'Archives', 'Program Files'].join('|') + ')\\b', 'i')],
    ['private_uri', new RegExp('\\b(?:' + ['app', 'plugin'].join('|') + '):\\/\\/', 'i')],
    ['delegation_markup', new RegExp('<(?:codex_' + 'delegation|source_thread_' + 'id)>', 'i')],
    ['credential_assignment', /\b(?:api[_-]?key|access[_-]?token|password|secret)\b\s*[:=]\s*["'][^"']+["']/i]
  ];
  return classes.filter(row => row[1].test(text)).map(row => row[0]);
}

function validateStage(stage) {
  ownerStatus(false);
  const files = walk(ROOT);
  const issues = [];
  let jsonParses = 0;
  let maxWords = 0;
  const privacy = [];
  for (const file of files) {
    const relative = path.relative(ROOT, file).replace(/\\/g, '/');
    const text = fs.readFileSync(file, 'utf8');
    maxWords = Math.max(maxWords, wordCount(text));
    if (file.endsWith('.json')) {
      try { JSON.parse(text); jsonParses += 1; } catch (error) { issues.push('json:' + relative + ':' + error.message); }
    }
    for (const label of privacyHits(text)) privacy.push({ path: relative, class: label });
  }
  if (files.length >= 2000) issues.push('file_ceiling');
  if (maxWords > 100000) issues.push('word_ceiling');
  if (privacy.length) issues.push('privacy_candidates');
  if (!fs.existsSync(path.join(ROOT, stage, 'manifest.json')) && stage !== 'final') issues.push('manifest_missing');
  if (stage === 'planning') {
    if (fs.existsSync(path.join(ROOT, 'x1')) || fs.existsSync(path.join(ROOT, 'x2')) || fs.existsSync(path.join(ROOT, 'final'))) issues.push('planning_contains_later_stage');
  }
  if (stage === 'x1') {
    if (fs.existsSync(path.join(ROOT, 'x2')) || fs.existsSync(path.join(ROOT, 'final'))) issues.push('x1_contains_later_stage');
    const summary = readJson(path.join(ROOT, 'x1', 'summary.json'));
    if (summary.models !== 15 || summary.consultation.send_attempts !== 1 || !summary.consultation.acknowledged) issues.push('x1_requirements');
  }
  if (stage === 'x2') {
    if (fs.existsSync(path.join(ROOT, 'final'))) issues.push('x2_contains_final');
    const summary = readJson(path.join(ROOT, 'x2', 'summary.json'));
    if (summary.models !== 15 || summary.consultation.send_attempts !== 1 || !summary.consultation.acknowledged) issues.push('x2_requirements');
  }
  if (stage === 'final') {
    const truth = readJson(path.join(ROOT, 'final', 'phase-truth.json'));
    const metadata = readJson(path.join(ROOT, 'final', 'handoff-metadata.json'));
    if (metadata.words < 2000 || metadata.words > 100000) issues.push('baton_bounds');
    if (truth.terminal_verdict !== 'NOT_READY_FOR_STAGE_20') issues.push('stage20');
    if (truth.consultation.x1_messages !== 1 || truth.consultation.x2_messages !== 1) issues.push('consultation_counts');
  }
  const manifestPath = stage === 'final' ? path.join(ROOT, 'final', 'manifest.json') : path.join(ROOT, stage, 'manifest.json');
  let manifestEntries = 0;
  if (fs.existsSync(manifestPath)) {
    const m = readJson(manifestPath);
    manifestEntries = m.entries.length;
    for (const entry of m.entries) {
      const file = path.join(REPO, entry.path);
      if (!fs.existsSync(file)) issues.push('manifest_missing_file:' + entry.path);
      else {
        const bytes = fs.readFileSync(file);
        if (bytes.length !== entry.bytes || sha(bytes) !== entry.sha256) issues.push('manifest_mismatch:' + entry.path);
      }
    }
  }
  const receipt = { schema: 'caelen.stage-validation.v1', owner: OWNER, phase: PHASE, stage, valid: issues.length === 0, issues, owner_files: files.length, json_parses: jsonParses, maximum_words: maxWords, privacy_candidates: privacy, manifest_entries: manifestEntries, head: git(['rev-parse', 'HEAD']), boundary: BOUNDARY };
  bankJson('validation-' + stage + '.json', receipt);
  console.log(JSON.stringify(receipt));
  if (!receipt.valid) process.exitCode = 1;
}

function canonicalRun() {
  if (!BANK) throw new Error('bank_required');
  for (const key of ['plan', 'x1', 'x2', 'final']) if (!ARGS[key]) throw new Error(key + '_required');
  const directory = path.join(BANK, 'canonical');
  const latch = path.join(directory, 'invocation.json');
  if (fs.existsSync(latch)) throw new Error('canonical_latch_exists');
  fs.mkdirSync(directory, { recursive: true });
  fs.writeFileSync(latch, JSON.stringify({ state: 'INVOKED_NO_RETRY', invocations: 1, successes: 0, replays: 0, exact_final: ARGS.final }, null, 2) + '\n');
  const checks = [];
  const check = (name, passed, observed) => checks.push({ name, passed: Boolean(passed), observed });
  const head = git(['rev-parse', 'HEAD']);
  const upstream = git(['rev-parse', '@{upstream}']);
  const tracking = git(['rev-parse', 'refs/remotes/origin/' + BRANCH]);
  const live = git(['ls-remote', 'origin', 'refs/heads/' + BRANCH]).split(/\s+/)[0];
  check('branch', git(['branch', '--show-current']) === BRANCH, git(['branch', '--show-current']));
  check('head', head === ARGS.final, head);
  check('upstream', upstream === ARGS.final, upstream);
  check('tracking', tracking === ARGS.final, tracking);
  check('fresh_live', live === ARGS.final, live);
  check('clean', git(['status', '--porcelain=v1']) === '', git(['status', '--porcelain=v1']));
  const chain = [ARGS.plan, ARGS.x1, ARGS.x2, ARGS.final];
  const parents = [SOURCE, ARGS.plan, ARGS.x1, ARGS.x2];
  chain.forEach((commit, index) => check('direct_parent_' + (index + 1), git(['rev-parse', commit + '^']) === parents[index], git(['rev-parse', commit + '^'])));
  check('commit_count', Number(git(['rev-list', '--count', SOURCE + '..' + ARGS.final])) === 4, Number(git(['rev-list', '--count', SOURCE + '..' + ARGS.final])));
  const mergeRows = git(['rev-list', '--merges', SOURCE + '..' + ARGS.final]);
  check('zero_merges', mergeRows === '', mergeRows);
  const changed = git(['diff', '--name-only', SOURCE + '..' + ARGS.final]).split(/\r?\n/).filter(Boolean);
  check('owner_only_delta', changed.every(row => row.startsWith(PREFIX + '/')), changed.length);
  const tree = git(['ls-tree', '-r', '--name-only', ARGS.final]).split(/\r?\n/).filter(Boolean);
  const ownerFiles = tree.filter(row => row.startsWith(PREFIX + '/'));
  check('tree_ceiling', tree.length < 2000, tree.length);
  check('owner_files', ownerFiles.length < 400, ownerFiles.length);
  const manifests = [[ARGS.plan, 'planning/manifest.json'], [ARGS.x1, 'x1/manifest.json'], [ARGS.x2, 'x2/manifest.json'], [ARGS.final, 'final/manifest.json']];
  let bindings = 0;
  for (const [commit, relative] of manifests) {
    const m = JSON.parse(gitBlob(commit, PREFIX + '/' + relative).toString('utf8'));
    let passed = true;
    for (const entry of m.entries) {
      const bytes = gitBlob(commit, entry.path);
      bindings += 1;
      if (bytes.length !== entry.bytes || sha(bytes) !== entry.sha256) passed = false;
    }
    check('manifest_' + relative, passed, m.entries.length);
  }
  let jsonParses = 0;
  let maximumWords = 0;
  const privacy = [];
  for (const relative of ownerFiles) {
    const text = gitBlob(ARGS.final, relative).toString('utf8');
    maximumWords = Math.max(maximumWords, wordCount(text));
    if (relative.endsWith('.json')) { JSON.parse(text); jsonParses += 1; }
    for (const label of privacyHits(text)) privacy.push({ path: relative, class: label });
  }
  check('strict_json', jsonParses > 0, jsonParses);
  check('word_cap', maximumWords <= 100000, maximumWords);
  check('privacy', privacy.length === 0, privacy);
  const truth = JSON.parse(gitBlob(ARGS.final, PREFIX + '/final/phase-truth.json').toString('utf8'));
  check('source_fold', truth.accounting.source_fold_count === 1, truth.accounting.source_fold_count);
  check('models', truth.models.x1 === 15 && truth.models.x2 === 15, truth.models);
  check('consultation', truth.consultation.x1_messages === 1 && truth.consultation.x2_messages === 1, truth.consultation);
  check('route_held', truth.route.state === 'PREPARED_NOT_SENT', truth.route.state);
  check('stage20', truth.terminal_verdict === 'NOT_READY_FOR_STAGE_20', truth.terminal_verdict);
  check('source_replays', truth.source_domain_replays === 0 && truth.source_canonical_replays === 0, { domain: truth.source_domain_replays, canonical: truth.source_canonical_replays });
  const passed = checks.filter(row => row.passed).length;
  const payload = {
    state: passed === checks.length ? 'VALID_EXACT_FINAL_OWNER_SCOPED_METADATA_CANONICAL' : 'FAILED_ZERO_CANONICAL_SUCCESS_CREDIT',
    exact_final: ARGS.final, checks, passed, total: checks.length, owner_files: ownerFiles.length, repository_files: tree.length,
    raw_blob_bindings: bindings, json_files: jsonParses, maximum_words: maximumWords, invocations: 1,
    successes: passed === checks.length ? 1 : 0, replays: 0, domain_replayed: false, source_canonical_replayed: false, boundary: BOUNDARY
  };
  const payloadText = JSON.stringify(payload, null, 2) + '\n';
  fs.writeFileSync(path.join(directory, 'payload.json'), payloadText, 'utf8');
  const receipt = { state: payload.state, exact_final: ARGS.final, invocations: 1, successes: payload.successes, replays: 0, checks_passed: passed, checks_total: checks.length, payload_sha256: sha(payloadText) };
  fs.writeFileSync(path.join(directory, 'receipt.json'), JSON.stringify(receipt, null, 2) + '\n', 'utf8');
  fs.writeFileSync(latch, JSON.stringify({ state: payload.successes === 1 ? 'SUCCESS_LATCHED_NO_REPLAY' : 'FAILED_ZERO_SUCCESS_CREDIT_NO_REPLAY', invocations: 1, successes: payload.successes, replays: 0, exact_final: ARGS.final }, null, 2) + '\n');
  console.log(JSON.stringify(receipt));
  if (payload.successes !== 1) process.exitCode = 1;
}

if (COMMAND === 'plan') plan();
else if (COMMAND === 'x1' || COMMAND === 'x2') runSession(COMMAND);
else if (COMMAND === 'final') finalBuild();
else if (COMMAND === 'validate') validateStage(ARGS.stage);
else if (COMMAND === 'canonical') canonicalRun();
else throw new Error('command must be plan, x1, x2, final, validate, or canonical');
