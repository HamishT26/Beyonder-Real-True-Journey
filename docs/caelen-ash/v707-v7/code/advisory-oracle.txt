function canonical(value) {
  if (Array.isArray(value)) return '[' + value.map(canonical).join(',') + ']';
  if (value && typeof value === 'object') return '{' + Object.keys(value).sort().map(key => JSON.stringify(key) + ':' + canonical(value[key])).join(',') + '}';
  return JSON.stringify(value);
}

function sha(value) {
  return crypto.createHash('sha256').update(Buffer.isBuffer(value) ? value : typeof value === 'string' ? value : canonical(value)).digest('hex');
}

function writeJson(file, value) {
  fs.mkdirSync(path.dirname(file), { recursive: true });
  fs.writeFileSync(file, JSON.stringify(value, null, 2) + '\n', 'utf8');
}

function readJson(file) {
  return JSON.parse(fs.readFileSync(file, 'utf8'));
}

function implementationQueueA() {
  const seen = new Map([['r1', sha('corrected-payload')]]);
  const exact = seen.get('r1') === sha('corrected-payload') ? 'no_new_effect' : 'unexpected';
  const changed = seen.get('r1') !== sha('changed-payload') ? 'explicit_conflict' : 'unexpected';
  return {
    original_status: 'failed_unchanged', correction_parent: 'r0', inherited_completion_credit: 0,
    queue: { active: 'r1', waiting: 'c0', overflow: 'explicit_backpressure' }, appeal_renews_grant: false,
    publication: { r1: 'denied_after_revocation', c0: 'published_exactly_once' },
    retention: { payload_available: false, metadata_retained: true, verified_on_current_read: false },
    retry: { exact_replay: exact, changed_same_key: changed }, pending_at_25: 0
  };
}

function implementationQueueB() {

    ['active', 'r1'], ['waiting', 'c0'], ['overflow', 'explicit_backpressure'], ['appeal_renews_grant', false],
    ['r1', 'denied_after_revocation'], ['c0', 'published_exactly_once'], ['payload_available', false],
    ['metadata_retained', true], ['verified_on_current_read', false], ['exact_replay', 'no_new_effect'],
    ['changed_same_key', 'explicit_conflict'], ['pending_at_25', 0]

    original_status: row.original_status, correction_parent: row.correction_parent, inherited_completion_credit: row.inherited_completion_credit,

    retry: { exact_replay: row.exact_replay, changed_same_key: row.changed_same_key }, pending_at_25: row.pending_at_25
  };
}

function walk(directory) {
  const rows = [];
  for (const entry of fs.readdirSync(directory, { withFileTypes: true })) {
    const full = path.join(directory, entry.name);
    if (entry.isDirectory()) rows.push(...walk(full)); else rows.push(full);
  }
  return rows;
}
