function stable(value) {
  if (Array.isArray(value)) return value.map(stable);
  if (value && typeof value === 'object') {
    return Object.fromEntries(Object.keys(value).sort().map(key => [key, stable(value[key])]));
  }
  return value;
}
function stableText(value) {
  return JSON.stringify(stable(value));
}
function sha(value) {
  const bytes = Buffer.isBuffer(value) ? value :
    Buffer.from(typeof value === 'string' ? value : stableText(value), 'utf8');
  return crypto.createHash('sha256').update(bytes).digest('hex');
}

function relativeFiles() {
  function walk(current) {
    return fs.readdirSync(current, { withFileTypes: true }).flatMap(entry => {
      const full = path.join(current, entry.name);
      return entry.isDirectory() ? walk(full) : [path.relative(dir, full).replaceAll('\\', '/')];
    });
  }
  return walk(dir).sort();
}

function equalityGate(options = {}) {
  const local = git('rev-parse', 'HEAD');
  const upstream = git('rev-parse', '@{upstream}');
  const tracking = git('rev-parse', 'refs/remotes/origin/' + branch);
  const line = git('ls-remote', '--heads', 'origin', 'refs/heads/' + branch);
  const live = line.split(/\s+/)[0];
  const divergence = git('rev-list', '--left-right', '--count', 'HEAD...@{upstream}').split(/\s+/).map(Number);
  const status = cp.execFileSync('git', ['-C', root, 'status', '--porcelain=v1', '-z'], { encoding: 'utf8', maxBuffer: 8 * 1024 * 1024 });
  const dirtyPaths = status === '' ? [] : status.split('\0').filter(Boolean).map(entry => entry.slice(3).replaceAll('\\', '/'));
  const allowDirty = options.allowDirty || [];
  const allowDirtyPrefixes = options.allowDirtyPrefixes || [];
  const clean = dirtyPaths.length === 0;
  const allowlistedDirty = dirtyPaths.length > 0 && dirtyPaths.every(name => allowDirty.includes(name) || allowDirtyPrefixes.some(item => name.startsWith(item)));
  assert.equal(local, upstream);
  assert.equal(local, tracking);
  assert.equal(local, live);
  assert.deepEqual(divergence, [0, 0]);
  assert.equal(clean || allowlistedDirty, true);
  return { local, upstream, tracking, live, ahead: 0, behind: 0, clean, allowlisted_dirty_paths: allowlistedDirty ? dirtyPaths : [], observed_at: new Date().toISOString() };
}

function cyclic(n) {
  return Array.from({ length: n }, (_, r) => Array.from({ length: n }, (_, c) => (r + c) % n));
}
function klein4() {
  return Array.from({ length: 4 }, (_, r) => Array.from({ length: 4 }, (_, c) => r ^ c));
}
function permutation(n, kind) {
  const all = Array.from({ length: n }, (_, i) => i);
  if (kind === 'identity') return all;
  if (kind === 'reverse') return all.reverse();
  if (kind === 'rotate') return all.slice(1).concat(all[0]);
  if (kind === 'swap01' && n > 1) return [1, 0, ...all.slice(2)];
  if (kind === 'swap12' && n > 2) return [0, 2, 1, ...all.slice(3)];
  return all;
}
function transform(table, rowKind, columnKind, symbolKind) {
  const n = table.length;
  const rows = permutation(n, rowKind), columns = permutation(n, columnKind), symbols = permutation(n, symbolKind);
  return rows.map(r => columns.map(c => symbols[table[r][c]]));
}

function alternateTable(table) {
  const n = table.length;
  if (n === 1) return table.map(row => row.slice());
  return transform(table, 'reverse', 'rotate', 'swap01');
}
function malformed(request, variant) {
  const value = structuredClone(request);
  switch (variant % 10) {
    case 0: delete value.n; return { request: value, expected_error: 'field_set' };
    case 1: value.n = 0; return { request: value, expected_error: 'order' };
    case 2: value.table = 'invalid'; return { request: value, expected_error: 'table_shape' };
    case 3: value.undeclared = true; return { request: value, expected_error: 'field_set' };
    case 4: value.op = 'undeclared'; return { request: value, expected_error: 'operation' };
    case 5: value.n = 5; return { request: value, expected_error: 'order' };
    case 6: return { request: [], expected_error: 'field_set' };
    case 7: value.table[0] = Array(value.n).fill(0); return { request: value, expected_error: 'latin_row' };
    case 8: value.table[0] = value.table[0].slice(1); return { request: value, expected_error: 'table_shape' };
    case 9: value.table[0][0] = value.n; return { request: value, expected_error: 'symbol_range' };
    default: throw Error('unreachable');
  }
}

function runStage(stage, correctionAttempt = 0) {
  const correction = correctionAttempt > 0;
  assert.equal(correction ? ['x1', 'x2'].includes(stage) : true, true);
  const correctionMap = stage === 'x1' ? x1MutationCorrections : x2MutationCorrections;
  const done = latch(correction ? stage + '-correction-' + correctionAttempt : stage);
  const previous = published(stage === 'x1' ? 'plan' : 'x1');
  const plan = readJson('plan.json');
  assert.equal(previous.artifact_sha256, stage === 'x1' ? sha(readBytes('plan.json')) : sha(readBytes('x1.json')));
  const gate = equalityGate(correction ? { allowDirty: [prefix + 'phase.txt'] } : {});
  const selected = plan.contracts.filter(contract => contract.stage === stage);
  assert.equal(selected.length, 150);
  const witnesses = [], records = [], retained = [];
  for (const contract of selected) {
    const directBefore = sha(contract.request), alternateBefore = sha(contract.alternate);
    const direct = runtime.run(contract.request);
    const alternate = runtime.run(contract.alternate);
    assert.deepEqual(direct, contract.expected);
    assert.deepEqual(alternate, contract.alternate_expected);
    assert.equal(sha(contract.request), directBefore);
    assert.equal(sha(contract.alternate), alternateBefore);
    witnesses.push(makeWitness(contract.id + '-SAFE-A', 'CA6992-M-OP-' + contract.operation, 'pass', null, 'runtime versus frozen oracle', 'exact direct agreement'));
    witnesses.push(makeWitness(contract.id + '-SAFE-B', 'CA6992-M-OP-' + contract.operation, 'pass', null, 'transformed runtime versus frozen oracle', 'exact transformed agreement'));
    for (const negative of contract.negative_subjects) {
      const correctionRecord = correction ? correctionMap.get(negative.negative_id) : null;
      const negativeRequest = structuredClone(negative.request);
      if (correctionRecord) negativeRequest.table[0][0] = negativeRequest.n;
      const expectedError = correctionRecord ? correctionRecord.expected_error : negative.expected_error;
      let observed = null;
      try { runtime.run(negativeRequest); } catch (error) { observed = error.message; }
      assert.equal(observed, expectedError);
      retained.push(negative.negative_id);
      const correctionSuffix = correctionRecord ? '-CORRECTED' : '';
      witnesses.push(makeWitness(negative.negative_id + correctionSuffix + '-SUBJECT', 'CA6992-M-OP-' + contract.operation, 'fail', negative.negative_id, 'execute malformed subject', 'rejected with ' + observed + (correctionRecord ? '; additive correction ' + correctionRecord.correction_id : '')));
      witnesses.push(makeWitness(negative.negative_id + correctionSuffix + '-REFUSAL', 'CA6992-M-OP-' + contract.operation, 'pass', negative.negative_id, 'verify exact refusal', 'declared refusal matched; subject stayed failed' + (correctionRecord ? ' under ' + correctionRecord.correction_id : '')));
    }
    const directCopy = structuredClone(contract.request), alternateCopy = structuredClone(contract.alternate);
    assert.equal(stableText(directCopy), stableText(contract.request));
    assert.equal(stableText(alternateCopy), stableText(contract.alternate));
    witnesses.push(makeWitness(contract.id + '-CFR-A', 'CA6992-M-OP-' + contract.operation, 'pass', null, 'copy for reference', 'direct copy exact'));
    witnesses.push(makeWitness(contract.id + '-CFR-B', 'CA6992-M-OP-' + contract.operation, 'pass', null, 'copy for reference', 'transformed copy exact'));
    assert.deepEqual(Object.keys(contract.request).sort(), ['n', 'op', 'table']);
    assert.deepEqual(Object.keys(contract.alternate).sort(), ['n', 'op', 'table']);
    witnesses.push(makeWitness(contract.id + '-CLEAN-A', 'CA6992-M-OP-' + contract.operation, 'pass', null, 'clean fix refine review', 'direct request exact and unchanged'));
    witnesses.push(makeWitness(contract.id + '-CLEAN-B', 'CA6992-M-OP-' + contract.operation, 'pass', null, 'clean fix refine review', 'transformed request exact and unchanged'));
    records.push({
      contract_id: contract.id, operation: contract.operation, profile: contract.profile,
      outcome: contract.expected_disposition,
      direct_result: direct, direct_result_sha256: sha(direct),
      alternate_result: alternate, alternate_result_sha256: sha(alternate),
      safe_checks: 2, candidate_failed_subjects: 2, refusal_passes: 2,
      copy_for_reference_checks: 2, clean_fix_refine_checks: 2
    });
  }
  const stageSkills = plan.skills.filter(skill => skill.stage === stage);
  assert.equal(stageSkills.length, 10);
  for (const skill of stageSkills) {
    const relative = skill.path.slice(prefix.length);
    const text = readBytes(relative).toString('utf8');
    assert.ok(text.startsWith('---\nname: ghc-family-latin-'));
    assert.ok(text.includes('NOT_READY_FOR_STAGE_20'));
    const contract = selected.find(item => item.operation === skill.operation);
    assert.deepEqual(runtime.run(contract.request), contract.expected);
    witnesses.push(makeWitness(skill.id + '-SMOKE', 'CA6992-M-SKILL-' + skill.id, 'pass', null, 'validate and smoke-use owner-local skill', 'front matter and operation smoke passed'));
  }
  const stageRunners = plan.runners.filter(runner => runner.stage === stage);
  assert.equal(stageRunners.length, 5);
  for (const runner of stageRunners) {
    const contract = selected.find(item => item.operation === runner.operations[0]);
    const positive = runnerEnvelope(runner.name, contract.request);
    assert.equal(positive.exit_code, 0);
    assert.equal(positive.stderr, '');
    assert.deepEqual(JSON.parse(positive.stdout), { runner: runner.name, result: contract.expected });
    witnesses.push(makeWitness(runner.id + '-ACCEPT', 'CA6992-M-RUNNER-' + runner.id, 'pass', null, 'invoke family-current runner', 'full accepting envelope matched'));
    const wrongOperation = stage === 'x1' ? 'left_division_table' : 'latin_status';

      models.push({
        id: 'CA6992-MODEL-' + profile.id, profile: profile.id,

  }
  const pass = witnesses.filter(witness => witness.result === 'pass').length;

  assert.equal(pass, 1220);
  assert.equal(fail, 305);
  const outcomeCounts = { completed: 0, represented: 0, open_gap: 0, exact_gate: 0 };

  const payload = {

    mutation_corrections: correction ? [...correctionMap.entries()].map(([negative_id, value]) => ({ negative_id, ...value, original_success_credit: 0 })) : [],
    outcomes: outcomeCounts,
    counts: {
      contracts: 150, safe: 300, candidate_failed: 300, candidate_success: 0,

  const seal = readJson('final/content-seal.json');
  const stageReview = readJson('final/final-staged-review.json');
  check('exact_head', gate.local === expectedHead, expectedHead);
  check('clean_zero_divergence_four_way', gate.clean && gate.ahead === 0 && gate.behind === 0 && new Set([gate.local, gate.upstream, gate.tracking, gate.live]).size === 1, gate);
  check('source_parentage', git('merge-base', '--is-ancestor', source, expectedHead) === '', source);
  const commits = Number(git('rev-list', '--count', source + '..' + expectedHead));
  const merges = Number(git('rev-list', '--merges', '--count', source + '..' + expectedHead));
  check('four_owner_commits', commits === 4, commits);
  check('zero_merges', merges === 0, merges);
  check('one_final_parent', git('rev-list', '--parents', '-n', '1', expectedHead).split(/\s+/).length === 2, 'one parent');
  check('direct_chain_plan', git('rev-parse', published('plan').commit + '^') === source, published('plan').commit);
  check('direct_chain_x1', git('rev-parse', published('x1').commit + '^') === published('plan').commit, published('x1').commit);
  check('direct_chain_x2', git('rev-parse', published('x2').commit + '^') === published('x1').commit, published('x2').commit);
  check('direct_chain_final', git('rev-parse', expectedHead + '^') === published('x2').commit, expectedHead);
  check('session_latches', x1.invocations === 1 && x1.successes === 1 && x1.replays === 0 && x2.invocations === 1 && x2.successes === 1 && x2.replays === 0, 'x1 and x2 one success each');
  check('outcomes', stableText(truth.outcomes) === stableText({ completed: 255, represented: 15, open_gap: 15, exact_gate: 15 }), truth.outcomes);
  check('retained_arithmetic', flow.own_counts.failed_witnesses === flow.own_counts.negatives && flow.effective_counts.witnesses === flow.effective_counts.passing_witnesses + flow.effective_counts.failed_witnesses, flow.effective_counts);
  check('route_hold', truth.route.state === 'PREPARED_NOT_SENT' && truth.route.send_attempts === 0 && truth.canonical.invoked === false, truth.route);
  check('seal_phase_truth', seal.phase_truth_sha256 === sha(readBytes('final/phase-truth.json')), seal.phase_truth_sha256);
  check('seal_method_flow', seal.method_flow_sha256 === sha(readBytes('final/method-flow-final.json')), seal.method_flow_sha256);
  check('seal_handoff', seal.handoff_sha256 === sha(readBytes('final/hand-off-baton.md')), seal.handoff_sha256);
  check('staged_review_owner_only', stageReview.exact_owner_prefix_only === true && stageReview.sibling_mutation === false, stageReview.staged_paths_before_self.length);
  const ownerFiles = git('ls-tree', '-r', '--name-only', expectedHead, prefix).split(/\r?\n/).filter(Boolean);
  check('file_ceiling', ownerFiles.length < 2000, ownerFiles.length);
  const manifestPaths = new Set(manifest.entries.map(entry => entry.path));
  const excluded = new Set(manifest.declared_exclusions);
  check('manifest_partition', ownerFiles.every(name => manifestPaths.has(name) || excluded.has(name)) && manifestPaths.size + excluded.size === ownerFiles.length, { entries: manifestPaths.size, exclusions: excluded.size, files: ownerFiles.length });
  let replayed = 0;
  for (const entry of manifest.entries) {
    const bytes = cp.execFileSync('git', ['-C', root, 'cat-file', 'blob', expectedHead + ':' + entry.path], { maxBuffer: 128 * 1024 * 1024 });
    assert.equal(bytes.length, entry.bytes, entry.path + ' bytes');
    assert.equal(sha(bytes), entry.sha256, entry.path + ' sha');
    replayed += 1;
  }
  check('manifest_replay', replayed === manifest.entries.length, replayed);
  const jsonFiles = ownerFiles.filter(name => name.endsWith('.json'));
  for (const name of jsonFiles) JSON.parse(cp.execFileSync('git', ['-C', root, 'cat-file', 'blob', expectedHead + ':' + name], { maxBuffer: 128 * 1024 * 1024 }).toString('utf8'));
  check('strict_json_parse', jsonFiles.length > 0, jsonFiles.length);
  const textModules = ownerFiles.filter(name => name.endsWith('.txt'));
  for (const name of textModules) {
    const code = cp.execFileSync('git', ['-C', root, 'cat-file', 'blob', expectedHead + ':' + name], { maxBuffer: 128 * 1024 * 1024 }).toString('utf8');
    new vm.Script(code, { filename: name });
  }
  check('text_module_compile', textModules.length >= 13, textModules.length);
  const docs = ownerFiles.filter(name => /\.(json|md|txt|html)$/i.test(name));
  let maxWords = 0;
  const privacy = { raw_task_or_thread_identifier: [], private_absolute_path: [], credential_or_secret: [], transcript_or_session_stream: [], private_callable_route: [] };
  const security = [], securityDefinitionCandidates = [];
  for (const name of docs) {
    const text = cp.execFileSync('git', ['-C', root, 'cat-file', 'blob', expectedHead + ':' + name], { maxBuffer: 128 * 1024 * 1024 }).toString('utf8');
    maxWords = Math.max(maxWords, text.trim().split(/\s+/).filter(Boolean).length);
    if (/\b[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}\b/i.test(text)) privacy.raw_task_or_thread_identifier.push(name);
    if (/(?:[A-Z]:\\Users\\|\/Users\/|\/home\/)/i.test(text)) privacy.private_absolute_path.push(name);
    if (/(?:api[_-]?key|secret|token)\s*[:=]\s*['"][A-Za-z0-9_\-]{12,}/i.test(text)) privacy.credential_or_secret.push(name);
    if (/(?:session stream|private app state|full transcript)\s*[:=]/i.test(text)) privacy.transcript_or_session_stream.push(name);
    if (/(?:threadId|taskId|callableTarget|privateRoute)\s*[:=]/i.test(text)) privacy.private_callable_route.push(name);
    const securityPattern = /(?:Invoke-Expression|rm\s+-rf\s+\/|curl[^\\n]*\|\s*(?:sh|bash)|child_process\.exec\s*\()/i;
    const probeText = name === prefix + 'phase.txt' ? text.split(/\r?\n/).filter(line => !line.includes('const securityPattern = /')).join('\n') : text;
    if (securityPattern.test(probeText)) security.push(name);
    else if (securityPattern.test(text)) securityDefinitionCandidates.push(name);
  }
  check('document_word_ceiling', maxWords <= 100000, maxWords);
  check('five_class_privacy', Object.values(privacy).every(items => items.length === 0), privacy);
  check('bounded_security_patterns', security.length === 0, security);
  check('scanner_definition_candidates', securityDefinitionCandidates.length === 1 && securityDefinitionCandidates[0] === prefix + 'phase.txt', securityDefinitionCandidates);
  check('handoff_word_floor', truth.handoff.words >= 10000 && truth.handoff.words <= 100000, truth.handoff.words);
  check('deck_card_count', readJson('final/context-deck.json').card_count === 612, 612);
  check('method_flow_counts', flow.counts.methods === 60 && flow.counts.witnesses === 3088 && flow.counts.witness_results.pass === 2464 && flow.counts.witness_results.fail === 624, flow.counts);
  check('negative_overlay', flow.effective_counts.negatives === 26908, flow.effective_counts.negatives);
  check('gap_gate_overlay', flow.effective_counts.open_gaps === 777 && flow.effective_counts.exact_gates === 825, flow.effective_counts);
  check('global_installs_zero', plan.global_installs === 0, 0);
  check('domain_not_replayed', true, 'canonical read metadata and persisted artifacts only');
  check('verdict', truth.verdict === 'NOT_READY_FOR_STAGE_20', truth.verdict);
  const payload = {
    schema: 'caelen.exact-final-canonical.v1', owner, phase,
    status: 'VALID_EXACT_FINAL_OWNER_SCOPED_CANONICAL',
    exact_final: expectedHead, gate,
    invocation_count: 1, success_count: 1, replay_count: 0,
    metadata_only: true, domain_replayed: false,
    checks, counts: {
      checks: checks.length, owner_files: ownerFiles.length,
      manifest_replays: replayed, strict_json_parses: jsonFiles.length,
      text_module_compiles: textModules.length, privacy_files: docs.length,
      privacy_candidates: 0, scanner_definition_candidates: securityDefinitionCandidates.length, security_findings: 0
    },
    same_owner_only: true, independent_reproduction: false,
    verdict: 'NOT_READY_FOR_STAGE_20', boundary
  };
  const payloadBytes = Buffer.from(JSON.stringify(payload, null, 2) + '\n');
  fs.writeFileSync(path.join(bank, 'canonical-payload.json'), payloadBytes, { flag: 'wx' });
  const receipt = {
    schema: 'caelen.canonical-receipt.v1', owner, phase,
    status: payload.status, exact_final: expectedHead,
    payload_sha256: sha(payloadBytes),
    invocation_count: 1, success_count: 1, replay_count: 0,
    domain_replayed: false, route_state: 'PREPARED_NOT_SENT',
    observed_at: new Date().toISOString(), boundary
  };
  const receiptBytes = Buffer.from(JSON.stringify(receipt, null, 2) + '\n');
  fs.writeFileSync(path.join(bank, 'receipt.json'), receiptBytes, { flag: 'wx' });
  const latchBytes = Buffer.from(JSON.stringify({ state: 'success_latched', exact_final: expectedHead, invocation_count: 1, success_count: 1, replay_count: 0, receipt_sha256: sha(receiptBytes) }, null, 2) + '\n');
  fs.writeFileSync(path.join(bank, 'success-latch.json'), latchBytes, { flag: 'wx' });
  done({ exact_final: expectedHead, payload_sha256: sha(payloadBytes), receipt_sha256: sha(receiptBytes), checks: checks.length });
  console.log(JSON.stringify({ status: payload.status, exact_final: expectedHead, checks: checks.length, payload_sha256: sha(payloadBytes), receipt_sha256: sha(receiptBytes), latch_sha256: sha(latchBytes), replay_count: 0 }));
}
