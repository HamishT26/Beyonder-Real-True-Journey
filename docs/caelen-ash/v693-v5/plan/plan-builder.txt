function ensure(dir) { fs.mkdirSync(dir, { recursive: true }); }
function normalize(value) {
  if (Array.isArray(value)) return value.map(normalize);
  if (value && typeof value === "object") {
    const out = {};
    for (const key of Object.keys(value).sort()) out[key] = normalize(value[key]);
    return out;
  }
  return value;
}
function canonical(value) { return JSON.stringify(normalize(value)); }
function shaBytes(bytes) { return crypto.createHash("sha256").update(bytes).digest("hex"); }
function shaObject(value) { return shaBytes(Buffer.from(canonical(value), "utf8")); }
function writeJson(relative, value) { const target = path.join(PLAN, relative); ensure(path.dirname(target)); fs.writeFileSync(target, JSON.stringify(value, null, 2) + "\n", "utf8"); }
function writeText(relative, value) { const target = path.join(PLAN, relative); ensure(path.dirname(target)); const clean = String(value).replace(/\r\n/g, "\n").replace(/\r/g, "\n"); fs.writeFileSync(target, clean.endsWith("\n") ? clean : clean + "\n", "utf8"); }
function git(...args) { const out = cp.spawnSync("git", ["-C", REPO, ...args], { encoding: "utf8", maxBuffer: 128 * 1024 * 1024 }); if (out.status !== 0) throw new Error(`git ${args.join(" ")} failed: ${out.stderr}`); return out.stdout.trim(); }
function sorted(values) { return [...new Set(values)].sort((a, b) => String(a).localeCompare(String(b))); }
function setEqual(a, b) { return canonical(sorted(a)) === canonical(sorted(b)); }
function subsets(values) { const out = []; for (let mask = 0; mask < 2 ** values.length; mask++) out.push(values.filter((_, i) => (mask & (1 << i)) !== 0)); return out; }
function framework(argumentsList, attacks) { return { arguments: argumentsList, attacks }; }
function attacksFrom(fw, fromSet, target) { const sources = new Set(fromSet); return fw.attacks.some(([from, to]) => sources.has(from) && to === target); }
function attackers(fw, argument) { return sorted(fw.attacks.filter(([, to]) => to === argument).map(([from]) => from)); }
function conflictFree(fw, candidate) { const s = new Set(candidate); return !fw.attacks.some(([from, to]) => s.has(from) && s.has(to)); }
function defends(fw, candidate, argument) { return attackers(fw, argument).every(attacker => attacksFrom(fw, candidate, attacker)); }
function defended(fw, candidate) { return fw.arguments.filter(argument => defends(fw, candidate, argument)).sort(); }
function admissible(fw, candidate) { return conflictFree(fw, candidate) && candidate.every(argument => defends(fw, candidate, argument)); }
function completeExtension(fw, candidate) { return admissible(fw, candidate) && setEqual(candidate, defended(fw, candidate)); }
function grounded(fw) { let current = []; for (let i = 0; i <= fw.arguments.length; i++) { const next = defended(fw, current); if (setEqual(next, current)) return sorted(next); current = next; } return sorted(current); }
function maximalSets(sets) { return sets.filter(a => !sets.some(b => a.length < b.length && a.every(x => b.includes(x)))).map(sorted).sort((a, b) => canonical(a).localeCompare(canonical(b))); }
function preferred(fw) { return maximalSets(subsets(fw.arguments).filter(set => admissible(fw, set))); }
function stable(fw) { return subsets(fw.arguments).filter(set => conflictFree(fw, set) && fw.arguments.filter(x => !set.includes(x)).every(x => attacksFrom(fw, set, x))).map(sorted).sort((a, b) => canonical(a).localeCompare(canonical(b))); }
function naive(fw) { return maximalSets(subsets(fw.arguments).filter(set => conflictFree(fw, set))); }
function completeExtensions(fw) { return subsets(fw.arguments).filter(set => completeExtension(fw, set)).map(sorted).sort((a, b) => canonical(a).localeCompare(canonical(b))); }
function extensions(fw, semantics) {
  if (semantics === "grounded") return [grounded(fw)];
  if (semantics === "preferred") return preferred(fw);
  if (semantics === "stable") return stable(fw);
  if (semantics === "complete") return completeExtensions(fw);
  if (semantics === "naive") return naive(fw);
  throw new Error(`unknown semantics ${semantics}`);
}
function shortestPath(fw, source, target) {
  if (!fw.arguments.includes(source) || !fw.arguments.includes(target)) return null;
  const queue = [[source]]; const seen = new Set([source]);
  while (queue.length) { const p = queue.shift(); const last = p.at(-1); if (last === target) return p; for (const next of sorted(fw.attacks.filter(([from]) => from === last).map(([, to]) => to))) if (!seen.has(next)) { seen.add(next); queue.push([...p, next]); } }
  return null;
}
function reach(fw, start) { const out = new Set([start]); const q = [start]; while (q.length) { const x = q.shift(); for (const y of fw.attacks.filter(([a]) => a === x).map(([, b]) => b)) if (!out.has(y)) { out.add(y); q.push(y); } } return out; }
function sccs(fw) {
  const remaining = new Set(fw.arguments); const parts = [];
  while (remaining.size) { const first = [...remaining].sort()[0]; const forward = reach(fw, first); const group = [...remaining].filter(x => forward.has(x) && reach(fw, x).has(first)).sort(); parts.push(group); group.forEach(x => remaining.delete(x)); }
  return parts.sort((a, b) => canonical(a).localeCompare(canonical(b)));
}

function evaluate(request) {
  const fw = request.framework;
  switch (request.op) {
    case "argument_framework_structure": {
      const args = fw.arguments; const duplicateArguments = args.filter((x, i) => args.indexOf(x) !== i); const unknown = fw.attacks.flatMap(([a, b], i) => [a, b].filter(x => !args.includes(x)).map(x => ({ attack: i, argument: x })));
      return { arguments: sorted(args), attacks: fw.attacks.map(x => [...x]), duplicate_arguments: sorted(duplicateArguments), unknown_attack_endpoints: unknown, valid: duplicateArguments.length === 0 && unknown.length === 0, real_claims_interpreted: false };
    }
    case "conflict_free_check": { const s = new Set(request.candidate); const internal = fw.attacks.filter(([a, b]) => s.has(a) && s.has(b)); return { candidate: sorted(request.candidate), conflict_free: internal.length === 0, internal_attacks: internal, unknown_arguments: sorted(request.candidate.filter(x => !fw.arguments.includes(x))) }; }
    case "attackers_of_argument": return { argument: request.argument, exists: fw.arguments.includes(request.argument), attackers: fw.arguments.includes(request.argument) ? attackers(fw, request.argument) : [] };
    case "defended_arguments": return { defender_set: sorted(request.defender_set), defended: defended(fw, request.defender_set), external_truth_claim: false };
    case "admissible_extension_check": { const cf = conflictFree(fw, request.candidate); const undefended = request.candidate.filter(x => !defends(fw, request.candidate, x)).sort(); return { candidate: sorted(request.candidate), conflict_free: cf, undefended, admissible: cf && undefended.length === 0 }; }
    case "argument_range_projection": { const attacked = sorted(fw.attacks.filter(([a]) => request.candidate.includes(a)).map(([, b]) => b)); return { candidate: sorted(request.candidate), attacked, range: sorted([...request.candidate, ...attacked]) }; }
    case "complete_extension_check": { const defendedSet = defended(fw, request.candidate); return { candidate: sorted(request.candidate), defended: defendedSet, admissible: admissible(fw, request.candidate), complete: completeExtension(fw, request.candidate) }; }
    case "grounded_extension": return { extension: grounded(fw), semantics: "grounded", extensions_enumerated: 1 };
    case "preferred_extensions": { const rows = preferred(fw); return { extensions: rows, count: rows.length, semantics: "preferred" }; }
    case "stable_extensions": { const rows = stable(fw); return { extensions: rows, count: rows.length, semantics: "stable", existence: rows.length > 0 }; }
    case "naive_extensions": { const rows = naive(fw); return { extensions: rows, count: rows.length, semantics: "naive" }; }
    case "skeptical_acceptance": { const rows = extensions(fw, request.semantics); return { argument: request.argument, semantics: request.semantics, extension_count: rows.length, accepted: rows.length > 0 && rows.every(row => row.includes(request.argument)), vacuous_policy: "false_when_no_extensions" }; }
    case "credulous_acceptance": { const rows = extensions(fw, request.semantics); return { argument: request.argument, semantics: request.semantics, extension_count: rows.length, accepted: rows.some(row => row.includes(request.argument)) }; }
    case "semantics_comparison": { const rows = Object.fromEntries(request.semantics.map(s => [s, extensions(fw, s)])); return { semantics: request.semantics, extensions: rows, counts: Object.fromEntries(request.semantics.map(s => [s, rows[s].length])), convention_is_external_truth: false }; }
    case "attack_path_trace": { const p = shortestPath(fw, request.source, request.target); return { source: request.source, target: request.target, reachable: p !== null, path: p, length: p ? p.length - 1 : null, shortest: p !== null }; }
    case "strong_component_decomposition": { const rows = sccs(fw); return { components: rows, count: rows.length, cyclic_components: rows.filter(row => row.length > 1 || (row.length === 1 && fw.attacks.some(([a, b]) => a === row[0] && b === row[0]))) }; }
    case "accessible_extension_summary": { const rows = extensions(fw, request.semantics); return { semantics: request.semantics, language: request.language, rows: rows.map((row, i) => ({ extension: i + 1, arguments: row, status_text: `${row.length} arguments` })), row_count: rows.length, uses_color_only: false, manual_review: false, assistive_technology_review: false }; }
    case "argument_handover_projection": return { framework_digest: shaObject(fw), arguments: sorted(fw.arguments), attack_count: fw.attacks.length, holds: sorted(request.holds), next_owner_label: request.next_owner_label, external_transfer: false, acknowledged: false, source_preserved: true };
    case "real_deliberation_evidence_gate": { const required = sorted(request.required); return { required, real_rows: request.rows.length, review_state: request.review, missing: required, qualified: false, inference: false, release: false }; }
    case "decision_authority_gate": { const requirements = { publish: ["rights_holder", "affected_people", "privacy_review"], adjudicate: ["competent_decision_authority", "appeal_and_recourse"], production_identity: ["standards_conformant_keys", "interoperability", "trust_governance"], legal_interpretation: ["competent_legal_authority", "affected_people"], maori_wording: ["maori_authority", "affected_people", "data_governance"] }; return { requested: request.requested, target: request.target, required: requirements[request.requested], released: false, assertions_verified: false, external_authority: false }; }

  }
}
