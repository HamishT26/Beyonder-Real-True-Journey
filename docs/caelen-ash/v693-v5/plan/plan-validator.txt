function check(name, condition, detail = null) { checks.push({name,pass:Boolean(condition),detail}); if(!condition) errors.push(`${name}${detail?`: ${detail}`:""}`); }

function normalize(v){if(Array.isArray(v))return v.map(normalize);if(v&&typeof v==="object"){const o={};for(const k of Object.keys(v).sort())o[k]=normalize(v[k]);return o;}return v;}

function walk(dir){const out=[];for(const e of fs.readdirSync(dir,{withFileTypes:true}).sort((a,b)=>a.name.localeCompare(b.name))){const f=path.join(dir,e.name);if(e.isDirectory())out.push(...walk(f));else out.push(f);}return out;}

const cr=[],uuid=[],privatePaths=[],credentials=[],routeIds=[],emails=[];
