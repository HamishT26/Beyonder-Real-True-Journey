'use strict';
const http=require('http'),fs=require('fs'),path=require('path');const file=path.join(__dirname,'evidence-observatory.html');const server=http.createServer((req,res)=>{if(req.method!=='GET'||req.url!=='/'){res.writeHead(404,{'content-type':'text/plain'});res.end('Not found');return;}res.writeHead(200,{'content-type':'text/html; charset=utf-8','cache-control':'no-store','x-content-type-options':'nosniff'});res.end(fs.readFileSync(file));});server.listen(0,'127.0.0.1',()=>console.log(JSON.stringify({port:server.address().port,loopback:true,fixed_file:true})));process.on('SIGINT',()=>server.close(()=>process.exit(0)));
