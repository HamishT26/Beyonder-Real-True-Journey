'use strict';
const fs=require('fs'),path=require('path'),cp=require('child_process'),assert=require('assert/strict'),S=require('../x1/shared.txt');
const phase=path.resolve(__dirname,'..'),[globalRoot,skillRoot,python]=process.argv.slice(2),packageRoot=path.join(globalRoot,'family-evidence-observatory-v2');
const groups=[['evidence-revisions',[0,1,2,3]],['evidence-dependencies',[4,5,6,9]],['evidence-integrity',[7,8,15,16]],['delivery-state',[10,11,17,18]],['delivery-admission',[12,13,14,19]]];
const proposals=JSON.parse(fs.readFileSync(path.join(phase,'plan/new-proposals.json'))).proposals;
if(fs.existsSync(packageRoot))throw Error('global package exists');for(const [name]of groups)if(fs.existsSync(path.join(skillRoot,'ghc-family-'+name+'-v2')))throw Error('global skill collision');
function put(p,b){const f=path.join(phase,'x2/promotions',p);fs.mkdirSync(path.dirname(f),{recursive:true});fs.writeFileSync(f,typeof b==='string'||Buffer.isBuffer(b)?b:JSON.stringify(b,null,2)+'\n',{flag:'wx'});return f;}
const prepared=[];for(const rel of ['x1/shared.txt','x1/engine.txt','x2/engine.txt']){const bytes=fs.readFileSync(path.join(phase,rel));prepared.push({source:put('package/'+rel,bytes),dest:path.join(packageRoot,rel),destination:'global-tools/family-evidence-observatory-v2/'+rel,kind:'support'});}
const validations=[],smokes=[];
for(const [name,indices]of groups){const ops=indices.map(i=>S.OPS[i]),skillName='ghc-family-'+name+'-v2';const sources=ops.map(op=>{const stage=S.OPS.indexOf(op)<10?'x1':'x2',rel=stage+'/skills/ghc-family-'+op.replaceAll('_','-')+'-r2/SKILL.md';return {op,source:rel,sha256:S.hash(fs.readFileSync(path.join(phase,rel))),caller:stage+'/build-stage.txt'};});
 const code=`'use strict';\nconst S=require('../x1/shared.txt'),A=require('../x1/engine.txt'),B=require('../x2/engine.txt');\nconst allowed=${JSON.stringify(ops)};\nfunction run(req){if(!req||!allowed.includes(req.op))return {refused:true,code:'unsupported_operation'};return S.OPS.indexOf(req.op)<10?A.run(req):B.run(req);}\nmodule.exports={run,allowed};if(require.main===module)S.cli(run,allowed);\n`;
 prepared.push({source:put('package/runners/'+name+'.txt',code),dest:path.join(packageRoot,'runners',name+'.txt'),destination:'global-tools/family-evidence-observatory-v2/runners/'+name+'.txt',kind:'runner'});
 const guide=`---\nname: ${skillName}\ndescription: Inspect ${name.replaceAll('-',' ')} using four bounded JSON operations and retained evidence limits.\n---\n\n# ${name.replaceAll('-',' ')}\n\nUse this package for ${ops.join(', ')}. It consolidates four exact operation guides; the original local callers remain compatible. Read references/merge-sources.json for source hashes and phase examples.\n\nRun Node with family-evidence-observatory-v2/runners/${name}.txt beneath the configured D-first global-tools root. Supply one UTF-8 JSON object on standard input with op and input. The interface returns one complete disposition/op/result/boundary envelope, or a refused/code envelope and exit status two. Request bytes are capped at 262,144 and nesting at 32; records, events and projections have separate finite limits. Unknown fields and duplicate JSON keys are refused.\n\nBefore interpreting a result, distinguish a content hash from authentication, a projection from activation, a rejected subject from its passing refusal guard, and accepted delivery from completed work. Unknown or pending delivery requires status reconciliation. This package performs no network, account, credential or task mutation and cannot itself authorize a send.\n\nOn a failed result, retain the source input and exact package version. Make the smallest compatible correction and retain a separate witness. Use the previous compatible caller for rollback; do not erase a predecessor or overwrite this immutable package. The current family workflow and the latest direct user instruction govern real actions.\n\nThis is same-owner synthetic software evidence, not independent reproduction, empirical physics, production identity assurance, professional or public authority, complete accessibility/security, consciousness or Stage 20 readiness.\n`;
 const sourceGuide=put('skills/'+skillName+'/SKILL.md',guide),sourceRefs=put('skills/'+skillName+'/references/merge-sources.json',{owner:'Ceryn Alder',phase:'v696-v6-r2',sources,operations:ops,novelty:'Additive consolidation of current local interfaces; prior capabilities retained.',rollback:'Select previous compatible runner. Do not delete evidence.'});
 prepared.push({source:sourceGuide,dest:path.join(skillRoot,skillName,'SKILL.md'),destination:'skills/'+skillName+'/SKILL.md',kind:'skill'},{source:sourceRefs,dest:path.join(skillRoot,skillName,'references/merge-sources.json'),destination:'skills/'+skillName+'/references/merge-sources.json',kind:'provenance'});
 const result=cp.execFileSync(python,['-X','utf8','-B',path.join(skillRoot,'.system/skill-creator/scripts/quick_validate.py'),path.dirname(sourceGuide)],{encoding:'utf8',env:{...process.env,PYTHONDONTWRITEBYTECODE:'1'},maxBuffer:2e6});validations.push({skill:skillName,location:'prepared',pass:true,output:result.trim()});
}
// Prepared callers can be tested without touching a global installation.
for(const [name,indices]of groups){const runner=require(path.join(phase,'x2/promotions/package/runners',name+'.txt'));for(const i of indices){const p=proposals.find(p=>p.request.op===S.OPS[i]);assert.deepEqual(runner.run(p.request),p.expected);smokes.push({name,operation:p.request.op,location:'prepared',pass:true});}assert.deepEqual(runner.run({op:'outside_group',input:{}}),{refused:true,code:'unsupported_operation'});}
const bindings=[];for(const p of prepared){const b=fs.readFileSync(p.source);fs.mkdirSync(path.dirname(p.dest),{recursive:true});fs.writeFileSync(p.dest,b,{flag:'wx'});assert.equal(S.hash(fs.readFileSync(p.dest)),S.hash(b));bindings.push({source:path.relative(phase,p.source).replaceAll('\\','/'),destination:p.destination,sha256:S.hash(b),bytes:b.length,kind:p.kind,no_overwrite:true});}
for(const [name,indices]of groups){const runnerPath=path.join(packageRoot,'runners',name+'.txt'),runner=require(runnerPath);for(const i of indices){const p=proposals.find(p=>p.request.op===S.OPS[i]);assert.deepEqual(runner.run(p.request),p.expected);smokes.push({name,operation:p.request.op,location:'installed',pass:true});}const p=proposals.find(p=>p.request.op===S.OPS[indices[0]]);const output=JSON.parse(cp.execFileSync(process.execPath,[runnerPath],{input:JSON.stringify(p.request),encoding:'utf8',maxBuffer:2e6}));assert.deepEqual(output,p.expected);smokes.push({name,operation:'CLI',location:'installed',pass:true});const bad=runner.run({op:'outside_group',input:{}});assert.deepEqual(bad,{refused:true,code:'unsupported_operation'});smokes.push({name,operation:'unsupported subject',location:'installed',subject_result:'fail',original_credit:0,refusal_guard_pass:true});
 const skill='ghc-family-'+name+'-v2';const result=cp.execFileSync(python,['-X','utf8','-B',path.join(skillRoot,'.system/skill-creator/scripts/quick_validate.py'),path.join(skillRoot,skill)],{encoding:'utf8',env:{...process.env,PYTHONDONTWRITEBYTECODE:'1'},maxBuffer:2e6});validations.push({skill,location:'installed',pass:true,output:result.trim()});}
fs.writeFileSync(path.join(phase,'x2/global-installation.json'),JSON.stringify({package:'family-evidence-observatory-v2',bindings,validations,smokes,merged_local_guides:20,global_skills:5,global_runners:5,dependency_files:3,old_interfaces_retained:true,third_party_package_installs:0,spend_usd:0,rollback:'Keep the additive installation and select previous compatible callers; deletion is not required.'},null,2)+'\n',{flag:'wx'});console.log(JSON.stringify({bindings:bindings.length,global_skills:5,global_runners:5,prepared_operation_smokes:20,installed_operation_smokes:20,installed_cli_smokes:5,failed_unsupported_subjects:5,passing_refusals:5,skill_validations:10,spend_usd:0}));
