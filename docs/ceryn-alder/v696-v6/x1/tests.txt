'use strict';
const A=require('assert'),E=require('./engine.txt'),R=require('../plan/tools/rational.txt');
function main(){const tests=[];const test=(id,fn)=>{fn();tests.push({id,passed:true});};const f=(a,b=['1'],t='0')=>({a,b,t});const run=(operation,fixture)=>E.run({operation,fixture}).value;
test('zero-degree',()=>A.deepEqual(run('poly_normalize',f(['0','0'])),{coefficients:[],degree:-1}));
test('exact-rational-sum',()=>A.deepEqual(run('poly_add',f(['1/2'],['1/3'])).coefficients,['5/6']));
test('ordered-subtraction',()=>A.deepEqual(run('poly_subtract',f(['1/2'],['1'])).coefficients,['-1/2']));
test('difference-of-squares',()=>A.deepEqual(run('poly_multiply',f(['-1','1'],['1','1'])).coefficients,['-1','0','1']));
test('constant-derivative',()=>A.deepEqual(run('poly_derivative',f(['7/3'])).coefficients,[]));
test('integration-constant',()=>A.deepEqual(run('poly_antiderivative',f(['2','3'],['1'],'-2')).coefficients,['-2','2','3/2']));
test('exact-evaluation',()=>A.equal(run('poly_evaluate',f(['1','2','3'],['1'],'1/2')).value,'11/4'));
test('translation-inverse',()=>{const p=['1','-3','2'],q=run('poly_translate',f(p,['1'],'2/3')).coefficients;A.deepEqual(run('poly_translate',f(q,['1'],'-2/3')).coefficients,p);});
test('zero-dilation-collapse',()=>A.deepEqual(run('poly_dilate',f(['4','2','3'],['1'],'0')).coefficients,['4']));
test('composition-not-commutative',()=>{const a=run('poly_compose',f(['1','1'],['0','0','1'])).coefficients,b=run('poly_compose',f(['0','0','1'],['1','1'])).coefficients;A.notDeepEqual(a,b);});
test('derivative-antiderivative',()=>{const p=['1/2','-3','0','2'],q=run('poly_antiderivative',f(p,['1'],'4')).coefficients;A.deepEqual(run('poly_derivative',f(q)).coefficients,p);});
test('noncanonical-fraction-refused',()=>A.equal(E.run({operation:'poly_add',fixture:f(['2/4'])}).error,'invalid_rational'));
test('negative-zero-refused',()=>A.equal(E.run({operation:'poly_add',fixture:f(['-0'])}).error,'invalid_rational'));
test('degree-budget-refused',()=>A.equal(E.run({operation:'poly_add',fixture:f(Array(10).fill('1'))}).error,'polynomial_bounds'));
test('input-preserved',()=>{const q={operation:'poly_compose',fixture:f(['1','2'],['1','0','1'])},before=JSON.stringify(q);E.run(q);A.equal(JSON.stringify(q),before);});
test('shared-rational-reduction',()=>A.equal(R.out(R.pair(-8n,-12n)),'2/3'));
return{count:tests.length,passed:tests.length,failed:0,tests};}
module.exports={main};if(require.main===module)console.log(JSON.stringify(main()));
