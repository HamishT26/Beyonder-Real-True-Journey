'use strict';
const C=require('../plan/tools/common.txt'),assert=require('assert');
function main(stage,python,skillValidator){assert(['x1','x2'].includes(stage));const rows=C.json('plan/new-proposals.json').records.filter(r=>r.stage===stage),ops=C.json('plan/operations.json').operations.filter(r=>r.stage===stage),engine=require(C.path.join(C.BASE,stage,'engine.txt'));assert.equal(rows.length,150);const gate=C.json(stage+'/start-gate.json');assert.equal(C.git(['rev-parse','HEAD']),stage==='x1'?gate.planning:gate.x1);assert.equal(C.git(['status','--porcelain=v1','--untracked-files=no']),'');
const notes={poly_normalize:'The zero polynomial is the empty vector and has degree -1. Preserve the original vector separately.',poly_add:'Align coefficients by exponent, including missing zero terms.',poly_subtract:'Order matters: return a minus b, with exact rational signs.',poly_multiply:'Convolve exponents. A zero factor yields the empty vector.',poly_derivative:'Constants map to zero; coefficient exponents are formal, not measured rates.',poly_antiderivative:'Use the declared t as integration constant. Differentiation forgets that constant.',poly_evaluate:'Evaluate with exact rationals; no decimal approximation enters the witness.',poly_translate:'The result represents a(x+t). Its inverse uses -t.',poly_dilate:'The result represents a(t*x). t=0 collapses information and has no inverse.',poly_compose:'The result is a(b(x)); composition generally does not commute.',poly_divrem:'Require a nonzero divisor and certify a=b*q+r with deg(r)<deg(b).',poly_gcd:'Return the monic common factor; units and zero conventions are explicit.',poly_square_free:'Remove repeated factors over characteristic zero. Do not infer root locations.',poly_taylor:'The finite polynomial expansion is exact; this says nothing about a non-polynomial physical model.',poly_finite_difference:'Use a(x+t)-a(x), including t=0. The degree drops for nonconstant polynomials.',poly_product_rule:'Compare both complete coefficient vectors, not only one sampled point.',poly_sampled_range:'Finite sample extrema do not bound the polynomial between or outside sample points.',poly_potential_units:'Unit exponents are declared formal bookkeeping, not empirical confirmation of a potential.',poly_observation_vacancy:'Keep missing observations null. Synthetic coefficients cannot fill measurement prerequisites.',poly_consequential_gate:'Represent exact missing authority and execute no consequential action.'};
const capMethods=[],capWitnesses=[],capResults=[];for(let i=0;i<10;i++){const o=ops[i],name='ghc-family-'+o.operation.replaceAll('_','-')+'-v1',rel=stage+'/skills/'+name+'/SKILL.md';C.write(rel,`---\nname: ${name}\ndescription: ${o.title} for bounded canonical rational coefficient vectors. Use when this exact polynomial contract is requested.\n---\n\n# ${o.title}\n\n${notes[o.operation]}\n\nRead the frozen operation request and expected envelope in the phase planning records. Call the matching family runner with an object containing operation and fixture; the fixture has a, b, and t. Coefficients are in ascending exponent order, each a reduced rational string. Input degree is at most eight and integer numerator/denominator magnitudes at most one million. Preserve the input and compare the complete typed envelope. Reject extra fields and invalid fractions.\n\nA refusal witness leaves its malformed subject failed. Retain every earlier result and use additive corrections. ${C.BOUNDARY}\n`);}
for(let g=0;g<5;g++){const supported=ops.slice(g*2,g*2+2).map(x=>x.operation),rel=stage+'/runners/ghc_family_polynomial_'+stage+'_'+(g+1)+'.txt';C.write(rel,`'use strict';\nconst E=require('../engine.txt');const operations=${JSON.stringify(supported)};function run(request){if(!request||!operations.includes(request.operation))return E.reject('unsupported_group_operation');return E.run(request);}if(require.main===module){let result;try{result=run(JSON.parse(require('fs').readFileSync(0,'utf8')));}catch{result=E.reject('invalid_json');}process.stdout.write(JSON.stringify(result)+'\\n');}module.exports={operations,run};\n`);const runner=require(C.path.join(C.BASE,rel)),p=rows.find(x=>x.operation===supported[0]),smoke=runner.run(p.request);assert.deepEqual(smoke,p.expected);const unknown={operation:'outside_this_group',fixture:p.request.fixture},bad=runner.run(unknown);assert.equal(bad.error,'unsupported_group_operation');const mid='CA6966-'+stage.toUpperCase()+'-CAP'+(g+1),negative=mid+'-N',ids=[negative];capWitnesses.push(C.witness(negative,mid,'fail','Unsupported group subject',true,bad.ok,[negative]));let wi=0;const pass=(procedure,expected,observed)=>{const id=mid+'-P'+(++wi);ids.push(id);assert.deepEqual(observed,expected);capWitnesses.push(C.witness(id,mid,'pass',procedure,expected,observed,[negative]));};pass('Positive runner smoke',p.expected,smoke);pass('Unsupported operation refusal','unsupported_group_operation',bad.error);for(let k=0;k<2;k++){const name='ghc-family-'+supported[k].replaceAll('_','-')+'-v1',dir=C.path.join(C.BASE,stage,'skills',name);const result=C.cp.spawnSync(python,['-B','-X','utf8',skillValidator,dir],{encoding:'utf8'});assert.equal(result.status,0,result.stdout+result.stderr);pass('Official skill validator: '+name,0,result.status);}const cli=C.cp.spawnSync(process.execPath,[C.path.join(C.BASE,rel)],{input:JSON.stringify(p.request),encoding:'utf8'});assert.equal(cli.status,0);pass('JSON stdin CLI envelope',p.expected,JSON.parse(cli.stdout));capMethods.push(C.method(mid,'Bounded runner and two operation skill contracts',[negative],ids.filter(x=>x!==negative)));capResults.push({runner:rel,supported,positive_smoke:true,unknown_refusal:true,skill_validations:2,cli:true});}
const contracts=[],safe=[],candidates=[],refusals=[],cfr=[],cleanup=[],chunks=[];
for(const op of ops){const rs=rows.filter(r=>r.operation===op.operation),mid='CA6966-'+stage.toUpperCase()+'-'+op.operation,ws=[],neg=[];for(const p of rs){const before=C.stable(p.request),observed=engine.run(p.request),prefix=stage.toUpperCase()+'-'+p.proposal_id;const equal=C.stable(observed)===C.stable(p.expected),preserved=C.stable(p.request)===before;contracts.push({proposal_id:p.proposal_id,definition_sha256:p.definition_sha256,request_sha256:C.sha(before),expected:p.expected,observed,envelope_equal:equal,input_preserved:preserved,outcome:equal&&preserved?p.expected_disposition:'open_gap'});assert(equal,p.proposal_id+' envelope');assert(preserved,p.proposal_id+' input');for(const [suffix,condition]of [['envelope',equal],['input',preserved]]){const id=prefix+'-SAFE-'+suffix;safe.push({id,proposal_id:p.proposal_id,passed:condition});ws.push(C.witness(id,mid,'pass',suffix,true,condition));}
for(const m of p.malformed){const id=prefix+'-BAD-'+m.id_suffix,initial=C.stable(m.subject),actual=engine.run(m.subject);assert.deepEqual(actual,m.expected,id);candidates.push({id,proposal_id:p.proposal_id,subject:m.subject,observed:actual,result:'fail',original_success_credit:0});neg.push(id);ws.push(C.witness(id,mid,'fail','Malformed subject eligible for acceptance',true,actual.ok,[id]));const rid=id+'-REFUSAL';refusals.push({id:rid,failed_subject:id,expected:m.expected,observed:actual,passed:true});ws.push(C.witness(rid,mid,'pass','Exact refusal envelope',m.expected,actual,[id]));const cid=id+'-CFR';const kept=C.stable(m.subject)===initial&&candidates.at(-1).original_success_credit===0;assert(kept);cfr.push({id:cid,failed_subject:id,repair:m.repair,subject_retained:true,original_success_credit:0,passed:kept});ws.push(C.witness(cid,mid,'pass','Retained subject and repair separation',true,kept,[id]));}
const d={...p};delete d.definition_sha256;for(const [kind,condition]of [['definition',C.sha(C.stable(d))===p.definition_sha256],['protected-gates',p.protected_gates.length===C.GATES.length&&observed.external_credit===false]]){assert(condition);const id=prefix+'-CLEAN-'+kind;cleanup.push({id,proposal_id:p.proposal_id,review:kind,disposition:'keep_current',passed:condition});ws.push(C.witness(id,mid,'pass',kind,true,condition));}}
const m=C.method(mid,op.title,neg,ws.filter(w=>w.result==='pass').map(w=>w.witness_id)),rel=stage+'/method-flow/'+op.operation+'.json';C.write(rel,C.ledger(stage,[m],ws));chunks.push({path:C.PREFIX+rel,...C.ledger(stage,[m],ws).counts,sha256:C.binding(rel).sha256});}
const capRel=stage+'/method-flow/capabilities.json';const capLedger=C.ledger(stage,capMethods,capWitnesses);C.write(capRel,capLedger);chunks.push({path:C.PREFIX+capRel,...capLedger.counts,sha256:C.binding(capRel).sha256});
for(const [name,records]of Object.entries({'contract-results':contracts,'safe-results':safe,'candidate-results':candidates,'refusal-results':refusals,'cfr-results':cfr,'cleanup-results':cleanup,'capability-results':capResults}))C.write(stage+'/'+name+'.json',{stage,count:records.length,records,boundary:C.BOUNDARY});
C.write(stage+'/method-flow-index.json',{schema:'ceryn.chunked-method-flow.v1',stage,chunks,counts:{methods:chunks.reduce((s,x)=>s+x.methods,0),witnesses:chunks.reduce((s,x)=>s+x.witnesses,0),pass:chunks.reduce((s,x)=>s+x.witness_results.pass,0),fail:chunks.reduce((s,x)=>s+x.witness_results.fail,0)},all_failures_retained:true});
C.write(stage+'/phase-truth.json',{stage,core_contracts:150,outcomes:contracts.reduce((a,r)=>(a[r.outcome]=(a[r.outcome]||0)+1,a),{}),safe:300,candidate_failures:300,refusal_passes:300,cfr:300,cleanup:300,skills:10,runners:5,exact_packets_executed:0,blocked_packets_executed:0,source_credit:0,route:'PREPARED_NOT_SENT',canonical_invoked:false,terminal_verdict:'NOT_READY_FOR_STAGE_20'});
C.write(stage+'/overview.md',`# Ceryn ${stage} polynomial evidence\n\nThe stage compares 150 frozen contracts using a separate production implementation. It records 300 safe checks, 300 failed malformed subjects, 300 passing refusal guards, 300 CFR integrity reviews, and 300 cleanup reviews. Ten operation skills and five paired runners passed bounded smokes and official skill validation.\n\n${C.BOUNDARY}\n`);
console.log(JSON.stringify({stage,contracts:150,safe:300,candidates:300,refusals:300,cfr:300,cleanup:300,capabilities:capResults.length,method_flow:C.json(stage+'/method-flow-index.json').counts}));}
