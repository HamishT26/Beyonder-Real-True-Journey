'use strict';
const fs=require('fs'),path=require('path'),crypto=require('crypto'),cp=require('child_process');
const BASE=path.resolve(__dirname,'../..'),ROOT=path.resolve(BASE,'../../..');
const PREFIX='docs/ceryn-alder/v696-v6/',SOURCE='1b2ff9eedddced4144345fae95d2efdddc7bc58a';
const BOUNDARY='Same-owner finite synthetic software only; no empirical, production, authority, identity, independent-reproduction, or Stage 20 credit. Maori concepts remain under Maori authority.';
const GATES=['empirical','production','professional','legal','cultural','Maori-authority','affected-party','privacy-complete','accessibility-complete','exhaustive-security','independent-reproduction','identity','consciousness-personhood','Theory-of-Everything','Stage-20'];
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
const stable=x=>Array.isArray(x)?'['+x.map(stable).join(',')+']':x&&typeof x==='object'?'{'+Object.keys(x).sort().map(k=>JSON.stringify(k)+':'+stable(x[k])).join(',')+'}':JSON.stringify(x);
const json=r=>JSON.parse(fs.readFileSync(path.join(BASE,r),'utf8'));
function write(r,x){const p=path.join(BASE,r);fs.mkdirSync(path.dirname(p),{recursive:true});fs.writeFileSync(p,typeof x==='string'?x.replace(/\r\n/g,'\n'):JSON.stringify(x,null,2)+'\n',{flag:'wx'});}
function files(dir=BASE){return fs.readdirSync(dir,{withFileTypes:true}).flatMap(e=>e.isDirectory()?files(path.join(dir,e.name)):[path.relative(BASE,path.join(dir,e.name)).replaceAll('\\','/')]).sort();}
const git=a=>cp.execFileSync('git',a,{cwd:ROOT,encoding:'utf8',maxBuffer:32*1024*1024}).trim();
const binding=r=>{const b=fs.readFileSync(path.join(BASE,r));return{path:PREFIX+r,bytes:b.length,sha256:sha(b)};};
function manifest(stage){const list=files().filter(x=>x.startsWith(stage+'/'));write(stage+'/manifest.json',{schema:'ceryn.manifest.v1',stage,source:SOURCE,hash_domain:'UTF8 LF working bytes and raw Git blob bytes',entries:list.map(binding),exclusions:[PREFIX+stage+'/manifest.json']});}
function method(id,title,failIds,passIds){return{method_id:id,title,failure_signature:title,trigger_preconditions:['current owner phase','declared bounded contract'],privacy_class:'sanitized_public',approval_class:'safe_now',candidate_workaround:'Use the declared bounded procedure and preserve original failures.',validation_witness_ids:passIds,recurrence_guard:'Require exact typed comparison and retain every failed witness.',rollback:'Stop the affected owner operation; retain inputs and receipts; correct additively.',recommendation_state:'preferred',supersedes:[],protected_gates:GATES,retained_negative_ids:failIds,scope_boundary:BOUNDARY};}
function witness(id,mid,result,procedure,expected,observed,negative=[]){return{witness_id:id,method_id:mid,procedure,scope:'ceryn-owner-only',expected,observed,result,same_owner_only:true,independent_reproduction:false,retained_negative_ids:negative,boundary:BOUNDARY};}
function ledger(stage,methods,witnesses){return{schema:'ghc.family.method-flow-state.v1',phase:'v696-v6',owner:'Ceryn Alder',stage,identity_boundary:BOUNDARY,execution_authority:'owner_self_scoped_delta',methods,witnesses,state_events:methods.map(m=>({method_id:m.method_id,from:'candidate',to:'preferred',note:'Declared passing witness observed; failed subjects remain retained.'})),recommendations:methods.map(m=>({method_id:m.method_id,state:'preferred',recommendation:m.recurrence_guard})),counts:{methods:methods.length,witnesses:witnesses.length,state_events:methods.length,recommendations:methods.length,states:{preferred:methods.length},witness_results:{pass:witnesses.filter(w=>w.result==='pass').length,fail:witnesses.filter(w=>w.result==='fail').length}},boundary:BOUNDARY};}
module.exports={fs,path,cp,BASE,ROOT,PREFIX,SOURCE,BOUNDARY,GATES,sha,stable,json,write,files,git,binding,manifest,method,witness,ledger};
