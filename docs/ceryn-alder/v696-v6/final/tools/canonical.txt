'use strict';
const C=require('../../plan/tools/common.txt'),A=require('assert'),V=require('./validate-final.txt');
function main(receiptRoot,expectedFinal){A(receiptRoot);A(/^[a-f0-9]{40}$/.test(expectedFinal));const root=C.path.resolve(receiptRoot);A(!root.startsWith(C.ROOT+C.path.sep),'receipts must be external');C.fs.mkdirSync(root,{recursive:true});const attempt=C.path.join(root,'canonical-attempt.json'),receipt=C.path.join(root,'canonical-receipt.json'),latch=C.path.join(root,'canonical-success-latch.json');for(const p of [attempt,receipt,latch])A(!C.fs.existsSync(p),'one-shot latch already claimed');A.equal(C.git(['rev-parse','HEAD']),expectedFinal);A.equal(C.git(['status','--porcelain=v1','--untracked-files=all']),'');C.fs.writeFileSync(attempt,JSON.stringify({owner:'Ceryn Alder',phase:'v696-v6',final:expectedFinal,invocation:1,state:'CLAIMED'},null,2)+'\n',{flag:'wx'});let payload;try{payload=V.exact(expectedFinal);}catch(e){payload={status:'INVALID_EXACT_FINAL_OWNER_SCOPED_CANONICAL',owner:'Ceryn Alder',phase:'v696-v6',final:expectedFinal,error:String(e.message),invocation:1,success:0,replay:0};}const b=Buffer.from(JSON.stringify(payload,null,2)+'\n');C.fs.writeFileSync(receipt,b,{flag:'wx'});const digest=C.sha(b),payloadDigest=C.sha(JSON.stringify(payload));if(payload.status==='VALID_EXACT_FINAL_OWNER_SCOPED_CANONICAL')C.fs.writeFileSync(latch,JSON.stringify({owner:'Ceryn Alder',phase:'v696-v6',final:expectedFinal,state:'SUCCESS_LATCHED',invocation:1,success:1,replay:0,receipt_sha256:digest,payload_sha256:payloadDigest},null,2)+'\n',{flag:'wx'});return{status:payload.status,final:expectedFinal,checks_passed:payload.checks_passed,selected_tests:payload.selected_tests?.reduce((s,x)=>s+x.count,0),owner_files:payload.owner_files,manifest_bindings:payload.manifest_bindings,strict_json:payload.strict_json,receipt_sha256:digest,payload_sha256:payloadDigest,invocation:1,success:payload.status.startsWith('VALID_')?1:0,replay:0,error:payload.error};}
module.exports={main};if(require.main===module)console.log(JSON.stringify(main(process.argv[2],process.argv[3])));
