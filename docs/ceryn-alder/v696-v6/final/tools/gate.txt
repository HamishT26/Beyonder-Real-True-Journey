'use strict';
function validate(f){const errors=[];if(!f||typeof f!=='object')return{ok:false,errors:['object']};if(f.clean!==true)errors.push('clean');if(!Array.isArray(f.divergence)||f.divergence.length!==2||f.divergence.some(x=>!Number.isInteger(x)||x!==0))errors.push('divergence');if(f.parent_count!==1||f.commit_count!==4||f.merges!==0)errors.push('lineage');if(!/^[a-f0-9]{40}$/.test(f.final)||!Array.isArray(f.equal_refs)||f.equal_refs.length!==4||f.equal_refs.some(x=>x!==f.final))errors.push('equality');if(f.previous_success!==false||f.previous_attempt!==false)errors.push('latch');if(!Number.isInteger(f.files)||f.files<1||f.files>2000)errors.push('file_budget');if(!Number.isInteger(f.maximum_words)||f.maximum_words>100000)errors.push('document_budget');return{ok:errors.length===0,errors};}
function tests(){const A=require('assert'),f=require('fs'),p=require('path'),fixtures=JSON.parse(f.readFileSync(p.join(__dirname,'../gate-fixtures.json'))),results=fixtures.map(x=>{const r=validate(x.fixture);A.equal(r.ok,x.expected_ok,x.id);return{id:x.id,passed:true,subject_accepted:r.ok,errors:r.errors};});return{count:results.length,passed:results.length,failed:0,tests:results};}
module.exports={validate,tests};
