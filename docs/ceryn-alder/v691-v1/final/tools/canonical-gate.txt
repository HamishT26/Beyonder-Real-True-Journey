const C=require('../../plan/tools/common.txt');
function gate(s){
 return s!==null&&typeof s==='object'&&/^[0-9a-f]{40}$/.test(s.head)&&s.expected_head===s.head&&[s.local,s.upstream,s.tracking,s.fresh_live].every(h=>h===s.head)&&s.clean===true&&s.ahead===0&&s.behind===0&&s.invocations===0&&s.source_replayed===false&&s.sibling_mutation===false&&Array.isArray(s.paths)&&s.paths.length>0&&s.paths.length<2000&&new Set(s.paths).size===s.paths.length&&s.paths.every(p=>p.startsWith('docs/ceryn-alder/v691-v1/')&&!p.split('/').includes('..'))&&C.equal(s.lineage,[[s.head,s.x2],[s.x2,s.x1],[s.x1,s.planning],[s.planning]]);
}
