'use strict';
const fs=require('node:fs'),path=require('node:path'),cp=require('node:child_process');
const C=require('../plan/tools/common.txt');
const [skillBank,python]=process.argv.slice(2),prefix='docs/ceryn-alder/v691-v1',base=path.resolve(prefix),out=path.join(base,'x2/promotions');
const groups=C.read(path.join(base,'plan/tool-promotion-design.json')).groups,defs=C.read(path.join(base,'plan/definitions.json')).proposals;
const globalBase=path.join(out,'family-raster-lab');
for(const p of ['plan/tools/common.txt','x1/engine.txt','x2/engine.txt'])C.put(path.join(globalBase,p),fs.readFileSync(path.join(base,p),'utf8'));
const receipts=[],cards=[];
const card=(id,name,kind,rel,ops,status='current')=>({card_id:id,name,kind,source_path:prefix+'/'+rel,sha256:C.hash(fs.readFileSync(path.join(base,rel))),status,evidence_state:'validated',owner_scope:'Ceryn Alder v691-v1',triggers:[...new Set(ops.flatMap(x=>x.split('_')))].sort(),operations:ops,caller_paths:[prefix+'/x2/promotion-candidate-receipts.json'],rollback:'Select the retained local caller; no deletion or overwrite.',protected_gates:['no_empirical_or_authority_promotion','retain_failed_subjects','no_sibling_mutation'],execution_authority:'owner_self_scoped_delta',repository_scan:false,module_scan:true,cross_lane_scan:false,unchanged_history_scan:false,sibling_lane_mutation:false,supported_endpoint_kinds:['not_applicable'],route_controller_scope:'not_applicable',delivery_ack_required:false});
for(const[gi,g]of groups.entries()){
 const name=g.name,runner=name.replaceAll('-','_')+'.txt',rel='x2/promotions/family-raster-lab/runners/'+runner;
 C.put(path.join(base,rel),`'use strict';\nconst C=require('../plan/tools/common.txt');\nconst all={...require('../x1/engine.txt').operations,...require('../x2/engine.txt').operations};\nconst names=${JSON.stringify(g.operations)};\nconst selected=Object.fromEntries(names.map(n=>[n,all[n]]));\nmodule.exports={run:r=>C.dispatcher(selected,r)};\nif(require.main===module)C.cli(selected);\n`);
 const guide=`---\nname: ${name}\ndescription: Evaluate ${name.replace('ghc-family-raster-','raster ').replaceAll('-',' ')} using four bounded grid contracts, with typed refusals and explicit evidence ceilings.\n---\n\n# ${name.replace('ghc-family-','').replaceAll('-',' ')}\n\nUse this package for ${g.operations.join(', ')}. It merges four source guides while preserving their local callers and evidence. Read [the operation contracts](references/contracts.md) and select the narrowest operation.\n\nResolve family-raster-lab from the installation receipt and run ${runner} with Node, passing one JSON request on standard input. The public runner performs the actual algorithms. The adjacent plan/tools/common.txt and both x1/engine.txt and x2/engine.txt are required dependencies. It writes only a JSON envelope to standard output.\n\nKeep the exact field names and integer/Boolean limits. Input is bounded to one MiB. Duplicate JSON keys, unknown operations, unknown fields, invalid dimensions and implicit missing-support policies are refused. Compare the complete output and input digest; retain every rejected subject at zero acceptance credit. Reuse source evidence without calling it a new independent experiment.\n\nReview the source fixture for the chosen operation. A represented arithmetic model stays represented; a real-evaluation vacancy remains open; publication authority remains gated. Software never supplies competent, affected-party or Maori authority. Retain the earlier caller for additive rollback.\n\n${C.BOUNDARY}\n`;
 const guideRel='x2/promotions/skills/'+name+'/SKILL.md';C.put(path.join(base,guideRel),guide);
 const contracts=g.operations.map(op=>{const p=defs.find(p=>p.operation===op);return`## ${p.title}\n\n${p.hypothesis}\n\nCeiling: ${p.expected_disposition}.\n\nRequest:\n\n\`\`\`json\n${JSON.stringify(p.request,null,2)}\n\`\`\`\n\nExpected full output:\n\n\`\`\`json\n${JSON.stringify(p.expected_output,null,2)}\n\`\`\`\n\nAn added unreviewed_authority field yields an unknown_field envelope with authority false. Preserve the original phase guide ghc-family-${op.replaceAll('_','-')} and its source hash.\n`;}).join('\n');
 C.put(path.join(out,'skills',name,'references/contracts.md'),'# Four retained operation contracts\n\n'+contracts+'\n'+C.BOUNDARY+'\n');
 const vr=cp.spawnSync(python,['-B','-X','utf8',path.join(skillBank,'.system/skill-creator/scripts/quick_validate.py'),path.join(out,'skills',name)],{encoding:'utf8'});C.ensure(vr.status===0,'Candidate skill failed');
 const module=require(path.join(base,rel)),checks=[];
 for(const op of g.operations){const p=defs.find(p=>p.operation===op);const actual=module.run(structuredClone(p.request)),invalid=module.run(structuredClone(p.candidate_subject));checks.push({operation:op,positive:C.equal(actual,p.expected_output),refusal:C.equal(invalid,p.candidate_expected),source_proposal:p.proposal_id});}
 C.ensure(checks.every(x=>x.positive&&x.refusal),'Candidate contract mismatch');
 receipts.push({name,runner,source_guides:g.operations.map(op=>'ghc-family-'+op.replaceAll('_','-')),quick_validate:true,checks,valid:true});
 cards.push(card('CA6911-GSKILL-'+gi,name,'skill',guideRel,g.operations,'candidate'),card('CA6911-GRUNNER-'+gi,runner,'runner',rel,g.operations,'candidate'));
}
for(const[stage,si]of [['x1',0],['x2',1]]){
 const f=C.read(path.join(base,'plan/families.json')).families.filter(f=>f.session===stage);
 for(const[i,r]of f.entries()){const name='ghc-family-'+r.operation.replaceAll('_','-');cards.push(card('CA6911-LOCAL-SKILL-'+(si*10+i),name,'skill',stage+'/skills/'+name+'/SKILL.md',[r.operation]));}
 for(let i=0;i<5;i++){const name='ghc_family_'+stage+'_grid_pair_'+(i+1)+'.txt';cards.push(card('CA6911-LOCAL-RUNNER-'+(si*5+i),name,'runner',stage+'/runners/'+name,f.slice(i*2,i*2+2).map(x=>x.operation)));}
}
C.put(path.join(base,'x2/promotion-candidate-receipts.json'),{valid:true,groups:receipts,installed:false,boundary:C.BOUNDARY});
C.put(path.join(base,'x2/meta-tool-catalogue.json'),{schema:'ghc.family.meta-tool-box.catalogue.v2',owner:'Ceryn Alder',phase:'v691-v1',cards,boundary:C.BOUNDARY});
const script=path.join(skillBank,'ghc-family-meta-tool-box/scripts/ghc_family_meta_tool_box.py'),cat=path.join(base,'x2/meta-tool-catalogue.json');
for(const mode of ['validate','collisions']){const p=path.join(base,'x2/catalogue-'+mode+'.json'),r=cp.spawnSync(python,['-B','-X','utf8',script,mode,'--catalogue',cat,'--output',p],{encoding:'utf8'});C.ensure(r.status===0,'Catalogue '+mode+' failed');}
for(const c of cards.filter(c=>c.status==='candidate')){const p=path.join(base,'x2/promotion-readiness',c.card_id+'.json'),r=cp.spawnSync(python,['-B','-X','utf8',script,'promotion','--catalogue',cat,'--card-id',c.card_id,'--output',p],{encoding:'utf8'});C.ensure(r.status===0&&C.read(p).state==='ready','Promotion readiness failed');}
const collisions=C.read(path.join(base,'x2/catalogue-collisions.json'));
const summary=collisions.findings.map(f=>{const a=cards.find(c=>c.card_id===f.left),b=cards.find(c=>c.card_id===f.right);return{...f,left_name:a.name,right_name:b.name,shared_operations:a.operations.filter(x=>b.operations.includes(x))};});
C.put(path.join(base,'x2/collision-review-candidate.json'),{findings:summary,selection_performed:false,review_required:true,boundary:C.BOUNDARY});
console.log(JSON.stringify({candidate_groups:5,cards:cards.length,catalogue_valid:C.read(path.join(base,'x2/catalogue-validate.json')).valid,collision_count:summary.length,shared_contract_pairs:summary.filter(r=>r.shared_operations.length).length,disjoint_contract_pairs:summary.filter(r=>!r.shared_operations.length),installed:false}));
