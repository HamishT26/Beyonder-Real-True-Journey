const path=require('node:path'),C=require('../plan/tools/common.txt'),{run}=require('./engine.txt');
const plan=C.read(path.join(__dirname,'supplement-plan.json'));
const rows=plan.cases.map(p=>{const observed=p.requests.map(r=>run(structuredClone(r)));let passes;
 if(p.id==='CA6911-S001')passes=C.equal(observed[0].value.interval,p.expected_enclosure)&&!C.equal(p.expected_enclosure,p.expected_coupled_range);
 if(p.id==='CA6911-S002')passes=C.equal(observed[0],observed[1])&&!C.equal(p.requests[0],p.requests[1]);

 return{...p,observed,claim_result:false,guard_passed:passes,original_claim_success_credit:0,source_discovery_credit:0};});

C.ensure(rows.every(r=>r.guard_passed),'Supplement mismatch');console.log(JSON.stringify({valid:true,claims_retained_false:3,passing_refuters:3,core_credit:0}));
