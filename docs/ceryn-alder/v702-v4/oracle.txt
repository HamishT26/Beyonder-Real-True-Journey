'use strict';
const Q=require('./q.txt');const {q,ca,cs,cm,cj,scale,zero,ipow,vec,out,energy,str}=Q;
const ops=['vector_shape','sum','difference','hadamard','conjugate','reverse','shift','dft','inverse','convolution','correlation','parseval','convolution_theorem','shift_theorem','modulation','circulant_action','filter_projection','accessibility_reservation','observation_vacancy','authority_gate'];
const dft=(v,sign=-1)=>v.map((_,k)=>v.reduce((s,z,j)=>ca(s,cm(z,ipow(sign*k*j))),zero()));
const inverse=v=>dft(v,1).map(z=>scale(z,q(1,4)));
const convolution=(a,b)=>a.map((_,n)=>a.reduce((s,z,j)=>ca(s,cm(z,b[(n-j+4)%4])),zero()));
const shift=(a,k)=>a.map((_,n)=>a[(n-k+4)%4]);
function reference(r){const a=vec(r.a),b=vec(r.b),A=dft(a),B=dft(b),k=r.shift;let value;switch(r.operation){
case'vector_shape':value={length:4,field:'Q(i)',a:out(a),b:out(b)};break;
case'sum':value=out(a.map((z,j)=>ca(z,b[j])));break;
case'difference':value=out(a.map((z,j)=>cs(z,b[j])));break;
case'hadamard':value=out(a.map((z,j)=>cm(z,b[j])));break;
case'conjugate':value=out(a.map(cj));break;
case'reverse':value=out(a.map((_,j)=>a[(4-j)%4]));break;
case'shift':value=out(shift(a,k));break;
case'dft':value=out(A);break;
case'inverse':value=out(inverse(a));break;
case'convolution':value=out(convolution(a,b));break;
case'correlation':value=out(a.map((_,l)=>a.reduce((s,z,j)=>ca(s,cm(cj(z),b[(j+l)%4])),zero())));break;
case'parseval':value={time_energy:str(energy(a)),frequency_energy:str(energy(A)),scale:'4',equal:str(energy(A))===str(Q.mul(q(4),energy(a)))};break;
case'convolution_theorem':value={direct_spectrum:out(dft(convolution(a,b))),spectral_product:out(A.map((z,j)=>cm(z,B[j])))};break;
case'shift_theorem':value={shifted_spectrum:out(dft(shift(a,k))),phase_product:out(A.map((z,j)=>cm(z,ipow(-j*k))))};break;
case'modulation':value={modulated:out(a.map((z,j)=>cm(z,ipow(j*k)))),spectrum:out(A.map((_,j)=>A[(j-k+4)%4]))};break;
case'circulant_action':{const matrix=a.map((_,i)=>a.map((_,j)=>a[(i-j+4)%4]));value={matrix:matrix.map(out),action:out(matrix.map(row=>row.reduce((s,z,j)=>ca(s,cm(z,b[j])),zero())))};break;}
case'filter_projection':{const p=inverse(A.map((z,j)=>r.mask[j]?z:zero()));value={projected:out(p),before_energy:str(energy(a)),after_energy:str(energy(p)),mask:r.mask};break;}
case'accessibility_reservation':value={vector:out(a),text_coordinates_available:true,screen_reader_review:'not_performed',participant_evidence:false,complete_accessibility:false};break;
case'observation_vacancy':value={synthetic_spectrum:out(A),measured:false,calibration:null,physical_likelihood:null};break;
case'authority_gate':value={operation:'withheld',competent_authority_evidence:false,production_identity:false,public_allocation:false};break;
default:throw Error('UNKNOWN_OPERATION');}
return{operation:r.operation,value,evidence_scope:'finite exact synthetic software witness',broader_authority_credit:0};}
module.exports={ops,reference,dft,inverse,convolution};
