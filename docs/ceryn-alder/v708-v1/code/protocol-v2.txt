'use strict';
const P=require('./protocol.txt'),{hash,canonical}=require('./common.txt');
function currentEvidence(s){return s&&Number.isSafeInteger(s.epoch)&&s.epoch>=0&&Number.isSafeInteger(s.now)&&s.now>=0&&typeof s.granted==='boolean'&&s.tickets instanceof Map&&s.receipts instanceof Map;}
function apply(s,r,t,checkEpoch){let decision,receipt=null;const rd=P.validRequest(r)?hash(r):null;if(!currentEvidence(s))return{decision:'MALFORMED_EVIDENCE',reason:'MISSING_OR_INVALID_CURRENT_STATE',effects:s?.effects?.length||0,new_permission_granted:false};if(!P.validRequest(r))decision='MALFORMED_REQUEST';else if(!t)decision='NO_TICKET';else if(!P.validTicket(t))decision='MALFORMED_EVIDENCE';else if(s.receipts.has(t.nonce)){const prior=s.receipts.get(t.nonce);if(prior.request_digest===rd&&prior.ticket_binding_sha256===hash(t)){decision='DUPLICATE_HISTORY';receipt=structuredClone(prior);}else decision='NONCE_CONFLICT';}else if(!s.tickets.has(t.nonce)||canonical(t)!==canonical(s.tickets.get(t.nonce)))decision='TICKET_MISMATCH';else if(t.request_digest!==rd)decision='REQUEST_BINDING_MISMATCH';else if(checkEpoch&&t.epoch!==s.epoch)decision='STALE_EPOCH';else if(t.scope!==s.scope||t.context!==s.context||t.evidence_digest!==s.evidence_digest)decision='CURRENT_BINDING_MISMATCH';else if(!s.granted)decision='REVOKED';else if(s.now>=t.expires)decision='EXPIRED';else{decision='COMMITTED';receipt={receipt_id:'synthetic-receipt-'+(s.effects.length+1),sequence:s.effects.length+1,nonce:t.nonce,request_digest:rd,ticket_binding_sha256:hash(t),created_at:s.now,epoch:s.epoch,expires:t.expires,kind:'synthetic_compute_only'};s.effects.push(structuredClone(receipt));s.receipts.set(t.nonce,structuredClone(receipt));}if(!['COMMITTED','DUPLICATE_HISTORY'].includes(decision))s.failures.push({decision,nonce:t?.nonce||null,original_success_credit:0});return{decision,receipt,effects:s.effects.length,historical_only:decision==='DUPLICATE_HISTORY',new_permission_granted:false};}
const commit=(s,r,t)=>apply(s,r,t,true);
// Deliberately faulty control for the frozen laboratory only; never a live admission path.
const mutantCommitIgnoringEpoch=(s,r,t)=>apply(s,r,t,false);
module.exports={initial:P.initial,request:P.request,prepare:P.prepare,validRequest:P.validRequest,validTicket:P.validTicket,currentEvidence,commit,mutantCommitIgnoringEpoch};
