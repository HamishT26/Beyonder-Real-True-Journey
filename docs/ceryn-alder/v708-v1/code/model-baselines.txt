'use strict';
const near=(a,b,t=1e-9)=>Math.abs(a-b)<=t,sum=a=>a.reduce((s,x)=>s+x,0);
function baseline(r){const f=r.frames,m=r.metrics,q=r.request,n=q.steps,v=q.variant;const all=f.flat(),nonneg=all.every(p=>p.value>=-1e-12);switch(q.model){case'heat':{const a=f[0].map(p=>p.value),lo=Math.min(...a),hi=Math.max(...a);return f.every(a=>near(sum(a.map(p=>p.value)),m.initial_mass))&&all.every(p=>p.value>=lo-1e-12&&p.value<=hi+1e-12);}case'wave':return f.every(a=>near(sum(a.map(p=>p.value))/a.length,0))&&all.every(p=>Math.abs(p.value)<1.2);case'oscillator':return all.every(p=>p.value>=0&&p.value<=0.53);case'reaction':return all.every(p=>p.value>=-1e-12&&p.value<=1+1e-12);case'entropy':return nonneg&&f.every(a=>near(sum(a.map(p=>p.value)),1))&&m.entropy_trace.every((e,i,a)=>i===0||e>=a[i-1]-1e-12);case'queue':return m.total_arrivals-m.total_served===m.backlog&&all.every(p=>Number.isSafeInteger(p.value)&&p.value>=0);case'replication':return f.every((a,i)=>i===0||a.every((p,j)=>p.x>=f[i-1][j].x&&p.y>=f[i-1][j].y));case'retry':return near(m.expected_attempts,(1-(1-m.success_probability)**(n+1))/m.success_probability)&&m.duplicate_after_ack===false;case'graph':return nonneg&&f.every(a=>near(sum(a.map(p=>p.value)),1));case'coding':return all.every(p=>p.value===((p.y.toString(2).split('1').length-1)%2));case'consent':return all.every(p=>p.value===(p.t<m.expires&&p.t<m.withdraw_at?1:0));case'allocation':return f.every((a,t)=>a.every(p=>p.value>=-1e-9&&p.value<=p.y+1e-9)&&near(sum(a.map(p=>p.value)),t/n*sum(m.demand)));case'voting':return v===0?m.cycle===true&&m.condorcet_winner===null:m.cycle===false&&m.condorcet_winner===0;case'bayes':{const prior=v?2:1,beta=prior+Math.floor(n/3),alpha=prior+n-Math.floor(n/3);return m.alpha===alpha&&m.beta===beta&&near(m.mean,alpha/(alpha+beta));}case'remedy':return m.original_preserved&&m.original_success_credit===0&&m.records[0].kind==='failure'&&m.records.every((x,i)=>x.original_success_credit===0&&(i===0?x.parent===null:x.parent===m.records[i-1].id));default:throw Error('UNKNOWN_BASELINE');}}
module.exports={baseline};
