const C=require('./common.txt');
function audit(){
 const checks=[],check=(name,ok,detail)=>{checks.push({name,passed:!!ok,detail});if(!ok)throw Error('CANONICAL_CHECK '+name+' '+JSON.stringify(detail));};

 let bindings=0;
 for(let i=0;i<receipts.length;i++){const r=receipts[i];check(stages[i]+' receipt binding',r.stage===stages[i]&&r.parent===(i?receipts[i-1].head:C.BASE)&&r.source_content===C.SOURCE&&r.source_is_git_parent===false&&r.push_exit===0,r.head);const m=JSON.parse(C.cp.execFileSync('git',['show',r.head+':'+r.manifest],{cwd:C.REPO}));let at=0;const raw=C.cp.execFileSync('git',['cat-file','--batch'],{cwd:C.REPO,input:m.entries.map(e=>r.head+':'+e.path).join('\n')+'\n',maxBuffer:128*1024*1024});for(const e of m.entries){const end=raw.indexOf(10,at),header=raw.subarray(at,end).toString().split(' '),n=Number(header[2]);at=end+1;const b=raw.subarray(at,at+n);at+=n;if(header[1]!=='blob'||raw[at++]!==10||b.length!==e.bytes||C.hash(b)!==e.sha256)throw Error('MANIFEST_BYTE_BINDING '+e.path);bindings++;}check(stages[i]+' original-domain manifest',m.entries.length===r.entries&&m.parent===r.parent&&m.source_content===C.SOURCE,m.entries.length);}
 const ownerPaths=C.git(['ls-tree','-r','--name-only',last.head,'--',C.PREFIX]).split('\n'),ownerBytes=new Map(),allRaw=C.cp.execFileSync('git',['cat-file','--batch'],{cwd:C.REPO,input:ownerPaths.map(p=>last.head+':'+p).join('\n')+'\n',maxBuffer:128*1024*1024});let offset=0;for(const p of ownerPaths){const end=allRaw.indexOf(10,offset),h=allRaw.subarray(offset,end).toString().split(' '),n=Number(h[2]);offset=end+1;const b=allRaw.subarray(offset,offset+n);offset+=n;if(h[1]!=='blob'||allRaw[offset++]!==10)throw Error('OWNER_BLOB');ownerBytes.set(p,b);}let json=0;for(const p of ownerPaths){const b=ownerBytes.get(p);if(p.endsWith('.json')){JSON.parse(b);json++;}if(!/\.(json|md|txt)$/.test(p)||b.toString().trim().split(/\s+/).length>100000)throw Error('FORMAT_WORD_BOUND');}
 check('file and format budget',ownerPaths.length===last.owner_files&&C.git(['ls-files']).split('\n').length<=2000,{owner:ownerPaths.length,json,total:C.git(['ls-files']).split('\n').length});
 const blob=p=>{const b=ownerBytes.get(C.PREFIX+p);if(!b)throw Error('MISSING_BOUND_BLOB '+p);return b;},read=p=>JSON.parse(blob(p)),seal=read('final/content-seal.json');check('content seal',seal.targets.every(t=>{const b=blob(t.path);return b.length===t.bytes&&C.hash(b)===t.sha256;}),seal.targets.length);
 const contract=read('final/canonical-contract.json');check('literal canonical contract',contract.required_literals.every(x=>x.tokens.every(t=>blob(x.path).toString().includes(t))),contract.required_literals);
 const baton=blob('final/handoff-baton.md').toString(),words=baton.trim().split(/\s+/).length;check('baton through literal EOF',words>=2000&&words<=100000&&baton.trimEnd().endsWith('LITERAL_EOF_CERYN_V708_V1_HANDOFF'),words);
 const assertions=['x1/test-inventory.json','x2/test-inventory.json'].flatMap(p=>read(p).tests);check('saved assertion inventory',assertions.length===158&&new Set(assertions.map(t=>t.id)).size===158&&assertions.every(t=>t.passed===true),assertions.length);
 const methodPaths=ownerPaths.filter(p=>p.endsWith('.json')).filter(p=>JSON.parse(ownerBytes.get(p)).schema==='ghc.family.method-flow-state.v1'),ledgers=methodPaths.map(p=>JSON.parse(ownerBytes.get(p))),own=ledgers.reduce((a,l)=>({methods:a.methods+l.counts.methods,witnesses:a.witnesses+l.witnesses.length,passing:a.passing+l.witnesses.filter(w=>w.result==='pass').length,failed:a.failed+l.witnesses.filter(w=>w.result==='fail').length}),{methods:0,witnesses:0,passing:0,failed:0});
 const negatives=[...new Set(ledgers.flatMap(l=>l.witnesses.filter(w=>w.result==='fail').flatMap(w=>w.retained_negative_ids)))];check('method ledger totals',own.methods===17&&own.witnesses===307&&own.passing===211&&own.failed===96&&negatives.length===96,{...own,effective_negatives:negatives.length});
 const validations=read('x2/ledger-validations.json').validations;check('saved global ledger validation',validations.length===methodPaths.length&&validations.every(v=>v.valid&&v.issue_count===0&&C.hash(blob(v.path))===v.sha256),validations.length);
 const accounting=read('final/source-faithful-ledger.json');check('single baseline fold',accounting.source_fold_count===1&&accounting.external_overlay_counts_added===0&&Object.keys(accounting.baseline).every(k=>accounting.combined[k]===accounting.baseline[k]+accounting.own_delta[k])&&Object.keys(own).every(k=>accounting.own_delta[k]===own[k]),accounting.combined);
 const truth=read('final/phase-truth.json'),outcomes=truth.projects.reduce((a,p)=>(a[p.disposition]++,a),{completed:0,represented:0,open_gap:0,exact_gate:0});check('core outcomes',truth.projects.length===20&&C.canonical(outcomes)===C.canonical(contract.expected_core_outcomes),outcomes);
 const packets=read('final/gate-register.json');check('held real-world gates',packets.exact_obligations===11&&packets.exact_packets.length===5&&packets.blocked_packets.length===5&&[...packets.exact_packets,...packets.blocked_packets].every(p=>p.executed===false)&&packets.verdict==='NOT_READY_FOR_STAGE_20',11);
 check('model sessions and extra pair',read('x1/model-summary.json').models===15&&read('x2/model-summary.json').models===15&&read('x2/equal-output-models.json').models.length===2&&truth.models.total===32,32);
 const advice1=read('x1/advisory-disposition.json'),advice2=read('x2/advisory-disposition.json');check('adviser and hook limits',advice1.send_invocations===1&&advice1.native_acknowledged===true&&advice1.reply_received===true&&advice1.resends===0&&advice2.sends===1&&advice2.reply_state==='RECEIVED'&&advice2.resends===0&&['x1','x2'].every(s=>read(s+'/hook-smokes.json').distinct_hooks===5&&read(s+'/hook-smokes.json').host_observed_events===0),{adviser_sends:2,manual_hooks:20,host_events:0});
 const research=read('x2/research-dispositions.json');check('research boundaries',research.law_hypotheses.length===15&&research.open_problem_probes.length===15&&research.open_problem_probes.every(p=>p.solved===false&&p.missing_dependency),{hypotheses:15,open_probes:15});
 const syntax=read('final/static-validation.json').syntax,allow=read('final/executable-allowlist.json');check('explicit executable inventory',syntax.length===allow.syntax_only_paths.length+allow.retained_failed_helpers.length&&syntax.every(s=>s.exit===0&&C.hash(blob(s.path))===s.sha256)&&!allow.syntax_only_paths.some(p=>!p.startsWith('code/')),syntax.length);
 const source=read('planning/source-readback.json');check('source provenance without execution',source.source===C.SOURCE&&source.owner_main_base===C.BASE&&source.source_code_executions===0&&source.source_gate==='PASS_BEFORE_MUTATION'&&source.batons.every(b=>b.read_through_eof)&&truth.source_is_git_parent===false,source.source);
 const sourceReceipt=C.fs.readFileSync(C.BANK.replace('ceryn-alder-v708-v1','orin-thale-v707-v8')+'/corrective-canonical-receipt.json');check('source corrected receipt raw binding',C.hash(sourceReceipt)===source.canonical_receipt_sha256,source.canonical_receipt_sha256);
 const plan=read('planning/plan.json');check('live v20 control bindings',plan.controls.every(c=>C.hash(C.fs.readFileSync('C:/Users/hamis/.codex/skills/'+c.path))===c.sha256),plan.controls.map(c=>c.name));
 check('failure and continuation preservation',read('x2/failed-fixture-attempt.json').original_success_credit===0&&read('x2/failed-fixture-attempt.json').saved_domain_results_before_failure===0&&C.hash(blob('code/x2-fixtures.txt'))===read('x2/failed-fixture-attempt.json').helper_sha256&&read('x2/continuation-plan.json').continuation_id==='CA7081-X2-R1'&&truth.successful_session_replays===0,{failed_attempt:1,named_continuation:1});
 check('no premature route or recipient credit',truth.route_state==='HELD_UNTIL_TERMINAL'&&truth.recipient_completion==='NOT_CLAIMED'&&truth.shared_mutation===false&&truth.sibling_mutation===false&&truth.spend_usd===0,truth.verdict);
 return{schema:'ceryn.exact-owner-canonical.v1',head:last.head,source_content:C.SOURCE,owner_base:C.BASE,source_is_git_parent:false,checks,check_count:checks.length,passed:checks.filter(c=>c.passed).length,lifecycle_bindings:bindings,content_seal_bindings:seal.targets.length,owner_files:ownerPaths.length,json_files:json,stored_assertions:assertions.length,own,combined:accounting.combined,baton_words:words,equality,source_executions:0,domain_executions:0,domain_replays:0,verdict:'VALID_EXACT_FINAL_OWNER_SCOPED_CANONICAL',boundary:C.boundary};
}

module.exports={audit};if(require.main===module)main();
