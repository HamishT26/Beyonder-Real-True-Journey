'use strict';
const {hash,canonical}=require('./common.txt');
function keys(x,expected){return x&&Object.getPrototypeOf(x)===Object.prototype&&Object.keys(x).sort().join('|')===[...expected].sort().join('|');}
function validRequest(r){return keys(r,['id','body','scope','context','evidence_digest'])&&['id','body','scope','context'].every(k=>typeof r[k]==='string'&&r[k].length>0&&r[k].length<=128)&&/^[a-f0-9]{64}$/.test(r.evidence_digest);}
function validTicket(t){return keys(t,['nonce','epoch','request_digest','evidence_digest','scope','context','expires'])&&/^[a-z0-9-]{1,40}$/.test(t.nonce)&&[t.epoch,t.expires].every(x=>Number.isSafeInteger(x)&&x>=0&&x<=1000000)&&[t.request_digest,t.evidence_digest].every(x=>typeof x==='string'&&/^[a-f0-9]{64}$/.test(x))&&[t.scope,t.context].every(x=>typeof x==='string'&&x.length>0&&x.length<=128);}
function initial(){const evidence_digest=hash('synthetic evidence A');return{epoch:0,now:0,granted:true,expires:5,scope:'synthetic_compute',context:'context-A',evidence_digest,tickets:new Map(),receipts:new Map(),effects:[],failures:[],corrections:[]};}
function request(s){return{id:'request-A',body:'finite-compute',scope:s.scope,context:s.context,evidence_digest:s.evidence_digest};}
function prepare(s,r,nonce){if(!validRequest(r))return{decision:'MALFORMED_REQUEST',ticket:null};if(!s.granted||s.now>=s.expires||r.scope!==s.scope||r.context!==s.context||r.evidence_digest!==s.evidence_digest)return{decision:'PREPARE_DENIED',ticket:null};if(s.tickets.has(nonce))return{decision:'NONCE_ALREADY_PREPARED',ticket:null};const t={nonce,epoch:s.epoch,request_digest:hash(r),evidence_digest:s.evidence_digest,scope:s.scope,context:s.context,expires:s.expires};s.tickets.set(nonce,structuredClone(t));return{decision:'PREPARED',ticket:structuredClone(t)};}
function commit(s,r,t){let decision;if(!validRequest(r))decision='MALFORMED_REQUEST';else if(!t)decision='NO_TICKET';else if(!validTicket(t))decision='MALFORMED_EVIDENCE';else if(!s.tickets.has(t.nonce)||canonical(t)!==canonical(s.tickets.get(t.nonce)))decision='TICKET_MISMATCH';else if(s.receipts.has(t.nonce))decision=s.receipts.get(t.nonce).request_digest===hash(r)?'DUPLICATE_HISTORY':'NONCE_CONFLICT';else if(t.request_digest!==hash(r))decision='REQUEST_BINDING_MISMATCH';else if(t.epoch!==s.epoch)decision='STALE_EPOCH';else if(t.scope!==s.scope||t.context!==s.context||t.evidence_digest!==s.evidence_digest)decision='CURRENT_BINDING_MISMATCH';else if(!s.granted)decision='REVOKED';else if(s.now>=t.expires)decision='EXPIRED';else{decision='COMMITTED';const e={sequence:s.effects.length+1,nonce:t.nonce,request_digest:hash(r),epoch:s.epoch,kind:'synthetic_compute_only'};s.effects.push(e);s.receipts.set(t.nonce,structuredClone(e));}if(!['COMMITTED','DUPLICATE_HISTORY'].includes(decision))s.failures.push({decision,nonce:t?.nonce||null,request_digest:validRequest(r)?hash(r):null,original_success_credit:0});return{decision,effects:s.effects.length,historical_only:decision==='DUPLICATE_HISTORY',new_permission_granted:false};}
function scenario(def){if(!Array.isArray(def.events)||def.events.length>16)throw Error('EVENT_BOUND');const s=initial();let r=request(s),ticket=null,last=null;const trace=[];for(const event of def.events){let result=null;switch(event){case'prepare':result=prepare(s,r,'nonce-original');ticket=result.ticket;break;case'prepare_new':result=prepare(s,r,'nonce-refreshed');ticket=result.ticket;break;case'commit':case'commit_new':result=commit(s,r,ticket);last=result.decision;break;case'revoke':s.granted=false;s.epoch++;break;case'grant':s.granted=true;s.epoch++;break;case'replace':s.evidence_digest=hash('synthetic evidence B');s.epoch++;break;case'change_scope':s.scope='synthetic_archive';s.epoch++;break;case'change_context':r={...r,context:'context-B'};break;case'alter_payload':r={...r,body:'changed-finite-compute'};break;case'corrupt_ticket':ticket={...ticket,expires:6};break;case'expire':s.now=5;break;case'near_expiry':s.now=4;break;case'record_correction':s.corrections.push({failed_index:0,original_success_credit:0,proposed_next_step:'Separate eligibility refresh before any new ticket.'});break;default:throw Error('UNKNOWN_EVENT');}trace.push({event,epoch:s.epoch,now:s.now,granted:s.granted,effects:s.effects.length,decision:result?.decision||null});}return{id:def.id,name:def.name,events:def.events,trace,last_decision:last,effects:s.effects,retained_failures:s.failures,corrections:s.corrections,external_actions:0,real_permission_grants:0};}
function booleanMutant(){const s=initial(),r=request(s),t=prepare(s,r,'nonce-original').ticket;s.granted=false;s.epoch++;s.granted=true;s.epoch++;const accepted=s.granted&&s.now<t.expires&&r.evidence_digest===s.evidence_digest;return{accepted,epoch_prepared:t.epoch,epoch_current:s.epoch,claimed_effects:accepted?1:0,bug:'Ignores the policy epoch after revoke/regrant.',original_success_credit:0};}
module.exports={keys,validRequest,validTicket,initial,request,prepare,commit,scenario,booleanMutant};
