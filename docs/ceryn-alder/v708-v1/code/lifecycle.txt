'use strict';
const C=require('./common.txt');
function seal(stage){C.admitted();const {fs,path,hash,git,walk,write}=C,stages=['planning','x1','x2','final'],i=stages.indexOf(stage);if(i<0)throw Error('STAGE');const receipt=C.BANK+'/'+stage+'-receipt.json';if(fs.existsSync(receipt))throw Error('SUCCESSFUL_STAGE_ALREADY_SEALED');const parent=i?JSON.parse(fs.readFileSync(C.BANK+'/'+stages[i-1]+'-receipt.json')).head:C.BASE;if(git(['rev-parse','HEAD'])!==parent)throw Error('STAGE_PARENT');const manifest=stage==='final'?'final/owner-manifest.json':stage+'/manifest.json';if(fs.existsSync(C.ROOT+'/'+manifest))throw Error('MANIFEST_EXISTS_INSPECT_BEFORE_RETRY');const entries=walk().map(p=>{const b=fs.readFileSync(p),relative=path.relative(C.REPO,p).replaceAll('\\','/');if(!/\.(json|md|txt|html)$/.test(relative))throw Error('FORMAT');if(b.toString().trim().split(/\s+/).length>100000)throw Error('WORD_BUDGET '+relative);if(p.endsWith('.json'))JSON.parse(b);return{path:relative,bytes:b.length,sha256:hash(b)};});write(manifest,{schema:'ceryn.lifecycle-manifest.v1',stage,parent,source_content:C.SOURCE,byte_domain:'raw Git blob; stage with core.autocrlf=false',self_excluded:C.PREFIX+manifest,entries});git(['-c','core.autocrlf=false','add','--sparse','--',C.PREFIX]);const changed=git(['diff','--cached','--name-only']).split('\n');if(changed.some(p=>!p.startsWith(C.PREFIX)))throw Error('OWNER_SCOPE');const tracked=git(['ls-files']).split('\n').length;if(tracked>2000)throw Error('FILE_CEILING');git(['commit','-m','Ceryn Alder v708-v1 '+stage+' evidence admission']);const head=git(['rev-parse','HEAD']);let at=0;const raw=C.cp.execFileSync('git',['cat-file','--batch'],{cwd:C.REPO,input:entries.map(e=>head+':'+e.path).join('\n')+'\n',maxBuffer:64*1024*1024});for(const e of entries){const end=raw.indexOf(10,at),h=raw.subarray(at,end).toString().split(' '),n=Number(h[2]);if(h[1]!=='blob')throw Error('BLOB');at=end+1;const b=raw.subarray(at,at+n);at+=n;if(raw[at++]!==10||n!==e.bytes||hash(b)!==e.sha256)throw Error('BYTE_BINDING');}git(['push','origin',C.BRANCH]);const equality=C.gate(),r={schema:'ceryn.stage-receipt.v1',stage,head,parent,source_content:C.SOURCE,push_exit:0,manifest:C.PREFIX+manifest,entries:entries.length,owner_files:walk().length,tracked_files:tracked,equality,source_is_git_parent:false,created_utc:new Date().toISOString()};fs.writeFileSync(receipt,JSON.stringify(r,null,2)+'\n',{flag:'wx'});console.log(JSON.stringify(r));}
module.exports={seal};if(require.main===module)seal(process.argv[2]);
