function northWest(supplyInput, demandInput) {
  const supply = supplyInput.slice();
  const demand = demandInput.slice();
  const table = [[0, 0, 0], [0, 0, 0], [0, 0, 0]];
  let row = 0, column = 0;
  while (row < 3 && column < 3) {
    const value = Math.min(supply[row], demand[column]);
    table[row][column] += value;
    supply[row] -= value;
    demand[column] -= value;
    if (supply[row] === 0) row += 1;
    if (demand[column] === 0) column += 1;
  }
  return table;
}

function write(file, value) {
  const full = path.join(dir, file);
  fs.mkdirSync(path.dirname(full), {recursive: true});
  let content = typeof value === 'string' ? value : JSON.stringify(value, null, 2) + '\n';
  const countWords = text => text.trim() ? text.trim().split(/\s+/).length : 0;
  if (countWords(content) > 100000 && typeof value !== 'string') content = JSON.stringify(value) + '\n';
  assert(countWords(content) <= 100000, 'document ceiling ' + file);
  fs.writeFileSync(full, content, {flag: 'wx'});
}

function ownerFiles() {
  const found = [];
  function walk(folder) {
    for (const entry of fs.readdirSync(folder, {withFileTypes: true})) {
      const target = path.join(folder, entry.name);
      if (entry.isDirectory()) walk(target);
      else found.push(path.relative(dir, target).replaceAll('\\', '/'));
    }
  }
  walk(dir);
  return found.sort();
}

function gate() {
  const head = git('rev-parse', 'HEAD');
  const upstream = git('rev-parse', '@{u}');
  const tracking = git('rev-parse', 'refs/remotes/origin/' + branch);
  const live = git('ls-remote', '--heads', 'origin', 'refs/heads/' + branch).split(/\s+/)[0];
  const divergence = git('rev-list', '--left-right', '--count', 'HEAD...@{u}').split(/\s+/).map(Number);
  assert.equal(git('status', '--porcelain=v1'), '');
  assert.equal(git('branch', '--show-current'), branch);
  for (const value of [upstream, tracking, live]) assert.equal(value, head);
  assert.deepEqual(divergence, [0, 0]);
  return {head, upstream, tracking, remote: live, ahead: 0, behind: 0, clean: true, checked_at: new Date().toISOString()};
}

function runBoundTool(file, args, input) {
  const tools = external('toolchain.json');
  const name = Object.keys(tools.bindings).find(key => tools[key] === file);
  assert(name, 'Unbound helper ' + file);
  assert.equal(sha(fs.readFileSync(file)), tools.bindings[name], 'helper hash ' + name);
  const result = cp.spawnSync(tools.python, ['-X', 'utf8', '-B', file, ...args], {
    input, encoding: 'utf8', timeout: 60000,
    env: {...process.env, PYTHONDONTWRITEBYTECODE: '1'}
  });
  return {
    exit_code: result.status, stdout: result.stdout.trim(), stderr: result.stderr.trim(),
    signal: result.signal, error: result.error ? String(result.error.message) : null
  };
}

function invalid(request, selector) {
  const subject = clone(request);
  const kind = selector % 12;
  if (kind === 0) { delete subject.table; return {request: subject, error: 'field_set'}; }
  if (kind === 1) { subject.extra = true; return {request: subject, error: 'field_set'}; }
  if (kind === 2) { subject.op = 'unregistered'; return {request: subject, error: 'operation'}; }
  if (kind === 3) { subject.record_id = 'invalid'; return {request: subject, error: 'record_id'}; }
  if (kind === 4) { subject.supply[0] = 0; return {request: subject, error: 'positive_histogram'}; }
  if (kind === 5) { subject.table[0] = [0, 1]; return {request: subject, error: 'table_shape'}; }
  if (kind === 6) { subject.table[0][0] = -1; return {request: subject, error: 'table_shape'}; }
  if (kind === 7) { subject.table[0][0] = true; return {request: subject, error: 'table_shape'}; }
  if (kind === 8) { subject.table[0][0] += 1; return {request: subject, error: 'row_margin'}; }
  if (kind === 9) {
    const from = subject.table[0].findIndex(value => value > 0), to = (from + 1) % 3;
    subject.table[0][from] -= 1; subject.table[0][to] += 1;
    return {request: subject, error: 'column_margin'};
  }
  if (kind === 10) { subject.demand[0] += 1; return {request: subject, error: 'mass_bound'}; }
  subject.supply = [4, 4, 4]; subject.demand = [4, 4, 4];
  return {request: subject, error: 'mass_bound'};
}

function stageRequest(op, profile) {

function transposeRequest(request) {
  return {
    op: request.op, record_id: request.record_id,
    supply: request.demand, demand: request.supply,
    table: request.table[0].map((_, column) => request.table.map(row => row[column]))

function permuteRequest(request) {
  const rows = [2, 0, 1], columns = [1, 2, 0];
  return {
    op: request.op, record_id: request.record_id,
    supply: rows.map(index => request.supply[index]),
    demand: columns.map(index => request.demand[index]),
    table: rows.map(row => columns.map(column => request.table[row][column]))
  };
}

      'Successful bounded preflight', 'Retained operational failure', [negative.negative_id]));

  const plan = {

function refreshPlanFlow() {
  assert(!fs.existsSync(path.join(dir, 'x1')));
  assert(!fs.existsSync(path.join(dir, 'x2')));
  const startup = external('startup-negatives.json');
  const witnesses = [];
  for (const negative of startup) {
    const method = negative.negative_id + '-M';
    witnesses.push(witness(negative.negative_id, method, 'fail', negative.subject,
      'Successful bounded preflight', 'Retained operational failure', [negative.negative_id]));
    witnesses.push(witness(negative.negative_id + '-RECOVERY', method, 'pass', negative.recovery,
      'Smallest bounded recovery', 'Recovered without source execution or mutation', [negative.negative_id]));
  }
  const ledger = makeLedger('planning', witnesses);
  fs.writeFileSync(path.join(dir, 'planning', 'method-flow.json'),

    schema: 'sable.stage-manifest.v1', owner, phase, stage: 'planning',
    entries, excluded_self: prefix + 'planning/manifest.json'

  console.log(JSON.stringify({

    startup_negatives: startup.length,
    methods: ledger.counts.methods,
    witnesses: ledger.counts.witnesses,
    manifest_entries: entries.length
  }));
}

function execute(stage) {
  verifyFrozen();
  const preGate = gate();
  assert.equal(preGate.head, external(stage === 'x1' ? 'plan-published.json' : 'x1-published.json').commit);
  const success = latch(stage), plan = json('planning/plan.json'), runtime = require('./runtime.txt');
  const rows = contracts().filter(item => item.stage === stage);
  const safe = [], failures = [], refusals = [], repairs = [], core = [], witnesses = [];
  for (const contract of rows) {
    const method = 'SR7047-M-' + contract.operation;
    const variants = [{request: contract.request, expected: contract.expected}, ...contract.alternatives];
    variants.forEach((variant, index) => {
      const before = sha(variant.request), result = runtime.run(variant.request);
      assert.deepEqual(runtime.project(result), variant.expected, contract.id + ' safe ' + index);
      assert.equal(sha(variant.request), before);
      const id = contract.id + '-SAFE' + (index + 1);
      safe.push({id, contract: contract.id, request: variant.request, result,
        original_input_unchanged: true, new_proposal_credit: index === 0 ? 1 : 0});
      witnesses.push(witness(id, method, 'pass', 'Exact projected planning-reference comparison',
        sha(variant.expected), sha(runtime.project(result))));
      if (index === 0) core.push({
        id: contract.id, profile: contract.profile, operation: contract.operation,
        request: contract.request, result
      });
    });
    contract.negative_subjects.forEach(negative => {
      let observed = null;
      try { runtime.run(negative.request); } catch (error) { observed = error.message; }
      assert.equal(observed, negative.error, contract.id + ' malformed');
      failures.push({negative_id: negative.negative_id, subject: negative.request,
        expected_error: negative.error, observed_error: observed, original_success_credit: 0, retained: true});
      witnesses.push(witness(negative.negative_id, method, 'fail', 'Malformed frozen request subject',
        'Valid bounded request', observed, [negative.negative_id]));
      const refusalId = negative.negative_id + '-REFUSAL';
      refusals.push({id: refusalId, negative_id: negative.negative_id,
        expected: negative.error, observed, passed: true});
      witnesses.push(witness(refusalId, method, 'pass', 'Separate refusal guard witness',
        negative.error, observed, [negative.negative_id]));
      const repaired = clone(contract.request); runtime.validate(repaired);
      assert.equal(sha(repaired), contract.request_sha256);
      const repairId = negative.negative_id + '-CFR';
      repairs.push({id: repairId, negative_id: negative.negative_id,
        repair: 'Create a separate request from the frozen template and retain the original subject.',
        repaired_request_sha256: sha(repaired), original_success_credit: 0});
      witnesses.push(witness(repairId, method, 'pass', 'Separate CLEAN/FIX/REFINE template recovery',
        contract.request_sha256, sha(repaired), [negative.negative_id]));
    });
  }
  assert.equal(safe.length, 400); assert.equal(failures.length, 300);
  assert.equal(refusals.length, 300); assert.equal(repairs.length, 300);

  const runnerReceipts = [];
  for (const runner of plan.runners.filter(item => item.stage === stage)) {
    const contract = rows.find(item => item.operation === runner.operations[0]);
    const invoke = request => {
      const result = cp.spawnSync(process.execPath, [path.join(dir, runner.path)], {
        input: JSON.stringify(request), encoding: 'utf8', timeout: 10000
      });
      return {exit_code: result.status, stdout: result.stdout.trim(), stderr: result.stderr.trim(), signal: result.signal};
    };
    const positive = invoke(contract.request);
    const expected = {runner: runner.id, ok: true,
      value: core.find(item => item.id === contract.id).result, error: null};
    assert.deepEqual(positive, {exit_code: 0, stdout: JSON.stringify(expected), stderr: '', signal: null});
    const bad = {...contract.request, op: 'unregistered'}, rejection = invoke(bad);
    assert.deepEqual(rejection, {
      exit_code: 2, stdout: JSON.stringify({runner: runner.id, ok: false, value: null, error: 'runner_operation'}),
      stderr: '', signal: null
    });
    const negativeId = runner.id + '-N', method = 'SR7047-M-' + runner.id;
    failures.push({negative_id: negativeId, subject: bad, original_success_credit: 0,
      retained: true, observed_error: 'runner_operation'});
    witnesses.push(witness(runner.id + '-ACCEPT', method, 'pass',
      'Standalone runner acceptance envelope', 0, positive.exit_code, [negativeId]));
    witnesses.push(witness(negativeId, method, 'fail',
      'Unsupported runner operation', 'Supported operation', 'runner_operation', [negativeId]));
    witnesses.push(witness(runner.id + '-REFUSE', method, 'pass',
      'Standalone runner refusal envelope', 2, rejection.exit_code, [negativeId]));
    runnerReceipts.push({id: runner.id, positive, rejection});
  }

  const skillReceipts = [];
  for (const skill of plan.skills.filter(item => item.stage === stage)) {
    const result = runBoundTool(external('toolchain.json').skill_validator,
      [path.dirname(path.join(dir, skill.path))]);
    assert.equal(result.exit_code, 0, result.stdout + result.stderr);
    skillReceipts.push({id: skill.id, exit_code: result.exit_code,
      stdout: result.stdout, stderr: result.stderr});
    const contract = rows.find(item => item.operation === skill.op);
    witnesses.push(witness(skill.id + '-VALID', 'SR7047-M-' + skill.op, 'pass',
      'Official skill quick validator', 0, result.exit_code,
      [contract.negative_subjects[0].negative_id]));
  }

  const allCore = stage === 'x2' ? [...json('x1/core-results.json'), ...core] : core;
  const get = (op, profile) => allCore.find(item => item.operation === op && item.profile === profile).result.data;
  const tests = [];
  for (const profile of plan.profiles) {
    const retained = rows.find(item => item.profile === profile.id).negative_subjects[0].negative_id;
    if (stage === 'x1') {
      const sparse = get('sparse_encode', profile.id), decoded = get('sparse_decode', profile.id);
      const dense = get('dense_flatten', profile.id), graph = get('support_graph', profile.id);
      const components = get('support_components', profile.id), rank = get('support_cycle_rank', profile.id);
      const rowData = get('row_profile', profile.id), columnData = get('column_profile', profile.id);
      const lineage = get('cell_lineage', profile.id);
      const checks = {
        sparse_roundtrip: decoded.roundtrip && JSON.stringify(decoded.matrix) === JSON.stringify(profile.table),
        dense_cardinality: dense.cell_count === 9 && dense.values.length === 9,
        edge_parity: graph.edges.length === sparse.nonzero_cells,
        component_coverage: new Set(components.components.flat()).size === 6,
        cycle_rank: rank.cycle_rank === rank.edge_count - rank.vertex_count + rank.component_count,
        row_conservation: rowData.every((row, index) => row.total === profile.supply[index]),
        column_conservation: columnData.every((column, index) => column.total === profile.demand[index]),
        lineage_complete: lineage.length === 9 && new Set(lineage.map(item => item.lineage_id)).size === 9
      };
      assert(Object.values(checks).every(Boolean));
      const id = 'SR7047-X1-TEST-' + profile.id;
      tests.push({id, profile: profile.id, checks, domain_reexecution: false});
      witnesses.push(witness(id, 'SR7047-M-sparse_decode', 'pass',
        'Persisted finite representation identities', true, true, [retained]));
    } else {
      const permutation = get('permutation_covariance', profile.id);
      const transpose = get('transpose_covariance', profile.id);
      const mask = get('support_mask', profile.id), zero = get('zero_elision_guard', profile.id);
      const order = get('canonical_entry_order', profile.id), correction = get('correction_patch', profile.id);
      const summary = get('accessible_summary', profile.id), uncertainty = get('uncertainty_annotation', profile.id);
      const gap = get('external_evidence_gap', profile.id), authority = get('catalog_authority_gate', profile.id);
      const checksA = {
        permutation_support: permutation.support_count_preserved,
        transpose_involution: transpose.involution && transpose.support_count_preserved,
        mask_fidelity: mask.every((row, i) => row.every((value, j) => value === (profile.table[i][j] > 0))),
        zero_roundtrip: zero.encoded_zero_count === 0 && zero.roundtrip,
        canonical_order: order.row_major && new Set(order.coordinate_order).size === order.coordinate_order.length
      };
      const rowTotals = correction.result.map(row => sum(row));
      const columnTotals = [0, 1, 2].map(column => correction.result.reduce((total, row) => total + row[column], 0));
      const checksB = {
        correction_conservation: correction.margins_preserved &&
          JSON.stringify(rowTotals) === JSON.stringify(profile.supply) &&
          JSON.stringify(columnTotals) === JSON.stringify(profile.demand),
        accessible_saved: summary.record_id === profile.id && summary.mass_units === sum(profile.supply),
        uncertainty_held: uncertainty.represented_only &&
          uncertainty.annotations.every(item => item.observation_status === 'not_observed'),
        gap_zero_credit: gap.external_credit === 0 && gap.missing.length === 3,
        authority_held: authority.action_performed === false
      };
      assert(Object.values(checksA).every(Boolean)); assert(Object.values(checksB).every(Boolean));
      const idA = 'SR7047-X2-TEST-A-' + profile.id, idB = 'SR7047-X2-TEST-B-' + profile.id;
      tests.push({id: idA, profile: profile.id, checks: checksA, domain_reexecution: false});
      tests.push({id: idB, profile: profile.id, checks: checksB, domain_reexecution: false});
      witnesses.push(witness(idA, 'SR7047-M-permutation_covariance', 'pass',
        'Persisted covariance and encoding identities', true, true, [retained]));
      witnesses.push(witness(idB, 'SR7047-M-correction_patch', 'pass',
        'Persisted correction and authority-boundary identities', true, true, [retained]));
    }
  }

  const models = stage === 'x2' ? plan.profiles.map(profile => ({
    id: 'SR7047-MODEL-' + profile.id, record_id: profile.id,
    coordinates: [
      get('sparse_encode', profile.id).nonzero_cells,
      get('support_components', profile.id).count,
      get('support_cycle_rank', profile.id).cycle_rank
    ],
    axes: ['nonzero cells', 'support components', 'cycle rank'],
    supply: profile.supply, demand: profile.demand, table: profile.table,
    nonphysical: true, domain_reexecution: false, outcome: 'represented'
  })) : [];
  for (const model of models) {
    const negative = rows.find(item => item.profile === model.record_id).negative_subjects[0].negative_id;
    witnesses.push(witness(model.id, 'SR7047-M-uncertainty_annotation', 'pass',
      'Persisted three-coordinate nonphysical projection', 3, model.coordinates.length, [negative]));
  }

  const hookReceipts = [];
  let pluginValidation = null;
  if (stage === 'x2') {
    pluginValidation = runBoundTool(external('toolchain.json').plugin_validator, [path.join(dir, plan.plugin_path)]);
    assert.equal(pluginValidation.exit_code, 0, pluginValidation.stdout + pluginValidation.stderr);
    for (const topic of plan.hooks) {
      const method = 'SR7047-HOOK-' + topic, negativeId = method + '-N';
      failures.push({negative_id: negativeId, subject: {topic, violation: true},
        subject_kind: 'synthetic declared violation, not a performed external action',
        original_success_credit: 0, retained: true});
      witnesses.push(witness(negativeId, method, 'fail', 'Synthetic declared boundary violation',
        'Boundary-preserving state', 'violation declared', [negativeId]));
      for (const violation of [false, true]) {
        const result = cp.spawnSync(process.execPath, [
          path.join(dir, plan.plugin_path, 'scripts', 'catalog-advisories.txt'), topic
        ], {input: JSON.stringify({violation}), encoding: 'utf8', timeout: 10000});
        assert.equal(result.status, 0); assert.equal(result.stderr, '');
        const value = JSON.parse(result.stdout);
        assert.equal(value.continue, true); assert.equal(value.advisory_active, violation);
        hookReceipts.push({topic, violation, exit_code: result.status,
          result: value, live_lifecycle_observation: false});
        witnesses.push(witness(method + '-' + (violation ? 'ADVERSE' : 'BENIGN'),
          method, 'pass', 'Manual nonblocking hook fixture', violation,
          value.advisory_active, [negativeId]));
      }
    }
  }

  for (let index = 0; index < 4; index += 1) {
    write(stage + '/safe-' + String(index + 1).padStart(2, '0') + '.json',
      safe.slice(index * 100, index * 100 + 100));
  }
  write(stage + '/core-results.json', core);
  write(stage + '/candidate-failures.json', failures);
  write(stage + '/refusals.json', refusals);
  write(stage + '/cfr.json', repairs);
  write(stage + '/tests.json', tests);
  write(stage + '/runner-smokes.json', runnerReceipts);
  write(stage + '/skill-validation.json', skillReceipts);
  if (stage === 'x2') {
    write(stage + '/models.json', models);
    const html = require('./viewer.txt')(models);
    const scripts = [...html.matchAll(/<script>([\s\S]*?)<\/script>/g)];
    assert.equal(scripts.length, 1); new vm.Script(scripts[0][1]);
    write(stage + '/viewer/index.html', html);
    write(stage + '/viewer-structure.json', {
      model_records: models.length, exact_coordinate_count: models.every(item => item.coordinates.length === 3),
      inline_script_syntax: true, table_fallback: true, aria_live: true,
      domain_replay: false, complete_accessibility_claim: false
    });
    write(stage + '/hook-smokes.json', hookReceipts);
    write(stage + '/plugin-validation.json', pluginValidation);
  }
  const ledger = makeLedger(stage, witnesses);
  write(stage + '/method-flow.json', ledger);
  write(stage + '/method-flow-validation.json', validateFlow(stage + '/method-flow.json'));
  const summary = {
    owner, phase, stage, definition_commit: preGate.head, pre_execution_gate: preGate,
    contracts: core.length, safe: safe.length, malformed_candidates: 300,

function publish(stage) {
  const state = gate();
  const artifact = stage === 'plan' ? 'planning/manifest.json' :
    stage === 'final' ? 'final/content-seal.json' : stage + '/manifest.json';
  const expectedParent = stage === 'plan' ? source :
    stage === 'x1' ? external('plan-published.json').commit :
    stage === 'x2' ? external('x1-published.json').commit :
    external('x2-published.json').commit;
  assert.equal(git('show', '-s', '--format=%P', state.head), expectedParent);
  const receipt = {
    schema: 'sable.stage-publication.v1', owner, phase, stage,
    commit: state.head, direct_parent: expectedParent, gate: state,
    artifact: prefix + artifact, artifact_sha256: sha(bytes(artifact)), clean: true
  };
  fs.writeFileSync(path.join(bank, stage + '-published.json'),
    JSON.stringify(receipt, null, 2) + '\n', {flag: 'wx'});
  console.log(JSON.stringify(receipt));
}

function canonical() {
  verifyFrozen();
  const state = gate(), published = external('final-published.json');
  assert.equal(state.head, published.commit);
  const success = latch('canonical'), checks = [];
  const check = (name, observed, expected) => {
    assert.deepEqual(observed, expected, name);
    checks.push({name, passed: true, observed});
  };
  const plan = json('planning/plan.json'), truth = json('final/phase-truth.json');
  const chain = [source, truth.lifecycle.plan, truth.lifecycle.x1, truth.lifecycle.x2, state.head];
  for (let index = 1; index < chain.length; index += 1) {
    check('direct_parent_' + index, git('show', '-s', '--format=%P', chain[index]), chain[index - 1]);
  }
  check('commit_count', Number(git('rev-list', '--count', source + '..' + state.head)), 4);
  check('zero_merges', Number(git('rev-list', '--count', '--merges', source + '..' + state.head)), 0);
  check('owner_only_delta', git('diff', '--name-only', source, state.head).split('\n').filter(Boolean)
    .every(file => file.startsWith(prefix)), true);
  const repositoryFiles = git('ls-tree', '-r', '--name-only', 'HEAD').split('\n').filter(Boolean);
  check('repository_ceiling', repositoryFiles.length <= 2000, true);
  check('owner_ceiling', ownerFiles().length <= 170, true);
  check('source_unchanged', git('diff', source, state.head, '--', '.', ':(exclude)' + prefix), '');
  const manifests = [
    ['planning/manifest.json', truth.lifecycle.plan], ['x1/manifest.json', truth.lifecycle.x1],
    ['x2/manifest.json', truth.lifecycle.x2], ['final/content-seal.json', state.head]
  ];
  let bindings = 0;
  for (const [file, commit] of manifests) {
    const declared = JSON.parse(git('show', commit + ':' + prefix + file));
    for (const entry of declared.entries) {
      const blob = cp.execFileSync('git', ['-C', root, 'show', commit + ':' + entry.path],
        {maxBuffer: 64 * 1024 * 1024});
      assert.equal(sha(blob), entry.sha256, entry.path);
      assert.equal(blob.length, entry.bytes, entry.path);
      bindings += 1;
    }
    check('manifest_' + file, declared.entries.length, declared.entries.length);
  }
  let jsonParses = 0;
  for (const file of ownerFiles()) {
    const content = bytes(file), words = content.toString().trim() ?
      content.toString().trim().split(/\s+/).length : 0;
    assert(words <= 100000, file + ' word ceiling');
    assert(!file.endsWith('.pdf'));
    if (file.endsWith('.json')) { JSON.parse(content); jsonParses += 1; }
  }
  check('strict_json_documents', jsonParses, jsonParses);
  check('content_seal_hash', sha(bytes('final/content-seal.json')), published.artifact_sha256);
  for (const stage of ['x1', 'x2']) {
    const summary = json(stage + '/summary.json'), invocation = external(stage + '-invocation.json');
    check(stage + '_latch', [
      summary.invocations, summary.successes, summary.replays,
      invocation.invocations, invocation.successes, invocation.replays
    ], [1, 1, 0, 1, 1, 0]);
    check(stage + '_manifest_hash', sha(bytes(stage + '/manifest.json')), invocation.manifest_sha256);
    check(stage + '_counts', [
      summary.safe, summary.malformed_candidates, summary.separate_refusals,
      summary.clean_fix_refine, summary.skills, summary.runners
    ], [400, 300, 300, 300, 10, 5]);
    check(stage + '_flow_valid', json(stage + '/method-flow-validation.json').valid, true);
  }
  check('planning_flow_valid', json('planning/method-flow-validation.json').valid, true);
  check('x1_tests', json('x1/tests.json').length, 15);
  check('x2_tests', json('x2/tests.json').length, 30);
  check('models', json('x2/models.json').length, 15);
  check('hook_fixtures', json('x2/hook-smokes.json').length, 20);
  check('plugin_validator', json('x2/plugin-validation.json').exit_code, 0);
  check('source_fold', truth.source_fold_count, 1);
  check('witness_sum', truth.owner_delta.witnesses,
    truth.owner_delta.failed_witnesses + truth.owner_delta.passing_witnesses);
  for (const [key, value] of Object.entries(truth.owner_delta)) {
    check('effective_' + key, truth.effective_totals[key], plan.source_baseline[key] + value);
  }
  check('source_execution', truth.source_execution_credit, 0);
  check('route_held', truth.route.state, 'PREPARED_NOT_SENT');
  check('stage20', truth.terminal_verdict, 'NOT_READY_FOR_STAGE_20');
  const findings = [], patterns = [
    /[A-Z]:[\\/](?:Users|GHC-Archives)/i,
    /-----BEGIN [A-Z ]*PRIVATE KEY-----/,
    /[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}/i,
    /(?:app|plugin):\/\//i, new RegExp('<codex_' + 'delegation>', 'i')
  ];
  for (const file of ownerFiles()) {
    const content = bytes(file).toString();
    if (patterns.some(pattern => pattern.test(content))) findings.push(file);
  }
  check('bounded_privacy_hits', findings, []);
  const receipt = {
    schema: 'sable.owner-canonical.v1', state: 'VALID_EXACT_FINAL_OWNER_SCOPED_CANONICAL',
    owner, phase, exact_final: state.head, gate: state, checks,
    passed: checks.length, total: checks.length, manifest_raw_bindings: bindings,
    owner_files: ownerFiles().length, repository_files: repositoryFiles.length,
    json_parses: jsonParses, invocations: 1, successes: 1, replays: 0,
    domain_replays: 0, source_replays: 0, skill_runner_hook_test_replays: 0,
    metadata_only: true, boundary
  };
  const serialized = JSON.stringify(receipt, null, 2) + '\n';
  fs.writeFileSync(path.join(bank, 'canonical-receipt.json'), serialized, {flag: 'wx'});
  success({exact_final: state.head, receipt_sha256: sha(serialized)});
  console.log(JSON.stringify({
    state: receipt.state, passed: checks.length, total: checks.length,
    exact_final: state.head, receipt_sha256: sha(serialized)
  }));
}
