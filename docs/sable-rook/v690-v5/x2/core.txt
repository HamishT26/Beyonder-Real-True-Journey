'use strict';
function keys(v,a){return Object.keys(v).every(k=>a.includes(k));}
const operations={
 audit_filter_projection:p=>({visible_ids:p.rows.filter(r=>p.statuses.includes(r.status)&&p.pillars.includes(r.pillar)).map(r=>r.id)}),
 uncertainty_narration:p=>({text:p.label+': '+p.value+' '+p.unit+'; bounded interval '+p.lower+' to '+p.upper+'; source '+p.source_status+'.',machine:{value:p.value,lower:p.lower,upper:p.upper,unit:p.unit}}),
 evidence_expiry_state:p=>{const state=p.revoked?'revoked':p.observed_at>=p.expires_at?'expired':'active';return{state,usable:state==='active'};},
 focus_order_trace:p=>{const active=p.elements.filter(r=>!r.hidden&&!r.disabled).sort((a,b)=>a.order-b.order||a.id.localeCompare(b.id)),orders=active.map(r=>r.order);return{focus_order:active.map(r=>r.id),duplicate_order:orders.some((v,i)=>i&&orders[i-1]===v)};},
 table_sort_stability:p=>{const rows=p.rows.map((r,i)=>({...r,_i:i})).sort((a,b)=>(a.value-b.value)*(p.direction==='desc'?-1:1)||a._i-b._i);return{row_ids:rows.map(r=>r.id),direction:p.direction};},
 error_summary_links:p=>{const linked=[],missing_targets=[];for(const e of p.errors)(p.targets.includes(e.target)?linked:missing_targets).push(e.field);return{linked:linked.sort(),missing_targets:missing_targets.sort()};},
 live_region_queue:p=>{const rows=[...p.events].sort((a,b)=>a.time-b.time||a.id.localeCompare(b.id));return{assertive:rows.filter(r=>r.urgency==='assertive').map(r=>r.id),polite:rows.filter(r=>r.urgency==='polite').map(r=>r.id),evaluation:'structural_proxy'};},
 locale_number_readback:p=>({canonical_decimal:(p.numerator/p.denominator).toFixed(3),requested_locale:p.locale,spoken_review:'reserved',evaluation:'structural_proxy'}),
 real_user_evaluation_gate:p=>({state:p.participants>0&&p.assistive_technologies.length>0?'evidence_present_requires_review':'OPEN_GAP',participant_rows:p.participants,manual_review:'reserved'}),
 authority_release_gate:p=>{const vacant=p.required_roles.filter(r=>r.status!=='authorized').map(r=>r.role).sort();return{state:vacant.length?'EXACT_GATE':'requires_live_authority_verification',vacant_roles:vacant,software_authority:false};}
};
const allowed={audit_filter_projection:['statuses','pillars','rows','case_token'],uncertainty_narration:['label','value','lower','upper','unit','source_status','case_token'],evidence_expiry_state:['observed_at','expires_at','revoked','case_token'],focus_order_trace:['elements','case_token'],table_sort_stability:['direction','rows','case_token'],error_summary_links:['targets','errors','case_token'],live_region_queue:['events','case_token'],locale_number_readback:['numerator','denominator','locale','case_token'],real_user_evaluation_gate:['participants','assistive_technologies','manual_review','intended_context','case_token'],authority_release_gate:['required_roles','case_token']};
function evaluate(request){if(!request||typeof request!=='object'||Array.isArray(request)||!keys(request,['op','payload'])||!operations[request.op]||!request.payload||typeof request.payload!=='object'||Array.isArray(request.payload))return{ok:false,error:'invalid_request',value:null};if(!keys(request.payload,allowed[request.op]))return{ok:false,error:'unknown_field',value:null};try{return{ok:true,error:null,value:operations[request.op](request.payload)};}catch{return{ok:false,error:'evaluation_error',value:null};}}
module.exports={evaluate,operations,allowed};if(require.main===module){const fs=require('node:fs'),input=process.argv[2]||fs.readFileSync(0,'utf8');process.stdout.write(JSON.stringify(evaluate(JSON.parse(input)))+'\n');}
