'use strict';
// Owner runtime, written separately from the exhaustive reference.
const OPS=['transitive_closure','cover_relations','order_matrix','minimal_elements','maximal_elements','rank_bounds','height','width','maximal_chains','linear_extensions','order_ideals','order_filters','principal_ideals','principal_filters','mobius_matrix','dual_profile','incomparability_edges','scene','external_gap','authority_gate'];
const outcome=op=>op==='scene'?'represented':op==='external_gap'?'open_gap':op==='authority_gate'?'exact_gate':'completed';
const pairSort=(a,b)=>a[0]-b[0]||a[1]-b[1];
const arraySort=(a,b)=>a.length-b.length||a.find((x,i)=>x!==b[i])-b[a.findIndex((x,i)=>x!==b[i])];
function validate(q){if(!q||Array.isArray(q)||typeof q!=='object'||JSON.stringify(Object.keys(q).sort())!==JSON.stringify(['n','op','relations']))throw Error('field_set');if(!OPS.includes(q.op))throw Error('operation');if(!Number.isInteger(q.n)||q.n<1||q.n>6)throw Error('order');if(!Array.isArray(q.relations)||q.relations.length>15)throw Error('relation_count');const seen=new Set();for(const p of q.relations){if(!Array.isArray(p)||p.length!==2||!p.every(Number.isInteger)||p[0]<0||p[1]<0||p[0]>=q.n||p[1]>=q.n||p[0]===p[1])throw Error('relation');const k=p.join(',');if(seen.has(k))throw Error('relation_unique');seen.add(k);}const c=closure(q.n,q.relations);if(c.some(([a,b])=>a===b))throw Error('acyclic');return true;}
function closure(n,relations){const adj=Array.from({length:n},()=>[]);for(const[a,b]of relations)adj[a].push(b);const pairs=[];for(let start=0;start<n;start++){const seen=new Set(),stack=[...adj[start]];while(stack.length){const x=stack.pop();if(seen.has(x))continue;seen.add(x);stack.push(...adj[x]);}for(const end of seen)pairs.push([start,end]);}return pairs.sort(pairSort);}
function core(n,relations){const strict=closure(n,relations),S=new Set(strict.map(p=>p.join(','))),lt=(a,b)=>S.has(a+','+b),le=(a,b)=>a===b||lt(a,b),elements=Array.from({length:n},(_,i)=>i),covers=strict.filter(([a,b])=>!elements.some(x=>lt(a,x)&&lt(x,b))),mins=elements.filter(x=>!elements.some(y=>lt(y,x))),maxs=elements.filter(x=>!elements.some(y=>lt(x,y)));const subsets=Array.from({length:1<<n},(_,m)=>elements.filter(i=>m&(1<<i)));const antichains=subsets.filter(s=>s.every((a,i)=>s.slice(i+1).every(b=>!lt(a,b)&&!lt(b,a)))),width=Math.max(...antichains.map(s=>s.length));const ideals=subsets.filter(s=>s.every(x=>elements.every(y=>!lt(y,x)||s.includes(y)))).sort(arraySort),filters=subsets.filter(s=>s.every(x=>elements.every(y=>!lt(x,y)||s.includes(y)))).sort(arraySort);const outgoing=new Map(elements.map(x=>[x,covers.filter(e=>e[0]===x).map(e=>e[1])]));const chains=[];function walk(path){const next=outgoing.get(path.at(-1));if(!next.length)chains.push(path);else for(const y of next)walk([...path,y]);}for(const m of mins)walk([m]);chains.sort(arraySort);const indegree=()=>new Map(elements.map(x=>[x,strict.filter(e=>e[1]===x).length]));const extensions=[];function topo(prefix,left,deg){if(!left.length){extensions.push(prefix);return;}for(const x of left.filter(y=>deg.get(y)===0)){const nextLeft=left.filter(y=>y!==x),nextDeg=new Map(deg);for(const[a,b]of strict)if(a===x&&nextLeft.includes(b))nextDeg.set(b,nextDeg.get(b)-1);topo([...prefix,x],nextLeft,nextDeg);}}topo([],elements,indegree());extensions.sort(arraySort);const lower=new Map(elements.map(x=>[x,1])),upper=new Map(elements.map(x=>[x,1]));for(const p of extensions[0])for(const[a,b]of covers)if(b===p)lower.set(b,Math.max(lower.get(b),lower.get(a)+1));for(const p of [...extensions[0]].reverse())for(const[a,b]of covers)if(a===p)upper.set(a,Math.max(upper.get(a),upper.get(b)+1));const topoOrder=extensions[0],mu=Array.from({length:n},()=>Array(n).fill(0));for(const a of topoOrder){mu[a][a]=1;for(const b of topoOrder)if(lt(a,b))mu[a][b]=-topoOrder.filter(k=>k!==b&&le(a,k)&&lt(k,b)).reduce((s,k)=>s+mu[a][k],0);}return {elements,strict,lt,le,covers,mins,maxs,chains,extensions,ideals,filters,width,antichains:antichains.filter(s=>s.length===width).sort(arraySort),height:Math.max(...chains.map(s=>s.length)),bounds:elements.map(element=>({element,lower_height:lower.get(element),upper_height:upper.get(element)})),mu};}
function render(op,n,relations){const d=core(n,relations);if(op==='transitive_closure')return {n,strict_relations:d.strict};if(op==='cover_relations')return {n,covers:d.covers};if(op==='order_matrix')return {matrix:d.elements.map(a=>d.elements.map(b=>d.le(a,b)?1:0))};if(op==='minimal_elements')return {elements:d.mins};if(op==='maximal_elements')return {elements:d.maxs};if(op==='rank_bounds')return {bounds:d.bounds};if(op==='height')return {height:d.height};if(op==='width')return {width:d.width,maximum_antichains:d.antichains};if(op==='maximal_chains')return {chains:d.chains};if(op==='linear_extensions')return {extensions:d.extensions};if(op==='order_ideals')return {ideals:d.ideals};if(op==='order_filters')return {filters:d.filters};if(op==='principal_ideals')return {ideals:d.elements.map(element=>({element,members:d.elements.filter(x=>d.le(x,element))}))};if(op==='principal_filters')return {filters:d.elements.map(element=>({element,members:d.elements.filter(x=>d.le(element,x))}))};if(op==='mobius_matrix')return {matrix:d.mu};if(op==='dual_profile'){const dual=core(n,relations.map(([a,b])=>[b,a]));return {strict_relations:dual.strict,height:dual.height,width:dual.width,minimal_elements:dual.mins,maximal_elements:dual.maxs};}if(op==='incomparability_edges')return {edges:d.elements.flatMap((a,i)=>d.elements.slice(i+1).filter(b=>!d.lt(a,b)&&!d.lt(b,a)).map(b=>[a,b]))};if(op==='scene')return {points:d.elements.map(i=>({id:i,x:i,y:d.bounds[i].lower_height-1,z:d.bounds[i].upper_height-1})),cover_edges:d.covers,coordinate_units:'declared index only',physical_measurement:false};if(op==='external_gap')return {state:'open_gap',missing:['independent implementation','external domain evidence','competent review'],real_rows:0};if(op==='authority_gate')return {state:'exact_gate',held:['identity action','legal interpretation','cultural decision','Maori authority','deployment'],actions_performed:0};throw Error('operation');}
function run(q){validate(q);const relations=q.relations.map(p=>p.slice()).sort(pairSort);return {outcome:outcome(q.op),data:render(q.op,q.n,relations)};}
module.exports={OPS,validate,run,closure};
if(require.main===module){try{const q=JSON.parse(require('node:fs').readFileSync(0,'utf8'));process.stdout.write(JSON.stringify(run(q))+'\n');}catch(e){process.stdout.write(JSON.stringify({error:e.message})+'\n');process.exitCode=2;}}
