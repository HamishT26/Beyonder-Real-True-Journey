'use strict';
// Same-owner exhaustive oracle used only while freezing the plan.
const assert=require('node:assert/strict');
const OPS=['transitive_closure','cover_relations','order_matrix','minimal_elements','maximal_elements','rank_bounds','height','width','maximal_chains','linear_extensions','order_ideals','order_filters','principal_ideals','principal_filters','mobius_matrix','dual_profile','incomparability_edges','scene','external_gap','authority_gate'];
const outcomes=op=>op==='scene'?'represented':op==='external_gap'?'open_gap':op==='authority_gate'?'exact_gate':'completed';
const pairSort=(a,b)=>a[0]-b[0]||a[1]-b[1];
const arraySort=(a,b)=>a.length-b.length||a.find((x,i)=>x!==b[i])-b[a.findIndex((x,i)=>x!==b[i])];
function exactFields(value,fields){if(!value||Array.isArray(value)||typeof value!=='object')throw Error('field_set');const got=Object.keys(value).sort(),want=[...fields].sort();if(JSON.stringify(got)!==JSON.stringify(want))throw Error('field_set');}
function closureFrom(n,relations){const set=new Set(relations.map(p=>p.join(',')));let changed=true;while(changed){changed=false;for(let a=0;a<n;a++)for(let b=0;b<n;b++)if(set.has(a+','+b))for(let c=0;c<n;c++)if(set.has(b+','+c)&&!set.has(a+','+c)){set.add(a+','+c);changed=true;}}return [...set].map(x=>x.split(',').map(Number)).sort(pairSort);}
function validate(request){exactFields(request,['op','n','relations']);if(!OPS.includes(request.op))throw Error('operation');if(!Number.isInteger(request.n)||request.n<1||request.n>6)throw Error('order');if(!Array.isArray(request.relations)||request.relations.length>15)throw Error('relation_count');const seen=new Set();for(const p of request.relations){if(!Array.isArray(p)||p.length!==2||!p.every(Number.isInteger))throw Error('relation');const [a,b]=p;if(a<0||b<0||a>=request.n||b>=request.n||a===b)throw Error('relation');const key=a+','+b;if(seen.has(key))throw Error('relation_unique');seen.add(key);}const c=closureFrom(request.n,request.relations);if(c.some(([a,b])=>a===b))throw Error('acyclic');return true;}
function subsets(n){return Array.from({length:1<<n},(_,m)=>Array.from({length:n},(_,i)=>i).filter(i=>m&(1<<i)));}
function permutations(xs){if(xs.length<2)return [xs.slice()];const out=[];for(let i=0;i<xs.length;i++)for(const tail of permutations(xs.slice(0,i).concat(xs.slice(i+1))))out.push([xs[i],...tail]);return out;}
function derived(n,relations){const strict=closureFrom(n,relations),keys=new Set(strict.map(p=>p.join(','))),lt=(a,b)=>keys.has(a+','+b),le=(a,b)=>a===b||lt(a,b);const elements=Array.from({length:n},(_,i)=>i);const covers=strict.filter(([a,b])=>!elements.some(z=>z!==a&&z!==b&&lt(a,z)&&lt(z,b)));const mins=elements.filter(x=>!elements.some(y=>lt(y,x))),maxs=elements.filter(x=>!elements.some(y=>lt(x,y)));const all=subsets(n);const chains=all.filter(s=>s.length>0&&s.every((a,i)=>s.slice(i+1).every(b=>le(a,b)||le(b,a))));const maximalChains=chains.filter(s=>!elements.some(x=>!s.includes(x)&&s.every(y=>le(x,y)||le(y,x)))).map(s=>s.slice().sort((a,b)=>lt(a,b)?-1:lt(b,a)?1:a-b)).sort(arraySort);const antichains=all.filter(s=>s.every((a,i)=>s.slice(i+1).every(b=>!lt(a,b)&&!lt(b,a))));const width=Math.max(...antichains.map(s=>s.length));const maxAntichains=antichains.filter(s=>s.length===width).sort(arraySort);const ideals=all.filter(s=>s.every(x=>elements.filter(y=>lt(y,x)).every(y=>s.includes(y)))).sort(arraySort);const filters=all.filter(s=>s.every(x=>elements.filter(y=>lt(x,y)).every(y=>s.includes(y)))).sort(arraySort);const extensions=permutations(elements).filter(p=>{const pos=new Map(p.map((x,i)=>[x,i]));return strict.every(([a,b])=>pos.get(a)<pos.get(b));}).sort(arraySort);const memo=new Map();function mu(a,b){const key=a+','+b;if(memo.has(key))return memo.get(key);if(a===b){memo.set(key,1);return 1;}if(!lt(a,b)){memo.set(key,0);return 0;}const value=-elements.filter(k=>k!==b&&le(a,k)&&lt(k,b)).reduce((s,k)=>s+mu(a,k),0);memo.set(key,value);return value;}const matrix=elements.map(a=>elements.map(b=>mu(a,b)));const lower=x=>Math.max(...chains.filter(s=>s.includes(x)&&s.every(y=>le(y,x))).map(s=>s.length));const upper=x=>Math.max(...chains.filter(s=>s.includes(x)&&s.every(y=>le(x,y))).map(s=>s.length));return {elements,strict,lt,le,covers,mins,maxs,chains:maximalChains,antichains:maxAntichains,width,height:Math.max(...chains.map(s=>s.length)),ideals,filters,extensions,matrix,bounds:elements.map(element=>({element,lower_height:lower(element),upper_height:upper(element)}))};}
function dataFor(op,n,relations){const d=derived(n,relations);switch(op){case'transitive_closure':return {n,strict_relations:d.strict};case'cover_relations':return {n,covers:d.covers};case'order_matrix':return {matrix:d.elements.map(a=>d.elements.map(b=>d.le(a,b)?1:0))};case'minimal_elements':return {elements:d.mins};case'maximal_elements':return {elements:d.maxs};case'rank_bounds':return {bounds:d.bounds};case'height':return {height:d.height};case'width':return {width:d.width,maximum_antichains:d.antichains};case'maximal_chains':return {chains:d.chains};case'linear_extensions':return {extensions:d.extensions};case'order_ideals':return {ideals:d.ideals};case'order_filters':return {filters:d.filters};case'principal_ideals':return {ideals:d.elements.map(element=>({element,members:d.elements.filter(x=>d.le(x,element))}))};case'principal_filters':return {filters:d.elements.map(element=>({element,members:d.elements.filter(x=>d.le(element,x))}))};case'mobius_matrix':return {matrix:d.matrix};case'dual_profile':{const dual=derived(n,relations.map(([a,b])=>[b,a]));return {strict_relations:dual.strict,height:dual.height,width:dual.width,minimal_elements:dual.mins,maximal_elements:dual.maxs};}case'incomparability_edges':return {edges:d.elements.flatMap((a,i)=>d.elements.slice(i+1).filter(b=>!d.lt(a,b)&&!d.lt(b,a)).map(b=>[a,b]))};case'scene':return {points:d.elements.map(i=>({id:i,x:i,y:d.bounds[i].lower_height-1,z:d.bounds[i].upper_height-1})),cover_edges:d.covers,coordinate_units:'declared index only',physical_measurement:false};case'external_gap':return {state:'open_gap',missing:['independent implementation','external domain evidence','competent review'],real_rows:0};case'authority_gate':return {state:'exact_gate',held:['identity action','legal interpretation','cultural decision','Maori authority','deployment'],actions_performed:0};default:throw Error('operation');}}
function evaluate(request){validate(request);const stable={op:request.op,n:request.n,relations:request.relations.map(p=>p.slice()).sort(pairSort)};return {outcome:outcomes(stable.op),data:dataFor(stable.op,stable.n,stable.relations)};}
module.exports={OPS,validate,evaluate,closureFrom};
if(require.main===module){const q=JSON.parse(require('node:fs').readFileSync(0,'utf8'));process.stdout.write(JSON.stringify(evaluate(q))+'\n');}
