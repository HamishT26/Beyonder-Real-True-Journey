'use strict';
const fs=require('node:fs'),path=require('node:path'),crypto=require('node:crypto');
const root=path.resolve(process.argv[2]||'.'),stage=process.argv[3];
if(!['plan','x1','x2'].includes(stage))throw new Error('STAGE');
const ownerRoot=path.join(root,'docs/sable-rook/v693-v4'),stageRoot=path.join(ownerRoot,stage),target=path.join(stageRoot,'manifest.json');
if(fs.existsSync(target))throw new Error('MANIFEST_EXISTS');
const excluded=[stage+'/manifest.json'];
const files=fs.readdirSync(stageRoot,{recursive:true}).filter(rel=>fs.statSync(path.join(stageRoot,rel)).isFile()).map(rel=>stage+'/'+rel.replaceAll('\\','/')).filter(rel=>!excluded.includes(rel)).sort();
const bindings=files.map(rel=>{const bytes=fs.readFileSync(path.join(ownerRoot,rel));return{path:rel,bytes:bytes.length,sha256:crypto.createHash('sha256').update(bytes).digest('hex')};});
const value={schema:'ghc.family.stage-manifest.v1',owner:'Sable Rook',phase:'v693-v4',stage,bindings,excluded,hash_domain:'raw Git blob bytes',stage_anchor:'Bound externally to the immutable stage commit',total_paths:bindings.length+excluded.length};
fs.writeFileSync(target,JSON.stringify(value,null,2)+'\n',{encoding:'utf8',flag:'wx'});
process.stdout.write(JSON.stringify({stage,bindings:bindings.length,excluded:excluded.length,total_paths:value.total_paths})+'\n');
