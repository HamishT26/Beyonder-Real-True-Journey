'use strict';
// Pure x2 extension factory. The caller supplies the immutable, hash-bound x1 API.
module.exports=base=>{
 const {need,own,plain,keys,name,canon,clone,equal,digest,validateJSON,vector,ge,dependencyView}=base.helpers;
 const {Refusal,envelope,parseStrict}=base,OPS={...base.operations};
 function versionedMap(value,code){need(plain(value),code);for(const [key,item] of Object.entries(value)){name(key,'FIELD');keys(item,['version','value'],'VERSIONED_FIELDS');need(Number.isSafeInteger(item.version)&&item.version>=0,'VERSION');validateJSON(item.value);}return value;}
 OPS.anti_entropy_delta=(record,args)=>{keys(record,['source','target'],'ANTI_ENTROPY_FIELDS');keys(args,[]);versionedMap(record.source,'SOURCE_MAP');versionedMap(record.target,'TARGET_MAP');const send=[],conflicts=[];for(const key of [...new Set([...Object.keys(record.source),...Object.keys(record.target)])].sort()){const source=record.source[key],target=record.target[key];if(source&&(!target||source.version>target.version))send.push(key);else if(source&&target&&source.version===target.version&&!equal(source.value,target.value))conflicts.push(key);}return{send,conflicts,network_transfer:false,source_preserved:true};};
 OPS.delta_application_guard=(record,args)=>{keys(args,['delta','current_sequence']);need(plain(args.delta),'DELTA');keys(args.delta,['base_digest','sequence','changes'],'DELTA_FIELDS');need(typeof args.delta.base_digest==='string'&&/^[0-9a-f]{64}$/.test(args.delta.base_digest),'DIGEST');need(Number.isSafeInteger(args.delta.sequence)&&args.delta.sequence>=0&&Number.isSafeInteger(args.current_sequence)&&args.current_sequence>=0,'SEQUENCE');need(plain(args.delta.changes),'CHANGES');const actual=digest(record);let reason=null;if(args.delta.base_digest!==actual)reason='BASE_DIGEST';else if(args.delta.sequence!==args.current_sequence+1)reason='SEQUENCE';const result=clone(record);if(reason===null)for(const key of Object.keys(args.delta.changes).sort()){name(key,'FIELD');const change=args.delta.changes[key];need(plain(change),'CHANGE');if(own(change,'remove')){keys(change,['remove'],'REMOVE_CHANGE');need(change.remove===true,'REMOVE_CHANGE');delete result[key];}else{keys(change,['value'],'VALUE_CHANGE');result[key]=clone(change.value);}}return{applied:reason===null,result:reason===null?result:clone(record),reason,actual_base_digest:actual,next_sequence:reason===null?args.delta.sequence:args.current_sequence,source_preserved:true};};
 OPS.tombstone_retention_check=(record,args)=>{keys(record,['clock','acknowledgements','required'],'TOMBSTONE_FIELDS');keys(args,[]);vector(record.clock);need(plain(record.acknowledgements),'ACK_MAP');for(const [replica,clock] of Object.entries(record.acknowledgements)){name(replica,'REPLICA');vector(clock);}need(Array.isArray(record.required)&&record.required.length<=128,'REQUIRED_REPLICAS');record.required.forEach(replica=>name(replica,'REPLICA'));need(new Set(record.required).size===record.required.length,'REQUIRED_DUPLICATE');const required=[...record.required].sort(),missing=required.filter(replica=>!own(record.acknowledgements,replica)),behind=required.filter(replica=>own(record.acknowledgements,replica)&&!ge(record.acknowledgements[replica],record.clock));return{retain:missing.length>0||behind.length>0,safe_to_collect:missing.length===0&&behind.length===0,missing,behind,required,production_collection:false};};
 OPS.lease_epoch_fence=(record,args)=>{keys(record,['current_epoch','current_holder','request'],'LEASE_FIELDS');keys(args,[]);need(Number.isSafeInteger(record.current_epoch)&&record.current_epoch>=0,'EPOCH');name(record.current_holder,'HOLDER');keys(record.request,['epoch','holder','action'],'LEASE_REQUEST_FIELDS');need(Number.isSafeInteger(record.request.epoch)&&record.request.epoch>=0,'EPOCH');name(record.request.holder,'HOLDER');name(record.request.action,'ACTION');const reason=record.request.epoch!==record.current_epoch?'STALE_EPOCH':record.request.holder!==record.current_holder?'WRONG_HOLDER':null;return{accepted:reason===null,reason,current_epoch:record.current_epoch,current_holder:record.current_holder,requested_action:record.request.action,external_lock:false,authority:false};};
 OPS.repair_plan_toposort=(record,args)=>{keys(record,['steps'],'REPAIR_FIELDS');keys(args,[]);need(Array.isArray(record.steps)&&record.steps.length<=128,'REPAIR_STEPS');const events=record.steps.map(step=>{keys(step,['id','requires'],'REPAIR_STEP_FIELDS');return{id:step.id,deps:step.requires};}),view=dependencyView(events);return{ready:view.closed,order:view.order,missing:view.missing,cycle_members:view.cycle_members,executed:false};};
 const claimRequirements={empirical_gmut:['real_data','frozen_analysis','uncertainty','independent_review'],professional_effectiveness:['real_operators','preregistered_comparison','safety_monitoring','independent_review'],production_identity:['real_keys','interoperability','privacy_security_review','trust_governance'],consciousness_personhood:['independent_valid_evidence','competent_interpretation'],stage20:['all_declared_external_gates','independent_review','authority']};
 OPS.causal_claim_firewall=(record,args)=>{keys(record,['claim','software_evidence'],'CLAIM_FIELDS');keys(args,[]);need(own(claimRequirements,record.claim),'CLAIM_TYPE');need(Array.isArray(record.software_evidence),'SOFTWARE_EVIDENCE');record.software_evidence.forEach(item=>name(item,'EVIDENCE_LABEL'));return{claim:record.claim,permitted:false,required:claimRequirements[record.claim],software_credit_only:true,authority:false};};
 OPS.accessible_consistency_summary=(record,args)=>{keys(record,['rows'],'SUMMARY_FIELDS');keys(args,[]);need(Array.isArray(record.rows)&&record.rows.length<=128,'SUMMARY_ROWS');const rows=record.rows.map(row=>{keys(row,['replica','key','state','detail'],'SUMMARY_ROW_FIELDS');return{replica:name(row.replica,'REPLICA'),key:name(row.key,'FIELD'),state:name(row.state,'STATE'),detail:typeof row.detail==='string'?row.detail:(()=>{throw new Refusal('DETAIL');})()};});return{rows,row_count:rows.length,uses_color_only:false,status_text:rows.length+' consistency rows',manual_review:false,assistive_technology_review:false};};
 OPS.handover_snapshot_projection=(record,args)=>{keys(record,['snapshot','holds','next_owner_label'],'HANDOVER_FIELDS');keys(args,[]);need(plain(record.snapshot),'SNAPSHOT');need(plain(record.snapshot.replicas),'SNAPSHOT_REPLICAS');need(Array.isArray(record.holds)&&record.holds.length<=128,'HOLDS');record.holds.forEach(hold=>name(hold,'HOLD'));name(record.next_owner_label,'NEXT_OWNER');return{snapshot_digest:digest(record.snapshot),replicas:Object.keys(record.snapshot.replicas).sort(),holds:[...record.holds].sort(),next_owner_label:record.next_owner_label,external_transfer:false,acknowledged:false,source_preserved:true};};
 OPS.real_cluster_observation_gate=(record,args)=>{keys(record,['required','rows','review'],'REAL_GATE_FIELDS');keys(args,[]);need(Array.isArray(record.required)&&Array.isArray(record.rows),'REAL_GATE_ARRAY');record.required.forEach(item=>name(item,'REQUIREMENT'));need(new Set(record.required).size===record.required.length,'REQUIREMENT_DUPLICATE');need(record.review===null||typeof record.review==='string','REVIEW_STATE');const missing=[];if(record.rows.length===0)missing.push('real_cluster_rows');if(record.review!=='independent_qualified')missing.push('independent_review');for(const item of record.required)if(!['real_cluster_rows','independent_review'].includes(item))missing.push(item);return{required:[...record.required].sort(),real_rows:record.rows.length,review_state:record.review,missing:[...new Set(missing)].sort(),qualified:false,inference:false,release:false};};
 const requiredByAction={deploy:['competent_operator','live_validation','rollback_evidence'],publish:['rights_holder','affected_people','privacy_review'],delete:['exact_destructive_authority','recoverable_source'],production_identity:['standards_conformant_keys','interoperability','trust_governance'],maori_label:['maori_authority','affected_people','data_governance']};
 OPS.distributed_release_authority_gate=(record,args)=>{keys(record,['target','authority','evidence'],'RELEASE_FIELDS');keys(args,['requested']);name(record.target,'TARGET');need(own(requiredByAction,args.requested),'RELEASE_ACTION');return{requested:args.requested,target:record.target,required:requiredByAction[args.requested],released:false,assertions_verified:false,external_authority:false};};
 const represented=new Set(['accessible_consistency_summary','handover_snapshot_projection']);
 function execute(request){try{keys(request,['op','record','args'],'REQUEST_FIELDS');need(typeof request.op==='string'&&own(OPS,request.op),'UNKNOWN_OPERATION');need(plain(request.record),'RECORD_TYPE');need(plain(request.args),'ARGS_TYPE');validateJSON(request);const value=OPS[request.op](request.record,request.args);validateJSON(value);const disposition=represented.has(request.op)?'represented':request.op==='real_cluster_observation_gate'?'open_gap':request.op==='distributed_release_authority_gate'?'exact_gate':'completed';return envelope(true,value,null,disposition);}catch(error){if(error instanceof Refusal)return envelope(false,null,error.code);throw error;}}
 return{execute,parseStrict,envelope,Refusal,operations:OPS,helpers:base.helpers};
};
