const cp = require('child_process');
const fs = require('fs');
const path = require('path');

const stage = process.argv[2];
const extra = process.argv.slice(3);
const scripts = {
  planning: 'build_planning.py',
  'validate-planning': 'validate_planning.py',
  'validate-staged': 'validate_staged.py',
  x1: 'build_x1.py',
  'validate-x1': 'validate_x1.py',
  x2: 'build_x2.py',
  'validate-x2': 'validate_x2.py',
  final: 'build_final.py',
  'validate-final': 'validate_final.py',
  canonical: 'canonical.py',
};

if (!Object.prototype.hasOwnProperty.call(scripts, stage)) {
  console.error(JSON.stringify({ok: false, error: 'unknown_stage', stage}));
  process.exit(2);
}
const script = path.join(__dirname, scripts[stage]);
if (!fs.existsSync(script)) {
  console.error(JSON.stringify({ok: false, error: 'stage_not_materialized', stage, script: path.basename(script)}));
  process.exit(3);
}
const run = cp.spawnSync('python', [script, ...extra], {
  cwd: path.resolve(__dirname, '../../../..'),
  encoding: 'utf8',
  stdio: ['ignore', 'pipe', 'pipe'],
  env: {...process.env, PYTHONDONTWRITEBYTECODE: '1', PYTHONUTF8: '1'},
});
if (run.stdout) process.stdout.write(run.stdout);
if (run.stderr) process.stderr.write(run.stderr);
process.exit(run.status === null ? 4 : run.status);
