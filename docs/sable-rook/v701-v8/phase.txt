function stable(value) {
  if (Array.isArray(value)) return value.map(stable);
  if (value && typeof value === 'object') {
    const out = {};
    for (const key of Object.keys(value).sort()) out[key] = stable(value[key]);
    return out;
  }
  return value;
}

function canonical(value) {
  return Buffer.from(JSON.stringify(stable(value)), 'utf8');
}

function shaBytes(bytes) {
  return crypto.createHash('sha256').update(bytes).digest('hex');
}

function popcount(mask) {
  let n = 0;
  while (mask) {
    n += mask & 1;
    mask >>>= 1;
  }
  return n;
}

function labels(mask) {
  return GROUND.filter(function(_, index) { return Boolean(mask & (1 << index)); });
}

function maskFromLabels(values) {
  let mask = 0;
  for (const value of values) {
    const index = GROUND.indexOf(value);
    assert(index >= 0, 'unknown_label');
    mask |= 1 << index;
  }
  return mask;
}

function sortedNumeric(values) {
  return Array.from(values).sort(function(a, b) { return a - b; });
}

function familyForPrerequisites(prerequisiteMasks) {
  const feasible = [];
  for (let mask = 0; mask <= FULL; mask += 1) {
    let ok = true;
    for (let element = 0; element < GROUND.length; element += 1) {
      if ((mask & (1 << element)) && (mask & prerequisiteMasks[element]) !== prerequisiteMasks[element]) {
        ok = false;
        break;
      }
    }
    if (ok) feasible.push(mask);
  }
  return feasible;
}

function checkFamily(groundMask, familyValues) {
  const family = new Set(familyValues);
  const failures = [];
  if (!family.has(0)) failures.push('empty_missing');
  if (!family.has(groundMask)) failures.push('ground_missing');
  for (const value of family) {
    if ((value & ~groundMask) !== 0) failures.push('outside_ground:' + value);
    if (value !== 0) {
      let removable = false;
      for (let element = 0; element < GROUND.length; element += 1) {
        const bit = 1 << element;
        if ((groundMask & bit) && (value & bit) && family.has(value ^ bit)) removable = true;
      }
      if (!removable) failures.push('inaccessible:' + value);
    }
  }
  const values = sortedNumeric(family);
  for (const left of values) {
    for (const right of values) {
      if (!family.has(left | right)) failures.push('union:' + left + ':' + right);
    }
  }
  return {valid: failures.length === 0, failures: Array.from(new Set(failures)), feasible_count: family.size};
}

function basisForFamily(familyValues, query) {
  let basis = 0;
  for (const feasible of familyValues) {
    if ((feasible & query) === feasible) basis |= feasible;
  }
  return basis;
}

function extensionsForFamily(familyValues, feasible) {
  const family = new Set(familyValues);
  const out = [];
  for (let element = 0; element < GROUND.length; element += 1) {
    const bit = 1 << element;
    if (!(feasible & bit) && family.has(feasible | bit)) out.push(element);
  }
  return out;
}

function basicWords(profile) {
  const family = new Set(profile.feasible_masks);
  const out = [];
  function walk(mask, word) {
    if (mask === FULL) {
      out.push(word.slice());
      return;
    }
    for (let element = 0; element < GROUND.length; element += 1) {
      const bit = 1 << element;
      if (!(mask & bit) && family.has(mask | bit)) walk(mask | bit, word.concat(GROUND[element]));
    }
  }
  walk(0, []);
  return out;
}

function closedSets(profile) {
  return sortedNumeric(profile.feasible_masks.map(function(mask) { return FULL ^ mask; }));
}

function closure(profile, query) {
  let out = FULL;
  for (const closed of closedSets(profile)) {
    if ((closed & query) === query) out &= closed;
  }
  return out;
}

function extremePoints(profile, query) {
  const out = [];
  for (let element = 0; element < GROUND.length; element += 1) {
    const bit = 1 << element;
    if ((query & bit) && !(closure(profile, query ^ bit) & bit)) out.push(element);
  }
  return out;
}

function antiExchange(profile) {
  let checked = 0;
  const failures = [];
  for (const closed of closedSets(profile)) {
    for (let x = 0; x < GROUND.length; x += 1) {
      const xb = 1 << x;
      if (closed & xb) continue;
      for (let y = 0; y < GROUND.length; y += 1) {
        const yb = 1 << y;
        if (x === y || (closed & yb)) continue;
        checked += 1;
        if ((closure(profile, closed | xb) & yb) && (closure(profile, closed | yb) & xb)) {
          failures.push({closed, x, y});
        }
      }
    }
  }
  return {valid: failures.length === 0, checked, failures};
}

function relabelMask(mask, mapping) {
  let out = 0;
  for (let oldIndex = 0; oldIndex < mapping.length; oldIndex += 1) {
    if (mask & (1 << oldIndex)) out |= 1 << mapping[oldIndex];
  }
  return out;
}

function makeProfiles() {
  const pairs = [];
  for (let child = 1; child < GROUND.length; child += 1) {
    for (let parent = 0; parent < child; parent += 1) pairs.push([parent, child]);
  }
  const profiles = [];
  const signatures = new Set();
  for (let candidate = 1; profiles.length < 30 && candidate < (1 << pairs.length); candidate += 1) {
    const prerequisites = Array(GROUND.length).fill(0);
    const edges = [];
    for (let index = 0; index < pairs.length; index += 1) {
      if (!(candidate & (1 << index))) continue;
      const parent = pairs[index][0];
      const child = pairs[index][1];
      prerequisites[child] |= 1 << parent;
      edges.push([GROUND[parent], GROUND[child]]);
    }
    const feasible = familyForPrerequisites(prerequisites);
    const signature = feasible.join(',');
    if (signatures.has(signature) || feasible.length < 8 || feasible.length > 48) continue;
    signatures.add(signature);
    const ordinal = profiles.length + 1;
    const query = ((candidate * 37) + (ordinal * 11)) & FULL;
    const internalFeasible = feasible.filter(function(mask) { return mask !== FULL; });
    const seedFeasible = internalFeasible[(candidate + ordinal) % internalFeasible.length];
    const rotation = (candidate + ordinal) % GROUND.length;
    const mapping = GROUND.map(function(_, index) { return (index + rotation) % GROUND.length; });
    const profile = {
      profile_id: 'SR7018-AM-' + String(ordinal).padStart(2, '0'),
      ground: GROUND,
      prerequisite_masks: prerequisites,
      prerequisite_edges: edges,
      feasible_masks: feasible,
      query_mask: query,
      seed_feasible_mask: seedFeasible,
      trace_mask: 31,
      delete_element: (candidate + 2 * ordinal) % GROUND.length,
      relabel_mapping: mapping,
      subclass: 'poset_shelling_antimatroid',
      synthetic: true,
      empirical: false,
      authority: false,
    };
    profile.profile_sha256 = shaBytes(canonical(profile));
    profiles.push(profile);
  }
  assert(profiles.length === 30, 'profile_count');
  return profiles;
}

function validateRequest(request, plan) {
  assert(request && typeof request === 'object' && !Array.isArray(request), 'request_object');
  assert(JSON.stringify(Object.keys(request).sort()) === JSON.stringify(['op', 'profile_id', 'profile_sha256']), 'field_set');
  const operation = operationByName(request.op);
  assert(operation, 'operation');
  const profile = plan.profiles.find(function(item) { return item.profile_id === request.profile_id; });
  assert(profile, 'profile');
  assert(request.profile_sha256 === profile.profile_sha256, 'profile_sha256');
  return {operation, profile};
}

function malformedRequest(request, index) {
  const variant = index % 5;
  if (variant === 0) return {op: request.op, profile_id: request.profile_id};
  if (variant === 1) return Object.assign({}, request, {extra: true});
  if (variant === 2) return Object.assign({}, request, {op: 'am_unknown'});
  if (variant === 3) return Object.assign({}, request, {profile_id: 'SR7018-AM-UNKNOWN'});
  return Object.assign({}, request, {profile_sha256: '0'.repeat(64)});
}

function resultFor(operation, profile) {
  const family = profile.feasible_masks;
  const familySet = new Set(family);
  if (operation.name === 'am_validate') {
    return checkFamily(FULL, family);
  }
  if (operation.name === 'am_feasible_sets') {
    return {count: family.length, masks: family, labels: family.map(labels)};
  }
  if (operation.name === 'am_accessibility_witnesses') {
    return {
      witnesses: family.filter(function(mask) { return mask !== 0; }).map(function(mask) {
        return {
          feasible_mask: mask,
          removable: GROUND.filter(function(_, index) {
            const bit = 1 << index;
            return Boolean((mask & bit) && familySet.has(mask ^ bit));
          }),
        };
      }),
    };
  }
  if (operation.name === 'am_union_closure_certificate') {
    let checked = 0;
    const failures = [];
    for (const left of family) {
      for (const right of family) {
        checked += 1;
        if (!familySet.has(left | right)) failures.push([left, right]);
      }
    }
    return {checked, failures, valid: failures.length === 0};
  }
  if (operation.name === 'am_extensions') {
    const extensions = extensionsForFamily(family, profile.seed_feasible_mask);
    return {feasible_mask: profile.seed_feasible_mask, feasible_labels: labels(profile.seed_feasible_mask), extensions: extensions.map(function(index) { return GROUND[index]; })};
  }
  if (operation.name === 'am_rank') {
    const basis = basisForFamily(family, profile.query_mask);
    return {query_mask: profile.query_mask, query_labels: labels(profile.query_mask), rank: popcount(basis)};
  }
  if (operation.name === 'am_basis') {
    const basis = basisForFamily(family, profile.query_mask);
    return {query_mask: profile.query_mask, basis_mask: basis, basis_labels: labels(basis), feasible: familySet.has(basis)};
  }
  if (operation.name === 'am_basic_words') {
    const words = basicWords(profile);
    return {count: words.length, first: words[0], last: words[words.length - 1], all: words};
  }
  if (operation.name === 'am_trace') {
    const traced = sortedNumeric(new Set(family.map(function(mask) { return mask & profile.trace_mask; })));
    return {ground_mask: profile.trace_mask, family: traced, certificate: checkFamily(profile.trace_mask, traced)};
  }
  if (operation.name === 'cg_closure') {
    const closed = closure(profile, profile.query_mask);
    return {query_mask: profile.query_mask, closure_mask: closed, closure_labels: labels(closed), extensive: (closed & profile.query_mask) === profile.query_mask};
  }
  if (operation.name === 'cg_closed_sets') {
    const closed = closedSets(profile);
    return {count: closed.length, masks: closed, labels: closed.map(labels)};
  }
  if (operation.name === 'cg_extreme_points') {
    const closed = closure(profile, profile.query_mask);
    const extreme = extremePoints(profile, closed);
    return {closed_mask: closed, extreme_points: extreme.map(function(index) { return GROUND[index]; })};
  }
  if (operation.name === 'cg_anti_exchange') {
    return antiExchange(profile);
  }
  if (operation.name === 'cg_lattice_meet_join') {
    const closed = closedSets(profile);
    const left = closed[1];
    const right = closed[closed.length - 2];
    const meet = left & right;
    const join = closure(profile, left | right);
    return {left, right, meet, join, meet_closed: closed.includes(meet), join_closed: closed.includes(join)};
  }
  if (operation.name === 'am_relabel_covariance') {
    const mapped = sortedNumeric(family.map(function(mask) { return relabelMask(mask, profile.relabel_mapping); }));
    return {mapping: profile.relabel_mapping, original_count: family.length, mapped_count: mapped.length, certificate: checkFamily(FULL, mapped)};
  }
  if (operation.name === 'am_deletion') {
    const bit = 1 << profile.delete_element;
    const ground = FULL ^ bit;
    const deleted = family.filter(function(mask) { return !(mask & bit); });
    return {deleted_element: GROUND[profile.delete_element], ground_mask: ground, family: deleted, certificate: checkFamily(ground, deleted)};
  }
  if (operation.name === 'am_sensitivity') {
    const bit = 1 << profile.delete_element;
    const deletedFamily = family.filter(function(mask) { return !(mask & bit); });
    const reducedQuery = profile.query_mask & ~bit;
    const originalBasis = basisForFamily(family, profile.query_mask);
    const reducedBasis = basisForFamily(deletedFamily, reducedQuery);
    return {
      deleted_element: GROUND[profile.delete_element],
      original_rank: popcount(originalBasis),
      reduced_rank: popcount(reducedBasis),
      rank_delta: popcount(originalBasis) - popcount(reducedBasis),
      original_basis: originalBasis,
      reduced_basis: reducedBasis,
    };
  }
  if (operation.name === 'am_accessible_explanation') {
    return {
      disposition: 'represented',
      summary: 'The declared family has ' + family.length + ' feasible states and ' + basicWords(profile).length + ' basic words.',
      facts: [
        'Every nonempty feasible state has at least one declared removable element.',
        'Every union of two declared feasible states remains feasible.',
        'The complementary closed sets pass the finite anti-exchange check.',
      ],
      reservations: ['manual review', 'assistive-technology review', 'affected-user review', 'Maori-language review'],
    };
  }
  if (operation.name === 'am_real_learning_gap') {
    return {
      disposition: 'open_gap',
      real_rows: 0,
      participants: 0,
      institutions: 0,
      calibrated_prerequisites: 0,
      effectiveness_estimates: 0,
      reason: 'No real learning, curriculum, measurement, consent, safety, or independent-review evidence exists.',
    };
  }
  if (operation.name === 'am_authority_gate') {
    return {
      disposition: 'exact_gate',
      authorized_actions: 0,
      held: true,
      required: ['competent professional authority', 'affected-party participation', 'privacy and accessibility review', 'legal and cultural review where applicable', 'Maori authority for Maori concepts and data'],
    };
  }
  throw new Error('unknown_operation');
}

function predecessorGate(expectedCommits, expectedChanged) {
  assert(gitText('branch', '--show-current') === BRANCH, 'gate_branch');
  assert(gitText('status', '--porcelain=v1', '-uall') === '', 'gate_clean');
  const head = gitText('rev-parse', 'HEAD');
  const upstreamName = gitText('rev-parse', '--abbrev-ref', '--symbolic-full-name', '@{u}');
  const upstream = gitText('rev-parse', '@{u}');
  const tracking = gitText('rev-parse', 'refs/remotes/origin/' + BRANCH);
  const liveRows = gitText('ls-remote', '--heads', 'origin', 'refs/heads/' + BRANCH).split(/\r?\n/).filter(Boolean);
  assert(liveRows.length === 1, 'gate_live_unique');
  const live = liveRows[0].split(/\s+/)[0];
  const divergence = gitText('rev-list', '--left-right', '--count', 'HEAD...@{u}');
  assert(head === upstream && head === tracking && head === live && divergence === '0\t0', 'gate_four_way');
  const commits = gitText('rev-list', '--reverse', SOURCE + '..' + head).split(/\r?\n/).filter(Boolean);
  const changed = gitText('diff', '--name-only', SOURCE + '..' + head).split(/\r?\n/).filter(Boolean);
  assert(commits.length === expectedCommits, 'gate_commit_count');
  assert(changed.length === expectedChanged && changed.every(function(file) { return file.startsWith(PREFIX); }), 'gate_changed');
  assert(upstreamName === 'origin/' + BRANCH, 'gate_upstream_name');
  return {head, upstream, tracking, live, divergence, commits, changed};
}

function privacyHits(files) {
  const patterns = [
    {class: 'raw_uuid', re: /\b[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}\b/ig},
    {class: 'private_path', re: new RegExp(['(?:[A-Z]:', '\\\\', '(?:Users|GHC-Archives))'].join(''), 'ig')},
    {class: 'credential', re: /\b(?:api[_-]?key|password|secret|token)\s*[:=]\s*["'][^"']+["']/ig},
    {class: 'private_route', re: new RegExp(['(?:plugin', ':', '\\/\\/', '|codex', ':', '\\/\\/', 'threads', '\\/)'].join(''), 'ig')},
    {
      class: ['session', 'stream'].join('_'),
      re: new RegExp(
        '\\b(?:' +
        ['source', 'thread', 'id'].join('_') + '|' +
        ['private', 'callable', 'identifier'].join('_') + '|' +
        ['session', 'stream'].join('_') +
        ')\\b',
        'ig'
      ),
    },
  ];
  const hits = [];
  for (const file of files) {
    const text = fs.readFileSync(file, 'utf8');
    for (const pattern of patterns) {
      const matches = text.match(pattern.re) || [];
      if (matches.length) hits.push({file: relative(file), class: pattern.class, count: matches.length});
    }
  }
  return hits;
}

  add('C14', 'strict JSON parse count', [plan, x1, x2, final].every(Boolean), 4);

  add('C16', 'outcome counts', plan.outcomes.completed === 510 && plan.outcomes.represented === 30 && plan.outcomes.open_gap === 30 && plan.outcomes.exact_gate === 30, plan.outcomes);
  for (const pair of [['x1', x1], ['x2', x2]]) {
    const stageName = pair[0];
    const stage = pair[1];
    add(stageName + '-01', stageName + ' contracts', stage.contracts.length === 300);
    add(stageName + '-02', stageName + ' safe', stage.portfolios.safe_checks === 300);
    add(stageName + '-03', stageName + ' candidate', stage.portfolios.malformed_candidate_subjects === 300);
    add(stageName + '-04', stageName + ' refusal', stage.portfolios.refusal_checks === 300);
    add(stageName + '-05', stageName + ' copy reference', stage.portfolios.copy_from_reference_checks === 300);
    add(stageName + '-06', stageName + ' cleanup', stage.portfolios.clean_fix_refine_reviews === 300);
    add(stageName + '-07', stageName + ' skills', stage.skills.length === 10 && stage.skills.every(function(skill) { return skill.used && skill.validation_passed; }));
    add(stageName + '-08', stageName + ' runners', stage.runners.length === 5 && stage.runners.every(function(runner) { return runner.used && runner.positive_smoke.pass && runner.refusal_check.pass; }));
    add(stageName + '-09', stageName + ' latch', stage.session.invocations === 1 && stage.session.successes === 1 && stage.session.replays === 0);
  }
  add('C17', 'twenty additional tests', x2.tests.length === 20 && x2.tests.every(function(test) { return test.pass; }));
  add('C18', 'five nonphysical models', x2.models.length === 5 && x2.models.every(function(model) { return model.points.length === 6 && !model.physical && !model.empirical; }));
  add('C19', 'held approval packets', plan.approval_packets.exact.length === 50 && plan.approval_packets.blocked.length === 30 && final.approval_packets.executed === 0);
  add('C20', 'logical skills and runners', plan.skills.length === 20 && plan.runners.length === 10 && final.skills.global_installs === 0);
  const flow = final.method_flow;
  const passing = flow.witnesses.filter(function(witness) { return witness.result === 'pass'; }).reduce(function(sum, witness) { return sum + witness.atomic_count; }, 0);
  const failed = flow.witnesses.filter(function(witness) { return witness.result === 'fail'; }).reduce(function(sum, witness) { return sum + witness.atomic_count; }, 0);
  add('C21', 'method count', flow.methods.length === flow.counts.methods, {actual: flow.methods.length, declared: flow.counts.methods});
  add('C22', 'passing witness arithmetic', passing === flow.counts.passing_witnesses, {passing, declared: flow.counts.passing_witnesses});
  add('C23', 'failed witness arithmetic', failed === flow.counts.failed_witnesses, {failed, declared: flow.counts.failed_witnesses});
  add('C24', 'negative uniqueness', new Set(flow.retained_negative_ids).size === flow.retained_negative_ids.length && flow.retained_negative_ids.length === flow.counts.negatives);
  add('C25', 'effective arithmetic', final.effective.failed_witnesses + final.effective.passing_witnesses === final.effective.witnesses);
  add('C26', 'source activation baseline folded once', final.source_baseline_fold_count === 1);
  add('C27', 'baton modules and length', final.handoff.modules.length === 13 && final.handoff.words >= 2000 && final.handoff.words <= 100000, final.handoff.words);
  add('C28', 'baton literal EOF', final.handoff.modules[final.handoff.modules.length - 1].endsWith(final.handoff.literal_eof));
  add('C29', 'exact terminal token and verdict', final.terminal_verdict === 'NOT_READY_FOR_STAGE_20' && final.handoff.modules.join('\n').includes('NOT_READY_FOR_STAGE_20'));
  const hits = privacyHits(files);
  add('C30', 'zero confirmed five-class privacy hits', hits.length === 0, hits);
  const sourceText = fs.readFileSync(path.join(ROOT, 'phase.txt'), 'utf8');
  const securityFindings = [];
  if (/shell\s*:\s*true/.test(sourceText)) securityFindings.push('shell_true');
  if (/\beval\s*\(/.test(sourceText)) securityFindings.push('eval');
  add('C31', 'zero bounded changed-source security findings', securityFindings.length === 0, securityFindings);
  add('C32', 'predecessor validators not replayed', !plan.source.source_terminal_validator_replayed_by_sable && !plan.source.source_domain_session_replayed_by_sable);
  add('C33', 'stage sessions not replayed', x1.session.replays === 0 && x2.session.replays === 0);
  add('C34', 'resource and format boundary', final.resource_closeout.tracked_after === 1423 && final.resource_closeout.global_installs === 0 && final.resource_closeout.shared_mutations === 0 && changed.concat(preflight ? [PREFIX + 'final.json'] : []).every(function(file) { return /\.(?:txt|json)$/.test(file); }));
  add('C35', 'v15 control bindings', plan.controls.workflow_sha256 === CONTROL_HASHES.workflow_v15 && plan.controls.roster_sha256 === CONTROL_HASHES.roster_v15 && plan.controls.authorization_sha256 === CONTROL_HASHES.authorization_v15);
  add('C36', 'source terminal validation retained', final.source_validation.invocations === 1 && final.source_validation.successes === 1 && final.source_validation.replays === 0);
  add('C37', 'source delivery retained separately', plan.source.delivery_state === 'SENT_ONCE_ACKNOWLEDGED' && final.route.state === 'PREPARED_NOT_SENT');
  add('C38', 'novelty collision retained', final.operational_failures.some(function(failure) { return failure.id === 'SR7018-OP-N001' && failure.original_success_credit === 0; }));
  add('C39', 'antimatroid novelty replacement', plan.novelty_audit.repository_hits === 0 && plan.novelty_audit.shared_skill_hits === 0);
  const failures = checks.filter(function(check) { return !check.pass; });
  return {head, upstream, tracking, live, divergence, tracked, changed, checks, failures, privacy_hits: hits, security_findings: securityFindings};
}

function validateCommand() {
  const finalExists = fs.existsSync(path.join(ROOT, 'final.json'));
  if (!finalExists) {
    const plan = readJson(path.join(ROOT, 'plan.json'));
    const output = {stage: 'plan', checks: makePlanChecks(plan), engine_fixed: bindFile(path.join(ROOT, 'phase.txt')).sha256 === plan.engine.sha256};
    if (fs.existsSync(path.join(ROOT, 'x1.json'))) output.x1 = readJson(path.join(ROOT, 'x1.json')).checks;
    if (fs.existsSync(path.join(ROOT, 'x2.json'))) output.x2 = readJson(path.join(ROOT, 'x2.json')).checks;
    assert(output.engine_fixed && output.checks.every(function(check) { return check.pass; }) && (!output.x1 || Object.values(output.x1).every(Boolean)) && (!output.x2 || Object.values(output.x2).every(Boolean)), 'validation');
    console.log(JSON.stringify(output));
    return;
  }
  const validation = validateRepository({remote: false, preflight: true});
  if (validation.failures.length) throw new Error('validation_failures:' + JSON.stringify(validation.failures));
  console.log(JSON.stringify({state: 'VALID_OWNER_WORKTREE_PREFLIGHT', checks: validation.checks.length, tracked: validation.tracked, changed: validation.changed.length, privacy_hits: validation.privacy_hits.length, security_findings: validation.security_findings.length}));
}

function exportFlow() {
  const final = readJson(path.join(ROOT, 'final.json'));
  const file = path.join(BANK, 'method-flow.json');
  assert(!fs.existsSync(file), 'flow_export_exclusive');
  writeExclusive(file, final.method_flow);
  console.log(JSON.stringify({state: 'METHOD_FLOW_EXPORTED', sha256: bindExternal(file).sha256, methods: final.method_flow.counts.methods, witnesses: final.method_flow.counts.witnesses}));
}

function canonicalRun(exactFinal) {
  ensureDir(path.join(BANK, 'canonical'));
  const latchFile = path.join(BANK, 'canonical', 'invocation-latch.json');
  const payloadFile = path.join(BANK, 'canonical', 'canonical-payload.json');
  const receiptFile = path.join(BANK, 'canonical', 'canonical-receipt.json');
  assert(!fs.existsSync(latchFile) && !fs.existsSync(payloadFile) && !fs.existsSync(receiptFile), 'canonical_exclusive_latch');
  writeExclusive(latchFile, {schema: 'ghc.family.canonical-invocation-latch.v15', owner: 'Sable Rook', phase: 'v701-v8', invoked_utc: new Date().toISOString(), successful_replays: 0});
  let validation;
  try {
    validation = validateRepository({remote: true, exactFinal});
  } catch (error) {
    writeExclusive(path.join(BANK, 'canonical', 'canonical-failure.json'), {schema: 'ghc.family.owner-canonical-failure.v15', error: error.message, successful_invocations: 0, successful_replays: 0, repository_mutated: false, retained: true});
    throw error;
  }
  if (validation.failures.length) {
    writeExclusive(path.join(BANK, 'canonical', 'canonical-failure.json'), {schema: 'ghc.family.owner-canonical-failure.v15', failures: validation.failures, successful_invocations: 0, successful_replays: 0, repository_mutated: false, retained: true});
    throw new Error('canonical_failures:' + JSON.stringify(validation.failures));
  }
  const final = readJson(path.join(ROOT, 'final.json'));
  const payload = {
    schema: 'ghc.family.owner-canonical-payload.v15',
    state: 'VALID_EXACT_FINAL_OWNER_SCOPED_CANONICAL',
    owner: 'Sable Rook',
    phase: 'v701-v8',
    branch: BRANCH,
    source: SOURCE,
    head: validation.head,
    check_count: validation.checks.length,
    checks: validation.checks,
    owner_files: validation.changed.length,
    tracked_files: validation.tracked,
    strict_json_parses: 4,
    manifest_replays: final.manifest.entries.length,
    privacy_hits: validation.privacy_hits,
    security_findings: validation.security_findings,
    successful_invocations: 1,
    successful_replays: 0,
    predecessor_terminal_replay: false,
    predecessor_domain_replay: false,
    x1_session_replay: false,
    x2_session_replay: false,
    same_owner_only: true,
    independent_reproduction: false,
    full_repository_suite: false,
    repository_mutated: false,
    terminal_verdict: 'NOT_READY_FOR_STAGE_20',
    boundary: BOUNDARY,
  };
  const payloadBytes = canonical(payload);
  const payloadSha = shaBytes(payloadBytes);
  fs.writeFileSync(payloadFile, payloadBytes, {flag: 'wx'});
  const receipt = {schema: 'ghc.family.owner-canonical-receipt.v15', payload, payload_sha256: payloadSha, invoked_utc: new Date().toISOString(), repository_mutated: false};
  writeExclusive(receiptFile, receipt);
  console.log(JSON.stringify({state: payload.state, check_count: payload.check_count, payload_sha256: payloadSha, receipt_sha256: bindExternal(receiptFile).sha256, head: payload.head, tracked_files: payload.tracked_files}));
}

const command = process.argv[process.argv.length - 1];
if (command === 'plan') makePlan();
else if (command === 'x1' || command === 'x2') runStage(command);
else if (command === 'final') makeFinal();
else if (command === 'validate') validateCommand();
else if (command === 'export-flow') exportFlow();
else if (command && command.startsWith('canonical=')) canonicalRun(command.slice('canonical='.length));
else throw new Error('usage: plan | x1 | x2 | final | validate | export-flow | canonical=<exact-final>');
