def result_summary(op,r):
 if op=='word_lengths':return 'Lengths are '+str(r['lengths'])+', ranging from '+str(r['minimum'])+' to '+str(r['maximum'])+' bits.'
 if op in ['prefix_conflicts','suffix_conflicts']:return 'The exact conflict count is '+str(len(r['conflicts']))+'. Relations and residuals: '+stable(r['conflicts'])+'.'
 if op=='kraft_measure':return 'The exact Kraft sum is '+r['numerator']+'/'+r['denominator']+', '+r['relation_to_one']+' one. This measure alone does not decide this codebook\'s unique decodability.'
 if op=='trie_vertices':return 'The trie contains '+str(len(r['vertices']))+' exact prefixes. Terminal bindings are '+stable(r['vertices'])+'.'
 if op=='trie_edges':return 'The trie contains '+str(len(r['edges']))+' parent-bit-child edges: '+stable(r['edges'])+'.'
 if op=='trie_degrees':return 'There are '+str(r['vertices'])+' vertices and '+str(r['edges'])+' edges. Counts for outdegree zero, one and two are '+str(r['histogram'])+'.'
 if op in ['common_prefix_lengths','common_suffix_lengths']:return 'The complete ordered length matrix is '+stable(r['matrix'])+'. Diagonal entries retain each word\'s own length.'
 if op=='pair_collisions':return 'There are '+str(len(r['collisions']))+' colliding encoded strings among two-symbol messages: '+stable(r['collisions'])+'. No broader absence claim follows from this bounded comparison.'
 if op=='residual_closure':return 'The initial residuals are '+str(r['initial'])+'; the closed residual set is '+str(r['residuals'])+'. Empty-residual reachability is '+str(r['epsilon_reachable']).lower()+'.'
 if op=='unique_decodability':return 'The complete finite residual criterion returns unique='+str(r['unique']).lower()+', using '+str(r['residual_count'])+' reachable residuals. This is a mathematical code property only.'
 if op=='decode_probe':return 'Probe '+r['probe']+' has '+r['count']+' exact parses. The retained first two index messages are '+stable(r['first_two'])+'.'
 if op=='encode_probe':return 'Message '+str(r['sequence'])+' encodes to '+r['bits']+'. Exact bit boundaries are '+str(r['boundaries'])+'.'
 if op=='length_counts':return 'Message counts at bit lengths zero through '+str(r['limit'])+' are '+str(r['counts'])+'. These count symbol messages, including coincident bit encodings.'
 if op=='shortest_ambiguity':
  return ('The shortest witness through '+str(r['search_bound'])+' bits is '+r['bits']+', decoded as '+stable(r['first_two'])+'.' if r['found'] else 'No ambiguity was found through '+str(r['search_bound'])+' bits; this operation makes no global uniqueness claim.')
 if op=='kraft_completion':return ('Completion at depth '+str(r['depth'])+' adds '+str(len(r['added_words']))+' words: '+str(r['added_words'])+'. The resulting exact Kraft sum is one.' if r['applicable'] else 'Completion is inapplicable because the supplied code has prefix conflicts. The exact conflicts are '+stable(r['conflicts'])+'.')

 if op=='external_decoder_gap':return 'Separate implementation, separate operator and external evaluation are absent. The codebook digest is '+r['codebook_sha256']+'. A local pass does not supply these missing inputs.'
