# Python 3. Attributable stage execution; successful stage latches forbid replay.
from pathlib import Path
from datetime import datetime, timezone
import collections, copy, hashlib, json, runpy, subprocess, sys

ROOT=Path(__file__).resolve().parent; REPO=ROOT.parents[2]
MODE,STAGE=sys.argv[1:3]; BANK=Path(sys.argv[3])
assert STAGE in ['x1','x2']
PREFIX='docs/orren-pike/v699-v3/'
BRANCH='codex/GHC-Family/orren-pike-v699-v3-full-tools'
PLAN=json.loads((ROOT/'planning/plan.json').read_text(encoding='utf-8'))
BOUNDARY=PLAN['boundary']; SCOPE='finite_binary_codebook_only'
ROWS=[x for x in PLAN['contracts'] if x['stage']==STAGE]
GROUPS=['word_prefix','suffix_kraft','trie_topology','trie_comparison','suffix_collision',
        'residual_decoding','message_codec','bounded_language','capacity_scene','evidence_reservations']
def stable(v):return json.dumps(v,ensure_ascii=False,sort_keys=True,separators=(',',':'))
def sha(b):return hashlib.sha256(b).hexdigest()
def h(v):return sha(stable(v).encode())
def read(p):return json.loads(p.read_text(encoding='utf-8'))
def write(p,v):
 p.parent.mkdir(parents=True,exist_ok=True)
 with p.open('x',encoding='utf-8',newline='\n') as f:f.write((v.rstrip() if isinstance(v,str) else stable(v))+'\n')
def git(*a):return subprocess.check_output(['git','-C',str(REPO),*a],stderr=subprocess.PIPE).decode().strip()
def error(code):return {'ok':False,'error':code,'outcome':'exact_gate','scope':SCOPE,'independent_reproduction':False}
def caller(path,stdin,extra=()):
 p=subprocess.run(['node','-e','require(require("node:path").resolve(process.argv[1])).cli()',str(path),*extra],
                  input=stdin,capture_output=True,text=True,encoding='utf-8',timeout=15)
 return {'status':p.returncode,'stdout':p.stdout,'stderr':p.stderr,'signal':None,'timeout':False}
previous=read(BANK/('plan-published.json' if STAGE=='x1' else 'x1-published.json'))
assert git('rev-parse','HEAD')==previous['commit'] and git('branch','--show-current')==BRANCH
assert sha((ROOT/'planning/plan.json').read_bytes())==read(BANK/'plan-built.json')['plan_sha256']
assert not git('diff','--','docs/orren-pike/v699-v3/planning')
if STAGE=='x1':assert not (ROOT/'x2').exists()
else:assert (BANK/'x1-success.json').exists()
assert not (BANK/(STAGE+'-success.json')).exists(), 'successful_stage_replay_forbidden'
ops=[x for x in PLAN['operations'] if x['stage']==STAGE]
if MODE=='prepare':
 validate=runpy.run_path(sys.argv[4])['validate_skill']
 capabilities={'schema':'orren.local-capabilities.v1','stage':STAGE,'skills':[],'runners':[],'global_installs':0,'boundary':BOUNDARY}
 for op in ops:
  name='ghc-family-codebook-'+op['operation'].replace('_','-')
  skill=ROOT/'skills'/op['operation']/'SKILL.md'
  body=('---\nname: '+name+'\ndescription: '+op['hypothesis']+' Use only the bounded binary-codebook request contract.\n---\n\n'
    '# '+op['operation'].replace('_',' ').capitalize()+'\n\n'
    'Use this phase-local guide when the exact responsibility is '+op['operation']+'. '+op['hypothesis']+'\n\n'
    'The request has exactly op, codewords, probe, sequence and limit. Admit one to six distinct nonempty binary words of at most six bits, '
    'a binary probe of at most eighteen bits, one to three valid symbol indices and an integer search limit from one through twelve. '
    'Reject unknown fields, duplicate JSON keys, duplicate words, empty words and undeclared indices. The runner rejects operations outside its pair.\n\n'
    'Read the matching frozen planning contract, submit its complete request to the paired local runner, and compare the entire result and caller envelope. '
    'Record exit status, stdout, stderr, signal and timeout. Retain every malformed subject at zero original success credit beside its separate refusal. '
    'Preserve input order and the original expected output. Correct only a separately identified copy.\n\n'
    'Expected core disposition: '+op['outcome']+'. No finite search without a witness establishes a global decoding property. '
    'A Kraft sum at most one is insufficient to certify the supplied codebook. Declared coordinates have no physical interpretation. '
    'A passing caller check is bounded same-owner evidence. Successful session tranches and canonical aggregates must never be replayed.\n\n'+BOUNDARY+'\n')
  write(skill,body);valid,message=validate(skill.parent);assert valid,(name,message)
  capabilities['skills'].append({'operation':op['operation'],'path':skill.relative_to(REPO).as_posix(),
    'sha256':sha(skill.read_bytes()),'quick_validation':{'valid':valid,'message':message},'complete_read_required_before_use':True})
 for offset in range(0,10,2):
  index=(0 if STAGE=='x1' else 5)+offset//2; allowed=[x['operation'] for x in ops[offset:offset+2]]
  runner=ROOT/'runners'/('ghc_family_codebook_'+GROUPS[index]+'_runner.txt')
  write(runner,"// Node.js CommonJS. Phase-local caller; no historical caller changes.\n'use strict';\nconst engine=require('../"+STAGE+"/runtime.txt');\nmodule.exports.cli=()=>engine.cli("+json.dumps(allowed)+");\n")
  capabilities['runners'].append({'id':'OP6993-R%02d'%(index+1),'path':runner.relative_to(REPO).as_posix(),
    'sha256':sha(runner.read_bytes()),'operations':allowed})
 write(ROOT/STAGE/'capabilities.json',capabilities)
 print(json.dumps({'stage':STAGE,'prepared_skills':10,'quick_validated':10,'prepared_runners':5,'domain_executions':0}))
elif MODE=='execute':
 assert len(ROWS)==150
 write(BANK/(STAGE+'-invocation.json'),{'stage':STAGE,'state':'invoked_once','source_publication':previous['commit'],
     'observed_at':datetime.now(timezone.utc).isoformat(),'successes':0,'replays':0})
 try:
  capabilities=read(ROOT/STAGE/'capabilities.json')
  requests=[]
  for row in ROWS:requests.extend([row['request'],row['alternate'],*[n['request'] for n in row['negative_subjects']]])
  cp=subprocess.run(['node','-e','require(require("node:path").resolve(process.argv[1]))',str(ROOT/'session-probe.txt'),STAGE],
      input=stable(requests),capture_output=True,text=True,encoding='utf-8',timeout=60)
  envelope={'status':cp.returncode,'stdout':cp.stdout,'stderr':cp.stderr,'signal':None,'timeout':False}
  write(BANK/(STAGE+'-probe-envelope.json'),envelope)
  assert cp.returncode==0 and cp.stderr=='',{'status':cp.returncode,'stderr':cp.stderr[:1000]}
  observed=json.loads(cp.stdout)['results'];assert len(observed)==600
  witnesses=[];records=[];negatives=[]
  def witness(method,procedure,result,expected,actual,negative=()):
   wid='OP6993-'+STAGE.upper()+'-W%05d'%(len(witnesses)+1)
   witnesses.append({'witness_id':wid,'method_id':method,'procedure':procedure,'scope':STAGE,
    'expected':expected,'observed':actual,'result':result,'same_owner_only':True,'independent_reproduction':False,
    'retained_negative_ids':list(negative),'boundary':'Same-owner synthetic only; protected gates remain closed.'})
   return wid
  for i,row in enumerate(ROWS):
   method='OP6993-M-'+row['operation'];actual=observed[4*i:4*i+4]
   for j,key in enumerate(['expected','alternate_expected']):
    assert actual[j]['output']==row[key] and actual[j]['input_preserved'],{'id':row['id'],'check':key,'observed':actual[j]}
    witness(method,'exact_output_and_input_preservation','pass',h(row[key]),h(actual[j]['output']))
   for j,n in enumerate(row['negative_subjects']):
    got=actual[j+2];assert got['output']==error(n['error']) and got['input_preserved'],{'id':n['id'],'observed':got}
    negatives.append({'id':n['id'],'kind':'malformed_input','proposal':row['id'],'original_success_credit':0})
    witness(method,'malformed_subject','fail','admissible_request',n['error'],[n['id']])
    witness(method,'strict_refusal','pass',error(n['error']),got['output'],[n['id']])
   cloned=json.loads(stable(actual[0]['output']));assert cloned==actual[0]['output']
   witness(method,'clean_reference_copy','pass',h(actual[0]['output']),h(cloned))
   altered=copy.deepcopy(actual[0]['output']);altered['outcome']='unsupported_claim'
   assert altered!=row['expected'];nid=row['id']+'-CFR-N1'
   negatives.append({'id':nid,'kind':'altered_output','proposal':row['id'],'original_success_credit':0})
   witness(method,'altered_output_subject','fail',h(row['expected']),h(altered),[nid])
   repaired=copy.deepcopy(altered);repaired['outcome']=row['expected']['outcome'];assert repaired==row['expected']
   witness(method,'fix_refine_detached_copy','pass',h(row['expected']),h(repaired),[nid])
   records.append({'id':row['id'],'operation':row['operation'],'profile':row['profile'],'request_sha256':row['request_sha256'],
    'output':actual[0]['output'],'alternate_output':actual[1]['output'],'input_preserved':True,
    'malformed_outputs':[a['output'] for a in actual[2:]],'altered_output':altered,'repaired_output_sha256':h(repaired),
    'outcome':actual[0]['output']['outcome']})
  runner_records=[]
  for index,runner in enumerate(capabilities['runners']):
   method=runner['id'];path=REPO/runner['path']
   for op in runner['operations']:
    row=next(x for x in ROWS if x['operation']==op)
    skill=next(x for x in capabilities['skills'] if x['operation']==op)
    assert sha((REPO/skill['path']).read_bytes())==skill['sha256']
    result=caller(path,stable(row['request']))
    assert result['status']==0 and result['stderr']=='' and json.loads(result['stdout'])==row['expected'],{'runner':runner['id'],'op':op,'result':result}
    wid=witness(method,'paired_runner_and_skill_use','pass',row['expected_sha256'],h(json.loads(result['stdout'])))
    runner_records.append({'runner':runner['id'],'operation':op,'guide_sha256':skill['sha256'],'witness_id':wid,'caller':result})
   row=next(x for x in ROWS if x['operation']==runner['operations'][0]);q=copy.deepcopy(row['request'])
   if index==0:q['undeclared']=True;wire=stable(q);code='field_set'
   elif index==1:wire=stable(q)[:-1]+',"op":'+json.dumps(q['op'])+'}';code='duplicate_key'
   elif index==2:q['op']=next(o['operation'] for o in ops if o['operation'] not in runner['operations']);wire=stable(q);code='runner_operation'
   elif index==3:wire=' '*65537+stable(q);code='json_bytes'
   else:q['codewords'][0]='';wire=stable(q);code='codeword_value'
   result=caller(path,wire);assert result['status']==2 and result['stderr']=='' and json.loads(result['stdout'])==error(code),{'runner':runner['id'],'code':code,'result':result}
   nid=runner['id']+'-N1';negatives.append({'id':nid,'kind':'malformed_caller_frame','original_success_credit':0})
   witness(method,'malformed_caller_subject','fail','admissible_caller_frame',code,[nid])
   wid=witness(method,'caller_refusal','pass',error(code),json.loads(result['stdout']),[nid])
   runner_records.append({'runner':runner['id'],'negative_id':nid,'witness_id':wid,'caller':result,
                         'submitted_bytes':len(wire.encode()),'submitted_sha256':sha(wire.encode())})
  counts={'contracts':150,'safe_checks':300,'malformed_candidates':300,'candidate_success_credit':0,'refusal_checks':300,
    'clean_fix_refine_checks':300,'altered_output_subjects':150,'skill_uses':10,'runner_uses':5,'caller_checks':15,
    'witnesses':len(witnesses),'passing_witnesses':sum(w['result']=='pass' for w in witnesses),
    'failed_witnesses':sum(w['result']=='fail' for w in witnesses),'negatives':len(negatives)}
  assert counts['passing_witnesses']==915 and counts['failed_witnesses']==455 and counts['witnesses']==1370
  artifact={'schema':'orren.stage-execution.v1','owner':'Orren Pike','phase':'v699-v3','stage':STAGE,
    'previous_publication':previous,'plan_sha256':sha((ROOT/'planning/plan.json').read_bytes()),
    'runtime_sha256':sha((ROOT/STAGE/'runtime.txt').read_bytes()),'probe_sha256':sha((ROOT/'session-probe.txt').read_bytes()),
    'records':records,'runner_records':runner_records,'witnesses':witnesses,'retained_negatives':negatives,'counts':counts,
    'outcomes':dict(collections.Counter(r['outcome'] for r in records)),'invocations':1,'successes':1,'replays':0,
    'shared_witness_credit':'Each paired skill/runner call is counted once.','boundary':BOUNDARY}
  write(ROOT/STAGE/'execution.json',artifact)
  write(BANK/(STAGE+'-success.json'),{'stage':STAGE,'state':'success_latched','invocations':1,'successes':1,'replays':0,
      'execution_sha256':sha((ROOT/STAGE/'execution.json').read_bytes()),'runtime_sha256':artifact['runtime_sha256']})
  print(json.dumps({'stage':STAGE,'status':'SUCCESS_LATCHED_NO_REPLAY','counts':counts,'outcomes':artifact['outcomes']}))
 except Exception as e:
  write(BANK/(STAGE+'-failed-attempt-1.json'),{'stage':STAGE,'status':'FAILED_ZERO_SUCCESS_CREDIT','error':str(e),
      'partial_results_retained':(BANK/(STAGE+'-probe-envelope.json')).exists(),'successes':0,'replays':0})
  raise
else:raise ValueError('mode')
