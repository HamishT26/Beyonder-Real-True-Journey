def raw(cmd,input=None,cwd=None):
    r=subprocess.run(cmd,input=input,stdout=subprocess.PIPE,stderr=subprocess.PIPE,cwd=cwd)
    if r.returncode: raise RuntimeError('Child failed '+str(r.returncode)+': '+r.stderr.decode('utf8','replace')[:1500])

def privacy_scan(files):
    patterns={'raw_identifier':r'\b[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}\b','private_local_path':r'(?i)\b[A-Za-z]:[\\/](?:Users|GHC-Archives|Program Files)\b','private_uri':r'(?i)\b(?:codex|app|plugin)://(?:threads|tasks|sessions|agent)','delegation_markup':r'<(?:codex_delegation|source_thread_id)>','credential_assignment':r'''(?i)\b(?:api[_-]?key|access[_-]?token|password|secret)\b\s*[:=]\s*["'][A-Za-z0-9._-]{8,}["']'''}
    hits=[]
    for name,b in files:
        s=b.decode('utf8')
        for cls,pat in patterns.items():
            if re.search(pat,s):hits.append({'path':name,'class':cls})
    return hits
