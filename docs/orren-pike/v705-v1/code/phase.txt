'use strict';
const cp=require('child_process'),path=require('path');
if(process.env.GHC_MODEL_GUARD!=='v16')throw new Error('Use the registered phase launcher.');
if(!process.env.GHC_PYTHON)throw new Error('GHC_PYTHON required.');
const r=cp.spawnSync(process.env.GHC_PYTHON,['-X','utf8',path.join(__dirname,'phase-engine.txt'),...process.argv.slice(2)],{stdio:'inherit',env:{...process.env,PYTHONDONTWRITEBYTECODE:'1'},shell:false});
if(r.error)throw r.error;process.exitCode=r.status===null?1:r.status;
