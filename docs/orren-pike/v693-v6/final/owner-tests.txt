'use strict';
const fs=require('fs'),path=require('path'),cp=require('child_process'),api=require('../x2/aggregation-extension.txt');
const root=path.join(__dirname,'..'),read=n=>JSON.parse(fs.readFileSync(path.join(root,n),'utf8'));
function run(){const checks=[],check=(name,pass)=>checks.push({name,pass:Boolean(pass)}),defs=read('plan/proposals.json').proposals,observations=[...read('x1/observations.json').observations,...read('x2/observations.json').observations];
for(const p of defs){const input=structuredClone(p.request),before=api.canonical(input),observed=api.run(input);check(p.proposal_id+' full envelope',api.canonical(observed)===api.canonical(p.expected));check(p.proposal_id+' input preserved',before===api.canonical(input));const old=observations.find(x=>x.proposal_id===p.proposal_id);check(p.proposal_id+' sealed observation and ceiling',old&&api.canonical(old.observed)===api.canonical(observed)&&old.observed_outcome===p.expected_execution_disposition);for(const n of p.adverse){const subject=structuredClone(n.request),snapshot=api.canonical(subject),out=api.run(subject);check(n.negative_id+' refusal',api.canonical(out)===api.canonical(n.expected)&&snapshot===api.canonical(subject));}}
for(const stage of ['x1','x2']){for(const g of read(stage+'/additional-refusal-definitions.json').cases){let observed;try{observed=api.run(api.parseStrict(g.raw));}catch(e){observed={ok:false,value:null,error:e.message};}check(g.negative_id+' additional refusal',api.canonical(observed)===api.canonical(g.expected));}for(const r of read(stage+'/runner-validation.json').runners){const p=defs.find(p=>p.proposal_id===r.runs[0].proposal_id),file=path.join(root,stage,'runners',r.runner);for(const [tag,input,expected,exit] of [['positive',p.request,p.expected,0],['refusal',p.adverse[0].request,p.adverse[0].expected,2]]){const run=cp.spawnSync(process.execPath,[file],{input:JSON.stringify(input),encoding:'utf8',timeout:20000});let value;try{value=JSON.parse(run.stdout);}catch{value=null;}check(r.runner+' '+tag,run.status===exit&&api.canonical(value)===api.canonical(expected));}}}
const algebraRun=cp.spawnSync(process.execPath,[path.join(root,'x2/algebra-validator.txt')],{encoding:'utf8',timeout:30000,maxBuffer:4*1024*1024});let algebra;try{algebra=JSON.parse(algebraRun.stdout);}catch{algebra=null;}check('algebra process and declared case coverage',algebraRun.status===0&&algebra?.cases===135&&algebra?.checks_total===405);if(algebra)for(const row of algebra.rows)for(const [name,pass] of Object.entries(row.checks))check(row.case_id+' '+name,pass);
const model={owner_scope:true,clean:true,divergence:[0,0],fresh_equal:true,namespace_absent:true,attempts:0};const preflight=v=>v.owner_scope===true&&v.clean===true&&Array.isArray(v.divergence)&&v.divergence.length===2&&v.divergence.every(x=>Number.isInteger(x)&&x===0)&&v.fresh_equal===true&&v.namespace_absent===true&&v.attempts===0;check('canonical preflight synthetic positive',preflight(model));for(const [name,mutation] of [['dirty',{clean:false}],['nonzero divergence',{divergence:[1,0]}],['unequal fresh ref',{fresh_equal:false}],['used namespace',{namespace_absent:false}],['prior invocation',{attempts:1}]])check('canonical preflight refuses '+name,!preflight({...model,...mutation}));
return {schema:'ghc.family.owner-tests.v1',owner:'Orren Pike',phase:'v693-v6',valid:checks.every(c=>c.pass),tests_total:checks.length,tests_passed:checks.filter(c=>c.pass).length,tests_failed:checks.filter(c=>!c.pass).length,positive_contracts:100,malformed_contracts:200,additional_parser_subjects:38,local_runner_checks:20,finite_algebra_cases:135,finite_algebra_checks:405,canonical_structural_fixture_checks:6,scope:'exact owner source-to-final delta and declared dependencies',full_repository_suite:false,independent_reproduction:false,checks};}
module.exports={run};if(require.main===module){const result=run();process.stdout.write(JSON.stringify(result,null,2)+'\n');process.exitCode=result.valid?0:2;}
