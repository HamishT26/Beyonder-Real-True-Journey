def fixtures():
    fs=[]
    for k in range(15):
        n=1 if k<2 else 2+(k%3);g=[[0,1],[1,3],[1,2],[2,3],[3,4]][k%5];aa=[]
        for s in range(n):
            actions=[]
            for a in range(1 if k%5==4 and s==0 else 2):
                weights=[F(0)]*n
                weights[(s+a+k)%n]+=F(1,2);weights[(s+1+a)%n]+=F(1,2)
                reward=F((s+2*a+k)%7-3,1+(k%2))
                if k==0:reward=F(a+1)
                if k==1:reward=F(-1)
                actions.append({'reward':[reward.numerator,reward.denominator],'transition':[[p.numerator,p.denominator] for p in weights]})
            aa.append(actions)
        fs.append({'id':f'MDP{k+1:02d}','states':n,'discount':g,'initial':[[1,n] for _ in range(n)],'actions':aa})
    return fs

def publish(stage,bank):
    bank.mkdir(parents=True,exist_ok=True);receiptpath=bank/(stage+'-push.json');pending=bank/(stage+'-push-pending.json')
    assert not receiptpath.exists() and not pending.exists(),'publication already attempted; inspect exact pending state before continuation'
    branch='codex/GHC-Family/ilyan-reed-main';assert git('branch','--show-current')==branch
    sequence=['planning','x1','x2','final'];idx=sequence.index(stage)
    expected=PARENT if idx==0 else read(bank/(sequence[idx-1]+'-push.json'))['head']
    assert git('rev-parse','HEAD')==expected,'unexpected owner parent'
    manifest=read(ROOT/stage/'manifest.json')
    for e in manifest['entries']:
        b=(REPO/e['path']).read_bytes();assert digest(b)==e['sha256'] and len(b)==e['bytes']
    status=git('status','--porcelain')
    assert status and all(PREFIX+'/' in line for line in status.splitlines()),'nonowner or missing changes'
    git('add','--sparse','--',PREFIX+'/'+stage)
    git('commit','-m',OWNER+' '+PHASE+' '+stage+' exact finite decision evidence')
    head=git('rev-parse','HEAD');assert git('rev-parse','HEAD^')==expected
    write(pending,{'stage':stage,'head':head,'parent':expected,'started_at':utc(),'state':'PUSH_PENDING','duplicate_push_permitted':False})
    command=['git','-c','maintenance.auto=false','-c','gc.auto=0','-c','push.negotiate=true','-c','fetch.negotiationAlgorithm=skipping','-c','remote.origin.negotiationRestrict=refs/heads/'+branch,'-c','pack.threads=2','-C',str(REPO),'push','--no-thin','origin','HEAD:refs/heads/'+branch]
    with (bank/(stage+'-push-log.txt')).open('w',encoding='utf-8') as log:cp=subprocess.run(command,stdout=log,stderr=subprocess.STDOUT)

        write(bank/(stage+'-push-failure.json'),{'stage':stage,'head':head,'exit_code':cp.returncode,'original_success_credit':0});raise RuntimeError('push failed; retain pending and inspect')
    git('fetch','--no-tags','origin','refs/heads/'+branch+':refs/remotes/origin/'+branch)
