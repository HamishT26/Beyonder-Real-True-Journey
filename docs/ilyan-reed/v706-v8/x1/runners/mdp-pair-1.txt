import json,hashlib
from pathlib import Path
p=Path(__file__).resolve().parents[1]/'results.json'
b=p.read_bytes();d=json.loads(b)
ops=[1, 2]
rows=[r for r in d['records'] if r['operation'] in ops]
assert len(rows)==30
assert all(all(r['checks'].values()) for r in rows)
print(json.dumps({'operations':ops,'record_count':len(rows),'results_sha256':hashlib.sha256(b).hexdigest(),'domain_replayed':False,'passed':True}))
