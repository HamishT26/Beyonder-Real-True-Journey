const {spawnSync}=require('node:child_process');
const path=require('node:path');
if(process.env.GHC_MODEL_GUARD!=='v16') throw new Error('Registered admission required');
const name=process.argv[2];
if(!['assemble.txt','canonical.txt'].includes(name)) throw new Error('Unknown final entrypoint');
const r=spawnSync('python',['-B',path.join(__dirname,name),...process.argv.slice(3)],{stdio:'inherit',env:{...process.env,PYTHONUTF8:'1',PYTHONDONTWRITEBYTECODE:'1'}});
if(r.error) throw r.error;
process.exit(r.status===null?1:r.status);
