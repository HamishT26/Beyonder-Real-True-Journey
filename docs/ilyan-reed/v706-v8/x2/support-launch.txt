const {spawnSync}=require('node:child_process');
const path=require('node:path');
if(process.env.GHC_MODEL_GUARD!=='v16') throw new Error('Admission required');
const script=process.argv[2]||'build-viewer.txt';
if(!['build-viewer.txt','build-capabilities.txt'].includes(script)) throw new Error('Unknown support entrypoint');
const r=spawnSync('python',['-B',path.join(__dirname,script),...process.argv.slice(3)],{stdio:'inherit',env:{...process.env,PYTHONUTF8:'1',PYTHONDONTWRITEBYTECODE:'1'}});
if(r.error) throw r.error;
process.exit(r.status===null?1:r.status);
