"""Bounded five-element rough approximations. Runtime uses integer masks."""
import json,re
from math import gcd
OPS=['partition-shape','lower-table','upper-table','three-region-partition','complement-duality','monotone-inclusions','idempotent-fixed-points','lower-meet-upper-join','definable-boolean-algebra','rough-pair-census','rough-membership-ratios','accuracy-boundary-ratios','refinement-sandwich','attribute-positive-region','minimal-attribute-reducts','relabel-covariance','nonpreservation-witnesses','accessible-explanation','observation-remedy-gap','classification-authority-reservation']
def validate(p):
 if type(p) is not dict or set(p)!={'id','partition','refiner','target'}:raise ValueError('closed_profile_keys')
 if type(p['id']) is not str or not re.fullmatch(r'RS-[A-Z0-9-]{2,20}',p['id']):raise ValueError('profile_id')
 for name in ['partition','refiner']:
  v=p[name]
  if type(v) is not list or len(v)!=5:raise ValueError(name+'_length')
  if any(type(x) is not int or not 0<=x<5 for x in v):raise ValueError(name+'_labels')
 if type(p['target']) is not int or not 0<=p['target']<32:raise ValueError('target_mask')
 return True
def blocks(v):
 d={}
 for i,x in enumerate(v):d[x]=d.get(x,0)|(1<<i)
 return sorted(d.values())
def tables(bs):
 return ([sum(b for b in bs if b&m==b) for m in range(32)],
         [sum(b for b in bs if b&m) for m in range(32)])
def frac(n,d):
 if not d:return None
 g=gcd(n,d);return [n//g,d//g]
def calc(p,op):
 bs=blocks(p['partition']);lo,hi=tables(bs);t=p['target']
 if op==OPS[0]:return {'block_masks':bs,'universe':31,'size':5}
 if op==OPS[1]:return lo
 if op==OPS[2]:return hi
 if op==OPS[3]:return [[lo[m],hi[m]^lo[m],31^hi[m]] for m in range(32)]
 if op==OPS[4]:return {'holds':all(lo[m]==31^hi[31^m] and hi[m]==31^lo[31^m] for m in range(32)),'subsets':32}
 if op==OPS[5]:
  count=0;ok=True
  for a in range(32):
   for b in range(32):
    if a&b==a:
     count+=1;ok=ok and lo[a]&lo[b]==lo[a] and hi[a]&hi[b]==hi[a]
  return {'holds':ok,'nested_pairs':count}
 if op==OPS[6]:return {'holds':all(lo[lo[m]]==lo[m] and hi[hi[m]]==hi[m] for m in range(32)),'lower_fixed':[m for m in range(32) if lo[m]==m],'upper_fixed':[m for m in range(32) if hi[m]==m]}
 if op==OPS[7]:return {'holds':all(lo[a&b]==lo[a]&lo[b] and hi[a|b]==hi[a]|hi[b] for a in range(32) for b in range(32)),'ordered_pairs':1024}
 if op==OPS[8]:
  ds=[m for m in range(32) if lo[m]==hi[m]]
  return {'definable_masks':ds,'boolean_closed':all(31^m in ds for m in ds) and all(a&b in ds and a|b in ds for a in ds for b in ds),'count':len(ds)}
 if op==OPS[9]:
  pairs={}
  for m in range(32):pairs[(lo[m],hi[m])]=pairs.get((lo[m],hi[m]),0)+1
  return [[a,b,c] for (a,b),c in sorted(pairs.items())]
 if op==OPS[10]:
  return [frac((next(b for b in bs if b&(1<<i))&t).bit_count(),next(b for b in bs if b&(1<<i)).bit_count()) for i in range(5)]
 if op==OPS[11]:return {'accuracy':frac(lo[t].bit_count(),hi[t].bit_count()),'boundary_fraction':frac((hi[t]^lo[t]).bit_count(),5),'empty_upper':'undefined'}
 if op==OPS[12]:
  refined=blocks(list(zip(p['partition'],p['refiner'])));l2,h2=tables(refined)
  return {'refined_blocks':refined,'holds':all(lo[m]&l2[m]==lo[m] and l2[m]&m==l2[m] and m&h2[m]==m and h2[m]&hi[m]==h2[m] for m in range(32)),'lower':l2,'upper':h2}
 def pos(b):return sum(x for x in b if x&t in (0,x))
 if op==OPS[13]:
  v=pos(bs);return {'positive_region':v,'dependency':frac(v.bit_count(),5)}
 if op==OPS[14]:
  ps=[]
  for mask in range(4):
   values=[tuple(p['partition' if k==0 else 'refiner'][i] for k in range(2) if mask&(1<<k)) for i in range(5)]
   ps.append(pos(blocks(values)))
  good=[m for m in range(4) if ps[m]==ps[3]]
  minimal=[m for m in good if not any(v!=m and v&m==v for v in good)]
  return {'joint_positive_region':ps[3],'reducts':[[k for k in range(2) if m&(1<<k)] for m in minimal],'positive_regions':ps}
 if op==OPS[15]:
  perm=[2,4,1,0,3]
  def rename(m):return sum(1<<perm[i] for i in range(5) if m&(1<<i))
  l2,h2=tables([rename(b) for b in bs])
  return {'permutation':perm,'holds':all(l2[rename(m)]==rename(lo[m]) and h2[rename(m)]==rename(hi[m]) for m in range(32))}
 if op==OPS[16]:
  x=y=None
  for a in range(32):
   for b in range(32):
    if x is None and hi[a&b]!=hi[a]&hi[b]:x=[a,b,hi[a&b],hi[a]&hi[b]]
    if y is None and lo[a|b]!=lo[a]|lo[b]:y=[a,b,lo[a|b],lo[a]|lo[b]]
  return {'upper_intersection_counterexample':x,'lower_union_counterexample':y}
 if op==OPS[17]:
  ids=lambda m:[i for i in range(5) if m&(1<<i)]
  return {'certain':ids(lo[t]),'boundary':ids(hi[t]^lo[t]),'excluded':ids(31^hi[t]),'interpretation':'synthetic partition-relative description'}
 if op==OPS[18]:return {'missing':['real_observation','measurement_provenance','affected_party_review','remedy_trial'],'empirical_credit':0}
 if op==OPS[19]:return {'held':['real_classification','identity_or_eligibility_decision'],'authority_credit':0,'release_requires':['concrete_target','consent','competent_authority','appeal_and_rollback']}
 raise ValueError('unknown operation')
def run(p,op):
 try:
  validate(p)
  if op not in OPS:raise ValueError('operation_allowlist')
 except ValueError as e:return {'accepted':False,'operation_executed':False,'error':str(e),'original_success_credit':0}
 idx=OPS.index(op)
 return {'accepted':True,'operation_executed':True,'outcome':'completed' if idx<17 else ['represented','open_gap','exact_gate'][idx-17],'result':calc(p,op)}
