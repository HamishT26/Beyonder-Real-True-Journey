 for r in io.read(lane+'/skill-validation.json')['records']:witness(mid_for_op(skillmap[r['name']]),lane+'-'+r['name']+'-skill','Official skill validation',True,r['passed'],'pass' if r['passed'] else 'fail',io.PREFIX+lane+'/skill-validation.json')
 for r in io.read(lane+'/runner-smokes.json')['records']:witness(mid_for_op(runnermap[r['runner']]),lane+'-'+r['runner']+'-'+r['kind'],'New paired caller '+r['kind'],True,r['passed'],'pass' if r['passed'] else 'fail',io.PREFIX+lane+'/runner-smokes.json')
 ops=[]
 if lane=='x1':ops+=io.read('plan/startup-failures.json')['records']
 if (BASE/lane/'operational-corrections.json').exists():ops+=io.read(lane+'/operational-corrections.json')['records']
 if ops:
  ensure('IR6955-M06','Bounded source and runtime recovery',[r['id'] for r in ops])
  for r in ops:
   witness('IR6955-M06',r['id']+'-original','Retained operational failure','successful declared procedure',r['failure'],'fail',io.PREFIX+('plan/startup-failures.json' if 'START' in r['id'] else lane+'/operational-corrections.json'),[r['id']])
   if r['recovered']:witness('IR6955-M06',r['id']+'-recovery','Narrow dependency recovery','matching dependency resolved',r['recovery'],'pass',io.PREFIX+('plan/startup-failures.json' if 'START' in r['id'] else lane+'/operational-corrections.json'),[r['id']])
 if lane=='x1':
  for i,r in enumerate(io.read('x1/recent-overview-review.json')['records'],1):witness('IR6955-M06',f'IR6955-OV{i:02d}','Read-only key overview bound to exact Git blob','bounded source record read',{'source_commit':r['source_commit'],'sha256_raw':r['sha256_raw'],'source_credit':0},'pass',io.PREFIX+'x1/recent-overview-review.json')
 if lane=='x2' and (BASE/'x2/promotion-witnesses.json').exists():
  for r in io.read('x2/promotion-witnesses.json')['records']:witness(r['method_id'],r['id'],r['procedure'],r['expected'],r['observed'],'pass' if r['passed'] else 'fail',r['evidence_ref'])
 event_methods={e['method_id'] for e in ledger['state_events']}
 for m in ledger['methods']:
  if m['method_id'] not in event_methods:
   passing=next(w['witness_id'] for w in ledger['witnesses'] if w['method_id']==m['method_id'] and w['result']=='pass');mf.append_event(ledger,m['method_id'],'candidate','validated','Declared passing witness with retained original negative',passing);ledger['recommendations'].append({'method_id':m['method_id'],'preconditions':m['trigger_preconditions'],'recommendation':m['recurrence_guard'],'rollback':m['rollback']})
 mf.refresh_counts(ledger);valid=mf.validate_ledger(ledger);assert valid['valid'],valid;io.write(lane+'/method-flow.json',ledger);io.write(lane+'/method-flow-validation.json',valid);neg=sorted({n for m in ledger['methods'] for n in m['retained_negative_ids']});io.write(lane+'/negative-index.json',{'ids':neg,'owner_count':len(neg),'source_activation_negatives':218,'source_plus_owner_activation_negatives':218+len(neg),'source_selected_lineage_kept_separate':True,'original_subject_success_credit':0});print(json.dumps({'counts':ledger['counts'],'unique_negatives':len(neg),'valid':valid['valid']}))
if __name__=='__main__':main()
