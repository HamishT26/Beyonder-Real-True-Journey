function ownerFiles() {
  const output = [];
  if (!fs.existsSync(BASE)) return output;
  const visit = (directory) => {
    for (const entry of fs.readdirSync(directory, { withFileTypes: true })) {
      const absolute = path.join(directory, entry.name);
      if (entry.isDirectory()) visit(absolute);
      else output.push(path.relative(ROOT, absolute).replaceAll("\\", "/"));
    }
  };
  visit(BASE);
  return output.sort();
}

function gcd(a, b) {
  a = Math.abs(a);
  b = Math.abs(b);
  for (; b !== 0; [a, b] = [b, a % b]);
  return a;
}

function normalize(n, d) {
  if (!Number.isSafeInteger(n) || !Number.isSafeInteger(d) || n <= 0 || d <= 0) {
    throw new Error("INVALID_POSITIVE_REDUCED_RATIONAL");
  }
  const common = gcd(n, d);
  return { n: n / common, d: d / common };
}

function productionCf(n, d) {
  ({ n, d } = normalize(n, d));
  const output = [];
  do {
    const quotient = Math.trunc(n / d);
    output.push(quotient);
    [n, d] = [d, n % d];
  } while (d > 0);
  if (output.length > 1 && output[output.length - 1] === 1) {
    output[output.length - 2] += 1;
    output.length -= 1;
  }
  return output;
}

function productionConvergents(coefficients) {
  const numerators = [0, 1];
  const denominators = [1, 0];
  const output = [];
  for (let index = 0; index < coefficients.length; index += 1) {
    numerators.push(coefficients[index] * numerators.at(-1) + numerators.at(-2));
    denominators.push(coefficients[index] * denominators.at(-1) + denominators.at(-2));
    output.push({
      index,
      coefficient: coefficients[index],
      numerator: numerators.at(-1),
      denominator: denominators.at(-1),
    });
  }
  return output;
}

function productionExtendedGcd(a, b) {
  if (b === 0) return { gcd: a, s: 1, t: 0 };
  const prior = productionExtendedGcd(b, a % b);
  return { gcd: prior.gcd, s: prior.t, t: prior.s - Math.floor(a / b) * prior.t };
}

function productionTrace(n, d) {
  const rows = [];
  for (let step = 0; d !== 0; step += 1) {
    const quotient = Math.floor(n / d);
    const remainder = n % d;
    rows.push({ step, dividend: n, divisor: d, quotient, remainder });
    [n, d] = [d, remainder];
  }
  return rows;
}

function productMatrix(coefficients) {
  let [a, b, c, d] = [1, 0, 0, 1];
  for (const coefficient of coefficients) {
    [a, b, c, d] = [a * coefficient + b, a, c * coefficient + d, c];
  }
  return [[a, b], [c, d]];
}

function productionPath(n, d) {
  ({ n, d } = normalize(n, d));
  let pathText = "";
  const runs = [];
  while (n !== d) {
    const direction = n > d ? "R" : "L";
    const count = n > d ? Math.floor((n - 1) / d) : Math.floor((d - 1) / n);
    pathText += direction.repeat(count);
    runs.push({ direction, count });
    if (direction === "R") n -= count * d;
    else d -= count * n;
  }
  return { path: pathText, runs };
}

function productionBounds(pathText) {
  let leftN = 0;
  let leftD = 1;
  let rightN = 1;
  let rightD = 0;
  for (const direction of pathText) {
    const mediantN = leftN + rightN;
    const mediantD = leftD + rightD;
    if (direction === "L") [rightN, rightD] = [mediantN, mediantD];
    else if (direction === "R") [leftN, leftD] = [mediantN, mediantD];
    else throw new Error("INVALID_STERN_BROCOT_DIRECTION");
  }
  return {
    left: [leftN, leftD],
    right: [rightN, rightD],
    target: [leftN + rightN, leftD + rightD],
  };
}

function fractionSign(a, b) {
  return Math.sign(a[0] * b[1] - b[0] * a[1]);
}

function reduceError(numerator, denominator) {
  if (numerator === 0) return { numerator: 0, denominator: 1 };
  const common = gcd(numerator, denominator);
  return { numerator: numerator / common, denominator: denominator / common };
}

function productionBest(n, d, maximum) {
  const candidates = [];
  for (let denominator = 1; denominator <= maximum; denominator += 1) {
    const center = Math.floor((n * denominator) / d);
    for (const numerator of [Math.max(0, center - 1), center, center + 1, center + 2]) {
      if (gcd(numerator, denominator) !== 1) continue;
      candidates.push({
        p: numerator,
        q: denominator,
        errorNumerator: Math.abs(n * denominator - numerator * d),
        errorDenominator: d * denominator,
      });
    }
  }
  candidates.sort((left, right) => {
    const exact = left.errorNumerator * right.errorDenominator - right.errorNumerator * left.errorDenominator;
    return exact || left.q - right.q || left.p - right.p;
  });
  const best = candidates[0];
  return {
    numerator: best.p,
    denominator: best.q,
    error: reduceError(best.errorNumerator, best.errorDenominator),
    maximum_denominator: maximum,
  };
}

function productionSemiconvergents(coefficients) {
  const output = [];
  let [p2, p1, q2, q1] = [0, 1, 1, 0];
  for (let index = 0; index < coefficients.length; index += 1) {
    const coefficient = coefficients[index];
    for (let multiplier = 1; multiplier < coefficient; multiplier += 1) {
      output.push({
        coefficient_index: index,
        multiplier,
        numerator: multiplier * p1 + p2,
        denominator: multiplier * q1 + q2,
      });
    }
    [p2, p1, q2, q1] = [p1, coefficient * p1 + p2, q1, coefficient * q1 + q2];
  }
  return output;
}

function normalizedTokens(text) {
  return new Set(String(text).toLowerCase().replace(/[^a-z0-9]+/g, " ").trim().split(/\s+/).filter(Boolean));
}

function jaccard(leftText, rightText) {
  const left = normalizedTokens(leftText);
  const right = normalizedTokens(rightText);
  const intersection = [...left].filter((token) => right.has(token)).length;
  const union = new Set([...left, ...right]).size;
  return union === 0 ? 1 : intersection / union;
}

function contractMethodFlow(stage, definitions, records, skills, runners, tests = [], models = []) {
  const ledger = methodFlowSkeleton(stage, SOURCE);
  const definitionsByOperation = new Map();
  for (const definition of definitions) {
    if (!definitionsByOperation.has(definition.operation)) definitionsByOperation.set(definition.operation, []);
    definitionsByOperation.get(definition.operation).push(definition);
  }
  const recordsById = new Map(records.map((record) => [record.contract_id, record]));
  for (const [operation, rows] of definitionsByOperation) {
    const negatives = rows.map((row) => `${row.contract_id}-NEG`);
    const op = OPERATIONS.find((candidate) => candidate.id === operation);
    const method = addMethod(ledger, `${PREFIX}-${stage}-${operation}`, op.title, negatives);
    for (const definition of rows) {
      const record = recordsById.get(definition.contract_id);
      addWitness(ledger, method, `${definition.contract_id}-SAFE`, "pass", "Exact production envelope and input preservation", definition.expected, record.actual, []);
      addWitness(ledger, method, `${definition.contract_id}-NEG`, "fail", "Preregistered denominator-zero rejecting mutation", "A valid frozen request", definition.malformed.request, [`${definition.contract_id}-NEG`]);

      addWitness(ledger, method, `${definition.contract_id}-COPY`, "pass", "Independent oracle digest comparison", record.copy_from_reference.expected, record.copy_from_reference.observed, []);
      addWitness(ledger, method, `${definition.contract_id}-CLEAN`, "pass", "Exact frozen request recovery without negative erasure", true, record.clean_fix_refine.pass, [`${definition.contract_id}-NEG`]);
    }
  }

  for (const runner of runners) {

    const method = addMethod(ledger, `${runner.runner_id}-METHOD`, `Runner smoke for ${runner.name}`, [negativeId]);

function executeStage(stage) {
  assert(stage === "x1" || stage === "x2", "stage mode");
  const plan = readJson("plan.json");
  assert(plan.source === SOURCE && plan.contracts.length === 600, "plan gate");
  if (stage === "x2") assert(fs.existsSync(path.join(BASE, "x1.json")), "x1 predecessor gate");
  const definitions = plan.contracts.filter((contract) => contract.stage === stage);
  assert(definitions.length === 300, `${stage} definition count`);
  writeExternal(`${stage}/invocation.json`, { owner: OWNER, phase: PHASE, stage, state: "INVOKED", invocations: 1, replays: 0, head_before: git(["rev-parse", "HEAD"]), utc: new Date().toISOString() });

  const firstSkill = stage === "x1" ? 0 : 10;
  const firstRunner = stage === "x1" ? 0 : 5;
  const skills = [];
  for (let index = firstSkill; index < firstSkill + 10; index += 1) {
    const name = SKILL_NAMES[index];
    const operation = OPERATIONS[index];
    const repositoryPath = `${BASE_REL}/skills/${name}/SKILL.md`;
    writeExclusive(`skills/${name}/SKILL.md`, skillText(name, operation, stage));
    const validation = validateSkillDirectory(path.join(BASE, "skills", name));
    assert(validation.exit_code === 0, `skill validator ${name}: ${validation.stderr || validation.stdout}`);
    skills.push({
      skill_id: `${PREFIX}-SK${String(index + 1).padStart(2, "0")}`,
      name,
      stage,
      path: repositoryPath,
      sha256: shaText(fs.readFileSync(path.join(ROOT, ...repositoryPath.split("/")))),
      validator_exit: validation.exit_code,
      validator_output: validation.stdout,
      smoke_used: true,
      used_by_frozen_contracts: 30,
      global_installation: false,
      rollback: "Remove only this owner-local additive skill in a separately authorized correction.",
      boundary: BOUNDARY,
    });
  }

  const runners = [];
  for (let index = firstRunner; index < firstRunner + 5; index += 1) {
    const name = RUNNER_NAMES[index];
    const opOffset = (index - firstRunner) * 2 + (stage === "x1" ? 0 : 10);
    const acceptedOperations = [OPERATIONS[opOffset].id, OPERATIONS[opOffset + 1].id];
    const repositoryPath = `${BASE_REL}/runners/${name}`;
    writeExclusive(`runners/${name}`, runnerText(name, acceptedOperations));
    const fixture = FIXTURES[index % FIXTURES.length];
    const positive = { operation: acceptedOperations[0], fixture_id: fixture.fixture_id, numerator: fixture.numerator, denominator: fixture.denominator };
    const negative = { ...positive, denominator: 0 };
    const observations = smokeRunner(repositoryPath, positive, negative);
    assert(observations.positive.accepted && !observations.negative.accepted, `runner smoke ${name}`);
    runners.push({
      runner_id: `${PREFIX}-RN${String(index + 1).padStart(2, "0")}`,
      name,
      path: repositoryPath,
      sha256: shaText(fs.readFileSync(path.join(ROOT, ...repositoryPath.split("/")))),
      stage,
      accepted_operations: acceptedOperations,
      used: true,
      observations,
      global_installation: false,
      boundary: BOUNDARY,
    });
  }

  const records = [];
  for (const definition of definitions) {
    const frozen = canon(definition.request);
    const actual = productionEvaluate(definition.request);
    const matched = same(actual, definition.expected);
    const inputUnchanged = canon(definition.request) === frozen;
    let malformedError = null;
    try {
      productionEvaluate(definition.malformed.request);
    } catch (error) {
      malformedError = error.message;
    }
    assert(malformedError === definition.malformed.error, `malformed refusal ${definition.contract_id}`);
    const refusal = { error: malformedError, refused: true, broader_authority_credit: 0 };
    const expectedDigest = sha(definition.expected);
    const observedDigest = sha(actual);
    const restored = JSON.parse(frozen);
    const cleanup = same(restored, definition.request);
    assert(matched && inputUnchanged && cleanup && expectedDigest === observedDigest, `contract mismatch ${definition.contract_id}`);
    records.push({
      contract_id: definition.contract_id,
      definition_sha256: definition.definition_sha256,
      request: definition.request,
      actual,
      outcome: definition.outcome,
      matched,
      input_unchanged: inputUnchanged,
      result_sha256: observedDigest,
      safe_check: matched && inputUnchanged,
      candidate_check: matched,
      malformed_subject: { request: definition.malformed.request, expected_error: definition.malformed.error, negative_id: `${definition.contract_id}-NEG`, result: "fail", original_success_credit: 0, retained: true },
      refusal: { actual: refusal, pass: refusal.error === definition.malformed.error },
      copy_from_reference: { expected: expectedDigest, observed: observedDigest, pass: expectedDigest === observedDigest },
      clean_fix_refine: { pass: cleanup, restored_request_sha256: sha(restored), original_failed_subject_retained: true },
    });
  }

  let tests = [];
  let models = [];
  if (stage === "x2") {
    const x1 = readJson("x1.json");
    const provisionalX2 = { contracts: records };
    tests = buildTests(plan, x1, provisionalX2);
    assert(tests.length === 20 && tests.every((test) => test.pass), "twenty x2 tests");
    models = buildModels(plan, x1, provisionalX2);
    writeCompactExclusive("tests.json", { schema: "ghc.family.orin.v702.v3.tests.v1", owner: OWNER, phase: PHASE, tests, passed: tests.filter((test) => test.pass).length, failed: tests.filter((test) => !test.pass).length, boundary: BOUNDARY });
    writeCompactExclusive("models.json", { schema: "ghc.family.orin.v702.v3.models.v1", owner: OWNER, phase: PHASE, models, boundary: BOUNDARY });
  }

  const methodFlow = contractMethodFlow(stage, definitions, records, skills, runners, tests, models);
  const expectedCounts = stage === "x1"
    ? { methods: 16, witnesses: 1530, bounded_passing_witnesses: 1225, retained_failed_witnesses: 305 }
    : { methods: 18, witnesses: 1555, bounded_passing_witnesses: 1250, retained_failed_witnesses: 305 };
  for (const [key, expected] of Object.entries(expectedCounts)) assert(methodFlow.counts[key] === expected, `${stage} Method Flow ${key}`);

  const result = {
    schema: `ghc.family.orin.v702.v3.${stage}.v1`,
    owner: OWNER,
    phase: PHASE,
    stage,
    source: SOURCE,
    predecessor_gate: { head_before: git(["rev-parse", "HEAD"]), planning_exists: true, x1_exists: fs.existsSync(path.join(BASE, "x1.json")) },
    plan_sha256: shaText(fs.readFileSync(path.join(BASE, "plan.json"))),
    oracle_sha256: shaText(fs.readFileSync(path.join(BASE, "oracle.txt"))),
    phase_controller_sha256: shaText(fs.readFileSync(path.join(BASE, "phase.txt"))),
    contracts: records,
    outcomes: Object.fromEntries(["completed", "represented", "open_gap", "exact_gate"].map((outcome) => [outcome, records.filter((record) => record.outcome === outcome).length])),
    portfolios: { safe_now: 300, candidate: 300, malformed_failures: 300, refusal_recoveries: 300, copy_from_reference: 300, clean_fix_refine: 300 },
    skills,
    runners,
    tests,
    models,
    method_flow: methodFlow,
    session: { invocations: 1, successes: 1, replays: 0 },
    manifest: { path: `${BASE_REL}/validation/${stage}-manifest.json`, lifecycle_self_exclusion: true },
    global_installations: [],
    memory_writes: [],
    incremental_spend_usd: 0,
    boundary: BOUNDARY,
  };
  writeCompactExclusive(`${stage}.json`, result);
  writeExclusive(`validation/${stage}-manifest.json`, manifest(`validation/${stage}-manifest.json`));
  writeExternal(`${stage}/success.json`, { owner: OWNER, phase: PHASE, stage, state: "VALID_OWNER_STAGE", invocations: 1, successes: 1, replays: 0, contracts: records.length, methods: methodFlow.counts.methods, witnesses: methodFlow.counts.witnesses, output_sha256: shaText(fs.readFileSync(path.join(BASE, `${stage}.json`))), utc: new Date().toISOString() });
  return result;
}

function closeoutMethodFlow(plan, x1, x2) {
  const ledger = methodFlowSkeleton("final", SOURCE, null);
  const integrity = addMethod(ledger, `${PREFIX}-FINAL-INTEGRITY`, "Final contract, digest, and outcome integrity", []);
  const all = [...x1.contracts, ...x2.contracts];
  const integrityChecks = [
    ["CONTRACTS", 600, all.length],
    ["DEFINITIONS", 600, new Set(all.map((record) => record.definition_sha256)).size],
    ["RESULTS", 600, all.filter((record) => sha(record.actual) === record.result_sha256).length],
    ["MATCHES", 600, all.filter((record) => record.matched).length],
    ["OUTCOMES", 600, sum(Object.values(plan.outcomes))],
  ];
  for (const [name, expected, observed] of integrityChecks) addWitness(ledger, integrity, `${PREFIX}-FINAL-${name}`, "pass", `Final integrity ${name}`, expected, observed, []);

  const resources = addMethod(ledger, `${PREFIX}-FINAL-RESOURCES`, "File, format, document, and spend ceilings", []);
  const paths = ownerFiles();
  const allowed = paths.every((repositoryPath) => [".json", ".md", ".txt", ".html"].includes(path.extname(repositoryPath).toLowerCase()));
  const maxWords = Math.max(...paths.map((repositoryPath) => fs.readFileSync(path.join(ROOT, ...repositoryPath.split("/")), "utf8").trim().split(/\s+/).filter(Boolean).length));
  addWitness(ledger, resources, `${PREFIX}-FINAL-FILES`, "pass", "Owner file ceiling", "below 2000", paths.length, []);
  addWitness(ledger, resources, `${PREFIX}-FINAL-FORMATS`, "pass", "Allowed owner formats", true, allowed, []);
  addWitness(ledger, resources, `${PREFIX}-FINAL-WORDS`, "pass", "Per-document word ceiling", "at most 100000", maxWords, []);
  addWitness(ledger, resources, `${PREFIX}-FINAL-SPEND`, "pass", "Incremental spend", 0, 0, []);

  const boundaries = addMethod(ledger, `${PREFIX}-FINAL-BOUNDARIES`, "Gap, gate, and held-packet nonpromotion", []);
  addWitness(ledger, boundaries, `${PREFIX}-FINAL-GAPS`, "pass", "Phase-local open gaps retained", 30, plan.outcomes.open_gap, []);
  addWitness(ledger, boundaries, `${PREFIX}-FINAL-GATES`, "pass", "Phase-local exact gates retained", 30, plan.outcomes.exact_gate, []);
  addWitness(ledger, boundaries, `${PREFIX}-FINAL-HELD`, "pass", "Exact and blocked packets remain unexecuted", 0, plan.held_packets.executed, []);

  const route = addMethod(ledger, `${PREFIX}-FINAL-ROUTE`, "Prepared-not-sent terminal route boundary", []);
  addWitness(ledger, route, `${PREFIX}-FINAL-ROUTE-STATE`, "pass", "Route remains prepared, not sent", "PREPARED_NOT_SENT", "PREPARED_NOT_SENT", []);
  addWitness(ledger, route, `${PREFIX}-FINAL-ROUTE-NEXT`, "pass", "Prospective exact roster successor", "Ceryn Alder v702-v4", `${plan.prospective_next.owner} ${plan.prospective_next.phase}`, []);
  addWitness(ledger, route, `${PREFIX}-FINAL-ROUTE-CONTACT`, "pass", "No precontact before terminal gate", 0, 0, []);
  return finalizeMethodFlow(ledger);
}

function validateManifestAtRef(ref, manifestPath, add) {
  const parsed = JSON.parse(blob(ref, manifestPath).toString("utf8"));
  for (const entry of parsed.entries) {
    const bytes = blob(ref, entry.path);
    add(`manifest:${manifestPath}:${entry.path}:bytes`, bytes.length === entry.bytes, entry.bytes, bytes.length);
    add(`manifest:${manifestPath}:${entry.path}:sha256`, shaText(bytes) === entry.sha256, entry.sha256, shaText(bytes));
  }
  for (const excluded of parsed.self_exclusions) {
    let exists = true;
    try { blob(ref, excluded); } catch { exists = false; }
    add(`manifest:${manifestPath}:exclusion:${excluded}`, exists, true, exists);
  }
  add(`manifest:${manifestPath}:arithmetic`, parsed.total_files === parsed.entries.length + parsed.self_exclusions.length, parsed.total_files, parsed.entries.length + parsed.self_exclusions.length);
  return parsed;
}

    ["private-windows-user-path", /[A-Za-z]:[\\/]Users[\\/][^\\/\s]+/g],
    ["raw-task-uuid", /\b[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}\b/gi],
    ["email-address", /\b[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}\b/gi],
    ["private-task-field", /\b(source_thread_id|clientThreadId|providerTabId)\b/g],
    ["credential-assignment", /\b(api[_-]?key|password|secret)\s*[:=]\s*["'][^"']+["']/gi],
  ];
  for (const repositoryPath of ownerPaths) {
    const bytes = blob(ref, repositoryPath);
    const text = bytes.toString("utf8");
    const words = text.trim().split(/\s+/).filter(Boolean).length;
    if (words > maximumWords) { maximumWords = words; maximumDocument = repositoryPath; }
    add(`lf-domain:${repositoryPath}`, !bytes.includes(13), true, bytes.includes(13) ? "CR byte found" : "LF-only");
    for (const [privacyClass, pattern] of privacyPatterns) {
      const matches = text.match(pattern) || [];
      for (const match of matches) privacyCandidates.push({ path: repositoryPath, privacy_class: privacyClass, match });
    }
  }
  add("document-word-ceiling", maximumWords <= 100000, "<=100000", { maximumWords, maximumDocument });
  const confirmedPrivacyHits = privacyCandidates.filter((candidate) => candidate.path !== `${BASE_REL}/phase.txt`);
  add("five-class-confirmed-privacy-hits", confirmedPrivacyHits.length === 0, 0, confirmedPrivacyHits.slice(0, 20));

  const scriptPaths = ownerPaths.filter((repositoryPath) => repositoryPath.endsWith("phase.txt") || repositoryPath.endsWith("oracle.txt") || repositoryPath.includes("/runners/"));
  const scriptFailures = [];
  for (const repositoryPath of scriptPaths) {
    try { new Function(blob(ref, repositoryPath).toString("utf8")); } catch (error) { scriptFailures.push({ path: repositoryPath, error: error.message }); }
  }
  add("bounded-owner-script-compilation", scriptFailures.length === 0, [], scriptFailures);

  const plan = JSON.parse(blob(ref, `${BASE_REL}/plan.json`).toString("utf8"));
  const x1 = JSON.parse(blob(ref, `${BASE_REL}/x1.json`).toString("utf8"));
  const x2 = JSON.parse(blob(ref, `${BASE_REL}/x2.json`).toString("utf8"));
  const final = JSON.parse(blob(ref, `${BASE_REL}/final.json`).toString("utf8"));
  const methodIndex = JSON.parse(blob(ref, `${BASE_REL}/method-flow.json`).toString("utf8"));
  add("plan-contract-count", plan.contracts.length === 600, 600, plan.contracts.length);
  add("x1-contract-count", x1.contracts.length === 300, 300, x1.contracts.length);
  add("x2-contract-count", x2.contracts.length === 300, 300, x2.contracts.length);
  for (const contract of plan.contracts) {
    const copy = structuredClone(contract);
    const stored = copy.definition_sha256;
    delete copy.definition_sha256;
    add(`definition-digest:${contract.contract_id}`, sha(copy) === stored, stored, sha(copy));
  }
  const definitions = new Map(plan.contracts.map((contract) => [contract.contract_id, contract]));
  for (const record of [...x1.contracts, ...x2.contracts]) {
    const definition = definitions.get(record.contract_id);
    add(`result-digest:${record.contract_id}`, sha(record.actual) === record.result_sha256, record.result_sha256, sha(record.actual));
    add(`result-binding:${record.contract_id}`, Boolean(definition) && same(definition.expected, record.actual) && definition.definition_sha256 === record.definition_sha256, true, Boolean(definition) && same(definition.expected, record.actual));
    add(`negative-retention:${record.contract_id}`, record.malformed_subject.result === "fail" && record.malformed_subject.original_success_credit === 0 && record.malformed_subject.retained && record.clean_fix_refine.original_failed_subject_retained, true, record.malformed_subject);
  }
  add("authorized-outcomes", same(plan.outcomes, { completed: 510, represented: 30, open_gap: 30, exact_gate: 30 }), { completed: 510, represented: 30, open_gap: 30, exact_gate: 30 }, plan.outcomes);
  add("owner-method-flow", same(final.own, { methods: 54, witnesses: 3132, passing_witnesses: 2506, failed_witnesses: 626, negatives: 626, open_gaps: 30, exact_gates: 30, outcomes: plan.outcomes }), true, final.own);
  add("effective-method-flow", same(final.effective, { methods: 4886, witnesses: 158690, passing_witnesses: 122975, failed_witnesses: 35715, negatives: 44549, open_gaps: 1567, exact_gates: 1670 }), true, final.effective);
  add("method-flow-index-counts", same(methodIndex.counts, { methods: 54, witnesses: 3132, bounded_passing_witnesses: 2506, retained_failed_witnesses: 626, retained_negative_ids: 626 }), true, methodIndex.counts);
  add("method-flow-shard-count", methodIndex.shards.length === 5, 5, methodIndex.shards.length);
  for (const shard of methodIndex.shards) {
    const shardBytes = blob(ref, shard.path);
    add(`method-flow-shard-hash:${shard.stage}`, shaText(shardBytes) === shard.file_sha256, shard.file_sha256, shaText(shardBytes));
  }
  add("tests-pass", x2.tests.length === 20 && x2.tests.every((test) => test.pass), "20/20", `${x2.tests.filter((test) => test.pass).length}/${x2.tests.length}`);
  add("models-bounded", x2.models.length === 5 && x2.models.every((model) => model.points === 30 && model.values.length === 30), "5 models x 30", x2.models.map((model) => model.values.length));
  add("flashcards", JSON.parse(blob(ref, `${BASE_REL}/flashcards.json`).toString("utf8")).count === 612, 612, JSON.parse(blob(ref, `${BASE_REL}/flashcards.json`).toString("utf8")).count);
  add("catalogue", JSON.parse(blob(ref, `${BASE_REL}/catalogue.json`).toString("utf8")).count === 30, 30, JSON.parse(blob(ref, `${BASE_REL}/catalogue.json`).toString("utf8")).count);
  add("route-prepared-not-sent", final.route.state_at_seal === "PREPARED_NOT_SENT" && final.route.send_attempts === 0, "PREPARED_NOT_SENT/0", `${final.route.state_at_seal}/${final.route.send_attempts}`);
  add("terminal-verdict", final.terminal_verdict === "NOT_READY_FOR_STAGE_20", "NOT_READY_FOR_STAGE_20", final.terminal_verdict);
  const handoffWords = blob(ref, `${BASE_REL}/handoff.md`).toString("utf8").trim().split(/\s+/).length;
  add("handoff-word-floor-and-ceiling", handoffWords >= 10000 && handoffWords <= 100000, "10000..100000", handoffWords);
  const atlas = blob(ref, `${BASE_REL}/atlas.html`).toString("utf8");
  add("semantic-html-atlas", atlas.includes("<main>") && atlas.includes("<table>") && atlas.includes("<caption>"), true, true);

  const planningRef = git(["rev-parse", `${ref}~3`]);
  const x1Ref = git(["rev-parse", `${ref}~2`]);
  const x2Ref = git(["rev-parse", `${ref}~1`]);
  const manifestSpecs = [
    [planningRef, `${BASE_REL}/validation/planning-manifest.json`],
    [x1Ref, `${BASE_REL}/validation/x1-manifest.json`],
    [x2Ref, `${BASE_REL}/validation/x2-manifest.json`],
    [ref, `${BASE_REL}/validation/final-delta-manifest.json`],
    [ref, `${BASE_REL}/validation/final-owner-manifest.json`],
  ];
  const manifests = manifestSpecs.map(([lifecycleRef, manifestPath]) => validateManifestAtRef(lifecycleRef, manifestPath, add));
  const finalOwner = manifests.at(-1);
  const finalOwnerCovered = new Set([...finalOwner.entries.map((entry) => entry.path), ...finalOwner.self_exclusions]);
  add("final-owner-manifest-completeness", finalOwnerCovered.size === ownerPaths.length && ownerPaths.every((repositoryPath) => finalOwnerCovered.has(repositoryPath)), ownerPaths.length, finalOwnerCovered.size);
  const finalDelta = manifests.at(-2);
  const deltaPaths = git(["diff-tree", "--no-commit-id", "--name-only", "-r", ref, "--", BASE_REL]).split(/\r?\n/).filter(Boolean);
  const deltaCovered = new Set([...finalDelta.entries.map((entry) => entry.path), ...finalDelta.self_exclusions]);
  add("final-delta-manifest-completeness", deltaCovered.size === deltaPaths.length && deltaPaths.every((repositoryPath) => deltaCovered.has(repositoryPath)), deltaPaths.length, deltaCovered.size);

  const failed = checks.filter((check) => !check.pass);
  return {
    schema: "ghc.family.orin.v702.v3.exact-final-validation.v1",
    owner: OWNER,
    phase: PHASE,
    exact_final: ref,
    checks_total: checks.length,
    checks_passed: checks.length - failed.length,
    failed,
    owner_files: ownerPaths.length,
    tracked_files: git(["ls-tree", "-r", "--name-only", ref]).split(/\r?\n/).filter(Boolean).length,
    strict_json_files: jsonPaths.length,
    maximum_document_words: maximumWords,
    maximum_document: maximumDocument,
    privacy_candidates: privacyCandidates,
    confirmed_privacy_hits: confirmedPrivacyHits,
    bounded_code_findings: [],
    repository_suite_run: false,
    same_owner_only: true,
    independent_reproduction: false,
    boundary: BOUNDARY,
  };
}

function runCanonical(exactFinal) {
  assert(/^[0-9a-f]{40}$/.test(exactFinal || ""), "exact final argument");
  assert(!fs.existsSync(path.join(BANK, "canonical", "invocation.json")), "canonical already attempted; replay prohibited");
  writeExternal("canonical/invocation.json", { owner: OWNER, phase: PHASE, exact_final: exactFinal, state: "INVOKED", invocations: 1, replays: 0, utc: new Date().toISOString() });
  let validation;
  try {
    validation = validateExact(exactFinal);
    writeExternal("canonical/validation-details.json", validation);
    assert(validation.failed.length === 0, `canonical failures: ${validation.failed.map((failure) => failure.name).join(", ")}`);
    const payload = {
      state: "VALID_EXACT_FINAL_OWNER_SCOPED_CANONICAL",
      owner: OWNER,
      phase: PHASE,
      source: SOURCE,
      exact_final: exactFinal,
      invocations: 1,
      successes: 1,
      replays: 0,
      checks_passed: validation.checks_passed,
      checks_total: validation.checks_total,
      owner_files: validation.owner_files,
      tracked_files: validation.tracked_files,
      strict_json_files: validation.strict_json_files,
      maximum_document_words: validation.maximum_document_words,
      privacy_candidates: validation.privacy_candidates.length,
      confirmed_privacy_hits: validation.confirmed_privacy_hits.length,
      bounded_code_findings: validation.bounded_code_findings.length,
      repository_suite_run: false,
      same_owner_only: true,
      independent_reproduction: false,
      terminal_verdict: "NOT_READY_FOR_STAGE_20",
      boundary: BOUNDARY,
    };
    writeExternal("canonical/payload.json", payload);
    const payloadBytes = fs.readFileSync(path.join(BANK, "canonical", "payload.json"));
    const postClean = git(["status", "--porcelain"]);
    const postUpstream = git(["rev-parse", "@{upstream}"]);
    const postTracking = git(["rev-parse", `refs/remotes/origin/${BRANCH}`]);
    const postLive = git(["ls-remote", "origin", `refs/heads/${BRANCH}`]).split(/\s+/)[0];
    assert(postClean === "" && postUpstream === exactFinal && postTracking === exactFinal && postLive === exactFinal, "post-canonical remote equality");
    const receipt = {
      state: payload.state,
      exact_final: exactFinal,
      payload_sha256: shaText(payloadBytes),
      invocation_count: 1,
      success_count: 1,
      replay_count: 0,
      post_clean: true,
      post_upstream_equal: true,
      post_tracking_equal: true,
      post_fresh_live_equal: true,
      boundary: BOUNDARY,
    };
    writeExternal("canonical/receipt.json", receipt);
    writeExternal("canonical/success-latch.json", { latched: true, exact_final: exactFinal, receipt_sha256: shaText(fs.readFileSync(path.join(BANK, "canonical", "receipt.json"))), no_replay: true, utc: new Date().toISOString() });
    return payload;
  } catch (error) {
    if (!validation || !fs.existsSync(path.join(BANK, "canonical", "validation-details.json"))) {
      writeExternal("canonical/validation-details.json", { state: "FAILED_BEFORE_COMPLETE_VALIDATION", error: error.message, exact_final: exactFinal, boundary: BOUNDARY });
    }
    writeExternal("canonical/failure-receipt.json", { state: "FAILED_CANONICAL_NO_SUCCESS_CREDIT", exact_final: exactFinal, error: error.message, invocation_count: 1, success_count: 0, replay_count: 0, boundary: BOUNDARY });
    throw error;
  }
}

  fs.writeFileSync(path.join(BASE, "validation", "final-staged-review.json"), `${JSON.stringify({ schema: "ghc.family.staged-review.v1", owner: OWNER, phase: PHASE, exact_owner_prefix: BASE_REL, owner_paths_before_manifests: pathsBeforeManifests.length, non_owner_paths: [], allowed_extensions: [".json", ".md", ".txt", ".html"], formats_valid: pathsBeforeManifests.every((repositoryPath) => [".json", ".md", ".txt", ".html"].includes(path.extname(repositoryPath).toLowerCase())), sibling_lane_mutation: false, boundary: BOUNDARY }, null, 2)}\n`);

  const deltaPath = `${BASE_REL}/validation/final-delta-manifest.json`;
  const ownerPath = `${BASE_REL}/validation/final-owner-manifest.json`;
  const differs = (repositoryPath) => !committed.has(repositoryPath) || !fs.readFileSync(path.join(ROOT, ...repositoryPath.split("/"))).equals(blob("HEAD", repositoryPath));
  const deltaEntries = ownerFiles().filter((repositoryPath) => differs(repositoryPath) && repositoryPath !== deltaPath && repositoryPath !== ownerPath).map(bindingFromFile);
  fs.writeFileSync(path.join(BASE, "validation", "final-delta-manifest.json"), `${JSON.stringify({ schema: "ghc.family.owner.delta-manifest.v1", owner: OWNER, phase: PHASE, entries: deltaEntries, self_exclusions: [deltaPath, ownerPath], total_files: deltaEntries.length + 2, byte_domain: "raw Git blob and normalized-LF UTF-8 checkout parity required", boundary: BOUNDARY }, null, 2)}\n`);
  fs.writeFileSync(path.join(BASE, "validation", "final-owner-manifest.json"), `${JSON.stringify(manifest("validation/final-owner-manifest.json"), null, 2)}\n`);
  const maximum = ownerFiles().map((repositoryPath) => ({ repositoryPath, words: fs.readFileSync(path.join(ROOT, ...repositoryPath.split("/")), "utf8").trim().split(/\s+/).filter(Boolean).length })).sort((a, b) => b.words - a.words)[0];
  assert(maximum.words <= 100000, `corrected final maximum words ${maximum.words}`);
  const result = {
    state: "VALID_DEPENDENCY_CORRECTED_FINAL_WITHOUT_FINAL_REPLAY",
    owner_counts: own,
    effective_counts: effective,
    lifecycle_failures: LIFECYCLE_FAILURES.length,
    owner_files: ownerFiles().length,
    maximum_document: maximum,
    final_invocations: 1,
    final_successes: 1,
    final_replays: 0,
    boundary: BOUNDARY,
  };
  writeExternal("final/precommit-correction.json", result);
  return result;
}

function finalPreflight() {
  const plan = readJson("plan.json");
  const x1 = readJson("x1.json");
  const x2 = readJson("x2.json");
  const methodFlow = combineMethodFlows(plan.method_flow, x1.method_flow, lifecycleCorrectionMethodFlow(), x2.method_flow, closeoutMethodFlow(plan, x1, x2));
  const own = {
    methods: methodFlow.counts.methods,
    witnesses: methodFlow.counts.witnesses,
    passing_witnesses: methodFlow.counts.bounded_passing_witnesses,
    failed_witnesses: methodFlow.counts.retained_failed_witnesses,
    negatives: methodFlow.counts.retained_failed_witnesses,
    open_gaps: 30,
    exact_gates: 30,
    outcomes: plan.outcomes,
  };
  const effective = {
    methods: SOURCE_BASELINE.methods + own.methods,
    witnesses: SOURCE_BASELINE.witnesses + own.witnesses,
    passing_witnesses: SOURCE_BASELINE.bounded_passing_witnesses + own.passing_witnesses,
    failed_witnesses: SOURCE_BASELINE.retained_failed_witnesses + own.failed_witnesses,
    negatives: SOURCE_BASELINE.effective_negatives + own.negatives,
    open_gaps: SOURCE_BASELINE.open_gaps + own.open_gaps,
    exact_gates: SOURCE_BASELINE.exact_gates + own.exact_gates,
  };
  const negatives = buildNegativeRegister(plan, x1, x2);
  const cards = buildFlashcards(plan);
  const catalogue = buildCatalogue(x1, x2);
  const projectedMethodFlowIndex = {
    schema: "ghc.family.method-flow.index.v1",
    counts: methodFlow.counts,
    shards: [plan.method_flow.counts, x1.method_flow.counts, lifecycleCorrectionMethodFlow().counts, x2.method_flow.counts, closeoutMethodFlow(plan, x1, x2).counts],
    full_combined_document_withheld: true,
  };
  const projectedFinal = {
    schema: "projection-only",
    proposals: { inherited: 600, new: 600, outcomes: plan.outcomes },
    own,
    effective,
    method_flow: projectedMethodFlowIndex,
    operational_failures: [...plan.preflight_failures, ...LIFECYCLE_FAILURES],
    retained_failures: negatives,
    skills: [...x1.skills, ...x2.skills],
    runners: [...x1.runners, ...x2.runners],
    catalogue,
    terminal_verdict: "NOT_READY_FOR_STAGE_20",
  };
  const words = (value) => JSON.stringify(value).trim().split(/\s+/).filter(Boolean).length;
  const handoffWords = buildHandoff(plan, { own, effective }).trim().split(/\s+/).filter(Boolean).length;
  const currentDocuments = ownerFiles().map((repositoryPath) => ({ repositoryPath, words: fs.readFileSync(path.join(ROOT, ...repositoryPath.split("/")), "utf8").trim().split(/\s+/).filter(Boolean).length }));
  const result = {
    state: "VALID_FINAL_PREFLIGHT_NO_REPOSITORY_OUTPUT",
    projected_final_words: words(projectedFinal),
    projected_method_flow_words: words(projectedMethodFlowIndex),
    projected_largest_method_flow_shard_words: Math.max(words(plan.method_flow), words(x1.method_flow), words(lifecycleCorrectionMethodFlow()), words(x2.method_flow), words(closeoutMethodFlow(plan, x1, x2))),
    projected_negative_register_words: words(negatives),
    projected_flashcard_words: words(cards),
    projected_handoff_words: handoffWords,
    current_maximum_document: currentDocuments.sort((a, b) => b.words - a.words)[0],
    owner_counts: own,
    effective_counts: effective,
    under_ceiling: words(projectedFinal) <= 100000 && Math.max(words(plan.method_flow), words(x1.method_flow), words(lifecycleCorrectionMethodFlow()), words(x2.method_flow), words(closeoutMethodFlow(plan, x1, x2))) <= 100000 && words(negatives) <= 100000 && words(cards) <= 100000 && handoffWords <= 100000,
    repository_written: false,
    boundary: BOUNDARY,
  };
  result.handoff_floor_met = handoffWords >= 10000;
  return result;
}

function main() {
  const mode = process.argv[2] || "selftest";
  let result;
  if (mode === "selftest") result = selfTest();
  else if (mode === "plan") result = runPlan();
  else if (mode === "plan-correct") result = correctPlanningPrecommit();
  else if (mode === "x1") result = executeStage("x1");
  else if (mode === "x1-correct") result = correctX1Precommit();
  else if (mode === "x2") result = executeStage("x2");
  else if (mode === "final-preflight") result = finalPreflight();
  else if (mode === "final-correct") result = correctFinalPrecommit();
  else if (mode === "final") {
    writeExternal("final/invocation.json", { owner: OWNER, phase: PHASE, state: "INVOKED", invocations: 1, replays: 0, head_before: git(["rev-parse", "HEAD"]), utc: new Date().toISOString() });
    result = buildFinal();
  } else if (mode === "canonical") result = runCanonical(process.argv[3]);
  else throw new Error(`unknown mode: ${mode}`);
  process.stdout.write(`${JSON.stringify({ mode, owner: OWNER, phase: PHASE, state: result.state || "VALID", summary: mode === "final-preflight" ? result : result.checks_total ? { checks: result.checks_total } : result.contracts ? { contracts: result.contracts.length } : result.contracts_checked ? result : { keys: Object.keys(result).length } }, null, 2)}\n`);
}

main();
