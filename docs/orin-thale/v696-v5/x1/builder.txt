const sha = bytes => crypto.createHash('sha256').update(bytes).digest('hex');
const canon = value => {
  if (Array.isArray(value)) return value.map(canon);
  if (value && typeof value === 'object') return Object.fromEntries(Object.keys(value).sort().map(key => [key, canon(value[key])]));
  return value;
};
const canonical = value => JSON.stringify(canon(value));
function write(rel, value, pretty = false) {
  const dest = path.join(base, rel);
  if (!dest.startsWith(base + path.sep)) throw new Error('path escape');
  fs.mkdirSync(path.dirname(dest), { recursive: true });
  const text = typeof value === 'string' ? value : (pretty ? JSON.stringify(value, null, 2) : JSON.stringify(value)) + '\n';
  fs.writeFileSync(dest, text, { encoding: 'utf8', flag: 'wx' });
}
function git(args, buffer = false) {
  const result = cp.spawnSync('git', ['-C', root, ...args], { encoding: buffer ? null : 'utf8', maxBuffer: 64 * 1024 * 1024 });
  if (result.status !== 0) throw new Error(`git ${args.join(' ')} failed: ${String(result.stderr)}`);
  return result.stdout;
}
