'use strict';
const cp=require('node:child_process'),path=require('node:path');
function git(root,args){const r=cp.spawnSync('git',['-C',root,...args],{encoding:'utf8',maxBuffer:67108864});if(r.status!==0)throw Error(r.stderr);return r.stdout;}
function canon(v){if(Array.isArray(v))return v.map(canon);if(v&&typeof v==='object')return Object.fromEntries(Object.keys(v).sort().map(k=>[k,canon(v[k])]));return v;}
const c=v=>JSON.stringify(canon(v));
function run(root,ref){const proposals=JSON.parse(git(root,['show',ref+':docs/orin-thale/v696-v5/plan/new-proposals.json'])).records,x1=require(path.join(root,'docs','orin-thale','v696-v5','x1','core.txt')),x2=require(path.join(root,'docs','orin-thale','v696-v5','x2','correction','corrected-core.txt')),tests=[];const add=(name,pass)=>tests.push({name,result:pass?'pass':'fail'});for(const op of [...new Set(proposals.filter(p=>p.stage==='x1').map(p=>p.operation))]){const p=proposals.find(x=>x.operation===op);add('x1 '+op,c(x1.execute(p.request).result)===c(p.expected));}for(const op of [...new Set(proposals.filter(p=>p.stage==='x2').map(p=>p.operation))]){const p=proposals.find(x=>x.operation===op);add('x2 '+op,c(x2.execute(p.request).result)===c(p.expected));}const p156=proposals.find(p=>p.proposal_id==='OR6965-P156'),r156=x2.execute(p156.request).result;add('P156 corrected distance',r156.value.distance===1);add('P156 immutable mismatch',c(r156)!==c(p156.expected));const truth=JSON.parse(git(root,['show',ref+':docs/orin-thale/v696-v5/final/phase-truth.json']));add('corrected outcomes',c(truth.outcomes)===c({completed:239,represented:31,open_gap:15,exact_gate:15}));const deck=JSON.parse(git(root,['show',ref+':docs/orin-thale/v696-v5/final/deck/deck-index.json']));add('four tier deck',deck.count===312&&c(deck.tier_counts)===c({'1':1,'2':3,'3':8,'4':300}));const route=JSON.parse(git(root,['show',ref+':docs/orin-thale/v696-v5/final/routing-packet.json']));add('route held',route.state==='PREPARED_NOT_SENT'&&route.prospective_successor.owner==='Ceryn Alder');return {status:tests.every(t=>t.result==='pass')?'VALID_OWNER_TESTS':'INVALID',count:tests.length,passed:tests.filter(t=>t.result==='pass').length,failed:tests.filter(t=>t.result==='fail').length,tests};}
module.exports={run};
