'use strict';
const fs=require('node:fs'),path=require('node:path'),crypto=require('node:crypto');
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
function main(root){root=path.resolve(root||'.');const base=path.join(root,'docs','orin-thale','v696-v5','final','correction'),files=fs.readdirSync(path.join(root,'docs','orin-thale','v696-v5','final'),{recursive:true}).filter(rel=>fs.statSync(path.join(root,'docs','orin-thale','v696-v5','final',rel)).isFile()).map(String),json=files.filter(rel=>rel.endsWith('.json'));for(const rel of json)JSON.parse(fs.readFileSync(path.join(root,'docs','orin-thale','v696-v5','final',rel),'utf8'));const fm=JSON.parse(fs.readFileSync(path.join(base,'final-manifest.json'),'utf8')),om=JSON.parse(fs.readFileSync(path.join(base,'owner-manifest.json'),'utf8'));let failures=0;for(const row of [...fm.entries,...om.bindings]){const bytes=fs.readFileSync(path.join(root,...row.path.split('/')));if(bytes.length!==row.bytes||sha(bytes)!==row.sha256)failures++;}const baton=JSON.parse(fs.readFileSync(path.join(base,'baton-index.json'),'utf8')),truth=JSON.parse(fs.readFileSync(path.join(base,'phase-truth.json'),'utf8')),deck=JSON.parse(fs.readFileSync(path.join(base,'four-tier-validation.json'),'utf8'));return{status:failures===0&&baton.words>=2000&&baton.words<=100000&&truth.selected_lineage_effective_negatives===4479&&deck.status==='VALID_FOUR_TIER_PARENT_AND_SELECTION_SMOKE'?'VALID_TERMINAL_CORRECTION_PRECOMMIT':'INVALID',final_files:files.length,json:json.length,manifest_failures:failures,baton_words:baton.words,deck_status:deck.status};}
module.exports={main};
