'use strict';
const fs=require('node:fs'),path=require('node:path'),crypto=require('node:crypto');const sha=b=>crypto.createHash('sha256').update(b).digest('hex');function main(root){root=path.resolve(root||'.');const base=path.join(root,'docs','orin-thale','v696-v5','final'),files=fs.readdirSync(base,{recursive:true}).filter(r=>fs.statSync(path.join(base,r)).isFile()).map(String),json=files.filter(f=>f.endsWith('.json'));for(const rel of json)JSON.parse(fs.readFileSync(path.join(base,rel),'utf8'));const manifest=JSON.parse(fs.readFileSync(path.join(base,'manifest.json'),'utf8')),owner=JSON.parse(fs.readFileSync(path.join(base,'owner-manifest.json'),'utf8'));let failures=0;for(const e of manifest.entries){const b=fs.readFileSync(path.join(root,...e.path.split('/')));if(b.length!==e.bytes||sha(b)!==e.sha256)failures++;}for(const e of owner.bindings){const b=fs.readFileSync(path.join(root,...e.path.split('/')));if(b.length!==e.bytes||sha(b)!==e.sha256)failures++;}const baton=JSON.parse(fs.readFileSync(path.join(base,'baton-index.json'),'utf8')),truth=JSON.parse(fs.readFileSync(path.join(base,'phase-truth.json'),'utf8')),deck=JSON.parse(fs.readFileSync(path.join(base,'deck','deck-index.json'),'utf8'));return {status:failures===0&&baton.words>=2000&&baton.words<=100000&&truth.invalidated_completion==='OR6965-P156'&&deck.count===312?'VALID_FINAL_PRECOMMIT':'INVALID',files:files.length,json:json.length,manifest_failures:failures,baton_words:baton.words,deck_cards:deck.count};}module.exports={main};
