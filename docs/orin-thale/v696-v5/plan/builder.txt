}

const commonPrefix = (a, b) => {
  let i = 0;
  while (i < a.length && i < b.length && a[i] === b[i]) i++;
  return a.slice(0, i);
};
const commonSuffix = (a, b) => {
  let i = 0;
  while (i < a.length && i < b.length && a[a.length - 1 - i] === b[b.length - 1 - i]) i++;
  return i === 0 ? [] : a.slice(a.length - i);
};
function lcsTable(a, b) {
  const dp = Array.from({ length: a.length + 1 }, () => Array(b.length + 1).fill(0));
  for (let i = a.length - 1; i >= 0; i--) {
    for (let j = b.length - 1; j >= 0; j--) {
      dp[i][j] = a[i] === b[j] ? 1 + dp[i + 1][j + 1] : Math.max(dp[i + 1][j], dp[i][j + 1]);
    }
  }
  return dp;
}
function lcsWitness(a, b) {
  const dp = lcsTable(a, b);
  const sequence = [];
  const pairs = [];
  let i = 0;
  let j = 0;
  while (i < a.length && j < b.length) {
    if (a[i] === b[j]) {
      sequence.push(a[i]);
      pairs.push([i, j]);
      i++;
      j++;
    } else if (dp[i + 1][j] >= dp[i][j + 1]) {
      i++;
    } else {
      j++;
    }
  }
  return { sequence, pairs, length: sequence.length };
}
function editTable(a, b, costs = { insert: 1, delete: 1, substitute: 1 }) {
  const dp = Array.from({ length: a.length + 1 }, () => Array(b.length + 1).fill(0));
  for (let i = 1; i <= a.length; i++) dp[i][0] = i * costs.delete;
  for (let j = 1; j <= b.length; j++) dp[0][j] = j * costs.insert;
  for (let i = 1; i <= a.length; i++) {
    for (let j = 1; j <= b.length; j++) {
      dp[i][j] = Math.min(
        dp[i - 1][j] + costs.delete,
        dp[i][j - 1] + costs.insert,
        dp[i - 1][j - 1] + (a[i - 1] === b[j - 1] ? 0 : costs.substitute)
      );
    }
  }
  return dp;
}
function damerauDistance(a, b) {
  const dp = editTable(a, b);
  for (let i = 2; i <= a.length; i++) {
    for (let j = 2; j <= b.length; j++) {
      if (a[i - 1] === b[j - 2] && a[i - 2] === b[j - 1]) {
        dp[i][j] = Math.min(dp[i][j], dp[i - 2][j - 2] + 1);
      }
    }
  }
  return dp[a.length][b.length];
}
function shortestCommonSupersequence(a, b) {
  const dp = lcsTable(a, b);
  const out = [];
  let i = 0;
  let j = 0;
  while (i < a.length || j < b.length) {
    if (i === a.length) out.push(b[j++]);
    else if (j === b.length) out.push(a[i++]);
    else if (a[i] === b[j]) {
      out.push(a[i]);
      i++;
      j++;
    } else if (dp[i + 1][j] >= dp[i][j + 1]) out.push(a[i++]);
    else out.push(b[j++]);
  }
  return out;
}
function alignmentColumns(a, b) {
  const dp = editTable(a, b);
  const columns = [];
  let i = a.length;
  let j = b.length;
  while (i > 0 || j > 0) {
    if (i > 0 && j > 0 && dp[i][j] === dp[i - 1][j - 1] + (a[i - 1] === b[j - 1] ? 0 : 1)) {
      columns.push([a[i - 1], b[j - 1], a[i - 1] === b[j - 1] ? 'match' : 'substitute']);
      i--;
      j--;
    } else if (i > 0 && dp[i][j] === dp[i - 1][j] + 1) {
      columns.push([a[i - 1], null, 'delete']);
      i--;
    } else {
      columns.push([null, b[j - 1], 'insert']);
      j--;
    }
  }
  columns.reverse();
  return { columns, cost: dp[a.length][b.length] };
}
function bigrams(sequence) {
  const out = [];
  for (let i = 0; i + 1 < sequence.length; i++) out.push(sequence[i] + '\u241f' + sequence[i + 1]);
  return [...new Set(out)].sort();
}
function oracle(operation, fixture) {
  const a = fixture.a;
  const b = fixture.b;
  switch (operation) {
    case 'seq_shape': return { a_length: a.length, b_length: b.length, total: a.length + b.length, bounded: true };
    case 'seq_reverse': return { a_reversed: [...a].reverse(), b_reversed: [...b].reverse() };
    case 'seq_concat': return { ab: [...a, ...b], ba: [...b, ...a] };
    case 'seq_common_prefix': { const value = commonPrefix(a, b); return { value, length: value.length }; }
    case 'seq_common_suffix': { const value = commonSuffix(a, b); return { value, length: value.length }; }
    case 'seq_mismatch_profile': {
      const overlap = Math.min(a.length, b.length);
      const mismatch_indices = [];
      for (let i = 0; i < overlap; i++) if (a[i] !== b[i]) mismatch_indices.push(i);
      return { overlap, mismatch_indices, length_delta: b.length - a.length };
    }
    case 'seq_lcs_length': return { length: lcsTable(a, b)[0][0] };
    case 'seq_lcs_witness': return lcsWitness(a, b);
    case 'seq_levenshtein': { const table = editTable(a, b); return { distance: table[a.length][b.length] }; }
    case 'seq_edit_matrix': { const table = editTable(a, b); return { rows: table.length, columns: table[0].length, table }; }
    case 'seq_damerau_distance': return { distance: damerauDistance(a, b) };
    case 'seq_weighted_distance': {
      const costs = { insert: 2, delete: 2, substitute: 3 };
      const table = editTable(a, b, costs);
      return { costs, distance: table[a.length][b.length] };
    }
    case 'seq_shortest_common_supersequence': { const value = shortestCommonSupersequence(a, b); return { value, length: value.length }; }
    case 'seq_alignment_columns': return alignmentColumns(a, b);
    case 'seq_bigram_overlap': {
      const left = bigrams(a);
      const right = bigrams(b);
      const intersection = left.filter(x => right.includes(x));
      const union = [...new Set([...left, ...right])].sort();
      return { left, right, intersection, union, intersection_count: intersection.length, union_count: union.length };
    }
    case 'seq_bounded_transducer': {
      const map = { a: 'A', e: 'E', i: 'I', o: 'O', u: 'U' };
      return { a_output: a.map(x => map[x] || x), b_output: b.map(x => map[x] || x), rule: 'ascii-vowel-uppercase' };
    }
    case 'seq_accessible_summary': {
      const distance = editTable(a, b)[a.length][b.length];
      return { fixture: fixture.id, a_length: a.length, b_length: b.length, edit_distance: distance, summary: `${fixture.label}: lengths ${a.length} and ${b.length}; edit distance ${distance}. Synthetic token sequences only.`, human_evaluation: false };
    }
    case 'seq_provenance_handover': return { fixture: fixture.id, input_sha256: sha(Buffer.from(canonical({ a, b }))), correction_state: 'none', external_authority: false };
    case 'seq_observation_gate': return { fixture: fixture.id, observed_rows: 0, required: ['real language samples', 'declared segmentation profile', 'independent review'], gate_open: true };
    case 'seq_authority_gate': return { fixture: fixture.id, authority_present: false, required: ['competent domain authority', 'affected-party authority', 'legal and cultural authority', 'Maori authority where applicable'], release: false };
    default: throw new Error(`unknown operation ${operation}`);
  }
}
