'use strict';
const fs=require('node:fs'),path=require('node:path'),crypto=require('node:crypto');const sha=b=>crypto.createHash('sha256').update(b).digest('hex');function main(root){root=path.resolve(root||'.');const base=path.join(root,'docs','orin-thale','v696-v5'),terminal=path.join(base,'terminal'),files=fs.readdirSync(base,{recursive:true}).filter(rel=>fs.statSync(path.join(base,rel)).isFile()).map(String),json=files.filter(rel=>rel.endsWith('.json'));for(const rel of json)JSON.parse(fs.readFileSync(path.join(base,rel),'utf8'));const fm=JSON.parse(fs.readFileSync(path.join(terminal,'final-manifest.json'),'utf8')),om=JSON.parse(fs.readFileSync(path.join(terminal,'owner-manifest.json'),'utf8'));let failures=0;for(const row of [...fm.entries,...om.bindings]){const bytes=fs.readFileSync(path.join(root,...row.path.split('/')));if(bytes.length!==row.bytes||sha(bytes)!==row.sha256)failures++;}let over=0,maxWords=0;for(const rel of files){const text=fs.readFileSync(path.join(base,rel),'utf8'),words=(text.match(/\S+/g)||[]).length;maxWords=Math.max(maxWords,words);if(words>100000)over++;}const baton=JSON.parse(fs.readFileSync(path.join(terminal,'baton-index.json'),'utf8')),truth=JSON.parse(fs.readFileSync(path.join(terminal,'phase-truth.json'),'utf8')),chunks=JSON.parse(fs.readFileSync(path.join(terminal,'witness-chunk-manifest.json'),'utf8'));return{status:failures===0&&over===0&&baton.words>=2000&&baton.words<=100000&&truth.selected_lineage_effective_negatives===4481&&chunks.total_witnesses===3112?'VALID_CANONICAL_PRIVACY_CORRECTION_PRECOMMIT':'INVALID',owner_files:files.length,json:json.length,manifest_failures:failures,documents_over_ceiling:over,maximum_document_words:maxWords,baton_words:baton.words,witness_chunks:chunks.count};}module.exports={main};
