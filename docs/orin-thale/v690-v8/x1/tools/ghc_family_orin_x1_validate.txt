function check(name, condition, detail = null) {
  checks.push({ name, pass: Boolean(condition), detail });

}

function walk(dir) {
  const out = [];
  for (const entry of fs.readdirSync(dir, { withFileTypes: true }).sort((a, b) => a.name.localeCompare(b.name))) {
    const full = path.join(dir, entry.name);
    if (entry.isDirectory()) out.push(...walk(full)); else out.push(full);
  }
  return out;
}
function digest(bytes) { return crypto.createHash("sha256").update(bytes).digest("hex"); }

const cr = [];
const uuid = [];
const privatePath = [];
const credential = [];
const route = [];
const routePattern = new RegExp(`(?:${[["source", "thread", "id"].join("_"), ["provider", "Tab", "Id"].join(""), ["client", "Thread", "Id"].join(""), ["thread", "Id"].join("") + "\\s*[=:]"].join("|")})`);

  const rel = path.relative(x1Root, file).replaceAll("\\", "/");
  const text = bytes.toString("utf8");
  if (bytes.includes(13)) cr.push(rel);
  if (/\b[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}\b/i.test(text)) uuid.push(rel);
  if (/(?:[A-Z]:\\Users\\|[A-Z]:\\GHC-Archives\\|\/Users\/|\/home\/)/.test(text)) privatePath.push(rel);
  if (/(?:BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY|ghp_[A-Za-z0-9]{20,}|sk-[A-Za-z0-9]{20,})/.test(text)) credential.push(rel);
  if (routePattern.test(text)) route.push(rel);
}
