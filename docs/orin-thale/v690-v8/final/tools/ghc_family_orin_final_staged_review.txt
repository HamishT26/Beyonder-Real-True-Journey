"use strict";
const fs = require("fs");
const path = require("path");
const crypto = require("crypto");
const cp = require("child_process");
const finalRoot = path.resolve(__dirname, "..");
const ownerRoot = path.resolve(finalRoot, "..");
const repo = path.resolve(ownerRoot, "..", "..", "..");
const expectedParent = "279ae0e5eadf53587f0f1560465396e8e65c9558";
const receiptPath = path.join(finalRoot, "staged-review.json");
const receiptRepoPath = "docs/orin-thale/v690-v8/final/staged-review.json";
function git(args, encoding = "utf8") { return cp.execFileSync("git", ["-C", repo, ...args], { encoding, maxBuffer: 128 * 1024 * 1024 }); }
const head = git(["rev-parse", "HEAD"]).trim();
const stagedPaths = git(["diff", "--cached", "--name-only", "--diff-filter=ACMR", "-z"], "buffer").toString("utf8").split("\0").filter(Boolean).sort();
const selected = stagedPaths.filter((p) => p !== receiptRepoPath);
const invalid = selected.filter((p) => !p.startsWith("docs/orin-thale/v690-v8/final/"));
const entries = selected.map((p) => { const working = fs.readFileSync(path.join(repo, ...p.split("/"))); const staged = git(["show", `:${p}`], "buffer"); return { path: p, bytes: working.length, sha256: crypto.createHash("sha256").update(working).digest("hex"), staged_bytes: staged.length, staged_sha256: crypto.createHash("sha256").update(staged).digest("hex"), working_equals_staged: working.equals(staged) }; });
let diffCheckExit = 0; let diffCheckOutput = "";
try { diffCheckOutput = git(["diff", "--cached", "--check"]).trim(); } catch (error) { diffCheckExit = error.status ?? 1; diffCheckOutput = String(error.stdout || error.stderr || error.message).trim(); }
const receipt = { schema: "ghc.family.exact-staged-review.v5", owner: "Orin Thale", phase: "v690-v8", stage: "final", expected_parent: expectedParent, observed_parent: head, selected_count: entries.length, entries, self_exclusions: [receiptRepoPath], invalid_paths: invalid, working_staged_mismatches: entries.filter((row) => !row.working_equals_staged).map((row) => row.path), diff_check_exit: diffCheckExit, diff_check_output: diffCheckOutput, exact_allowlist_pass: head === expectedParent && invalid.length === 0 && entries.length > 0 && entries.every((row) => row.working_equals_staged) && diffCheckExit === 0, boundary: "Exact final-only staged review; no successor delivery, empirical, professional, production, legal, cultural, Maori-authority, independent-reproduction or Stage 20 claim." };
fs.writeFileSync(receiptPath, JSON.stringify(receipt, null, 2) + "\n", "utf8");
process.stdout.write(JSON.stringify({ valid: receipt.exact_allowlist_pass, selected_count: entries.length, expected_parent: expectedParent, observed_parent: head, invalid_paths: invalid.length }, null, 2) + "\n");
if (!receipt.exact_allowlist_pass) process.exitCode = 1;
