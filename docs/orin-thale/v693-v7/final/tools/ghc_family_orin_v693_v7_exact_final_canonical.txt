function gitBuffer(args) { return cp.execFileSync("git", ["-C", repo, ...args], { encoding: null }); }
function git(args) { return gitBuffer(args).toString("utf8").trim(); }
function blob(commit, rel) { return gitBuffer(["show", `${commit}:${rel}`]); }
function jsonBlob(commit, rel) { return JSON.parse(blob(commit, rel).toString("utf8")); }
function stable(value) { if (Array.isArray(value)) return value.map(stable); if (value && typeof value === "object") return Object.fromEntries(Object.keys(value).sort().map((key) => [key, stable(value[key])])); return value; }

function loadTextModule(bytes) { const module = { exports: {} }; new Function("require", "module", "exports", bytes.toString("utf8"))(require, module, module.exports); return module.exports; }
function equality() {
  const branch = git(["branch", "--show-current"]);
  const local = git(["rev-parse", "HEAD"]);
  const upstream = git(["rev-parse", "@{upstream}"]);
  const tracking = git(["rev-parse", `refs/remotes/origin/${branch}`]);
  const liveLine = git(["ls-remote", "--heads", "origin", `refs/heads/${branch}`]);
  const freshLive = liveLine.split(/\s+/)[0] || "";
  const divergence = git(["rev-list", "--left-right", "--count", `${local}...${upstream}`]).split(/\s+/).map(Number);
  const clean = git(["status", "--porcelain"]) === "";
  return { branch, local, upstream, tracking, fresh_live: freshLive, divergence, clean, four_way_equal: local === upstream && local === tracking && local === freshLive };
}
function manifestCheck(commit, rel) {
  const manifest = jsonBlob(commit, rel);
  const issues = [];
  for (const row of manifest.bindings) {
    let bytes;
    try { bytes = blob(commit, row.path); } catch { issues.push(`${row.path}:missing`); continue; }
    if (bytes.length !== row.bytes) issues.push(`${row.path}:bytes`);
    if (sha(bytes) !== row.sha256) issues.push(`${row.path}:sha256`);
  }
  return { rel, commit, bindings: manifest.bindings.length, exclusions: manifest.declared_exclusions || (manifest.declared_self_exclusion ? [manifest.declared_self_exclusion] : []), issues };
}
