function stable(value) { if (Array.isArray(value)) return value.map(stable); if (value && typeof value === "object") return Object.fromEntries(Object.keys(value).sort().map((k) => [k, stable(value[k])])); return value; }
function shaObject(value) { return crypto.createHash("sha256").update(JSON.stringify(stable(value)), "utf8").digest("hex"); }
function files(dir) { const out = []; for (const entry of fs.readdirSync(dir, { withFileTypes: true })) { const full = path.join(dir, entry.name); if (entry.isDirectory()) out.push(...files(full)); else out.push(full); } return out; }
function words(text) { return (String(text).match(/\S+/g) || []).length; }

check("four-stage lifecycle", JSON.stringify(profile.lifecycle) === JSON.stringify(["planning_freeze", "x1_execution", "x2_execution", "final_closeout"]));
check("owner lane reuse and explicit predecessor provenance", profile.prior_owner_final === "2dd71e0c6619d12e689c7082dcd66001e261565d" && profile.predecessor_provenance === "c3d8b45a08328ea7193f9a3e780241a3a1ef6628" && profile.predecessor_is_git_ancestor === false);
check("source exact anchors", source.exact_final === "c3d8b45a08328ea7193f9a3e780241a3a1ef6628" && source.planning === "b70e3ce4ad33847f1550a283b34ec9c3dc953c9e" && source.x1 === "f9e675e4bdb2c79042bcd4ed0007a1a2087ebacf" && source.x2 === "7940d83879c628b47c607a162d7707b9de5e77b6");
check("source canonical not replayed", source.canonical.invocations === 1 && source.canonical.successes === 1 && source.canonical.replays === 0 && source.canonical.replayed_by_orin === false);
check("source baton binding", source.baton.sha256 === "3e94a9ee5f7251f19cb6767cfa57ea92a5a7642346e5ac04f18ecbbdee2110b8" && source.baton.literal_eof_verified === true);
check("guidance skill hashes", guidance.skills.length === 18 && guidance.skills.every((row) => /^[0-9a-f]{64}$/.test(row.sha256) && row.read_through_eof === true));
check("guidance reference hashes", guidance.references.length === 16 && guidance.references.every((row) => /^[0-9a-f]{64}$/.test(row.sha256) && row.read_through_eof === true));
check("roster parse and edge", guidance.roster_rows_parsed === 269 && guidance.roster_validation_issues_after_recovery === 0 && guidance.current_edge === "Orin Thale v693-v7" && guidance.following_edge.includes("Liora Venn v693-v8"));
check("four practices and primary pillar", identity.practices.length === 4 && identity.primary_pillar === "GMUT Mind");
check("inherited reviews", inherited.count === 100 && inherited.records.length === 100 && inherited.records.every((row) => row.novelty_credit === 0 && row.execution_credit === 0));
check("twenty operation families", operations.count === 20 && new Set(operations.families.map((row) => row.operation)).size === 20);
check("one hundred unique proposals", rows.length === 100 && new Set(rows.map((row) => row.proposal_id)).size === 100);
check("fifty proposals per session", rows.filter((row) => row.session === "x1").length === 50 && rows.filter((row) => row.session === "x2").length === 50);
check("five fixtures per operation", operations.families.every((family) => rows.filter((row) => row.operation === family.operation).length === 5));
const allowedOutcomes = new Set(["completed", "represented", "open_gap", "exact_gate"]);
check("authorized outcome vocabulary", rows.every((row) => allowedOutcomes.has(row.expected_execution_disposition)));
check("outcome ceilings", rows.filter((row) => row.expected_execution_disposition === "completed").length === 80 && rows.filter((row) => row.expected_execution_disposition === "represented").length === 10 && rows.filter((row) => row.expected_execution_disposition === "open_gap").length === 5 && rows.filter((row) => row.expected_execution_disposition === "exact_gate").length === 5);

check("proposal definition hashes", rows.every((row) => row.definition_sha256 === shaObject({ proposal_id: row.proposal_id, operation: row.operation, request: row.request, expected_output: row.expected_output })));

check("startup Method Flow arithmetic", methodflow.counts.methods === 4 && methodflow.counts.witnesses === 8 && methodflow.counts.pass === 4 && methodflow.counts.fail === 4 && methodflow.counts.projected_negative_lineage === 2330);
check("failed witnesses retained", methodflow.witnesses.filter((row) => row.result === "fail").length === 4 && methodflow.witnesses.every((row) => row.same_owner_only === true && row.independent_reproduction === false));
check("route remains terminally gated", route.state === "PREPARED_NOT_SENT" && route.prospective_successor.title === "Liora Venn" && route.prospective_successor.phase === "v693-v8" && route.early_contact === false && route.maximum_send_attempts === 1);
check("manifest domain and exclusions", manifest.hash_domain === "working_tree_bytes_expected_to_equal_raw_git_blob_after_commit" && manifest.declared_exclusions.length === 3);
const manifestIssues = [];
for (const row of manifest.bindings) {
  const full = path.join(repo, ...row.path.split("/"));
  if (!fs.existsSync(full)) manifestIssues.push(`${row.path}:missing`);
  else {
    const bytes = fs.readFileSync(full);
    if (bytes.length !== row.bytes) manifestIssues.push(`${row.path}:bytes`);
    if (crypto.createHash("sha256").update(bytes).digest("hex") !== row.sha256) manifestIssues.push(`${row.path}:sha256`);
  }
}
check("planning manifest working-byte parity", manifestIssues.length === 0, manifestIssues);
check("staged allowlist owner-only", allowlist.paths.every((rel) => rel.startsWith(`${baseRel}/`)) && allowlist.execution_results_forbidden === true);

const forbiddenExtensions = allFiles.filter((file) => ![".json", ".md", ".txt", ".html", ".tex"].includes(path.extname(file).toLowerCase()));
check("authorized file types only", forbiddenExtensions.length === 0, forbiddenExtensions.map((file) => path.relative(repo, file)));
const overWords = allFiles.filter((file) => [".json", ".md", ".txt", ".html", ".tex"].includes(path.extname(file).toLowerCase())).map((file) => ({ file, words: words(fs.readFileSync(file, "utf8")) })).filter((row) => row.words > 100000);
check("document word ceiling", overWords.length === 0, overWords.map((row) => ({ path: path.relative(repo, row.file), words: row.words })));

  try { new vm.Script(`(function(require,module,exports){\n${fs.readFileSync(file, "utf8")}\n})`, { filename: file }); }
  catch (error) { check(`trusted TXT compiles ${path.basename(file)}`, false, error.message); }
}
check("trusted TXT compilation", !checks.some((row) => row.name.startsWith("trusted TXT compiles") && !row.pass));
const serialized = allFiles.map((file) => fs.readFileSync(file, "utf8")).join("\n");
const privacyPatterns = [
  ["private_task_uuid", /\b[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}\b/ig],

  ["private_c_path", /C:\\Users\\[^\s\"']+/ig],
  ["private_d_path", /D:\\GHC-Archives\\[^\s\"']+/ig],
  ["email", /\b[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}\b/ig]
];
const privacy = privacyPatterns.map(([kind, pattern]) => ({ kind, candidates: [...serialized.matchAll(pattern)].map((m) => m[0]).slice(0, 10) }));
