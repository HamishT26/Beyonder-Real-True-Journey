function stable(value) {
  if (Array.isArray(value)) return value.map(stable);
  if (value && typeof value === "object") {
    return Object.fromEntries(Object.keys(value).sort().map((key) => [key, stable(value[key])]));
  }
  return value;
}
function canonical(value) { return JSON.stringify(stable(value)); }

function shaObject(value) { return shaText(canonical(value)); }

function gcd(a, b) { a = a < 0n ? -a : a; b = b < 0n ? -b : b; while (b) [a, b] = [b, a % b]; return a; }
function F(n, d = 1n) {
  n = BigInt(n); d = BigInt(d);
  if (d === 0n) throw new Error("zero denominator");
  if (d < 0n) { n = -n; d = -d; }
  const g = gcd(n, d);
  return { n: n / g, d: d / g };
}
function parseF(token) {
  if (typeof token !== "string" || !/^-?\d+(\/\d+)?$/.test(token)) throw new Error("invalid rational token");
  const [a, b = "1"] = token.split("/");
  return F(BigInt(a), BigInt(b));
}
function add(a, b) { return F(a.n * b.d + b.n * a.d, a.d * b.d); }
function sub(a, b) { return F(a.n * b.d - b.n * a.d, a.d * b.d); }
function mul(a, b) { return F(a.n * b.n, a.d * b.d); }
function abs(a) { return F(a.n < 0n ? -a.n : a.n, a.d); }
function cmp(a, b) { const x = a.n * b.d - b.n * a.d; return x < 0n ? -1 : x > 0n ? 1 : 0; }
function fmt(a) { return a.d === 1n ? String(a.n) : `${a.n}/${a.d}`; }
function sum(xs) { return xs.reduce((acc, x) => add(acc, x), F(0)); }
function parseVector(xs) { return xs.map(parseF); }
function fmtVector(xs) { return xs.map(fmt); }
function validateDistribution(tokens) {
  const xs = parseVector(tokens);
  if (xs.some((x) => x.n < 0n)) throw new Error("E_PROBABILITY");
  if (cmp(sum(xs), F(1)) !== 0) throw new Error("E_PROBABILITY");
  return xs;
}
function tv(p, q) { return mul(F(1, 2), sum(p.map((x, i) => abs(sub(x, q[i]))))); }
function project(xs, partition) { return partition.map((block) => sum(block.map((i) => xs[i]))); }
function mix(a, b, alpha) { return a.map((x, i) => add(mul(alpha, x), mul(sub(F(1), alpha), b[i]))); }
function expectedLosses(dist, losses) { return losses.map((row) => sum(row.map((x, i) => mul(parseF(x), dist[i])))); }
function argmin(xs) { let index = 0; for (let i = 1; i < xs.length; i++) if (cmp(xs[i], xs[index]) < 0) index = i; return index; }

function oracle(req) {
  if (!allowedFields[req.op]) return { ok: false, value: null, error: "E_OPERATION" };
  if (Object.keys(req).some((key) => !allowedFields[req.op].includes(key)) || allowedFields[req.op].some((key) => !Object.hasOwn(req, key))) return { ok: false, value: null, error: "E_FIELDS" };
  let p, q;
  try {
    p = validateDistribution(req.p); q = validateDistribution(req.q);
    if (p.length !== q.length || p.length < 1 || p.length > 8) throw new Error("E_DIMENSION");
  } catch (error) { return { ok: false, value: null, error: error.message === "E_PROBABILITY" ? "E_PROBABILITY" : "E_DISTRIBUTION" }; }
  const residuals = p.map((x, i) => sub(x, q[i]));
  const absolute = residuals.map(abs);
  const totalVariation = tv(p, q);
  const event = Array.isArray(req.event) ? req.event : [];
  const eventP = event.length ? sum(event.map((i) => p[i])) : F(0);
  const eventQ = event.length ? sum(event.map((i) => q[i])) : F(0);
  const qdiff = sum(residuals.map((x) => mul(x, x)));
  if (req.op === "probability_total") return envelope({ p_total: fmt(sum(p)), q_total: fmt(sum(q)), p_nonnegative: true, q_nonnegative: true, dimensions_equal: true });
  if (req.op === "coordinate_residuals") return envelope({ residuals: fmtVector(residuals), residual_sum: fmt(sum(residuals)) });
  if (req.op === "l1_distance") return envelope({ absolute_residuals: fmtVector(absolute), l1: fmt(sum(absolute)) });
  if (req.op === "total_variation") return envelope({ total_variation: fmt(totalVariation), bounded: cmp(totalVariation, F(1)) <= 0 });
  if (req.op === "overlap_coefficient") return envelope({ overlap: fmt(sub(F(1), totalVariation)), complement_tv: fmt(totalVariation) });
  if (req.op === "maximum_coordinate_gap") {
    let index = 0; for (let i = 1; i < absolute.length; i++) if (cmp(absolute[i], absolute[index]) > 0) index = i;
    return envelope({ maximum: fmt(absolute[index]), witness_index: index, signed_residual: fmt(residuals[index]) });
  }
  if (req.op === "signed_event_gap") return envelope({ event: req.event, p_event: fmt(eventP), q_event: fmt(eventQ), signed_gap: fmt(sub(eventP, eventQ)) });
  if (req.op === "event_probability_table") return envelope({ event: req.event, complement: p.map((_, i) => i).filter((i) => !req.event.includes(i)), p_event: fmt(eventP), q_event: fmt(eventQ), absolute_gap: fmt(abs(sub(eventP, eventQ))) });
  if (req.op === "quadratic_discrepancy") return envelope({ squared_residuals: fmtVector(residuals.map((x) => mul(x, x))), quadratic_sum: fmt(qdiff) });
  if (req.op === "brier_score_pair") {
    if (!Number.isInteger(req.outcome) || req.outcome < 0 || req.outcome >= p.length) return { ok: false, value: null, error: "E_OUTCOME" };
    const score = (dist) => sum(dist.map((x, i) => mul(sub(x, F(i === req.outcome ? 1 : 0)), sub(x, F(i === req.outcome ? 1 : 0)))));
    const ps = score(p), qs = score(q);
    return envelope({ outcome: req.outcome, p_score: fmt(ps), q_score: fmt(qs), p_minus_q: fmt(sub(ps, qs)), lower: cmp(ps, qs) < 0 ? "p" : cmp(ps, qs) > 0 ? "q" : "tie" });
  }
  if (["expected_loss_table", "decision_regret", "minimax_model_loss"].includes(req.op)) {
    if (!Array.isArray(req.losses) || req.losses.length < 1 || req.losses.some((row) => !Array.isArray(row) || row.length !== p.length)) return { ok: false, value: null, error: "E_LOSSES" };
    let lp, lq; try { lp = expectedLosses(p, req.losses); lq = expectedLosses(q, req.losses); } catch { return { ok: false, value: null, error: "E_LOSSES" }; }
    if (req.op === "expected_loss_table") return envelope({ under_p: fmtVector(lp), under_q: fmtVector(lq), p_optimal: argmin(lp), q_optimal: argmin(lq) });
    if (!Number.isInteger(req.chosen_action) || req.chosen_action < 0 || req.chosen_action >= lp.length) return { ok: false, value: null, error: "E_ACTION" };
    if (req.op === "decision_regret") {
      const best = argmin(lp); return envelope({ chosen_action: req.chosen_action, optimal_action_under_p: best, chosen_loss: fmt(lp[req.chosen_action]), optimal_loss: fmt(lp[best]), regret: fmt(sub(lp[req.chosen_action], lp[best])) });
    }
    const worst = lp.map((x, i) => maxF(x, lq[i])); const best = argmin(worst);
    return envelope({ worst_case_losses: fmtVector(worst), chosen_action: req.chosen_action, minimax_action: best, chosen_worst_loss: fmt(worst[req.chosen_action]), minimax_loss: fmt(worst[best]), regret: fmt(sub(worst[req.chosen_action], worst[best])) });
  }
  if (["coarse_partition_projection", "coarse_tv_monotonicity"].includes(req.op)) {
    if (!Array.isArray(req.partition) || req.partition.flat().length !== p.length || new Set(req.partition.flat()).size !== p.length) return { ok: false, value: null, error: "E_PARTITION" };
    const cp = project(p, req.partition), cq = project(q, req.partition);
    if (req.op === "coarse_partition_projection") return envelope({ partition: req.partition, p_coarse: fmtVector(cp), q_coarse: fmtVector(cq), coarse_residuals: fmtVector(cp.map((x, i) => sub(x, cq[i]))) });
    const coarse = tv(cp, cq); return envelope({ fine_tv: fmt(totalVariation), coarse_tv: fmt(coarse), contraction_gap: fmt(sub(totalVariation, coarse)), monotone: cmp(coarse, totalVariation) <= 0 });
  }
  if (req.op === "mixture_tv_convexity") {
    let p2, q2, alpha; try { p2 = validateDistribution(req.p2); q2 = validateDistribution(req.q2); alpha = parseF(req.alpha); } catch { return { ok: false, value: null, error: "E_MIXTURE" }; }
    if (p2.length !== p.length || q2.length !== p.length || cmp(alpha, F(0)) < 0 || cmp(alpha, F(1)) > 0) return { ok: false, value: null, error: "E_MIXTURE" };
    const pm = mix(p, p2, alpha), qm = mix(q, q2, alpha); const lhs = tv(pm, qm); const rhs = add(mul(alpha, totalVariation), mul(sub(F(1), alpha), tv(p2, q2)));
    return envelope({ alpha: fmt(alpha), p_mixture: fmtVector(pm), q_mixture: fmtVector(qm), mixture_tv: fmt(lhs), convex_bound: fmt(rhs), slack: fmt(sub(rhs, lhs)), certified: cmp(lhs, rhs) <= 0 });
  }
  if (req.op === "discrepancy_explanation") return envelope({ states: p.length, total_variation: fmt(totalVariation), summary: cmp(totalVariation, F(0)) === 0 ? "The supplied finite distributions are equal." : "The supplied finite distributions differ; the reported value is model-local and not observational evidence.", manual_accessibility_review: "open_gap" });
  if (req.op === "discrepancy_provenance") return envelope({ p_sha256: hashVector(req.p), q_sha256: hashVector(req.q), state_count: p.length, governed_observations: 0, external_acceptance: false });
  if (req.op === "observation_gap") return envelope({ model_states: p.length, observed_rows: 0, missing: ["governed observations", "sampling and calibration assessment", "independent model comparison"], release: "open_gap" });
  if (req.op === "authority_reservation") return envelope({ action: req.action, authority_verified: false, external_action_performed: false, release: "exact_gate", required: ["competent and affected authority", "exact action scope", "appropriate evidence and review"] });
  return { ok: false, value: null, error: "E_OPERATION" };
}
