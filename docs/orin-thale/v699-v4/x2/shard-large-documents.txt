from pathlib import Path
from datetime import datetime, timezone
import hashlib, json

ROOT=Path(__file__).resolve().parents[1]
def stable(v):return json.dumps(v,sort_keys=True,separators=(',',':'),ensure_ascii=False)
def sha(raw):return hashlib.sha256(raw).hexdigest()
def read(name):return json.loads((ROOT/name).read_text(encoding='utf-8'))
def write(name,value):
 p=ROOT/name;p.parent.mkdir(parents=True,exist_ok=True);assert not p.exists(),name
 p.write_text(stable(value)+'\n',encoding='utf-8',newline='\n')
def shard_rows(stage,document,key,size):
 path=ROOT/stage/(document+'.json');raw=path.read_bytes();data=json.loads(raw);rows=data.pop(key)
 refs=[]
 for index,start in enumerate(range(0,len(rows),size),1):
  body={'schema':'orin.document-shard.v1','stage':stage,'document':document,'field':key,'index':index,
        'start':start,'count':len(rows[start:start+size]),'rows':rows[start:start+size]}
  name=f'{stage}/{document}-{key}-{index:02d}.json';write(name,body)
  refs.append({'path':name,'index':index,'start':start,'count':body['count'],'sha256':sha((ROOT/name).read_bytes())})
 data[key+'_count']=len(rows);data[key+'_shards']=refs
 data.setdefault('document_packaging',{})['pre_shard_sha256']=sha(raw)
 data['document_packaging']['lossless_row_sharding']=True
 data['document_packaging']['domain_replayed']=False
 path.write_text(stable(data)+'\n',encoding='utf-8',newline='\n')
 return {'path':f'{stage}/{document}.json','field':key,'pre_shard_sha256':sha(raw),'post_shard_sha256':sha(path.read_bytes()),
         'row_count':len(rows),'shards':refs}
