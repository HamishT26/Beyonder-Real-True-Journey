class Fault extends Error{constructor(code){super(code);this.code=code;}}
function fail(code){throw new Fault(code);}
function gcd(a,b){a=a<0n?-a:a;while(b){const r=a%b;a=b;b=r;}return a||1n;}
class Q{
 constructor(n,d=1n){if(d===0n)fail('E_RATIONAL');if(d<0n){n=-n;d=-d;}const k=gcd(n,d);this.n=n/k;this.d=d/k;}
 add(b){return new Q(this.n*b.d+b.n*this.d,this.d*b.d);}
 sub(b){return new Q(this.n*b.d-b.n*this.d,this.d*b.d);}
 mul(b){return new Q(this.n*b.n,this.d*b.d);}
 cmp(b){const x=this.n*b.d-b.n*this.d;return x<0n?-1:x>0n?1:0;}
 eq(b){return this.cmp(b)===0;}
 abs(){return new Q(this.n<0n?-this.n:this.n,this.d);}
 pow(k){return new Q(this.n**BigInt(k),this.d**BigInt(k));}
 toString(){return this.d===1n?String(this.n):this.n+'/'+this.d;}
}

function rational(s){if(typeof s!=='string'||s.length>30||!/^(-?(?:0|[1-9]\d*))(?:\/([1-9]\d*))?$/.test(s))fail('E_RATIONAL');const[nn,dd='1']=s.split('/'),n=BigInt(nn),d=BigInt(dd);if(d>1024n||n>1000n*d||n< -1000n*d)fail('E_RATIONAL');return new Q(n,d);}
function strictParse(text){
 if(typeof text!=='string'||Buffer.byteLength(text,'utf8')>131072)fail('E_JSON');let i=0;
 const ws=()=>{while(/[\x20\t\r\n]/.test(text[i]||'\0'))i++;};
 function string(){const start=i++;let closed=false;while(i<text.length){const c=text[i++];if(c==='\\'){i++;continue;}if(c==='"'){closed=true;break;}if(c<' ')fail('E_JSON');}if(!closed)fail('E_JSON');try{return JSON.parse(text.slice(start,i));}catch{fail('E_JSON');}}
 function value(depth){if(depth>32)fail('E_JSON');ws();const c=text[i];
  if(c==='"')return string();
  if(c==='{'){i++;const o=Object.create(null),keys=new Set();ws();if(text[i]==='}'){i++;return o;}while(true){ws();if(text[i]!=='"')fail('E_JSON');const k=string();if(keys.has(k))fail('E_DUPLICATE');keys.add(k);ws();if(text[i++]!==':')fail('E_JSON');o[k]=value(depth+1);ws();const end=text[i++];if(end==='}')return o;if(end!==',')fail('E_JSON');}}
  if(c==='['){i++;const a=[];ws();if(text[i]===']'){i++;return a;}while(true){a.push(value(depth+1));ws();const end=text[i++];if(end===']')return a;if(end!==',')fail('E_JSON');}}
  for(const[token,v]of[['true',true],['false',false],['null',null]])if(text.startsWith(token,i)){i+=token.length;return v;}
  const m=text.slice(i).match(/^-?(?:0|[1-9]\d*)(?:\.\d+)?(?:[eE][+-]?\d+)?/);if(!m)fail('E_JSON');i+=m[0].length;const n=Number(m[0]);if(!Number.isFinite(n))fail('E_JSON');return n;
 }

}
function object(x){return x!==null&&typeof x==='object'&&!Array.isArray(x);}
function fields(x,keys){if(!object(x)||Object.keys(x).sort().join('|')!==[...keys].sort().join('|'))fail('E_FIELDS');}
const CORE=['robust_validate','vertex_span','support_bounds','row_total_variation','one_step_interval','policy_lower_value','policy_upper_value','robust_bellman','optimistic_bellman','robust_action_sets'];
function input(d){

 if(!vec(d.vertices,n)||!d.vertices.every(s=>vec(s,2)&&s.every(a=>vec(a,2)&&a.every(r=>vec(r,n))))||!vec(d.rewards,n)||!d.rewards.every(r=>vec(r,2))||!vec(d.terminal,n)||!vec(d.initial,n))fail('E_DIMENSION');

function request(req,allowed=CORE){fields(req,['op','input']);if(typeof req.op!=='string'||!allowed.includes(req.op))fail('E_OPERATION');return input(req.input);}
function dynamic(d){
 const{n,H,P,R,z,g,policy}=d,L=[z.slice()],U=[z.slice()],PL=[z.slice()],PU=[z.slice()],qLower=[],ties=[];
 for(let h=1;h<=H;h++){
  const scores=(previous,adversary)=>Array.from({length:n},(_,s)=>[0,1].map(a=>adversary(P[s][a].map(row=>R[s][a].add(g.mul(dot(row,previous)))))));
  const ql=scores(L[h-1],min),qu=scores(U[h-1],max),qlp=scores(PL[h-1],min),qup=scores(PU[h-1],max);
  L.push(ql.map(max));U.push(qu.map(max));PL.push(qlp.map((row,s)=>row[policy[s]]));PU.push(qup.map((row,s)=>row[policy[s]]));
  qLower.push(ql);ties.push(ql.map((row,s)=>[0,1].filter(a=>row[a].eq(L[h][s]))));
 }

}
function encode(v){if(v instanceof Q)return v.toString();if(Array.isArray(v))return v.map(encode);if(object(v))return Object.fromEntries(Object.entries(v).map(([k,x])=>[k,encode(x)]));return v;}
function output(op,v){return{ok:true,op,outcome:op==='uncertainty_model'?'represented':op==='robust_evidence_gap'?'open_gap':op==='robust_authority_gate'?'exact_gate':'completed',value:encode(v),external_credit:false,scope:'finite_synthetic_rectangular_uncertainty'};}
function evaluate(req){const d=request(req),{n,H,P,R,z,g,policy,states}=d,V=dynamic(d);let v;
 switch(req.op){
 case'robust_validate':v={states:n,actions:2,vertices_per_row:2,horizon:H,rectangular:true};break;
 case'vertex_span':v={lower:P.map(s=>s.map(a=>Array.from({length:n},(_,j)=>min(a.map(r=>r[j]))))),upper:P.map(s=>s.map(a=>Array.from({length:n},(_,j)=>max(a.map(r=>r[j]))))),coordinate_box_is_not_declared_set:true};break;
