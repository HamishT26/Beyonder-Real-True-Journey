\[
S=\int d^4x\sqrt{-g}\left[\frac{R-2\Lambda}{2\kappa}
-\frac12g^{\mu\nu}\nabla_\mu\psi\nabla_\nu\psi-V(\psi)\right]+S_{\rm matter}.
\]
\[
G_{\mu\nu}+\Lambda g_{\mu\nu}
=\kappa(T^{\rm matter}_{\mu\nu}+T^\psi_{\mu\nu}),\qquad
T^\psi_{\mu\nu}=\nabla_\mu\psi\nabla_\nu\psi
-g_{\mu\nu}\left[\frac12(\nabla\psi)^2+V(\psi)\right].
\]
\[
\Box\psi-V'(\psi)=0,\qquad
\nabla^\mu T^\psi_{\mu\nu}=(\Box\psi-V'(\psi))\nabla_\nu\psi.
\]
