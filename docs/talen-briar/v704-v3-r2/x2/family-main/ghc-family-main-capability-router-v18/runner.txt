'use strict';
const fs=require('fs'),path=require('path'),crypto=require('crypto');
const [input,output]=process.argv.slice(2);if(!input||!output||process.argv.length!==4)throw Error('Two literal paths required');
const q=JSON.parse(fs.readFileSync(input,'utf8'));let result;
if(!q||Array.isArray(q)||Object.keys(q).sort().join('|')!=='limit|query'||typeof q.query!=='string'||q.query.length>200||!Number.isInteger(q.limit)||q.limit<1||q.limit>50){result={ok:false,error:'E_QUERY'};}
else{const index=JSON.parse(fs.readFileSync(path.join(__dirname,'index.json'),'utf8'));const records=[];for(const s of index.shards){const raw=fs.readFileSync(path.join(__dirname,s.path));if(crypto.createHash('sha256').update(raw).digest('hex')!==s.sha256)throw Error('Catalogue fixity failed');records.push(...JSON.parse(raw).records);}const terms=q.query.toLowerCase().trim().split(/\s+/).filter(Boolean);const matches=records.filter(r=>terms.every(t=>(r.skill+' '+r.description+' '+r.groups.join(' ')).toLowerCase().includes(t)));result={ok:true,total:matches.length,matches:matches.slice(0,q.limit),executed_skills:0};}
fs.writeFileSync(output,JSON.stringify(result,null,2)+'\n',{flag:'wx'});
