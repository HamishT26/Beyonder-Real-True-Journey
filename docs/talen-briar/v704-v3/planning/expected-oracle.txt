from fractions import Fraction as F
from functools import lru_cache
import json,hashlib,copy,itertools

def sha(b):return hashlib.sha256(b).hexdigest()


def canon(d):return json.dumps(d,sort_keys=True,separators=(',',':'),ensure_ascii=False).encode('utf8')


def emit(v):
 if isinstance(v,F):return str(v)
 if isinstance(v,list):return [emit(x) for x in v]
 if isinstance(v,tuple):return [emit(x) for x in v]
 if isinstance(v,dict):return {k:emit(x) for k,x in v.items()}
 return v


def basic_input(n=3):
 return {'states':[f's{i}' for i in range(n)],'transition':[[[str(int(i==j)) for j in range(n)] for a in range(2)] for i in range(n)],'rewards':[['0','0'] for i in range(n)],'terminal':['0']*n,'discount':'1','horizon':3,'policy':[i%2 for i in range(n)],'initial':['1']+['0']*(n-1),'shift':'1'}


def make_profile(pid,title,reason,rows,rewards,terminal=None,discount='1',horizon=3,policy=None,initial=None):
 d=basic_input(3);d['transition']=[[[str(F(x,sum(row))) for x in row] for row in state] for state in rows];d['rewards']=[[str(F(x)) for x in row] for row in rewards];d['terminal']=[str(F(x)) for x in (terminal or [0,0,0])];d['discount']=discount;d['horizon']=horizon
 if policy is not None:d['policy']=policy
 if initial is not None:d['initial']=[str(F(x)) for x in initial]
 return {'profile_id':pid,'title':title,'distinguishing_question':reason,'input':d,'synthetic':True,'external_credit':False}

@lru_cache(maxsize=2048)
def computed(key):
 d=json.loads(key);n=len(d['states']);H=d['horizon'];P=[[[F(x) for x in row] for row in state] for state in d['transition']];r=[[F(x) for x in row] for row in d['rewards']];z=[F(x) for x in d['terminal']];g=F(d['discount']);policy=d['policy'];mu=[F(x) for x in d['initial']]
 @lru_cache(None)
 def optimal(h,s):return z[s] if h==0 else max(r[s][a]+g*sum((P[s][a][j]*optimal(h-1,j) for j in range(n)),F(0)) for a in range(2))
 V=[[optimal(h,s) for s in range(n)] for h in range(H+1)]
 Q=[[[r[s][a]+g*sum((P[s][a][j]*V[h-1][j] for j in range(n)),F(0)) for a in range(2)] for s in range(n)] for h in range(1,H+1)]
 choices=[[[a for a in range(2) if Q[h-1][s][a]==V[h][s]] for s in range(n)] for h in range(1,H+1)]
 return n,H,P,r,z,g,policy,mu,V,Q,choices


def fixed_paths(d,policy=None):
 n=len(d['states']);H=d['horizon'];P=[[[F(x) for x in row] for row in state] for state in d['transition']];r=[[F(x) for x in row] for row in d['rewards']];z=[F(x) for x in d['terminal']];g=F(d['discount']);policy=d['policy'] if policy is None else policy;out=[]
 for start in range(n):
  total=F(0)
  for route in itertools.product(range(n),repeat=H):
   prob=F(1);value=F(0);state=start
   for t,nxt in enumerate(route):
    a=policy[state];value+=g**t*r[state][a];prob*=P[state][a][nxt];state=nxt
   value+=g**H*z[state];total+=prob*value
  out.append(total)
 return out


def expected(op,d):
 n,H,P,r,z,g,policy,mu,V,Q,choices=computed(canon(d).decode())
 if op=='mdp_validate':value={'states':n,'actions':2,'horizon':H,'finite':True,'terminal_included':True}
 elif op=='mdp_action_rewards':value=r
 elif op=='mdp_fixed_policy_transition':value=[P[s][policy[s]] for s in range(n)]
 elif op=='mdp_fixed_policy_reward':value=[r[s][policy[s]] for s in range(n)]

 elif op=='mdp_one_step_actions':value=[[r[s][a]+g*sum((P[s][a][j]*z[j] for j in range(n)),F(0)) for a in range(2)] for s in range(n)]

 elif op=='mdp_optimal_action_sets':value={'remaining_horizons':list(range(1,H+1)),'actions':choices}
 elif op=='mdp_optimal_policy':value={'remaining_horizons':list(range(1,H+1)),'policy':[[a[0] for a in row] for row in choices],'tie_counts':[[len(a) for a in row] for row in choices]}
 elif op=='mdp_bellman_residual':value={'residuals':[[V[h][s]-max(Q[h-1][s]) for s in range(n)] for h in range(1,H+1)],'zero':True}

 elif op=='mdp_advantage':value={'remaining_horizons':list(range(1,H+1)),'shortfalls':[[[V[h][s]-Q[h-1][s][a] for a in range(2)] for s in range(n)] for h in range(1,H+1)]}

   pol=[a[0] for a in choices[H-t-1]];stage.append(sum((occ[t][s]*r[s][pol[s]] for s in range(n)),F(0)));occ.append([sum((occ[t][s]*P[s][pol[s]][j] for s in range(n)),F(0)) for j in range(n)])
  terminal=sum((occ[H][s]*z[s] for s in range(n)),F(0))

   forward=sum((g**t*stage[t] for t in range(H)),F(0))+g**H*terminal;backward=sum((mu[s]*V[H][s] for s in range(n)),F(0));value={'stage_rewards':stage,'terminal_reward':terminal,'forward':forward,'backward':backward,'residual':forward-backward,'holds':forward==backward}
