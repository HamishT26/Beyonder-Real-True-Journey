const OPS=['interval_elements','incidence_zeta','incidence_delta','incidence_convolution','mobius_recursive','mobius_chain_sum','zeta_transform','mobius_inversion','open_interval_euler','convolution_associativity'];
function fail(code){throw new Error(code);}
function integer(n){if(!Number.isSafeInteger(n))fail('integer_range');return n;}
function keys(o,ks){if(!o||typeof o!=='object'||Array.isArray(o)||Object.keys(o).sort().join('|')!==ks.slice().sort().join('|'))fail('field_set');}
function parseStrict(text){

 const ws=()=>{while(/[\x20\t\r\n]/.test(text[p]||'!'))p++;};
 function str(){const start=p++;while(p<text.length){const c=text[p++];if(c==='"'){try{return JSON.parse(text.slice(start,p));}catch{fail('json_syntax');}}if(c==='\\')p++;}fail('json_syntax');}

  for(const [token,v] of [['true',true],['false',false],['null',null]])if(text.slice(p,p+token.length)===token){p+=token.length;return v;}

}
function order(p,cap=8){

function identity(n){return Array.from({length:n},(_,i)=>Array.from({length:n},(_,j)=>+(i===j)));}
function zero(n){return Array.from({length:n},()=>Array(n).fill(0));}
function transpose(a){return a[0].map((_,j)=>a.map(r=>r[j]));}
function mm(a,b){const n=a.length,c=zero(n);for(let i=0;i<n;i++)for(let j=0;j<n;j++)for(let k=0;k<n;k++)c[i][j]=integer(c[i][j]+integer(a[i][k]*b[k][j]));return c;}
function mv(a,b){return a.map(r=>r.reduce((s,v,i)=>integer(s+integer(v*b[i])),0));}
function matrix(a,z){const n=z.length;if(!Array.isArray(a)||a.length!==n||a.some((r,i)=>!Array.isArray(r)||r.length!==n||r.some((v,j)=>!Number.isSafeInteger(v)||Math.abs(v)>1000000||(!z[i][j]&&v!==0))))fail('matrix');return a;}
function vector(a,n){if(!Array.isArray(a)||a.length!==n||a.some(v=>!Number.isSafeInteger(v)||Math.abs(v)>1000000))fail('vector');return a;}
function mobius(z){const n=z.length,m=Array.from({length:n},()=>Array(n));function mu(i,j){if(m[i][j]!==undefined)return m[i][j];if(!z[i][j])return m[i][j]=0;if(i===j)return m[i][j]=1;let s=0;for(let k=0;k<n;k++)if(k!==j&&z[i][k]&&z[k][j])s=integer(s+mu(i,k));return m[i][j]=-s;}for(let i=0;i<n;i++)for(let j=0;j<n;j++)mu(i,j);return m;}

function envelope(value,disposition='completed'){return {ok:true,error:null,disposition,value,external_credit:false};}
function rejected(error){return {ok:false,error,disposition:'exact_gate',value:null,external_credit:false};}
function evaluate(input){keys(input,['op','poset','args']);if(!OPS.includes(input.op))fail('operation');const z=order(input.poset),n=z.length,a=input.args,op=input.op;let v,d='completed';

function fromText(text){try{return dispatch(parseStrict(text));}catch(e){return rejected(e.message);}}
