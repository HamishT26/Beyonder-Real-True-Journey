function stable(value) {
  if (Array.isArray(value)) return '[' + value.map(stable).join(',') + ']';
  if (value && typeof value === 'object') {
    return '{' + Object.keys(value).sort().map(k => JSON.stringify(k) + ':' + stable(value[k])).join(',') + '}';
  }
  return JSON.stringify(value);
}
function sha(value) {
  const data = Buffer.isBuffer(value) ? value : Buffer.from(typeof value === 'string' ? value : stable(value));
  return crypto.createHash('sha256').update(data).digest('hex');
}
function writeJson(relative, value) {
  const file = path.join(BASE, relative);
  fs.mkdirSync(path.dirname(file), {recursive: true});
  fs.writeFileSync(file, JSON.stringify(value, null, 2) + '\n', 'utf8');
}
function writeText(relative, value) {
  const file = path.join(BASE, relative);
  fs.mkdirSync(path.dirname(file), {recursive: true});
  fs.writeFileSync(file, value.replace(/\r\n/g, '\n').trimEnd() + '\n', 'utf8');
}
function tokens(text) {
  return new Set(text.toLowerCase().match(/[a-z0-9]+/g) || []);
}
function jaccard(a, b) {
  const x = tokens(a), y = tokens(b);
  const intersection = [...x].filter(v => y.has(v)).length;
  const union = new Set([...x, ...y]).size;
  return union ? intersection / union : 0;
}
function walk(dir) {
  if (!fs.existsSync(dir)) return [];
  const out = [];
  for (const name of fs.readdirSync(dir).sort()) {
    const file = path.join(dir, name);
    const stat = fs.statSync(file);
    if (stat.isDirectory()) out.push(...walk(file));
    else out.push(file);
  }
  return out;
}

function inputFor(operation, c) {
  const seq = sequenceCases[c - 1];
  switch (operation) {
    case 'fasta_header': return {identifier:headerCases[c-1][0], description:headerCases[c-1][1], maximumIdLength:25};
    case 'sequence_alphabet': return {sequence:c === 4 ? seq + '?' : seq, allowed:'ACGTRYSWKMBDHVN'};
    case 'line_wrap': return {sequence:seq.repeat(4), width:5 + c};
    case 'sequence_length': return {sequence:seq};
    case 'gc_fraction': return {sequence:seq.replace(/[^ACGT]/g, 'A')};
    case 'reverse_complement': return {sequence:seq, alphabet:'ACGTRYSWKMBDHVN'};
    case 'ambiguous_count': return {sequence:seq};
    case 'sequence_checksum': return {sequence:seq.toLowerCase(), normalization:'uppercase_no_whitespace'};
    case 'record_order': return {identifiers:c % 3 === 0 ? ['SYN-A'+c,'SYN-B'+c,'SYN-A'+c] : ['SYN-'+c,'SYN-'+(c+10),'SYN-'+(c+20)]};
    case 'record_manifest': return {entries:[
      {path:'records/synthetic-'+c+'.fa',bytes:10+c,digest:'a'.repeat(64)},
      {path:'records/synthetic-'+(c+10)+'.fa',bytes:20+c,digest:(c===7?'z':'b').repeat(64)}
    ]};
    case 'fastq_frame': return {header:(c===4?'bad':'@SYN'+c),sequence:seq.replace(/[^ACGTN]/g,'N'),plus:c===6?'minus':'+',quality:'I'.repeat(seq.length)};
    case 'quality_length': return {sequence:seq.replace(/[^ACGTN]/g,'N'),quality:'I'.repeat(Math.max(0,seq.length+(c===5?1:c===6?-1:0)))};
    case 'phred33_range': return {quality:c===8?'II\u0001I':String.fromCharCode(32+c).repeat(3)+String.fromCharCode(40+c)};
    case 'paired_read_binding': return {left:'SYN'+c+'/1',right:(c===6?'OTHER':'SYN'+c)+'/2'};
    case 'kmer_histogram': return {sequence:seq.replace(/[^ACGT]/g,'A'),k:1+(c%4)};
    case 'duplicate_record': return {records:[
      {identifier:'SYN'+c,sequence:seq},
      {identifier:c===7?'SYN'+c:'SYN'+(c+10),sequence:c===8?seq:sequenceCases[(c)%10]}
    ]};
    case 'provenance_chain': return {events:[
      {sequence:1,previous:'0'.repeat(64),payload:{action:'create',record:'SYN'+c}},
      {sequence:2,previous:c===9?'f'.repeat(64):null,payload:{action:'review',record:'SYN'+c}}
    ]};
    case 'evidence_ceiling': return {requested:['biological_origin','identity','clinical_effect','production_ready','population_frequency','physical_law','consent','ownership','independent_reproduction','Stage20'][c-1],available:['synthetic_fixture','same_owner_test']};

    case 'authority_gate': return {action:actions[c-1][0],requiredRoles:actions[c-1][1],verifiedRoles:[]};
    default: throw new Error('UNKNOWN_OPERATION');
  }
}
