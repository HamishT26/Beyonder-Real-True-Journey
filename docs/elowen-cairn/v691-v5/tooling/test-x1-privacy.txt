'use strict';

const fs = require('fs');
const path = require('path');

const ROOT = path.resolve(__dirname, '../../../..');
const BASE = path.join(ROOT, 'docs', 'elowen-cairn', 'v691-v5');
const X1 = path.join(BASE, 'x1');

function walk(directory) {
  if (!fs.existsSync(directory)) return [];
  const output=[];
  for (const name of fs.readdirSync(directory).sort()) {
    const file=path.join(directory,name);
    const stat=fs.statSync(file);
    if(stat.isDirectory()) output.push(...walk(file)); else output.push(file);
  }
  return output;
}

function count(regex, text) { return (text.match(regex) || []).length; }

function run() {
  const selected=walk(BASE).filter(file=>file.startsWith(X1+path.sep)||[
    'sequence-x1-engine.txt','build-x1.txt','test-x1.txt','test-x1-privacy.txt','finalize-x1.txt'
  ].includes(path.basename(file))).filter(file=>['.json','.md','.txt','.html'].includes(path.extname(file)));
  const records=[];
  for(const file of selected) {
    const relative=path.relative(ROOT,file).replaceAll('\\','/');
    const lines=fs.readFileSync(file,'utf8').split(/\r?\n/);
    lines.forEach((line,index)=>records.push({relative,line,index:index+1}));
  }
  const joined=records.map(row=>row.line).join('\n');
  const privateRouteCandidates=[];
  for(const row of records) {
    const matches=count(/(?:source_thread_id|providerTabId|private callable identifier|private application state|session stream)/gi,row.line);
    if(matches) {
      const scannerDefinition=row.relative.endsWith('/tooling/test-x1.txt')||row.relative.endsWith('/tooling/test-x1-privacy.txt');
      privateRouteCandidates.push({path:row.relative,line:row.index,count:matches,classification:scannerDefinition?'scanner_definition':'confirmed_payload'});
    }
  }
  const mediaCandidates=[];
  for(const row of records) {
    const matches=count(/(?:transcript|screenshot|screen capture)/gi,row.line);
    if(matches) {
      const policyMetadata=/new_screenshots.*false/i.test(row.line)||row.relative.endsWith('/tooling/test-x1-privacy.txt');
      mediaCandidates.push({path:row.relative,line:row.index,count:matches,classification:policyMetadata?'scanner_definition_or_policy_metadata':'confirmed_payload'});
    }
  }
  const result={
    schema:'ghc.family.elowen-x1-privacy-component.v1',
    files_scanned:selected.length,
    raw_uuid:{candidates:count(/\b[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}\b/gi,joined),confirmed:0},
    private_absolute_path:{candidates:count(/(?<![A-Za-z])[A-Za-z]:[\\/]/g,joined),confirmed:0},
    credential_assignment:{candidates:count(/(?:api[_-]?key|password|secret|access[_-]?token|authorization)\s*[:=]\s*[^\s,}]+/gi,joined),confirmed:0},
    private_route:{candidates:privateRouteCandidates.reduce((sum,row)=>sum+row.count,0),scanner_definitions:privateRouteCandidates.filter(row=>row.classification==='scanner_definition').reduce((sum,row)=>sum+row.count,0),confirmed:privateRouteCandidates.filter(row=>row.classification==='confirmed_payload').reduce((sum,row)=>sum+row.count,0)},
    media_or_transcript:{candidates:mediaCandidates.reduce((sum,row)=>sum+row.count,0),scanner_definition_or_policy_metadata:mediaCandidates.filter(row=>row.classification!=='confirmed_payload').reduce((sum,row)=>sum+row.count,0),confirmed:mediaCandidates.filter(row=>row.classification==='confirmed_payload').reduce((sum,row)=>sum+row.count,0)},
    private_route_candidates:privateRouteCandidates,
    media_candidates:mediaCandidates,
    same_owner_only:true,
    independent_reproduction:false
  };
  result.valid=result.raw_uuid.candidates===0&&result.private_absolute_path.candidates===0&&result.credential_assignment.candidates===0&&result.private_route.confirmed===0&&result.media_or_transcript.confirmed===0;
  return result;
}

module.exports={run};
if(require.main===module){const result=run();process.stdout.write(JSON.stringify(result)+'\n');if(!result.valid)process.exitCode=1;}
