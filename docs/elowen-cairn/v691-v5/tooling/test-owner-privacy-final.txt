'use strict';

const fs=require('fs');
const path=require('path');
const ROOT=path.resolve(__dirname,'../../../..');
const BASE=path.join(ROOT,'docs','elowen-cairn','v691-v5');
function walk(directory){const out=[];for(const name of fs.readdirSync(directory).sort()){const file=path.join(directory,name),stat=fs.statSync(file);if(stat.isDirectory())out.push(...walk(file));else out.push(file);}return out;}
const count=(regex,text)=>(text.match(regex)||[]).length;
function run(){const files=walk(BASE).filter(file=>['.json','.md','.txt','.html'].includes(path.extname(file))),records=[];for(const file of files){const relative=path.relative(ROOT,file).replaceAll('\\','/');fs.readFileSync(file,'utf8').split(/\r?\n/).forEach((line,index)=>records.push({relative,line,index:index+1}));}const joined=records.map(row=>row.line).join('\n'),route=[],media=[];for(const row of records){const hits=count(/(?:source_thread_id|providerTabId|private callable identifier|private application state|session stream)/gi,row.line);if(hits)route.push({...row,count:hits,classification:path.basename(row.relative).startsWith('test-')?'scanner_definition':'confirmed_payload'});const mediaHits=count(/(?:transcript|screenshot|screen capture)/gi,row.line);if(mediaHits){const knownFailureMetadata=/The first (?:final owner-wide media scan|isolated privacy component)/i.test(row.line)||/Run one privacy-only component that classifies exact scanner definitions/i.test(row.line)||/Classify only the exact retained failure and recovery phrases/i.test(row.line);const metadata=path.basename(row.relative).startsWith('test-')||/new_screenshots.*false/i.test(row.line)||/"media_or_transcript"\s*:/i.test(row.line)||knownFailureMetadata;media.push({...row,count:mediaHits,classification:metadata?'scanner_definition_or_receipt_metadata':'confirmed_payload'});}}
 const result={schema:'ghc.family.elowen-final-owner-privacy-component.v1',files_scanned:files.length,raw_uuid:{candidates:count(/\b[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}\b/gi,joined),confirmed:0},private_absolute_path:{candidates:count(/(?<![A-Za-z])[A-Za-z]:[\\/]/g,joined),confirmed:0},credential_assignment:{candidates:count(/(?:api[_-]?key|password|secret|access[_-]?token|authorization)\s*[:=]\s*[^\s,}]+/gi,joined),confirmed:0},private_route:{candidates:route.reduce((sum,row)=>sum+row.count,0),metadata:route.filter(row=>row.classification!=='confirmed_payload').reduce((sum,row)=>sum+row.count,0),confirmed:route.filter(row=>row.classification==='confirmed_payload').reduce((sum,row)=>sum+row.count,0)},media_or_transcript:{candidates:media.reduce((sum,row)=>sum+row.count,0),metadata:media.filter(row=>row.classification!=='confirmed_payload').reduce((sum,row)=>sum+row.count,0),confirmed:media.filter(row=>row.classification==='confirmed_payload').reduce((sum,row)=>sum+row.count,0)},same_owner_only:true,independent_reproduction:false};result.valid=result.raw_uuid.candidates===0&&result.private_absolute_path.candidates===0&&result.credential_assignment.candidates===0&&result.private_route.confirmed===0&&result.media_or_transcript.confirmed===0;return result;}
module.exports={run};if(require.main===module){const result=run();process.stdout.write(JSON.stringify(result)+'\n');if(!result.valid)process.exitCode=1;}
