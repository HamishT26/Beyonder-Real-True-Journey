function stable(value) {
  if (Array.isArray(value)) return '[' + value.map(stable).join(',') + ']';
  if (value && typeof value === 'object') {
    return '{' + Object.keys(value).sort().map(key => JSON.stringify(key) + ':' + stable(value[key])).join(',') + '}';
  }
  return JSON.stringify(value);
}

function sha256(value) {
  const bytes = Buffer.isBuffer(value) ? value : Buffer.from(typeof value === 'string' ? value : stable(value), 'utf8');
  return crypto.createHash('sha256').update(bytes).digest('hex');
}

function writeJson(relative, value) {
  const file = path.join(BASE, relative);
  fs.mkdirSync(path.dirname(file), {recursive: true});
  fs.writeFileSync(file, JSON.stringify(value, null, 2) + '\n', 'utf8');
}

function writeText(relative, value) {

  fs.mkdirSync(path.dirname(file), {recursive: true});
  fs.writeFileSync(file, value.replace(/\r\n/g, '\n').trimEnd() + '\n', 'utf8');
}

function walk(directory) {
  if (!fs.existsSync(directory)) return [];
  const output = [];
  for (const name of fs.readdirSync(directory).sort()) {
    const file = path.join(directory, name);
    const stat = fs.statSync(file);
    if (stat.isDirectory()) output.push(...walk(file));
    else output.push(file);
  }
  return output;
}

function mutation(operation, input) {
  const value = core.clone(input);
  switch (operation) {
    case 'fasta_header': value.identifier = 'bad identifier'; break;
    case 'sequence_alphabet': value.sequence = String(value.sequence ?? '') + '?'; break;
    case 'line_wrap': value.width = 0; break;
    case 'sequence_length': value.sequence = null; break;
    case 'gc_fraction': value.sequence = 'AX'; break;
    case 'reverse_complement': value.sequence = String(value.sequence ?? '') + '?'; break;
    case 'ambiguous_count': value.sequence = ['A']; break;
    case 'sequence_checksum': value.normalization = 'unspecified'; value.sequence = null; break;
    case 'record_order': value.identifiers = [...(Array.isArray(value.identifiers) ? value.identifiers : []), null]; break;
    case 'record_manifest': value.entries = [...(Array.isArray(value.entries) ? value.entries : []), {path: '../outside.fa', bytes: -1, digest: 'not-a-digest'}]; break;
    default: throw new Error('UNKNOWN_MUTATION ' + operation);
  }
  return value;
}

function expectedValidity(operation, input) {
  switch (operation) {
    case 'fasta_header': return typeof input.identifier === 'string' && input.identifier.length > 0 && input.identifier.length <= input.maximumIdLength && /^[A-Za-z0-9._:*#-]+$/.test(input.identifier);
    case 'sequence_alphabet': return typeof input.sequence === 'string' && [...input.sequence.toUpperCase().replace(/\s+/g, '')].every(symbol => input.allowed.includes(symbol));
    case 'line_wrap': return typeof input.sequence === 'string' && Number.isInteger(input.width) && input.width >= 1 && input.width <= 80;
    case 'sequence_length': return typeof input.sequence === 'string' && [...input.sequence.toUpperCase()].every(symbol => core.IUPAC.includes(symbol));
    case 'gc_fraction': return typeof input.sequence === 'string' && input.sequence.length > 0 && /^[ACGT]+$/i.test(input.sequence);
    case 'reverse_complement': return typeof input.sequence === 'string' && [...input.sequence.toUpperCase()].every(symbol => core.IUPAC.includes(symbol));
    case 'ambiguous_count': return typeof input.sequence === 'string' && [...input.sequence.toUpperCase()].every(symbol => core.IUPAC.includes(symbol));
    case 'sequence_checksum': return typeof input.sequence === 'string' && input.normalization === 'uppercase_no_whitespace';
    case 'record_order': return Array.isArray(input.identifiers) && input.identifiers.every(identifier => typeof identifier === 'string' && identifier.length > 0);
    case 'record_manifest': return Array.isArray(input.entries) && input.entries.every(entry => typeof entry.path === 'string' && entry.path && !entry.path.startsWith('/') && !entry.path.includes('..') && !entry.path.includes('\\') && Number.isInteger(entry.bytes) && entry.bytes >= 0 && typeof entry.digest === 'string' && /^[0-9a-f]{64}$/i.test(entry.digest)) && new Set(input.entries.map(entry => entry.path)).size === input.entries.length;
    default: return false;
  }
}

function invariant(operation, input, output) {
  if (output.valid !== expectedValidity(operation, input)) return false;
  if (!output.valid) return output.errors.length > 0;
  switch (operation) {
    case 'fasta_header': return output.result.definition_line.startsWith('>' + input.identifier);
    case 'sequence_alphabet': return output.result.unsupported.length === 0;
    case 'line_wrap': return output.result.roundtrip === input.sequence && output.result.lines.every(line => line.length <= input.width);
    case 'sequence_length': return output.result.total === input.sequence.length && output.result.canonical + output.result.ambiguous === input.sequence.length;
    case 'gc_fraction': return output.result.sequence_length === input.sequence.length && output.result.denominator > 0;
    case 'reverse_complement': return output.result.reverse_complement.length === input.sequence.length;
    case 'ambiguous_count': return output.result.ambiguous_total === [...input.sequence.toUpperCase()].filter(symbol => !'ACGT'.includes(symbol)).length;
    case 'sequence_checksum': return /^[0-9a-f]{64}$/.test(output.result.sha256);
    case 'record_order': return stable(output.result.identifiers) === stable(input.identifiers) && output.result.deduplicated === false;
    case 'record_manifest': return output.result.entry_count === input.entries.length;
    default: return false;
  }
}
