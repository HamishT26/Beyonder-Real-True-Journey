function ensureDir(p) { fs.mkdirSync(p, { recursive: true }); }
function writeJson(rel, value) {
  const p = path.join(root, rel);
  ensureDir(path.dirname(p));
  fs.writeFileSync(p, JSON.stringify(value, null, 2) + "\n", "utf8");
}
function writeText(rel, value) {
  const p = path.join(root, rel);
  ensureDir(path.dirname(p));
  fs.writeFileSync(p, value.replace(/\r\n/g, "\n"), "utf8");
}
function sha256Text(text) { return crypto.createHash("sha256").update(text, "utf8").digest("hex"); }
function gitShow(repo, commit, rel) {
  return cp.execFileSync("git", ["-C", repo, "show", `${commit}:${rel}`], { encoding: "utf8", maxBuffer: 64 * 1024 * 1024 });
}
function walk(dir) {
  if (!fs.existsSync(dir)) return [];
  const out = [];
  for (const ent of fs.readdirSync(dir, { withFileTypes: true })) {
    const p = path.join(dir, ent.name);
    if (ent.isDirectory()) out.push(...walk(p)); else out.push(p);
  }
  return out;
}

function makeModel(i) {
  const r = 2 + (i % 3);
  const n = 4 + (i % 3);
  const columns = [];
  for (let j = 0; j < n; j++) {
    if (j < r) columns.push(1 << j);
    else {
      const mod = 1 << r;
      const raw = ((i + 3) * (j + 5) + j * j + 7) % mod;
      columns.push((i % 7 === 0 && j === n - 1) ? 0 : raw);
    }
  }
  const labels = Array.from({ length: n }, (_, j) => `e${j + 1}`);
  const weights = Array.from({ length: n }, (_, j) => 1 + ((i * 5 + j * 3) % 13));
  return {
    profile_id: `MAT-${String(i + 1).padStart(2, "0")}`,
    ground_labels: labels,
    row_dimension: r,
    columns_gf2: columns,
    weights,
    query_subset_mask: ((i + 1) * 5 + 3) % (1 << n),
    minor_element_index: i % n,
    declared_scope: "finite synthetic binary matroid represented by a frozen GF(2) column matrix"
  };
}
