"use strict";
const fs=require("fs"),path=require("path"),vm=require("vm");
function load(file){const module={exports:{}};new vm.Script(fs.readFileSync(file,"utf8"),{filename:file}).runInContext(vm.createContext({module,exports:module.exports,require}));return module.exports;}
const mode=process.argv[2]||"valid";
const root=path.resolve(__dirname,"..");
const core=load(path.join(root,"ghc_family_binary_matroid_core.txt"));
const model=JSON.parse(fs.readFileSync(path.resolve(root,"..","plan","models.json"),"utf8")).records[1];
const operations=["independence-census","basis-census"];
if(mode==="valid"){const outputs=operations.map(op=>core.evaluate(op,model));process.stdout.write(JSON.stringify({mode,pass:outputs.length===2,operations}));}
else if(mode==="invalid"){const bad=JSON.parse(JSON.stringify(model));bad.ground_labels[1]=bad.ground_labels[0];let refused=0;for(const op of operations){try{core.evaluate(op,bad);}catch{refused++;}}process.stdout.write(JSON.stringify({mode,pass:refused===2,refused,operations}));if(refused!==2)process.exitCode=1;}
else{throw new Error("mode must be valid or invalid");}
