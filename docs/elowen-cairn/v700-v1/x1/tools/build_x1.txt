function ensureDir(p) { fs.mkdirSync(p, { recursive: true }); }
function writeJson(rel, value) {
  const p = path.join(root, rel);
  ensureDir(path.dirname(p));
  fs.writeFileSync(p, JSON.stringify(value, null, 2) + "\n", "utf8");
}
function writeText(rel, value) {
  const p = path.join(root, rel);
  ensureDir(path.dirname(p));
  fs.writeFileSync(p, value.replace(/\r\n/g, "\n"), "utf8");
}
function loadModule(file) {
  const source = fs.readFileSync(file, "utf8");
  const module = { exports: {} };
  const context = vm.createContext({ module, exports: module.exports, require });
  new vm.Script(source, { filename: file }).runInContext(context);
  return module.exports;
}
function walk(dir) {
  if (!fs.existsSync(dir)) return [];
  const out = [];
  for (const ent of fs.readdirSync(dir, { withFileTypes: true })) {
    const p = path.join(dir, ent.name);
    if (ent.isDirectory()) out.push(...walk(p)); else out.push(p);
  }
  return out;
}
function sha(file) { return crypto.createHash("sha256").update(fs.readFileSync(file)).digest("hex"); }

function malformedCase(index, operationId, model) {
  const bad = JSON.parse(JSON.stringify(model));
  const kind = index % 10;
  if (kind === 0) bad.ground_labels[1] = bad.ground_labels[0];
  else if (kind === 1) bad.columns_gf2[0] = 1 << bad.row_dimension;
  else if (kind === 2) bad.weights.pop();
  else if (kind === 3) bad.query_subset_mask = 1 << bad.ground_labels.length;
  else if (kind === 4) delete bad.profile_id;
  else if (kind === 5) bad.columns_gf2[0] = -1;
  else if (kind === 6) bad.row_dimension = 0;
  else if (kind === 7) operationId = `${operationId}-unknown`;
  else if (kind === 8) bad.minor_element_index = bad.ground_labels.length;
  else bad.ground_labels.pop();
  return { operationId, model: bad, malformed_kind: [
    "duplicate_label", "column_out_of_range", "weight_count_mismatch", "subset_mask_out_of_range",
    "missing_profile_id", "negative_column", "invalid_row_dimension", "unknown_operation",
    "minor_index_out_of_range", "label_count_mismatch"
  ][kind] };
}

  const rows = [];
  for (const proposal of proposalFile.records) {
    const model = models.find(x => x.profile_id === proposal.profile_id);
    const frozenBefore = JSON.stringify(model);
    const output = core.evaluate(op.operation_id, model);
    if (JSON.stringify(model) !== frozenBefore) throw new Error(`input mutated: ${proposal.proposal_id}`);
    rows.push({ proposal_id: proposal.proposal_id, operation_id: op.operation_id, profile_id: model.profile_id, outcome: "completed", pass: true, output });
    safe.push({ task_id: `${proposal.proposal_id}-SAFE`, proposal_id: proposal.proposal_id, pass: true, output_sha256: crypto.createHash("sha256").update(JSON.stringify(output)).digest("hex") });
    const malformed = malformedCase(contractIndex, op.operation_id, model);
    let refusal = null;
    try { core.evaluate(malformed.operationId, malformed.model); }
    catch (error) { refusal = error.message; }
    if (!refusal) throw new Error(`malformed subject accepted: ${proposal.proposal_id}`);
    candidate.push({
      task_id: `${proposal.proposal_id}-CANDIDATE`, proposal_id: proposal.proposal_id,
      malformed_kind: malformed.malformed_kind, invalid_subject_result: "fail",
      invalid_subject_credit: 0, refusal_guard_result: "pass", refusal
    });
    cfr.push({ task_id: `${proposal.proposal_id}-CFR`, proposal_id: proposal.proposal_id, pass: true, input_preserved: true, source_status: "finite_synthetic" });
    clean.push({ task_id: `${proposal.proposal_id}-CLEAN`, proposal_id: proposal.proposal_id, pass: true, destructive_mutation: false, private_material: false, overclaim: false });
    contractIndex++;
  }
  resultsByOperation.set(op.operation_id, rows);
  writeJson(path.join("results", `${op.operation_id}.json`), { owner, phase, stage: "x1", operation_id: op.operation_id, count: rows.length, records: rows });
}
