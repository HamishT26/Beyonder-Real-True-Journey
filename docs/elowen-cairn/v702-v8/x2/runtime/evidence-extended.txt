'use strict';
const core=require('../../x1/runtime/evidence-core.txt');
function ensure(condition,message){if(!condition)throw Error(message);}
const I=core.internals,ZERO=()=>I.rational(0n),ONE=()=>I.rational(1n);
function validateFrame(frame){ensure(Array.isArray(frame)&&frame.length===3&&frame.every(x=>typeof x==='string'&&x.length>0)&&new Set(frame).size===3,'valid three-label frame required');}
function validateKeys(request,allowed){ensure(JSON.stringify(Object.keys(request).sort())===JSON.stringify([...allowed].sort()),'closed request fields required');}
function base(request){return {operation:request.operation,frame:request.frame};}
function extend(values,mapping){const out=Array.from({length:16},ZERO);for(let mask=0;mask<values.length;mask++){let target=0;for(let i=0;i<3;i++)if(mask&(1<<i))target|=mapping[i];out[target]=I.plus(out[target],values[mask]);}return out;}
function marginalize(values,mapping){const out=Array.from({length:8},ZERO);for(let mask=0;mask<values.length;mask++){let source=0;for(let i=0;i<3;i++)if((mask&mapping[i])!==0)source|=1<<i;out[source]=I.plus(out[source],values[mask]);}return out;}
function validateMap(target,mapping){ensure(Array.isArray(target)&&target.length===4&&target.every(x=>typeof x==='string'&&x)&&new Set(target).size===4,'valid target frame');ensure(Array.isArray(mapping)&&mapping.length===3&&mapping.every(x=>Number.isInteger(x)&&x>0&&x<16),'valid refinement masks');ensure((mapping[0]&mapping[1])===0&&(mapping[0]&mapping[2])===0&&(mapping[1]&mapping[2])===0,'disjoint refinement blocks');ensure((mapping[0]|mapping[1]|mapping[2])===15,'complete refinement map');}
function pignistic(frame,values){const den=I.minus(ONE(),values[0]);ensure(den.n!==0n,'total conflict');return frame.map((label,index)=>{let value=ZERO();for(let mask=1;mask<8;mask++)if(mask&(1<<index))value=I.plus(value,I.over(values[mask],I.rational(BigInt(mask.toString(2).replace(/0/g,'').length))));return {label,value:I.format(I.over(value,den))};});}
const allowed={
  evidence_dempster:['operation','frame','mass','right_mass'],evidence_total_conflict:['operation','frame','left_mask','right_mask'],evidence_vacuous_extension:['operation','frame','mass','target_frame','mapping'],evidence_marginalize:['operation','frame','mass','target_frame','mapping'],evidence_refinement_roundtrip:['operation','frame','mass','target_frame','mapping'],evidence_pignistic:['operation','frame','mass'],evidence_credal_interval:['operation','frame','mass','query_mask'],evidence_trinity_bridge:['operation','frame'],evidence_observation_gap:['operation','frame'],evidence_authority_gate:['operation','frame']
};
function execute(request){
  ensure(request&&typeof request==='object'&&!Array.isArray(request),'request object required');const op=request.operation;ensure(allowed[op],'unsupported x2 operation');validateKeys(request,allowed[op]);validateFrame(request.frame);const outBase=base(request);
  if(op==='evidence_dempster'){const left=I.decodeMass(request.frame,request.mass),right=I.decodeMass(request.frame,request.right_mass),combined=I.combine(left,right),normalized=I.normalize(combined);return {...outBase,conflict:I.format(normalized.conflict),mass:I.encodeMass(normalized.mass)};}
  if(op==='evidence_total_conflict'){ensure(Number.isInteger(request.left_mask)&&Number.isInteger(request.right_mask)&&request.left_mask>0&&request.left_mask<8&&request.right_mask>0&&request.right_mask<8,'singleton masks');const left=Array.from({length:8},ZERO),right=Array.from({length:8},ZERO);left[request.left_mask]=ONE();right[request.right_mask]=ONE();const combined=I.combine(left,right);ensure(combined[0].n===1n&&combined[0].d===1n,'fixture must be total conflict');return {...outBase,status:'refused',reason:'total conflict gives zero normalization denominator',conflict:'1'};}
  if(['evidence_vacuous_extension','evidence_marginalize','evidence_refinement_roundtrip'].includes(op)){validateMap(request.target_frame,request.mapping);const mass=I.decodeMass(request.frame,request.mass),extended=extend(mass,request.mapping),back=marginalize(extended,request.mapping);if(op==='evidence_vacuous_extension')return {...outBase,target_frame:request.target_frame,mass:I.encodeMass(extended)};if(op==='evidence_marginalize')return {...outBase,target_frame:request.target_frame,marginal_mass:I.encodeMass(back)};return {...outBase,target_frame:request.target_frame,extended_mass:I.encodeMass(extended),recovered_mass:I.encodeMass(back),roundtrip:JSON.stringify(I.encodeMass(back))===JSON.stringify(I.encodeMass(mass))};}
  if(op==='evidence_pignistic'){const mass=I.decodeMass(request.frame,request.mass);return {...outBase,probabilities:pignistic(request.frame,mass),decision_authority:'reserved'};}
  if(op==='evidence_credal_interval'){ensure(Number.isInteger(request.query_mask)&&request.query_mask>=0&&request.query_mask<8,'query mask');const belief=core.execute({operation:'evidence_belief',frame:request.frame,mass:request.mass}).values,plausibility=core.execute({operation:'evidence_plausibility',frame:request.frame,mass:request.mass}).values;return {...outBase,query_mask:request.query_mask,lower:belief[request.query_mask].value,upper:plausibility[request.query_mask].value,interpretation:'finite formal interval only'};}
  if(op==='evidence_trinity_bridge')return {...outBase,status:'represented',claim:'Finite mass bookkeeping can represent an evidence-accounting analogy across GMUT, THOS, and Freed ID/CBR without measuring a mind, physical field, identity, or right.'};
  if(op==='evidence_observation_gap')return {...outBase,status:'open_gap',missing:['preregistered real observations','calibration','independent review','affected-user accessibility evaluation']};
  if(op==='evidence_authority_gate')return {...outBase,status:'exact_gate',required:['specific affected-party mandate','competent professional review','legal and cultural review where applicable','relevant Maori authority for Maori concepts or data']};
  throw Error('unreachable');
}
module.exports={execute,internals:{extend,marginalize,validateMap,pignistic}};
