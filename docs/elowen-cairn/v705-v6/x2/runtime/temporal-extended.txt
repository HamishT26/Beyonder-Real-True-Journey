'use strict';
const ALL=["before","meets","overlaps","starts","during","finishes","equals","finished_by","contains","started_by","overlapped_by","met_by","after"];
const CONVERSE={"before":"after","meets":"met_by","overlaps":"overlapped_by","starts":"started_by","during":"contains","finishes":"finished_by","equals":"equals","finished_by":"finishes","contains":"during","started_by":"starts","overlapped_by":"overlaps","met_by":"meets","after":"before"};
function ensure(c,m){if(!c)throw Error(m);}
function keys(o,a){ensure(o&&typeof o==='object'&&!Array.isArray(o),'request object');ensure(JSON.stringify(Object.keys(o).sort())===JSON.stringify([...a].sort()),'closed request fields');}
function interval(x){ensure(x&&Number.isInteger(x.start)&&Number.isInteger(x.end)&&x.start<x.end,'valid interval required');}
function relation(a,b){interval(a);interval(b);if(a.end<b.start)return'before';if(a.end===b.start)return'meets';if(a.start<b.start&&b.start<a.end&&a.end<b.end)return'overlaps';if(a.start===b.start&&a.end<b.end)return'starts';if(b.start<a.start&&a.end<b.end)return'during';if(b.start<a.start&&a.end===b.end)return'finishes';if(a.start===b.start&&a.end===b.end)return'equals';if(a.start<b.start&&a.end===b.end)return'finished_by';if(a.start<b.start&&b.end<a.end)return'contains';if(a.start===b.start&&b.end<a.end)return'started_by';if(b.start<a.start&&a.start<b.end&&b.end<a.end)return'overlapped_by';if(a.start===b.end)return'met_by';if(b.end<a.start)return'after';throw Error('unclassified');}
function intervals(max=7){const o=[];for(let s=0;s<max;s++)for(let e=s+1;e<=max;e++)o.push({start:s,end:e});return o;}const DOMAIN=intervals(7),CACHE=new Map();
function compose(r1,r2){ensure(ALL.includes(r1)&&ALL.includes(r2),'known relations');const k=r1+'|'+r2;if(CACHE.has(k))return CACHE.get(k);const f=new Set();for(const a of DOMAIN)for(const b of DOMAIN){if(relation(a,b)!==r1)continue;for(const c of DOMAIN)if(relation(b,c)===r2)f.add(relation(a,c));}const r=ALL.filter(x=>f.has(x));CACHE.set(k,r);return r;}
function rows(f){return[{pair:'AB',relations:[relation(f.A,f.B)]},{pair:'BC',relations:[relation(f.B,f.C)]},{pair:'AC',relations:[...ALL]}];}
function close(rs){const m=Object.fromEntries(rs.map(r=>[r.pair,[...r.relations]]));ensure(m.AB&&m.BC&&m.AC,'AB BC AC constraints');const p=[...new Set(m.AB.flatMap(a=>m.BC.flatMap(b=>compose(a,b))))];m.AC=ALL.filter(x=>m.AC.includes(x)&&p.includes(x));return['AB','BC','AC'].map(pair=>({pair,relations:m[pair]}));}
function scenarios(r1,r2){let count=0,witness=null;for(const a of DOMAIN)for(const b of DOMAIN){if(relation(a,b)!==r1)continue;for(const c of DOMAIN){if(relation(b,c)!==r2)continue;count++;if(!witness)witness={A:a,B:b,C:c,AC:relation(a,c)};}}return{bounded_domain:'integer endpoints 0..7',count,witness};}
const allowed={interval_record:['operation','intervals'],relation_classify:['operation','left','right'],relation_converse:['operation','relation'],endpoint_order:['operation','intervals'],relation_partition:['operation','left','right'],relation_symmetry:['operation','left','right'],duration_exact:['operation','intervals'],network_record:['operation','intervals'],pair_consistency:['operation','left','right','asserted'],relation_projection:['operation','left','right'],relation_compose:['operation','left_relation','right_relation'],weak_composition:['operation','left_relation','right_relation'],path_consistency_step:['operation','constraints'],path_consistency_closure:['operation','constraints'],contradiction_witness:['operation','left','right','asserted'],scenario_enumeration:['operation','left_relation','right_relation'],minimal_network:['operation','constraints'],temporal_trinity_bridge:['operation','fixture_id','synthetic'],temporal_observation_gap:['operation','fixture_id','synthetic'],temporal_authority_gate:['operation','fixture_id','synthetic']};
function execute(req){const op=req&&req.operation;ensure(allowed[op],'supported operation required');keys(req,allowed[op]);if(op==='interval_record'){ensure(Array.isArray(req.intervals)&&req.intervals.length===3,'three intervals');req.intervals.forEach(interval);return{operation:op,intervals:req.intervals,valid:true};}if(op==='relation_classify')return{operation:op,relation:relation(req.left,req.right)};if(op==='relation_converse'){ensure(ALL.includes(req.relation),'known relation');return{operation:op,relation:req.relation,converse:CONVERSE[req.relation]};}if(op==='endpoint_order'){const z=[];for(const[n,i]of Object.entries(req.intervals)){interval(i);z.push({value:i.start,label:n+'-start'},{value:i.end,label:n+'-end'});}z.sort((a,b)=>a.value-b.value||a.label.localeCompare(b.label));return{operation:op,order:z};}if(op==='relation_partition'){const r=relation(req.left,req.right);return{operation:op,matched:[r],exclusive:true,exhaustive_over_declared_pair:true};}if(op==='relation_symmetry'){const d={};for(const[k,i]of Object.entries(req.intervals)){interval(i);d[k]=i.end-i.start;}return{operation:op,durations:d,integer_endpoint_domain:true};}if(op==='network_record')return{operation:op,nodes:req.intervals,relations:{AB:relation(req.intervals.A,req.intervals.B),BC:relation(req.intervals.B,req.intervals.C),AC:relation(req.intervals.A,req.intervals.C)}};if(op==='pair_consistency'){const actual=relation(req.left,req.right);return{operation:op,asserted:req.asserted,actual,consistent:req.asserted===actual};}if(op==='relation_projection'){const r=relation(req.left,req.right);return{operation:op,relation:r,converse:CONVERSE[r],endpoints:[req.left.start,req.left.end,req.right.start,req.right.end]};}if(op==='relation_compose')return{operation:op,left_relation:req.left_relation,right_relation:req.right_relation,possible:compose(req.left_relation,req.right_relation),domain:'bounded exact integer witness 0..7'};if(op==='weak_composition')return{operation:op,possible:compose(req.left_relation,req.right_relation),claim_ceiling:'finite witness set; not an independent proof of the universal Allen table'};if(op==='path_consistency_step'){const c=close(req.constraints);return{operation:op,constraints:c,AC:c.find(x=>x.pair==='AC').relations};}if(op==='path_consistency_closure'){const actual=relation(req.left,req.right);return{operation:op,actual,asserted:req.asserted,contradiction:!req.asserted.includes(actual),witness:{left:req.left,right:req.right}};}if(op==='scenario_enumeration')return{operation:op,left_relation:req.left_relation,right_relation:req.right_relation,...scenarios(req.left_relation,req.right_relation)};if(op==='minimal_network'){const c=close(req.constraints);return{operation:op,constraints:c,nonuniversal_edges:c.filter(x=>x.relations.length<ALL.length).length,bounded_minimality_only:true};}if(op==='temporal_trinity_bridge')return{operation:op,status:'represented',gmUT:'typed temporal relation analogy only',thos:'synthetic scheduling protocol only',freed_id_cbr:'consent and authority reservations only'};if(op==='temporal_observation_gap')return{operation:op,status:'open_gap',missing:['real observations','calibrated clocks','participants','independent review'],network_calls:0,rows_ingested:0};if(op==='temporal_authority_gate')return{operation:op,status:'exact_gate',reserved:['professional scheduling','employment decisions','legal deadlines','cultural authority','Maori authority','affected-party consent']};}
module.exports={execute,relation,compose,ALL,CONVERSE};
