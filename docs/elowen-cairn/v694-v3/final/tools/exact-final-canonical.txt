function loadTxt(relative) { const filename = path.resolve(relative), mod = new Module(filename); mod.filename = filename; mod.paths = Module._nodeModulePaths(path.dirname(filename)); mod._compile(fs.readFileSync(filename, "utf8"), filename); return mod.exports; }
function replayManifest(manifest, head) {
  const issues = [];
  for (const binding of manifest.bindings) {
    let bytes;
    try { bytes = readBlob(binding.path, head); } catch { issues.push({ path: binding.path, issue: "missing" }); continue; }
    if (bytes.length !== binding.bytes) issues.push({ path: binding.path, issue: "bytes", expected: binding.bytes, observed: bytes.length });
    const observed = sha(bytes);
    if (observed !== binding.sha256) issues.push({ path: binding.path, issue: "sha256", expected: binding.sha256, observed });
  }
  return issues;
}
