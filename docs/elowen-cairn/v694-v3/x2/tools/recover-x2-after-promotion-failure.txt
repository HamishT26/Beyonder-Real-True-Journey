function smokeRunner(relative, operations) {
  const runnerPath = path.resolve(relative);
  const acceptingProposal = plan.rows.find((proposal) => operations.includes(proposal.operation));
  const rejectingProposal = plan.rows.find((proposal) => !operations.includes(proposal.operation));
  const accepting = JSON.parse(cp.execFileSync(process.execPath, ["-e", runnerLoader, runnerPath, JSON.stringify(acceptingProposal.request)], { cwd: root, encoding: "utf8" }).trim());
  let rejectingRaw = "", rejectingStatus = 0;
  try { rejectingRaw = cp.execFileSync(process.execPath, ["-e", runnerLoader, runnerPath, JSON.stringify(rejectingProposal.request)], { cwd: root, encoding: "utf8", stdio: ["ignore", "pipe", "pipe"] }).trim(); }
  catch (error) { rejectingStatus = error.status; rejectingRaw = String(error.stdout || "").trim(); }
  const rejecting = JSON.parse(rejectingRaw);
  return { accepting: { output: accepting, passed: accepting.ok === true }, rejecting: { output: rejecting, process_status: rejectingStatus, passed: rejecting.ok === false && rejecting.error === "E_SCOPE" } };
}
