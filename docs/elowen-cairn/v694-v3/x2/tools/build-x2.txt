function stable(value) {
  if (Array.isArray(value)) return value.map(stable);
  if (value && typeof value === "object") return Object.fromEntries(Object.keys(value).sort().map((key) => [key, stable(value[key])]));
  return value;
}
function canonical(value) { return JSON.stringify(stable(value)); }
function sha(bytes) { return crypto.createHash("sha256").update(bytes).digest("hex"); }
function readJson(relative) { return JSON.parse(fs.readFileSync(path.join(process.cwd(), relative), "utf8")); }
function ensureDir(relative) { fs.mkdirSync(path.join(process.cwd(), relative), { recursive: true }); }
function writeJson(relative, value) { ensureDir(path.dirname(relative)); fs.writeFileSync(path.join(process.cwd(), relative), `${JSON.stringify(value, null, 2)}\n`, "utf8"); }
function writeText(relative, value) { ensureDir(path.dirname(relative)); fs.writeFileSync(path.join(process.cwd(), relative), value.endsWith("\n") ? value : `${value}\n`, "utf8"); }
function git(...args) { return cp.execFileSync("git", args, { cwd: process.cwd(), encoding: "utf8" }).trim(); }
