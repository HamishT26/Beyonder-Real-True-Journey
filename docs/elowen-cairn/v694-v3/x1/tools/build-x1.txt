function stable(value) {
  if (Array.isArray(value)) return value.map(stable);
  if (value && typeof value === "object") return Object.fromEntries(Object.keys(value).sort().map((key) => [key, stable(value[key])]));
  return value;
}
function canonical(value) { return JSON.stringify(stable(value)); }
function sha(bytes) { return crypto.createHash("sha256").update(bytes).digest("hex"); }
function readJson(relative) { return JSON.parse(fs.readFileSync(path.join(process.cwd(), relative), "utf8")); }
function ensureDir(relative) { fs.mkdirSync(path.join(process.cwd(), relative), { recursive: true }); }
function writeJson(relative, value) { ensureDir(path.dirname(relative)); fs.writeFileSync(path.join(process.cwd(), relative), `${JSON.stringify(value, null, 2)}\n`, "utf8"); }
function writeText(relative, value) { ensureDir(path.dirname(relative)); fs.writeFileSync(path.join(process.cwd(), relative), value.endsWith("\n") ? value : `${value}\n`, "utf8"); }
function git(...args) { return cp.execFileSync("git", args, { cwd: process.cwd(), encoding: "utf8" }).trim(); }
function loadTxt(relative) {
  const filename = path.resolve(relative);
  const mod = new Module(filename);
  mod.filename = filename;
  mod.paths = Module._nodeModulePaths(path.dirname(filename));
  mod._compile(fs.readFileSync(filename, "utf8"), filename);
  return mod.exports;
}

const core = loadTxt(`${X1}/tools/transit-core.txt`);
const proposalResults = [];
const candidateResults = [];
const safeResults = [];
const cleanResults = [];
const receipts = [];
for (const proposal of x1Proposals) {
  const before = canonical(proposal.request);
  const output = core.evaluate(proposal.request);
  const after = canonical(proposal.request);
  const match = canonical(output) === canonical(proposal.expected_output);
  const immutable = before === after;
  proposalResults.push({ proposal_id: proposal.proposal_id, operation: proposal.operation, expected_disposition: proposal.expected_disposition, observed_disposition: match && immutable ? proposal.expected_disposition : null, complete_envelope_match: match, input_immutable: immutable, authority: output.authority, passed: match && immutable && output.authority === false, output });
  for (const subject of proposal.candidate_subjects) {
    const subjectBefore = canonical(subject.request);
    const observed = core.evaluate(subject.request);
    const subjectAfter = canonical(subject.request);
    const refusalPass = observed.ok === false && observed.error === subject.expected_rejection.error && subjectBefore === subjectAfter;
    candidateResults.push({ proposal_id: proposal.proposal_id, negative_id: subject.negative_id, expected_error: subject.expected_rejection.error, observed, request_immutable: subjectBefore === subjectAfter, refusal_pass: refusalPass, subject_success_credit: 0 });
  }
  const result = proposalResults.at(-1);
  safeResults.push(
    { task_id: `EC6943-X1-SAFE-${proposal.proposal_id.slice(-3)}-1`, proposal_id: proposal.proposal_id, check: "complete frozen envelope", passed: result.complete_envelope_match, credit: "bounded_same_owner_only" },
    { task_id: `EC6943-X1-SAFE-${proposal.proposal_id.slice(-3)}-2`, proposal_id: proposal.proposal_id, check: "request immutable and authority false", passed: result.input_immutable && result.authority === false, credit: "bounded_same_owner_only" }
  );
  cleanResults.push(
    { task_id: `EC6943-X1-CFR-${proposal.proposal_id.slice(-3)}-1`, proposal_id: proposal.proposal_id, check: "operation-specific strict field allowlist", passed: proposal.candidate_subjects[0].expected_rejection.error === "E_FIELDS", destructive: false },
    { task_id: `EC6943-X1-CFR-${proposal.proposal_id.slice(-3)}-2`, proposal_id: proposal.proposal_id, check: "authority boundary and exact outcome label", passed: proposal.expected_output.authority === false && ["completed", "represented", "open_gap", "exact_gate"].includes(proposal.expected_disposition), destructive: false }
  );
  const relatedCandidates = candidateResults.slice(-2);
  const receipt = { schema: "ghc.family.proposal-receipt.v9", owner: OWNER, phase: PHASE, session: "x1", proposal_id: proposal.proposal_id, definition_sha256: proposal.definition_sha256, positive: result, negatives: relatedCandidates, passed: result.passed && relatedCandidates.every((row) => row.refusal_pass), independent_reproduction: false, boundary: BOUNDARY };
  receipts.push(receipt);
  writeJson(`${X1}/evidence/${proposal.proposal_id}-receipt.json`, receipt);
}

const runnerSmokes = [];
for (const runner of runnerPlans) {
  const runnerPath = path.resolve(`${X1}/runners/${runner.name}.txt`);
  const acceptingProposal = x1Proposals.find((row) => runner.operations.includes(row.operation));
  const rejectingProposal = x1Proposals.find((row) => !runner.operations.includes(row.operation));
  const acceptingRaw = cp.execFileSync(process.execPath, ["-e", runnerLoader, runnerPath, JSON.stringify(acceptingProposal.request)], { cwd: process.cwd(), encoding: "utf8" }).trim();
  const accepting = JSON.parse(acceptingRaw);
  let rejectingRaw = "";
  let rejectingStatus = 0;
  try {
    rejectingRaw = cp.execFileSync(process.execPath, ["-e", runnerLoader, runnerPath, JSON.stringify(rejectingProposal.request)], { cwd: process.cwd(), encoding: "utf8", stdio: ["ignore", "pipe", "pipe"] }).trim();
  } catch (error) {
    rejectingStatus = error.status;
    rejectingRaw = String(error.stdout || "").trim();
  }
  const rejecting = JSON.parse(rejectingRaw);
  runnerSmokes.push({ runner: runner.name, operations: runner.operations, accepting: { output: accepting, passed: accepting.ok === true }, rejecting: { output: rejecting, process_status: rejectingStatus, passed: rejecting.ok === false && rejecting.error === "E_SCOPE" } });
}
