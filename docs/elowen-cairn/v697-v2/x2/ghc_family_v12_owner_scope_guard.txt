'use strict';
// Current weighted-profile reviewer. The historical v3 thirty-position reviewer remains unchanged.
function validate(profile,roster){const errors=[],check=(c,label)=>{if(!c)errors.push(label);},limits=profile?.limits||{},names=['inherited_proposals','new_proposals','safe_now_x1','safe_now_x2','candidate_x1','candidate_x2','clean_fix_refine_x1','clean_fix_refine_x2'];
check(typeof profile?.schema==='string'&&profile.schema.startsWith('ghc.family.workflow-profile.v12'),'profile_v12');check(roster?.schema==='ghc.family.weighted-route.v12','roster_v12');for(const key of names){const b=limits[key];check(Array.isArray(b)&&b.length===2&&b.every(Number.isInteger)&&b[0]===300&&b[1]>=b[0],'minimum_'+key);}
check(limits.own_practices===8&&limits.next_practices===4,'practices');check(Array.isArray(limits.baton_words)&&limits.baton_words[0]===2000&&limits.baton_words[1]===100000,'baton');check(limits.owner_file_ceiling===2000&&limits.materialized_file_ceiling===2000,'file_budget');check(profile.subagents===false&&profile.task_creation===false,'topology');check(profile.terminal_verdict==='NOT_READY_FOR_STAGE_20','truth');
check(JSON.stringify(profile.artifact_policy?.authored_extensions)===JSON.stringify(['.json','.md','.txt','.html']),'formats');check(profile.routing?.accepted_or_opaque_or_unresolved_stops_duplicates===true&&profile.routing?.blind_resend_quota_prohibited===true,'delivery_nonduplication');check(profile.routing?.minimum_bounded_discovery_or_read_recoveries_before_hold===5&&profile.routing?.definitive_preacceptance_rejection_resubmissions_maximum===3,'recovery_budget');
const cycle=roster.cycle,ids=roster.identities,rows=roster.assignments;check(Array.isArray(cycle)&&cycle.length===45&&new Set(cycle).size===30,'weighted_cycle');check(Array.isArray(ids)&&ids.length===30&&new Set(ids).size===30&&cycle.every(n=>ids.includes(n)),'identities');check(Array.isArray(rows)&&rows.length===235,'roster_rows');if(Array.isArray(rows)){const ordinal=s=>{const m=/^v(\d+)-v([1-8])$/.exec(s||'');return m?Number(m[1])*8+Number(m[2])-1:NaN;};check(rows.every((r,i)=>ordinal(r.phase)===ordinal(rows[0].phase)+i&&r.owner===cycle[(14+i)%45]),'consecutive_weighted_assignments');check(rows.at(-1)?.phase==='v725-v8'&&rows.at(-1)?.owner==='Teryn Halewick','terminal');}
check(roster.projection_is_activation===false&&roster.existing_model_settings_preserved===true,'projection_boundary');return {valid:errors.length===0,errors,scope:'new v12 owner profile and weighted projection only',permission_granted:false,delivery_verified:false,independent_reproduction:false};}
module.exports={validate};
