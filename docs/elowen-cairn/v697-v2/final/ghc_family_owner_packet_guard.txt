'use strict';
function validate(p){const errors=[],require=(c,label)=>{if(!c)errors.push(label);},record=(x,names)=>x&&typeof x==='object'&&!Array.isArray(x)&&Object.keys(x).length===names.length&&names.every(n=>Object.hasOwn(x,n));require(record(p,['owner','phase','namespace','content_source','git_parent','counts','outcomes','route','limits','controls','boundary']),'root_fields');if(errors.length)return {valid:false,errors};
require(typeof p.owner==='string'&&p.owner.length<=100,'owner');require(/^v\d+-v[1-8](?:-r\d+)?$/.test(p.phase),'phase');const slug=p.owner.toLowerCase().replace(/ /g,'-');require(p.namespace==='docs/'+slug+'/'+p.phase,'owner_namespace');require(/^[a-f0-9]{40}$/.test(p.content_source)&&/^[a-f0-9]{40}$/.test(p.git_parent),'anchors');require(p.content_source!==p.git_parent,'provenance_not_parent');
require(record(p.counts,['methods','witnesses','pass','fail'])&&Object.values(p.counts).every(n=>Number.isInteger(n)&&n>=0)&&p.counts.pass+p.counts.fail===p.counts.witnesses,'counts');require(record(p.outcomes,['completed','represented','open_gap','exact_gate'])&&Object.values(p.outcomes).every(n=>Number.isInteger(n)&&n>=0),'outcomes');require(record(p.route,['state','endpoint_kind','recipient','phase'])&&p.route.state==='PREPARED_NOT_SENT'&&p.route.endpoint_kind==='main_task'&&typeof p.route.recipient==='string'&&/^v\d+-v[1-8]$/.test(p.route.phase),'route_preparation');require(record(p.limits,['files','words','commits','spend_usd'])&&p.limits.files===2000&&p.limits.words===100000&&p.limits.commits===6&&p.limits.spend_usd===50,'limits');require(record(p.controls,['memory_write','package_updates','task_creation','subagents','sibling_mutation','source_replay','domain_replay'])&&Object.values(p.controls).every(v=>v===false),'protected_controls');require(typeof p.boundary==='string'&&p.boundary.includes('NOT_READY_FOR_STAGE_20'),'truth_boundary');return {valid:errors.length===0,errors,scope:'bounded owner packet structure only',delivery_verified:false,authority_granted:false,independent_reproduction:false};}
module.exports={validate};
