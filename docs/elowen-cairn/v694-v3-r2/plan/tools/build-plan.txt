function ensureDir(p) {
  fs.mkdirSync(p, { recursive: true });
}

function write(rel, value) {
  const target = path.join(ROOT, rel);
  ensureDir(path.dirname(target));
  const data = typeof value === 'string' ? value : JSON.stringify(value, null, 2) + '\n';
  fs.writeFileSync(target, data.replace(/\r\n/g, '\n'), 'utf8');
}

function shaText(value) {
  return crypto.createHash('sha256').update(value).digest('hex');
}

function fileSha(p) {
  return crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex');
}

function words(text) {
  const matches = String(text).match(/\S+/g);
  return matches ? matches.length : 0;
}

function round(value) {
  return Math.round((value + Number.EPSILON) * 1e12) / 1e12;
}

function nextPhase(phase) {
  const match = /^v(\d+)-v([1-8])$/.exec(phase);
  if (!match) throw new Error('invalid phase ' + phase);
  const version = Number(match[1]);
  const slot = Number(match[2]);
  return slot === 8 ? 'v' + (version + 1) + '-v1' : 'v' + version + '-v' + (slot + 1);
}

function oracle(operation, input) {
  if (operation === 'classify_claim_register') {
    const map = { physical: 'physical', informational: 'informational', normative: 'ethical', spiritual: 'ethical_metaphorical', mixed: 'requires_explicit_maps' };
    return { register: map[input.claim_type] || 'unresolved' };
  }
  if (operation === 'source_evidence_status') {
    const map = {
      historical_journey: 'historical_source_not_instruction',
      current_official: 'current_official_reference',
      primary_paper: 'primary_research_reference',
      inaccessible_external: 'open_gap_unreadable',
      private_or_secret_bearing: 'exact_gate_quarantine'
    };
    return { status: map[input.kind] || 'unresolved' };
  }
  if (operation === 'dimension_vector_equal') {
    const same = Array.isArray(input.left) && Array.isArray(input.right) &&
      input.left.length === input.right.length && input.left.every(function (v, i) { return v === input.right[i]; });
    return { equal: same };
  }
  if (operation === 'physical_equation_admission') {
    return input.term_register === 'physical'
      ? { admitted: true, reason: 'typed_physical_term' }
      : { admitted: false, reason: 'explicit_map_and_physical_evidence_required' };
  }
  if (operation === 'null_recovery_check') {
    return { recovered: input.alpha === 0 && input.phi_dot === 0 && input.eft_norm === 0 };
  }
  if (operation === 'coefficient_card_valid') {
    return { complete: Array.isArray(input.fields) && input.fields.length === 7 && input.fields.every(Boolean) };
  }
  if (operation === 'finite_difference') {
    return { differences: input.values.slice(1).map(function (v, i) { return round((v - input.values[i]) / input.dt); }) };
  }
  if (operation === 'symplectic_scalar_step') {
    const velocity = round(input.velocity - input.dt * input.mass2 * input.phi);
    const phi = round(input.phi + input.dt * velocity);
    return { phi, velocity };
  }
  if (operation === 'scalar_energy') {
    return { energy: round(0.5 * input.velocity * input.velocity + 0.5 * input.mass2 * input.phi * input.phi) };
  }
  if (operation === 'energy_drift') {
    const drift = round(Math.abs(input.after - input.before));
    return { drift, within_tolerance: drift <= input.tolerance };
  }
  if (operation === 'convergence_ratio') {
    const ratio = round(input.coarse / input.fine);
    return { ratio, meets_threshold: ratio >= input.threshold };
  }
  if (operation === 'sensitivity_delta') {
    const delta = round(input.variant - input.baseline);
    return { delta, relative: input.baseline === 0 ? null : round(delta / Math.abs(input.baseline)) };
  }
  if (operation === 'falsifier_matrix') {
    const residual = round(Math.abs(input.observed - input.predicted));
    return { residual, falsified: residual > input.tolerance };
  }
  if (operation === 'evidence_ceiling') {
    const map = {
      software: 'same_owner_software_only',
      citation: 'literature_support_only',
      synthetic: 'synthetic_evidence_only',
      independent: 'independent_reproduction_if_proven',
      empirical: 'empirical_result_with_declared_scope'
    };
    return { maximum_claim: map[input.evidence_class] || 'no_claim' };
  }
  if (operation === 'source_conflict_reconcile') {
    if (input.newer === 'equal_rank' && input.older === 'equal_rank') return { controlling: null, state: 'unresolved' };
    return { controlling: 'newer', state: 'prospective_override_with_historical_retention' };
  }
  if (operation === 'roster_next') {
    const index = input.position - 1;
    const exact = cycle[index] === input.owner;
    return exact
      ? { owner: cycle[(index + 1) % cycle.length], phase: nextPhase(input.phase), resolved: true }
      : { owner: null, phase: null, resolved: false };
  }
  if (operation === 'retry_decision') {
    if (['accepted', 'opaque_accepted', 'unresolved_after_submission'].includes(input.submission_state)) {
      return { action: 'stop_duplicates' };
    }
    if (input.submission_state === 'definitive_preacceptance_rejection') {
      return { action: input.rejected_resubmissions < 3 ? 'correct_and_resubmit' : 'hold_after_three_rejections' };
    }
    return { action: input.discovery_attempts < 5 ? 'bounded_discovery_retry' : 'hold_after_five_discovery_attempts' };
  }
  if (operation === 'delivery_latch') {
    if (['accepted', 'opaque_accepted', 'unresolved_after_submission'].includes(input.state)) return { send_allowed: false, latched: true };
    return { send_allowed: true, latched: false };
  }
  if (operation === 'accessibility_review_vacancy') {
    const missing = [];
    if (!input.automated) missing.push('automated_structure');
    if (!input.manual_keyboard) missing.push('manual_keyboard');
    if (!input.assistive_technology) missing.push('assistive_technology');
    if (!input.affected_user) missing.push('affected_user');
    return { open_gap: true, missing };
  }
  if (operation === 'authority_reservation') {
    return { exact_gate: !input.authority_present, domain: input.domain };
  }
  throw new Error('unknown operation ' + operation);
}

    cases[operation].forEach(function (input, caseIndex) {

      const canonicalRequest = { operation, input };

        execution_lane: lane,
        current_official_or_primary_source_needs: operationIndex <= 15 ? ['current source ledger and declared local evidence'] : ['live external evidence or authority remains absent'],
        concrete_artifacts: [
          'docs/elowen-cairn/v694-v3-r2/' + lane + '/evidence/' + proposalId + '-receipt.json'
        ],
        falsifier_or_acceptance_gate: 'Exact request and exact expected output must match; malformed siblings remain failed at zero subject credit.',
        rollback_or_recovery: 'Retain the failed subject, inspect the smallest dependency, and rerun only the failed dependency. Never replay a successful canonical.',
        protected_gates: PROTECTED,
        expected_disposition: disposition,
        canonical_request: canonicalRequest,
        expected_output: expectedOutput,
        semantic_fingerprint: shaText(JSON.stringify(canonicalRequest)),
        inherited_completion_credit: 0,
        independent_reproduction: false
      };

}

      protected_gates: PROTECTED
    };
  });
}

const manifestRows = [];
function walk(dir) {
  fs.readdirSync(dir, { withFileTypes: true }).forEach(function (entry) {
    const full = path.join(dir, entry.name);
    if (entry.isDirectory()) walk(full);
    else {
      const rel = path.relative(ROOT, full).split(path.sep).join('/');
      if (rel === 'manifest.json' || rel === 'planning-validation.json') return;
      const content = fs.readFileSync(full);
      manifestRows.push({ path: rel, bytes: content.length, sha256: fileSha(full) });
    }
  });
}
walk(ROOT);
manifestRows.sort(function (a, b) { return a.path.localeCompare(b.path); });
