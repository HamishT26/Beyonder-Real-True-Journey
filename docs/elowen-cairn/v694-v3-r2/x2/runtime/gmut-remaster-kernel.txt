function extensionExecute(request) {
  if (!request || typeof request !== 'object' || Array.isArray(request)) throw new TypeError('request_must_be_object');
  const operation = request.operation;
  const input = request.input;
  if (!input || typeof input !== 'object' || Array.isArray(input)) throw new TypeError('input_must_be_object');
  if (operation === 'convergence_ratio') {
    finite(input.coarse, 'coarse'); finite(input.fine, 'fine'); finite(input.threshold, 'threshold');
    if (input.coarse <= 0 || input.fine <= 0) throw new RangeError('positive_errors_required');
    const ratio = round(input.coarse / input.fine);
    return { ratio: ratio, meets_threshold: ratio >= input.threshold };
  }
  if (operation === 'sensitivity_delta') {
    finite(input.baseline, 'baseline'); finite(input.variant, 'variant');
    const delta = round(input.variant - input.baseline);
    return { delta: delta, relative: input.baseline === 0 ? null : round(delta / Math.abs(input.baseline)) };
  }
  if (operation === 'falsifier_matrix') {
    finite(input.observed, 'observed'); finite(input.predicted, 'predicted'); finite(input.tolerance, 'tolerance');
    if (input.tolerance < 0) throw new RangeError('tolerance_must_be_nonnegative');
    const residual = round(Math.abs(input.observed - input.predicted));
    return { residual: residual, falsified: residual > input.tolerance };
  }
  if (operation === 'evidence_ceiling') {
    const map = { software: 'same_owner_software_only', citation: 'literature_support_only', synthetic: 'synthetic_evidence_only', independent: 'independent_reproduction_if_proven', empirical: 'empirical_result_with_declared_scope' };
    if (!(input.evidence_class in map)) throw new RangeError('unknown_evidence_class');
    return { maximum_claim: map[input.evidence_class] };
  }
  if (operation === 'source_conflict_reconcile') {
    if (input.newer === 'equal_rank' && input.older === 'equal_rank') return { controlling: null, state: 'unresolved' };
    const accepted = ['direct_user', 'exact_git_blob', 'current_official', 'primary_research'];
    if (!accepted.includes(input.newer)) throw new RangeError('unsupported_newer_source');
    return { controlling: 'newer', state: 'prospective_override_with_historical_retention' };
  }
  if (operation === 'roster_next') {
    const index = input.position - 1;
    const exact = Number.isInteger(index) && index >= 0 && index < cycle.length && cycle[index] === input.owner;

  if (operation === 'retry_decision') {
    if (['accepted', 'opaque_accepted', 'unresolved_after_submission'].includes(input.submission_state)) return { action: 'stop_duplicates' };
    if (input.submission_state === 'definitive_preacceptance_rejection') return { action: input.rejected_resubmissions < 3 ? 'correct_and_resubmit' : 'hold_after_three_rejections' };
    if (input.submission_state !== 'not_submitted') throw new RangeError('unknown_submission_state');
    return { action: input.discovery_attempts < 5 ? 'bounded_discovery_retry' : 'hold_after_five_discovery_attempts' };
  }

    if (!input.assistive_technology) missing.push('assistive_technology');

  }
