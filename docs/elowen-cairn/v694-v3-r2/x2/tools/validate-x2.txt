function readJson(file) { return JSON.parse(fs.readFileSync(file, 'utf8')); }
function fileSha(file) { return crypto.createHash('sha256').update(fs.readFileSync(file)).digest('hex'); }
function write(rel, value) {
  const target = path.join(ROOT, rel);
  fs.mkdirSync(path.dirname(target), { recursive: true });
  fs.writeFileSync(target, JSON.stringify(value, null, 2) + '\n', 'utf8');
}
function loadTxt(file) {
  const mod = new Module(file);
  mod.filename = file;
  mod.paths = Module._nodeModulePaths(path.dirname(file));
  mod._compile(fs.readFileSync(file, 'utf8'), file);
  return mod.exports;
}
function walk(dir, rows) {
  fs.readdirSync(dir, { withFileTypes: true }).forEach(function (entry) {
    const full = path.join(dir, entry.name);
    if (entry.isDirectory()) walk(full, rows);
    else rows.push(full);
  });
}

function validateSkillTree(root, kind) {
  const rows = [];
  fs.readdirSync(root, { withFileTypes: true }).filter(function (entry) { return entry.isDirectory(); }).sort(function (a, b) { return a.name.localeCompare(b.name); }).forEach(function (entry) {
    const folder = path.join(root, entry.name);
    const quick = cp.spawnSync('python', [skillValidator, folder], { encoding: 'utf8' });
    const valid = !quick.error && quick.status === 0;
    if (!valid) issues.push(kind + '_skill_' + entry.name);
    const files = [];
    walk(folder, files);
    rows.push({
      name: entry.name,
      kind: kind,
      complete_read: readComplete,
      quick_validated: valid,
      quick_output: String(quick.stdout || quick.stderr || '').trim(),
      bindings: files.map(function (file) { return { path: path.relative(folder, file).split(path.sep).join('/'), bytes: fs.statSync(file).size, sha256: fileSha(file) }; }).sort(function (a, b) { return a.path.localeCompare(b.path); })
    });
  });
  return rows;
}

function validateRunners(root, expectedCount, kind) {
  const rows = [];
  fs.readdirSync(root).filter(function (name) { return name.startsWith('ghc_family_') && name.endsWith('.txt'); }).sort().forEach(function (name) {
    const runner = loadTxt(path.join(root, name));
    const operation = runner.operations[0];
    const proposal = proposals.rows.find(function (row) { return row.operation === operation; });
    const observed = runner.run(proposal.canonical_request);
    const positivePass = JSON.stringify(observed) === JSON.stringify(proposal.expected_output);
    let negativeFailed = false;
    try { runner.run({ operation: 'unsupported_runner_subject', input: {} }); } catch (error) { negativeFailed = true; }
    if (!positivePass || !negativeFailed) issues.push(kind + '_runner_' + name);
    rows.push({
      name: name,
      kind: kind,
      operations: runner.operations,
      positive_pass: positivePass,
      negative_subject_result: negativeFailed ? 'fail' : 'unexpected_pass',
      negative_subject_success_credit: 0,
      refusal_check_pass: negativeFailed,
      sha256: fileSha(path.join(root, name))
    });
  });
  if (rows.length !== expectedCount) issues.push(kind + '_runner_count');
  return rows;
}

}).map(function (file) {
  return { path: path.relative(ROOT, file).split(path.sep).join('/'), bytes: fs.statSync(file).size, sha256: fileSha(file) };
}).sort(function (a, b) { return a.path.localeCompare(b.path); });
