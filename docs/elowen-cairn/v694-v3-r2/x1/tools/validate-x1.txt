function readJson(p) {
  return JSON.parse(fs.readFileSync(p, 'utf8'));
}

function write(rel, value) {
  const target = path.join(ROOT, rel);
  fs.mkdirSync(path.dirname(target), { recursive: true });
  fs.writeFileSync(target, JSON.stringify(value, null, 2) + '\n', 'utf8');
}

function loadTxt(file) {
  const mod = new Module(file);
  mod.filename = file;
  mod.paths = Module._nodeModulePaths(path.dirname(file));
  mod._compile(fs.readFileSync(file, 'utf8'), file);
  return mod.exports;
}

function fileSha(file) {
  return crypto.createHash('sha256').update(fs.readFileSync(file)).digest('hex');
}

function walk(dir, rows) {
  fs.readdirSync(dir, { withFileTypes: true }).forEach(function (entry) {
    const full = path.join(dir, entry.name);
    if (entry.isDirectory()) walk(full, rows);
    else rows.push(full);
  });
}

  const skillText = fs.readFileSync(path.join(folder, 'SKILL.md'), 'utf8');

  let quick = { status: 1, stdout: '', stderr: 'validator unavailable' };
  if (skillValidator && fs.existsSync(skillValidator)) {
    quick = cp.spawnSync(process.execPath.replace(/node(?:\.exe)?$/i, 'python.exe'), [skillValidator, folder], { encoding: 'utf8' });
    if (quick.error || quick.status !== 0) {
      const py = cp.spawnSync('python', [skillValidator, folder], { encoding: 'utf8' });
      quick = py;
    }
  }
  const valid = !quick.error && quick.status === 0;
  if (!valid) issues.push('skill_' + entry.name);
  skillRows.push({
    name: entry.name,
    complete_read_required_before_use: true,
    complete_read_receipt_supplied_by_controller: process.env.SKILL_READ_COMPLETE === 'true',
    quick_validated: valid,
    quick_output: String(quick.stdout || quick.stderr || '').trim(),
    skill_sha256: crypto.createHash('sha256').update(skillText).digest('hex'),
    contract_sha256: fileSha(path.join(folder, 'references/contracts.json')),
    operations: contract.operations
  });
});

    runner.run({ operation: 'unsupported_runner_subject', input: {} });

    negativeFailed = true;

  if (!positivePass || !negativeFailed) issues.push('runner_' + name);
  runnerRows.push({
    name: name,
    operations: runner.operations,
    positive_pass: positivePass,
    negative_subject_result: negativeFailed ? 'fail' : 'unexpected_pass',
    negative_subject_success_credit: 0,

    sha256: fileSha(path.join(runnerRoot, name))

});

function addWitness(methodIndex, result, procedure, observed, negativeId) {
  const method = methods[methodIndex];
  const witness = {
    witness_id: 'EC6943R2-X1-W' + String(witnesses.length + 1).padStart(4, '0'),
    method_id: method.method_id,
    procedure: procedure,
    scope: 'owner_self_scoped_delta',
    expected: result === 'pass' ? 'bounded pass' : 'subject failure',
    observed: observed,
    result: result,
    same_owner_only: true,
    independent_reproduction: false,
    retained_negative_ids: negativeId ? [negativeId] : [],
    boundary: BOUNDARY
  };
  witnesses.push(witness);
  method.validation_witness_ids.push(witness.witness_id);
}
