function readJson(file) { return JSON.parse(fs.readFileSync(file, 'utf8')); }
function digest(buffer) { return crypto.createHash('sha256').update(buffer).digest('hex'); }
function loadTxt(file) {
  const mod = new Module(file);
  mod.filename = file;
  mod.paths = Module._nodeModulePaths(path.dirname(file));
  mod._compile(fs.readFileSync(file, 'utf8'), file);
  return mod.exports;
}
function walk(dir, rows) {
  fs.readdirSync(dir, { withFileTypes: true }).forEach(function (entry) {
    const full = path.join(dir, entry.name);
    if (entry.isDirectory()) walk(full, rows);
    else rows.push(full);
  });
}
function git(args) { return cp.execFileSync('git', args, { cwd: REPO, encoding: 'utf8' }).trim(); }
function check(name, pass, detail) {

}
function verifyManifest(rel) {
  const manifest = readJson(path.join(REPO, rel));
  const failures = [];
  manifest.entries.forEach(function (entry) {
    const full = entry.path.startsWith('docs/') ? entry.path : path.posix.join(path.posix.dirname(rel), entry.path);
    let blob;
    try { blob = cp.execFileSync('git', ['cat-file', 'blob', exactFinal + ':' + full], { cwd: REPO }); }
    catch (error) { failures.push(full + ':missing'); return; }
    if (blob.length !== entry.bytes || digest(blob) !== entry.sha256) failures.push(full + ':mismatch');
  });
  return { path: rel, bindings: manifest.entries.length, failures: failures };
}
