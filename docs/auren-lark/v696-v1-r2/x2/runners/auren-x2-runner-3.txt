"use strict";
const fs = require("fs");
const Module = require("module");
const path = require("path");
function loadTextModule(file) { const item = new Module(file, module); item.filename = file; item.paths = Module._nodeModulePaths(path.dirname(file)); item._compile(fs.readFileSync(file, "utf8"), file); return item.exports; }
const core = loadTextModule(path.resolve(__dirname, "../lib/conformance-extended.txt"));
const ALLOWED = ["harper_contraction","postulate_checks"];
function run(subject) { if (!subject || !subject.request || !ALLOWED.includes(subject.request.operation)) return {ok:false,error:"E_RUNNER_OPERATION",original_success_credit:0}; return core.run(subject); }
module.exports = { ALLOWED, run, runner: "auren-x2-runner-3.txt" };
if (require.main === module) { const input = JSON.parse(fs.readFileSync(0, "utf8")); process.stdout.write(JSON.stringify(run(input)) + "\n"); }
