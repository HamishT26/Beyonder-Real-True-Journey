function stable(value) {
  if (Array.isArray(value)) return value.map(stable);
  if (value && typeof value === 'object') {
    const out = {};
    for (const key of Object.keys(value).sort()) out[key] = stable(value[key]);
    return out;
  }
  return value;
}

function canonical(value) {
  return Buffer.from(JSON.stringify(stable(value)), 'utf8');
}

function shaBytes(bytes) {
  return crypto.createHash('sha256').update(bytes).digest('hex');
}

function shaValue(value) {
  return shaBytes(canonical(value));
}

function words(text) {
  const found = String(text).trim().match(/\S+/g);
  return found ? found.length : 0;
}

function ensureDir(dir) {
  fs.mkdirSync(dir, {recursive: true});
}

function writeExclusive(file, value) {
  if (fs.existsSync(file)) throw new Error(`exclusive_write:${path.relative(W, file)}`);
  ensureDir(path.dirname(file));
  const text = typeof value === 'string' ? value : `${JSON.stringify(value, null, 2)}\n`;
  if (words(text) > 100000) throw new Error(`word_ceiling:${path.relative(W, file)}:${words(text)}`);
  fs.writeFileSync(file, text, {encoding: 'utf8', flag: 'wx'});
}

function writeReplaceExternal(file, value) {
  ensureDir(path.dirname(file));
  const text = typeof value === 'string' ? value : `${JSON.stringify(value, null, 2)}\n`;
  fs.writeFileSync(file, text, {encoding: 'utf8'});
}

function git(args, options = {}) {
  const result = cp.spawnSync('git', ['-C', W, ...args], {
    encoding: options.encoding === null ? null : 'utf8',
    input: options.input,
    maxBuffer: 64 * 1024 * 1024,
    timeout: options.timeout || 300000,
    windowsHide: true,
  });
  if (result.error) throw result.error;
  if (result.status !== 0) throw new Error(`git:${args.join(' ')}:${String(result.stderr)}`);
  return result.stdout;
}

function makeProfiles() {
  const profiles = [];
  const seen = new Set();
  for (let seed = 1; profiles.length < 15 && seed < 200; seed += 1) {
    const edges = [];
    for (let from = 0; from < 6; from += 1) {
      for (let to = 0; to < 6; to += 1) {
        if (from === to) continue;
        const value = (seed * 11 + from * 7 + to * 13 + from * to * 3) % 19;
        if (value < 5 + (seed % 4)) edges.push([from, to]);
      }
    }
    const signature = edges.map(edge => edge.join('>')).join(',');
    if (!edges.length || edges.length === 30 || seen.has(signature)) continue;
    seen.add(signature);
    const profile = {
      profile_id: `AF6987-F${String(profiles.length + 1).padStart(2, '0')}`,
      arguments: argumentsList(),
      attacks: edges,
      synthetic: true,
      real_world_referent: false,
    };
    profile.profile_sha256 = shaValue(profile);
    profiles.push(profile);
  }
  assert(profiles.length === 15, 'profile_count');
  assert(new Set(profiles.map(row => row.profile_sha256)).size === 15, 'profile_uniqueness');
  return profiles;
}

function framework(profile) {
  const n = profile.arguments.length;
  const edgeSet = new Set(profile.attacks.map(([a, b]) => `${a}:${b}`));
  const attacks = (a, b) => edgeSet.has(`${a}:${b}`);
  const masks = Array.from({length: 1 << n}, (_, index) => index);
  const has = (mask, arg) => Boolean(mask & (1 << arg));
  const members = mask => Array.from({length: n}, (_, index) => index).filter(index => has(mask, index));
  const conflictFree = mask => profile.attacks.every(([a, b]) => !(has(mask, a) && has(mask, b)));
  const defends = (mask, arg) => {
    const attackers = Array.from({length: n}, (_, index) => index).filter(index => attacks(index, arg));
    return attackers.every(attacker => members(mask).some(defender => attacks(defender, attacker)));
  };
  const admissible = mask => conflictFree(mask) && members(mask).every(arg => defends(mask, arg));
  const complete = mask => admissible(mask) && Array.from({length: n}, (_, arg) => has(mask, arg) === defends(mask, arg)).every(Boolean);
  const conflictFreeSets = masks.filter(conflictFree);
  const admissibleSets = masks.filter(admissible);
  const completeSets = masks.filter(complete);
  const preferredSets = admissibleSets.filter(mask => !admissibleSets.some(other => other !== mask && (mask & other) === mask));
  const stableSets = conflictFreeSets.filter(mask => Array.from({length: n}, (_, arg) => has(mask, arg) || members(mask).some(source => attacks(source, arg))).every(Boolean));
  let grounded = 0;
  for (let round = 0; round <= n; round += 1) {
    let next = 0;
    for (let arg = 0; arg < n; arg += 1) if (defends(grounded, arg)) next |= 1 << arg;
    if (next === grounded) break;
    grounded = next;
  }
  const reach = Array.from({length: n}, (_, i) => Array.from({length: n}, (_, j) => i === j || attacks(i, j)));
  for (let k = 0; k < n; k += 1) for (let i = 0; i < n; i += 1) for (let j = 0; j < n; j += 1) reach[i][j] ||= reach[i][k] && reach[k][j];
  const unseen = new Set(Array.from({length: n}, (_, index) => index));
  const sccs = [];
  while (unseen.size) {
    const first = Math.min(...unseen);
    const component = Array.from(unseen).filter(other => reach[first][other] && reach[other][first]).sort((a, b) => a - b);
    component.forEach(item => unseen.delete(item));
    sccs.push(component);
  }
  const labelMask = mask => members(mask).map(index => profile.arguments[index]);
  return {n, edgeSet, attacks, masks, has, members, conflictFree, defends, admissible, complete, conflictFreeSets, admissibleSets, completeSets, preferredSets, stableSets, grounded, sccs, labelMask};
}

function cardinalityHistogram(masks) {
  const counts = {};
  for (const mask of masks) {
    const size = mask.toString(2).replace(/0/g, '').length;
    counts[size] = (counts[size] || 0) + 1;
  }
  return Object.entries(counts).map(([size, count]) => ({size: Number(size), count}));
}

function withAttacks(profile, attacks) {
  const next = {profile_id: profile.profile_id, arguments: [...profile.arguments], attacks: attacks.map(edge => [...edge]), synthetic: true, real_world_referent: false};
  next.profile_sha256 = shaValue(next);
  return next;
}

function expected(operationName, profile) {
  const af = framework(profile);
  let value;
  let outcome = OPERATIONS.find(item => item.name === operationName).outcome;
  if (operationName === 'af_conflict_free') value = af.conflictFreeSets.map(af.labelMask);
  else if (operationName === 'af_admissible') value = af.admissibleSets.map(af.labelMask);
  else if (operationName === 'af_complete') value = af.completeSets.map(af.labelMask);
  else if (operationName === 'af_grounded') value = af.labelMask(af.grounded);
  else if (operationName === 'af_preferred') value = af.preferredSets.map(af.labelMask);
  else if (operationName === 'af_stable') value = af.stableSets.map(af.labelMask);
  else if (operationName === 'af_scc') value = af.sccs.map(component => component.map(index => profile.arguments[index]));
  else if (operationName === 'af_degree_signature') {
    value = profile.arguments.map((argument, index) => ({
      argument,
      incoming: profile.attacks.filter(([, to]) => to === index).length,
      outgoing: profile.attacks.filter(([from]) => from === index).length,
    }));
  } else if (operationName === 'af_defense_map') {
    value = profile.arguments.map((argument, index) => ({argument, defending_masks: af.masks.filter(mask => af.defends(mask, index))}));
  } else if (operationName === 'af_semantics_profile') {
    value = {
      conflict_free: {count: af.conflictFreeSets.length, cardinalities: cardinalityHistogram(af.conflictFreeSets)},
      admissible: {count: af.admissibleSets.length, cardinalities: cardinalityHistogram(af.admissibleSets)},
      complete: {count: af.completeSets.length, cardinalities: cardinalityHistogram(af.completeSets)},
      preferred: {count: af.preferredSets.length, cardinalities: cardinalityHistogram(af.preferredSets)},
      stable: {count: af.stableSets.length, cardinalities: cardinalityHistogram(af.stableSets)},
      grounded: af.labelMask(af.grounded),
    };
  } else if (operationName === 'af_skeptical_preferred') {
    const intersection = af.preferredSets.reduce((acc, mask) => acc & mask, (1 << af.n) - 1);
    value = af.labelMask(intersection);
  } else if (operationName === 'af_credulous_preferred') {
    const union = af.preferredSets.reduce((acc, mask) => acc | mask, 0);
    value = af.labelMask(union);
  } else if (operationName === 'af_attack_deletion_sensitivity') {
    value = profile.attacks.map((edge, index) => {
      const attacks = profile.attacks.filter((_, candidate) => candidate !== index);
      const changed = framework(withAttacks(profile, attacks));
      return {deleted: edge, grounded: changed.labelMask(changed.grounded), changed: changed.grounded !== af.grounded};
    });
  } else if (operationName === 'af_argument_deletion_sensitivity') {
    value = profile.arguments.map((argument, removed) => {
      const survivors = Array.from({length: af.n}, (_, index) => index).filter(index => index !== removed);
      const remap = new Map(survivors.map((oldIndex, newIndex) => [oldIndex, newIndex]));
      const attacks = profile.attacks.filter(([from, to]) => from !== removed && to !== removed).map(([from, to]) => [remap.get(from), remap.get(to)]);
      const sub = {profile_id: `${profile.profile_id}-minus-${argument}`, arguments: survivors.map(index => profile.arguments[index]), attacks, synthetic: true, real_world_referent: false};
      sub.profile_sha256 = shaValue(sub);
      const changed = framework(sub);
      return {deleted: argument, grounded: changed.labelMask(changed.grounded)};
    });
  } else if (operationName === 'af_attack_reversal_sensitivity') {
    value = profile.attacks.map((edge, index) => {
      const reversed = [edge[1], edge[0]];
      const attacks = profile.attacks.filter((_, candidate) => candidate !== index);
      if (!attacks.some(([from, to]) => from === reversed[0] && to === reversed[1])) attacks.push(reversed);
      attacks.sort((a, b) => a[0] - b[0] || a[1] - b[1]);
      const changed = framework(withAttacks(profile, attacks));
      return {reversed: edge, grounded: changed.labelMask(changed.grounded), changed: changed.grounded !== af.grounded};
    });
  } else if (operationName === 'af_stable_repair_frontier') {
    if (af.stableSets.length) value = {distance: 0, edits: [], stable_extensions: af.stableSets.map(af.labelMask)};
    else {
      const edits = [];
      profile.attacks.forEach((edge, index) => {
        const changed = withAttacks(profile, profile.attacks.filter((_, candidate) => candidate !== index));
        const state = framework(changed);
        if (state.stableSets.length) edits.push({kind: 'delete', edge, stable_extensions: state.stableSets.map(state.labelMask)});
      });
      for (let from = 0; from < af.n; from += 1) for (let to = 0; to < af.n; to += 1) {
        if (from === to || af.attacks(from, to)) continue;
        const changed = withAttacks(profile, [...profile.attacks, [from, to]].sort((a, b) => a[0] - b[0] || a[1] - b[1]));
        const state = framework(changed);
        if (state.stableSets.length) edits.push({kind: 'add', edge: [from, to], stable_extensions: state.stableSets.map(state.labelMask)});
      }
      value = {distance: edits.length ? 1 : '>1', edits};
    }
  } else if (operationName === 'af_provenance_certificate') {
    const core = {
      profile_sha256: profile.profile_sha256,
      conflict_free: af.conflictFreeSets,
      admissible: af.admissibleSets,
      complete: af.completeSets,
      grounded: af.grounded,
      preferred: af.preferredSets,
      stable: af.stableSets,
    };
    value = {algorithm: 'sha256-canonical-json', certificate: shaValue(core), core};
  } else if (operationName === 'af_nonphysical_scene') {
    value = {
      coordinates: profile.arguments.map((argument, index) => [
        index,
        profile.attacks.filter(([, to]) => to === index).length,
        profile.attacks.filter(([from]) => from === index).length,
      ]),
      axes: ['declared argument index', 'finite incoming-attack count', 'finite outgoing-attack count'],
      physical: false,
      empirical_measurement: false,
    };
  } else if (operationName === 'af_evidence_gap') {
    value = {missing: ['empirical', 'participant', 'professional', 'independent-reproduction'], semantic_extensions_are_human_beliefs: false, release: false};
  } else if (operationName === 'af_authority_gate') {
    value = {closed: ['consent', 'legal', 'cultural', 'affected-party', 'Maori-authority'], finite_semantics_confer_authority: false, release: false};
  } else throw new Error(`unknown_operation:${operationName}`);
  return {ok: true, error: null, outcome, value, external_credit: false, same_owner_only: true};
}

}

function inheritedSelections() {
  const shardNames = ['proposals-x1-1.json', 'proposals-x1-2.json', 'proposals-x2-1.json', 'proposals-x2-2.json'];
  const selected = [];
  const byOperation = new Map();
  for (const shardName of shardNames) {
    const bytes = sourceBytes(`plan/${shardName}`);
    const parsed = JSON.parse(bytes.toString('utf8'));
    parsed.contracts.forEach((record, recordIndex) => {
      const count = byOperation.get(record.operation) || 0;
      if (count >= 15) return;
      byOperation.set(record.operation, count + 1);
      selected.push({
        source_path: `${SOURCE_PREFIX}plan/${shardName}`,
        source_blob_sha256: shaBytes(bytes),
        record_index: recordIndex,
        source_proposal_id: record.proposal_id,
        record_sha256: shaValue(record),
        novelty_credit: 0,
        completion_credit: 0,
        executed_as_auren_contract: false,
      });
    });
  }
  assert(byOperation.size === 20, 'inherited_operation_count');
  assert(Array.from(byOperation.values()).every(value => value === 15), 'inherited_per_operation');
  assert(selected.length === 300, 'inherited_count');
  return selected;
}

function validateRequest(request, plan) {
  const keys = Object.keys(request).sort();
  if (JSON.stringify(keys) !== JSON.stringify(['op', 'profile_id', 'profile_sha256'])) throw new Error('field_set');
  const operation = plan.operations.find(item => item.name === request.op);
  if (!operation) throw new Error('operation');
  const profile = plan.profiles.find(item => item.profile_id === request.profile_id);
  if (!profile || profile.profile_sha256 !== request.profile_sha256) throw new Error('profile');
  return {operation, profile};
}

function runTests(plan) {
  const states = plan.profiles.map(profile => ({profile, af: framework(profile)}));
  const checks = [
    ['T01', 'Every complete extension is admissible.', () => states.every(({af}) => af.completeSets.every(mask => af.admissible(mask)))],
    ['T02', 'The grounded extension is complete.', () => states.every(({af}) => af.complete(af.grounded))],
    ['T03', 'Every preferred extension is admissible.', () => states.every(({af}) => af.preferredSets.every(mask => af.admissible(mask)))],
    ['T04', 'No preferred extension has a strict admissible superset.', () => states.every(({af}) => af.preferredSets.every(mask => !af.admissibleSets.some(other => other !== mask && (mask & other) === mask)))],
    ['T05', 'Every stable extension is conflict-free.', () => states.every(({af}) => af.stableSets.every(mask => af.conflictFree(mask)))],
    ['T06', 'Every stable extension attacks each outside argument.', () => states.every(({af}) => af.stableSets.every(mask => Array.from({length: af.n}, (_, arg) => af.has(mask, arg) || af.members(mask).some(source => af.attacks(source, arg))).every(Boolean)))],
    ['T07', 'The grounded extension is contained in every complete extension.', () => states.every(({af}) => af.completeSets.every(mask => (af.grounded & mask) === af.grounded))],
    ['T08', 'Skeptically preferred arguments are credulously preferred.', () => states.every(({profile}) => {const s = expected('af_skeptical_preferred', profile).value; const c = new Set(expected('af_credulous_preferred', profile).value); return s.every(item => c.has(item));})],
    ['T09', 'Strongly connected components partition all arguments once.', () => states.every(({af}) => af.sccs.flat().sort((a, b) => a - b).join(',') === '0,1,2,3,4,5')],
    ['T10', 'Attack-deletion rows equal declared attacks.', () => states.every(({profile}) => expected('af_attack_deletion_sensitivity', profile).value.length === profile.attacks.length)],
    ['T11', 'Argument-deletion rows equal declared arguments.', () => states.every(({profile}) => expected('af_argument_deletion_sensitivity', profile).value.length === profile.arguments.length)],
    ['T12', 'Attack-reversal rows equal declared attacks.', () => states.every(({profile}) => expected('af_attack_reversal_sensitivity', profile).value.length === profile.attacks.length)],
    ['T13', 'Provenance certificates are deterministic.', () => states.every(({profile}) => shaValue(expected('af_provenance_certificate', profile)) === shaValue(expected('af_provenance_certificate', profile)))],
    ['T14', 'Repair-frontier edits are bounded to add or delete.', () => states.every(({profile}) => expected('af_stable_repair_frontier', profile).value.edits.every(edit => ['add', 'delete'].includes(edit.kind)))],
    ['T15', 'Every result uses one of the four exact outcome labels.', () => plan.contracts.every(contract => ['completed', 'represented', 'open_gap', 'exact_gate'].includes(contract.expected_outcome))],
  ].map(([id, claim, test]) => ({id, claim, pass: Boolean(test()), same_owner_only: true, independent_reproduction: false}));
  assert(checks.length === 15 && checks.every(check => check.pass), `additional_tests:${JSON.stringify(checks.filter(check => !check.pass))}`);
  return checks;
}

function buildModels(plan) {
  const rows = plan.profiles.map(profile => {
    const af = framework(profile);
    const deletion = expected('af_attack_deletion_sensitivity', profile).value;
    const reversal = expected('af_attack_reversal_sensitivity', profile).value;
    const argumentDeletion = expected('af_argument_deletion_sensitivity', profile).value;
    const repair = expected('af_stable_repair_frontier', profile).value;
    const skeptical = expected('af_skeptical_preferred', profile).value;
    const credulous = expected('af_credulous_preferred', profile).value;
    return {profile, af, deletion, reversal, argumentDeletion, repair, skeptical, credulous};
  });
  const models = [
    {
      id: 'AF6987-M01', title: 'Finite attack topology', axes: ['attack count', 'SCC count', 'grounded size'],
      points: rows.map(row => [row.profile.attacks.length, row.af.sccs.length, row.af.labelMask(row.af.grounded).length]),
    },
    {
      id: 'AF6987-M02', title: 'Finite semantic family counts', axes: ['admissible count', 'preferred count', 'stable count'],
      points: rows.map(row => [row.af.admissibleSets.length, row.af.preferredSets.length, row.af.stableSets.length]),
    },
    {
      id: 'AF6987-M03', title: 'Preferred acceptance envelope', axes: ['skeptical count', 'credulous count', 'declared minus credulous'],
      points: rows.map(row => [row.skeptical.length, row.credulous.length, 6 - row.credulous.length]),
    },
    {
      id: 'AF6987-M04', title: 'Finite edit sensitivity', axes: ['attack deletions changing grounded', 'reversals changing grounded', 'argument deletions'],
      points: rows.map(row => [row.deletion.filter(item => item.changed).length, row.reversal.filter(item => item.changed).length, row.argumentDeletion.length]),
    },
    {
      id: 'AF6987-M05', title: 'Bounded repair and gate view', axes: ['one-edit stable repairs', 'open evidence gates', 'exact authority gates'],
      points: rows.map(row => [row.repair.edits.length, 1, 1]),

  ].map(model => ({...model, points_count: model.points.length, physical: false, empirical_measurement: false, three_coordinate_representation: true, same_owner_only: true}));
  assert(models.length === 5 && models.every(model => model.points.length === 15 && model.points.every(point => point.length === 3)), 'model_shape');
  return models;
}

function runStage(stage) {
  assert(['x1', 'x2'].includes(stage), 'stage_name');
  const outputFile = path.join(ROOT, `${stage}.json`);
  assert(!fs.existsSync(outputFile), `${stage}_absent`);
  const plan = readJson(path.join(ROOT, 'plan.json'));
  assert(bindFile(path.join(ROOT, 'phase.txt')).sha256 === plan.engine.sha256, 'engine_fixed');
  assert(plan.contracts.length === 300 && plan.inherited.records.length === 300, 'plan_counts');
  const contracts = plan.contracts.filter(contract => contract.stage === stage);
  assert(contracts.length === 150, `${stage}_contracts`);
  const records = [];
  const negativeIds = [];
  for (const contract of contracts) {
    const {profile} = validateRequest(contract.request, plan);
    const result = expected(contract.operation, profile);
    const resultSha = shaValue(result);
    assert(resultSha === contract.expected_result_sha256, `${contract.proposal_id}:oracle`);
    assert(result.outcome === contract.expected_outcome, `${contract.proposal_id}:outcome`);
    const sourceDigest = shaValue(contract.request);
    const missing = {...contract.request}; delete missing.profile_id;
    const extra = {...contract.request, undeclared: true};
    const failedSubjects = [];
    for (const [index, subject] of [missing, extra].entries()) {
      let rejected = false;
      let error = null;
      try { validateRequest(subject, plan); } catch (caught) { rejected = true; error = caught.message; }
      assert(rejected && error === 'field_set', `${contract.proposal_id}:candidate:${index}`);
      const negativeId = contract.malformed_subjects[index].negative_id;
      negativeIds.push(negativeId);
      failedSubjects.push({negative_id: negativeId, observed_error: error, subject_result: 'fail', original_success_credit: 0, refusal_result: 'pass'});
    }
    const detached = JSON.parse(JSON.stringify(contract.request));
    const copyDigest = shaValue(detached);
    detached.local_only_mutation = true;
    const sourceAfter = shaValue(contract.request);
    assert(copyDigest === sourceDigest && sourceAfter === sourceDigest, `${contract.proposal_id}:cfr`);
    const rerun = expected(contract.operation, profile);
    assert(shaValue(rerun) === resultSha, `${contract.proposal_id}:determinism`);
    records.push({
      proposal_id: contract.proposal_id,
      operation: contract.operation,
      profile_id: contract.profile_id,
      outcome: result.outcome,
      result_sha256: resultSha,
      safe_witness_ids: [`${contract.proposal_id}-SAFE-A`, `${contract.proposal_id}-SAFE-B`],
      failed_subjects: failedSubjects,
      refusal_witness_ids: [`${contract.proposal_id}-REFUSE-A`, `${contract.proposal_id}-REFUSE-B`],
      cfr_witness_ids: [`${contract.proposal_id}-CFR-A`, `${contract.proposal_id}-CFR-B`],
      clean_fix_refine_ids: [`${contract.proposal_id}-CLEAN`, `${contract.proposal_id}-REFINE`],
      source_request_unchanged: sourceAfter === sourceDigest,
      external_credit: false,
    });
  }
  const stageOperations = plan.operations.filter(operation => operation.stage === stage);
  const skills = stageOperations.map(operation => {
    const contract = contracts.find(item => item.operation === operation.name);
    return {name: `auren-${operation.name.replaceAll('_', '-')}`, operation: operation.name, validation: 'pass', used: true, official_skill_package_validation_claimed: false};
  });
  const runners = plan.runners.filter(runner => runner.stage === stage).map((runner, index) => {
    const positive = runner.operations.every(operationName => {
      const contract = contracts.find(item => item.operation === operationName);
      const profile = plan.profiles.find(item => item.profile_id === contract.profile_id);
      return shaValue(expected(operationName, profile)) === contract.expected_result_sha256;
    });
    let rejected = false;
    try { validateRequest({op: runner.operations[0]}, plan); } catch (error) { rejected = error.message === 'field_set'; }
    assert(positive && rejected, `${runner.name}:smoke`);
    const negativeId = `AF6987-${stage}-RUN-N${String(index + 1).padStart(2, '0')}`;
    negativeIds.push(negativeId);
    return {name: runner.name, operations: runner.operations, positive_smoke: 'pass', negative_subject: 'fail', negative_id: negativeId, refusal_check: 'pass', original_success_credit: 0, used: true};
  });
  const tests = stage === 'x2' ? runTests(plan) : [];
  const models = stage === 'x2' ? buildModels(plan) : [];
  const outcomes = records.reduce((acc, record) => { acc[record.outcome] = (acc[record.outcome] || 0) + 1; return acc; }, {});
  const stageRecord = {
    schema: `ghc.family.auren-argumentation-${stage}.v14`,
    owner: 'Auren Lark',

    parent_expected: stage === 'x1' ? 'planning checkpoint' : 'frozen x1 checkpoint',

    portfolios: {
      contracts: records.length,
      safe_checks: records.length * 2,
      malformed_candidate_subjects: records.length * 2,
      refusal_checks: records.length * 2,
      copy_from_reference_checks: records.length * 2,
      clean_fix_refine_reviews: records.length * 2,
      filesystem_deletions: 0,
    },
    negative_ids: negativeIds,
    retained_negative_count: negativeIds.length,
    failed_subject_success_credit: 0,
    skills,
    runners,
    tests,
    models,
    checks: {
      contracts: records.length === 150,
      safe: records.length * 2 === 300,
      candidate: records.length * 2 === 300,
      refusal: records.length * 2 === 300,
      cfr: records.length * 2 === 300,
      clean_fix_refine: records.length * 2 === 300,
      skills: skills.length === 10,
      runners: runners.length === 5,
      tests: stage === 'x1' || tests.length === 15,
      models: stage === 'x1' || models.length === 5,
      no_replay: true,
    },
    exact_packets_executed: 0,
    blocked_packets_executed: 0,

function makeMethodFlow(plan, x1, x2, failures) {
  const methods = [];
  const witnesses = [];
  for (const operation of plan.operations) {
    const stageRecord = operation.stage === 'x1' ? x1 : x2;
    const rows = stageRecord.contracts.filter(contract => contract.operation === operation.name);
    const negatives = rows.flatMap(contract => contract.failed_subjects.map(subject => subject.negative_id));
    const methodId = `AF6987-METHOD-OP-${String(operation.index + 1).padStart(2, '0')}`;
    methods.push(methodRecord(methodId, operation.description, negatives));
    witnesses.push(witnessRecord(`${methodId}-PASS`, methodId, 'pass', rows.length * 8));
    witnesses.push(witnessRecord(`${methodId}-FAIL`, methodId, 'fail', rows.length * 2, negatives));
  }
  for (const [index, skill] of plan.skills.entries()) {
    const methodId = `AF6987-METHOD-SKILL-${String(index + 1).padStart(2, '0')}`;
    methods.push(methodRecord(methodId, `Validate and use logical owner-local skill module ${skill.name}.`));
    witnesses.push(witnessRecord(`${methodId}-PASS`, methodId, 'pass', 1));
  }
  for (const [index, runner] of plan.runners.entries()) {
    const stageRecord = runner.stage === 'x1' ? x1 : x2;
    const row = stageRecord.runners.find(item => item.name === runner.name);
    const methodId = `AF6987-METHOD-RUNNER-${String(index + 1).padStart(2, '0')}`;
    methods.push(methodRecord(methodId, `Run positive and malformed smokes for ${runner.name}.`, [row.negative_id]));
    witnesses.push(witnessRecord(`${methodId}-PASS`, methodId, 'pass', 2));
    witnesses.push(witnessRecord(`${methodId}-FAIL`, methodId, 'fail', 1, [row.negative_id]));
  }
  for (const [index, test] of x2.tests.entries()) {
    const methodId = `AF6987-METHOD-TEST-${String(index + 1).padStart(2, '0')}`;
    methods.push(methodRecord(methodId, test.claim));

function privacyHits(files) {
  const patterns = [
    ['raw_task_or_thread_id', /\b(?:task|thread)[_-]?id\s*[:=]\s*["']?[0-9a-f-]{16,}/ig],
    ['credential_material', /\b(?:api[_-]?key|access[_-]?token|private[_-]?key|password)\s*[:=]\s*["'][^"']+/ig],
    ['email_address', /\b[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}\b/ig],
    ['private_user_path', /[A-Z]:\\Users\\[^\\\s]+/ig],
    ['private_app_state', new RegExp([
      ['session', 'stream'].join(' '),
      ['private', 'callable', 'identifier'].join(' '),
      ['raw', 'transcript'].join(' '),
      ['screenshot', 'payload'].join(' '),
    ].join('|'), 'ig')],
  ];
  const hits = [];
  for (const file of files) {
    const text = fs.readFileSync(file, 'utf8');
    for (const [kind, regex] of patterns) {
      const found = text.match(regex) || [];
      for (const match of found) hits.push({file: path.relative(W, file).split(path.sep).join('/'), kind, digest: shaBytes(Buffer.from(match, 'utf8'))});
    }
  }
  return hits;
}

    upstream = gitText('rev-parse', '@{upstream}');
    tracking = gitText('rev-parse', `refs/remotes/origin/${BRANCH}`);
    const liveLines = gitText('ls-remote', '--heads', 'origin', `refs/heads/${BRANCH}`).split(/\r?\n/).filter(Boolean);
    live = liveLines.length === 1 ? liveLines[0].split(/\s+/)[0] : null;
    divergence = gitText('rev-list', '--left-right', '--count', `HEAD...refs/remotes/origin/${BRANCH}`);
    add('C04', 'upstream equality', upstream === head, upstream);
    add('C05', 'tracking equality', tracking === head, tracking);
    add('C06', 'fresh unique live equality', live === head, live);
    add('C07', 'typed zero divergence', /^0\s+0$/.test(divergence), divergence);
  }
  const commits = gitText('rev-list', '--reverse', `${SOURCE}..${head}`).split(/\r?\n/).filter(Boolean);
  const expectedCommitCount = preflight ? 3 : 4;
  add('C08', preflight ? 'three published predecessors before final commit' : 'four new commits', commits.length === expectedCommitCount, commits);
  let direct = commits.length === expectedCommitCount;
  let expectedParent = SOURCE;
  for (const commit of commits) {
    const parents = gitText('show', '-s', '--format=%P', commit).split(/\s+/).filter(Boolean);
    direct &&= parents.length === 1 && parents[0] === expectedParent;
    expectedParent = commit;
  }
  add('C09', 'direct single-parent lifecycle', direct);
  add('C10', 'zero merges', gitText('rev-list', '--merges', `${SOURCE}..${head}`) === '');
  const tracked = Number(gitText('ls-tree', '-r', '--name-only', head).split(/\r?\n/).filter(Boolean).length);
  add('C11', preflight ? 'projected tracked file ceiling exact' : 'tracked file ceiling exact', preflight ? tracked === 1999 && fs.existsSync(path.join(ROOT, 'final.json')) : tracked === 2000, tracked);
  const changed = gitText('diff', '--name-only', `${SOURCE}..${head}`).split(/\r?\n/).filter(Boolean);
  add('C12', preflight ? 'exact four published owner predecessors plus projected final' : 'exact five owner files', preflight ? changed.length === 4 && changed.every(item => item.startsWith(PREFIX)) && status === expectedPreflightStatus : changed.length === 5 && changed.every(item => item.startsWith(PREFIX)), changed);
  for (const entry of final.manifest.entries) {
    const file = path.join(W, ...entry.path.split('/'));
    const binding = bindFile(file);
    add(`M-${path.basename(entry.path)}`, `raw binding ${entry.path}`, binding.bytes === entry.bytes && binding.sha256 === entry.sha256, binding);
  }
  add('C13', 'manifest self exclusion', final.manifest.self_exclusions.length === 1 && final.manifest.self_exclusions[0] === `${PREFIX}final.json`);
  add('C14', 'strict JSON parse count', [plan, x1, x2, final].every(Boolean), 4);
  add('C15', 'proposal counts', plan.inherited.records.length === 300 && plan.contracts.length === 300);
  add('C16', 'outcome counts', JSON.stringify(plan.outcomes) === JSON.stringify({completed: 255, represented: 15, open_gap: 15, exact_gate: 15}), plan.outcomes);
  for (const [stageName, stage] of [['x1', x1], ['x2', x2]]) {
    add(`${stageName}-01`, `${stageName} contracts`, stage.contracts.length === 150);
    add(`${stageName}-02`, `${stageName} safe`, stage.portfolios.safe_checks === 300);
    add(`${stageName}-03`, `${stageName} candidate`, stage.portfolios.malformed_candidate_subjects === 300);
    add(`${stageName}-04`, `${stageName} refusal`, stage.portfolios.refusal_checks === 300);
    add(`${stageName}-05`, `${stageName} cfr`, stage.portfolios.copy_from_reference_checks === 300);
    add(`${stageName}-06`, `${stageName} cleanup`, stage.portfolios.clean_fix_refine_reviews === 300);
    add(`${stageName}-07`, `${stageName} skills`, stage.skills.length === 10);
    add(`${stageName}-08`, `${stageName} runners`, stage.runners.length === 5);
    add(`${stageName}-09`, `${stageName} latch`, stage.session.invocations === 1 && stage.session.successes === 1 && stage.session.replays === 0);
  }
  add('C17', 'additional tests', x2.tests.length === 15 && x2.tests.every(test => test.pass));
  add('C18', 'nonphysical models', x2.models.length === 5 && x2.models.every(model => model.points.length === 15 && model.physical === false));
  add('C19', 'held packets', plan.approval_packets.exact.length === 50 && plan.approval_packets.blocked.length === 30 && final.approval_packets.executed === 0);
  add('C20', 'logical skills and runners', plan.skills.length === 20 && plan.runners.length === 10);
  const flow = final.method_flow;
  const passing = flow.witnesses.filter(witness => witness.result === 'pass').reduce((sum, witness) => sum + witness.atomic_count, 0);
  const failed = flow.witnesses.filter(witness => witness.result === 'fail').reduce((sum, witness) => sum + witness.atomic_count, 0);
  add('C21', 'method count', flow.methods.length === flow.counts.methods, {actual: flow.methods.length, declared: flow.counts.methods});
  add('C22', 'passing witness arithmetic', passing === flow.counts.passing_witnesses, {passing, declared: flow.counts.passing_witnesses});
  add('C23', 'failed witness arithmetic', failed === flow.counts.failed_witnesses, {failed, declared: flow.counts.failed_witnesses});
  add('C24', 'retained negative uniqueness', new Set(flow.retained_negative_ids).size === flow.retained_negative_ids.length && flow.retained_negative_ids.length === flow.counts.negatives);
  add('C25', 'effective arithmetic', final.effective.failed_witnesses + final.effective.passing_witnesses === final.effective.witnesses);
  add('C26', 'source fold once', final.source_baseline_fold_count === 1);
  add('C27', 'baton modules and length', final.handoff.modules.length === 13 && final.handoff.words >= 2000 && final.handoff.words <= 100000, final.handoff.words);
  add('C28', 'baton EOF', final.handoff.modules.at(-1).endsWith(final.handoff.literal_eof));
  add('C29', 'terminal verdict', final.terminal_verdict === 'NOT_READY_FOR_STAGE_20');
  const hits = privacyHits(files);
  add('C30', 'zero confirmed five-class privacy hits', hits.length === 0, hits);
  const sourceText = fs.readFileSync(path.join(ROOT, 'phase.txt'), 'utf8');
  const securityFindings = [];
  if (/shell\s*:\s*true/.test(sourceText)) securityFindings.push('shell_true');
  if (/\beval\s*\(/.test(sourceText)) securityFindings.push('eval');
  if (/https?:\/\//.test(sourceText) && !sourceText.includes('research_sources')) securityFindings.push('unexpected_network_literal');
  add('C31', 'zero bounded changed-source security findings', securityFindings.length === 0, securityFindings);
  add('C32', 'source canonical not replayed', plan.source.source_canonical_executed_by_auren === false && plan.source.canonical_replays === 0);
  add('C33', 'stage sessions not replayed', x1.session.replays === 0 && x2.session.replays === 0);
  add('C34', 'resource and format boundary', final.resource_closeout.tracked_after === 2000 && final.resource_closeout.global_installs === 0 && final.resource_closeout.shared_mutations === 0);
  const failures = checks.filter(check => !check.pass);
  return {head, upstream, tracking, live, divergence, tracked, changed, checks, failures, privacy_hits: hits, security_findings: securityFindings};
}

function canonicalRun(exactFinal) {
  ensureDir(path.join(BANK, 'canonical'));
  const latchFile = path.join(BANK, 'canonical', 'invocation-latch.json');
  const payloadFile = path.join(BANK, 'canonical', 'canonical-payload.json');
  const receiptFile = path.join(BANK, 'canonical', 'canonical-receipt.json');
  assert(!fs.existsSync(latchFile) && !fs.existsSync(payloadFile) && !fs.existsSync(receiptFile), 'canonical_exclusive_latch');
  writeExclusive(latchFile, {schema: 'ghc.family.canonical-invocation-latch.v14', owner: 'Auren Lark', phase: 'v698-v7', invoked_utc: new Date().toISOString(), successful_replays: 0});
  let validation;
  try {
    validation = validateRepository({remote: true, exactFinal});
  } catch (error) {
    writeExclusive(path.join(BANK, 'canonical', 'canonical-failure.json'), {schema: 'ghc.family.owner-canonical-failure.v14', error: error.message, successful_invocations: 0, successful_replays: 0, repository_mutated: false, retained: true});
    throw error;
  }
  if (validation.failures.length) {
    writeExclusive(path.join(BANK, 'canonical', 'canonical-failure.json'), {schema: 'ghc.family.owner-canonical-failure.v14', failures: validation.failures, successful_invocations: 0, successful_replays: 0, repository_mutated: false, retained: true});
    throw new Error(`canonical_failures:${JSON.stringify(validation.failures)}`);
  }
  const final = readJson(path.join(ROOT, 'final.json'));
  const payload = {
    schema: 'ghc.family.owner-canonical-payload.v14',
    state: 'VALID_EXACT_FINAL_OWNER_SCOPED_CANONICAL',
    owner: 'Auren Lark',
    phase: 'v698-v7',
    branch: BRANCH,
    source: SOURCE,
    head: validation.head,
    check_count: validation.checks.length,
    checks: validation.checks,
    owner_files: validation.changed.length,
    tracked_files: validation.tracked,
    strict_json_parses: 4,
    manifest_replays: final.manifest.entries.length,
    privacy_hits: validation.privacy_hits,
    security_findings: validation.security_findings,
    successful_invocations: 1,
    successful_replays: 0,
    source_canonical_replay: false,
    x1_session_replay: false,
    x2_session_replay: false,
    same_owner_only: true,
    independent_reproduction: false,
    full_repository_suite: false,
    repository_mutated: false,
    terminal_verdict: 'NOT_READY_FOR_STAGE_20',
    boundary: BOUNDARY,
  };
  const payloadBytes = canonical(payload);
  const payloadSha = shaBytes(payloadBytes);
  fs.writeFileSync(payloadFile, payloadBytes, {flag: 'wx'});
  const receipt = {schema: 'ghc.family.owner-canonical-receipt.v14', payload, payload_sha256: payloadSha, invoked_utc: new Date().toISOString(), repository_mutated: false};
  writeExclusive(receiptFile, receipt);
  console.log(JSON.stringify({state: payload.state, check_count: payload.check_count, payload_sha256: payloadSha, receipt_sha256: bindFileExternal(receiptFile).sha256, head: payload.head, tracked_files: payload.tracked_files}));
}

function bindFileExternal(file) {
  const bytes = fs.readFileSync(file);
  return {bytes: bytes.length, sha256: shaBytes(bytes)};
}

function validateCommand() {
  const finalExists = fs.existsSync(path.join(ROOT, 'final.json'));
  if (!finalExists) {
    const plan = readJson(path.join(ROOT, 'plan.json'));
    const output = {stage: 'plan', checks: planChecks(plan), engine_fixed: bindFile(path.join(ROOT, 'phase.txt')).sha256 === plan.engine.sha256};
    if (fs.existsSync(path.join(ROOT, 'x1.json'))) output.x1 = readJson(path.join(ROOT, 'x1.json')).checks;
    if (fs.existsSync(path.join(ROOT, 'x2.json'))) output.x2 = readJson(path.join(ROOT, 'x2.json')).checks;
    assert(output.engine_fixed && output.checks.every(check => check.pass), 'validation');
    console.log(JSON.stringify(output));
    return;
  }
  const validation = validateRepository({remote: false, preflight: true});
  if (validation.failures.length) throw new Error(`validation_failures:${JSON.stringify(validation.failures)}`);
  console.log(JSON.stringify({state: 'VALID_OWNER_WORKTREE_PREFLIGHT', checks: validation.checks.length, tracked: validation.tracked, changed: validation.changed.length, privacy_hits: validation.privacy_hits.length, security_findings: validation.security_findings.length}));
}

const command = process.argv.at(-1);
if (command === 'plan') makePlan();
else if (command === 'x1' || command === 'x2') runStage(command);
else if (command === 'final') makeFinal();
else if (command === 'validate') validateCommand();
else if (command && command.startsWith('canonical=')) canonicalRun(command.slice('canonical='.length));
else throw new Error(`usage: plan | x1 | x2 | final | validate | canonical=<exact-final>`);
