function stable(value) {
  if (Array.isArray(value)) return value.map(stable);
  if (value && typeof value === 'object') {
    const out = {};
    for (const key of Object.keys(value).sort()) out[key] = stable(value[key]);
    return out;
  }
  return value;
}

function canonical(value) {
  return Buffer.from(JSON.stringify(stable(value)), 'utf8');
}

function shaBytes(bytes) {
  return crypto.createHash('sha256').update(bytes).digest('hex');
}

function gcd(a, b) {
  a = a < 0n ? -a : a;
  b = b < 0n ? -b : b;
  while (b) [a, b] = [b, a % b];
  return a || 1n;
}

function frac(n, d = 1n) {
  n = BigInt(n);
  d = BigInt(d);
  assert(d !== 0n, 'zero_denominator');
  if (d < 0n) { n = -n; d = -d; }
  const g = gcd(n, d);
  return {n: n / g, d: d / g};
}

function parseFrac(value) {
  if (typeof value === 'object' && value && 'n' in value && 'd' in value) return frac(value.n, value.d);
  const [n, d = '1'] = String(value).split('/');
  return frac(BigInt(n), BigInt(d));
}

function fstr(value) {
  const f = parseFrac(value);
  return f.d === 1n ? String(f.n) : `${f.n}/${f.d}`;
}

function fadd(a, b) {
  a = parseFrac(a); b = parseFrac(b);
  return frac(a.n * b.d + b.n * a.d, a.d * b.d);
}

function fsub(a, b) {
  a = parseFrac(a); b = parseFrac(b);
  return frac(a.n * b.d - b.n * a.d, a.d * b.d);
}

function fmul(a, b) {
  a = parseFrac(a); b = parseFrac(b);
  return frac(a.n * b.n, a.d * b.d);
}

function fdiv(a, b) {
  a = parseFrac(a); b = parseFrac(b);
  return frac(a.n * b.d, a.d * b.n);
}

function feq(a, b) {
  a = parseFrac(a); b = parseFrac(b);
  return a.n === b.n && a.d === b.d;
}

function fle(a, b) {
  a = parseFrac(a); b = parseFrac(b);
  return a.n * b.d <= b.n * a.d;
}

function bitCount(mask) {
  let count = 0;
  while (mask) { count += mask & 1; mask >>>= 1; }
  return count;
}

function massEntries(mass) {
  return Object.entries(mass).map(([mask, value]) => [Number(mask), parseFrac(value)]).sort((a, b) => a[0] - b[0]);
}

function addMass(out, mask, value) {
  out[mask] = fstr(fadd(out[mask] || '0', value));
}

function massSum(mass) {
  return massEntries(mass).reduce((sum, [, value]) => fadd(sum, value), frac(0n));
}

function makeMass(seed, offset) {
  const codes = new Set([15]);
  for (let k = 0; codes.size < 6 && k < 40; k += 1) codes.add(((seed * (k + 3) + offset * 11 + k * 5) % 15) + 1);
  const ordered = [...codes].sort((a, b) => a - b);
  const weights = ordered.map((mask, index) => 1 + ((seed * (index + 2) + offset * 7 + mask * 3) % 11));
  const total = weights.reduce((sum, value) => sum + value, 0);
  return Object.fromEntries(ordered.map((mask, index) => [String(mask), fstr(frac(BigInt(weights[index]), BigInt(total)))]));
}

function belief(mass, query) {
  return massEntries(mass).reduce((sum, [mask, value]) => (mask & query) === mask ? fadd(sum, value) : sum, frac(0n));
}

function plausibility(mass, query) {
  return massEntries(mass).reduce((sum, [mask, value]) => (mask & query) !== 0 ? fadd(sum, value) : sum, frac(0n));
}

function commonality(mass, query) {
  return massEntries(mass).reduce((sum, [mask, value]) => (mask & query) === query ? fadd(sum, value) : sum, frac(0n));
}

function allBeliefs(mass) {
  const out = {'0': '0'};
  for (let mask = 1; mask <= 15; mask += 1) out[String(mask)] = fstr(belief(mass, mask));
  return out;
}

function massFromBeliefs(beliefs) {
  const out = {};
  for (let mask = 1; mask <= 15; mask += 1) {
    let value = parseFrac(beliefs[String(mask)]);
    for (let sub = 1; sub < mask; sub += 1) if ((sub & mask) === sub && out[String(sub)]) value = fsub(value, out[String(sub)]);
    if (!feq(value, 0)) out[String(mask)] = fstr(value);
  }
  return out;
}

function mapMask(mask, mapping) {
  let out = 0;
  for (let i = 0; i < mapping.length; i += 1) if (mask & (1 << i)) out |= 1 << mapping[i];
  return out;
}

function relabelMass(mass, mapping) {
  const out = {};
  for (const [mask, value] of massEntries(mass)) addMass(out, mapMask(mask, mapping), value);
  return out;
}

function vacuousExtend(mass) {
  const out = {};
  for (const [mask, value] of massEntries(mass)) {
    let expanded = 0;
    for (let i = 0; i < 4; i += 1) if (mask & (1 << i)) expanded |= (1 << (2 * i)) | (1 << (2 * i + 1));
    addMass(out, expanded, value);
  }
  return out;
}

function coarsenMass(mass) {
  const out = {};
  for (const [mask, value] of massEntries(mass)) {
    let coarse = 0;
    for (let i = 0; i < 4; i += 1) if (mask & (1 << i)) coarse |= 1 << (i % 2);
    addMass(out, coarse, value);
  }
  return out;
}

function conjunctive(a, b) {
  const out = {};
  for (const [ma, va] of massEntries(a)) for (const [mb, vb] of massEntries(b)) addMass(out, ma & mb, fmul(va, vb));
  return out;
}

function disjunctive(a, b) {
  const out = {};
  for (const [ma, va] of massEntries(a)) for (const [mb, vb] of massEntries(b)) addMass(out, ma | mb, fmul(va, vb));
  return out;
}

function normalizeConjunctive(raw) {
  const conflict = parseFrac(raw['0'] || '0');
  const denominator = fsub(1, conflict);
  assert(!feq(denominator, 0), 'total_conflict');
  const out = {};
  for (const [mask, value] of massEntries(raw)) if (mask !== 0) out[String(mask)] = fstr(fdiv(value, denominator));
  return {conflict: fstr(conflict), mass: out};
}

function pignistic(mass) {
  const out = {'0': '0', '1': '0', '2': '0', '3': '0'};
  for (const [mask, value] of massEntries(mass)) {
    assert(mask !== 0, 'open_world_mass_not_supported');
    const share = fdiv(value, bitCount(mask));
    for (let i = 0; i < 4; i += 1) if (mask & (1 << i)) out[String(i)] = fstr(fadd(out[String(i)], share));
  }
  return out;
}

  return {operation, profile};
}

function privacyHits(files) {
  const patterns = [
    ['raw_task_or_thread_id', /\b(?:task|thread)[_-]?id\s*[:=]\s*["']?[0-9a-f-]{16,}/ig],
    ['credential_material', /\b(?:api[_-]?key|access[_-]?token|private[_-]?key|password)\s*[:=]\s*["'][^"']+/ig],
    ['email_address', /\b[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}\b/ig],
    ['private_user_path', /[A-Z]:\\Users\\[^\\\s]+/ig],
    ['private_app_state', new RegExp([
      ['session', 'stream'].join(' '),
      ['private', 'callable', 'identifier'].join(' '),
      ['raw', 'transcript'].join(' '),
      ['screenshot', 'payload'].join(' '),
    ].join('|'), 'ig')],
  ];
  const hits = [];
  for (const file of files) {
    const text = fs.readFileSync(file, 'utf8');
    for (const [kind, regex] of patterns) for (const match of text.match(regex) || []) hits.push({file: relative(file), kind, digest: shaBytes(Buffer.from(match, 'utf8'))});
  }
  return hits;
}

    const output = {stage: 'plan', checks: planChecks(plan), engine_fixed: bindFile(path.join(ROOT, 'phase.txt')).sha256 === plan.engine.sha256};
    if (fs.existsSync(path.join(ROOT, 'x1.json'))) output.x1 = readJson(path.join(ROOT, 'x1.json')).checks;
    if (fs.existsSync(path.join(ROOT, 'x2.json'))) output.x2 = readJson(path.join(ROOT, 'x2.json')).checks;
    assert(output.engine_fixed && output.checks.every(check => check.pass) && (!output.x1 || Object.values(output.x1).every(Boolean)) && (!output.x2 || Object.values(output.x2).every(Boolean)), 'validation');
    console.log(JSON.stringify(output));
    return;
  }
  const validation = validateRepository({remote: false, preflight: true});
  if (validation.failures.length) throw new Error(`validation_failures:${JSON.stringify(validation.failures)}`);
  console.log(JSON.stringify({state: 'VALID_OWNER_WORKTREE_PREFLIGHT', checks: validation.checks.length, tracked: validation.tracked, changed: validation.changed.length, privacy_hits: validation.privacy_hits.length, security_findings: validation.security_findings.length}));
}

function canonicalRun(exactFinal) {
  ensureDir(path.join(BANK, 'canonical'));
  const latchFile = path.join(BANK, 'canonical', 'invocation-latch.json');
  const payloadFile = path.join(BANK, 'canonical', 'canonical-payload.json');
  const receiptFile = path.join(BANK, 'canonical', 'canonical-receipt.json');
  assert(!fs.existsSync(latchFile) && !fs.existsSync(payloadFile) && !fs.existsSync(receiptFile), 'canonical_exclusive_latch');
  writeExclusive(latchFile, {schema: 'ghc.family.canonical-invocation-latch.v15', owner: 'Auren Lark', phase: 'v701-v7', invoked_utc: new Date().toISOString(), successful_replays: 0});
  let validation;
  try { validation = validateRepository({remote: true, exactFinal}); }
  catch (error) {
    writeExclusive(path.join(BANK, 'canonical', 'canonical-failure.json'), {schema: 'ghc.family.owner-canonical-failure.v15', error: error.message, successful_invocations: 0, successful_replays: 0, repository_mutated: false, retained: true});
    throw error;
  }
  if (validation.failures.length) {
    writeExclusive(path.join(BANK, 'canonical', 'canonical-failure.json'), {schema: 'ghc.family.owner-canonical-failure.v15', failures: validation.failures, successful_invocations: 0, successful_replays: 0, repository_mutated: false, retained: true});
    throw new Error(`canonical_failures:${JSON.stringify(validation.failures)}`);
  }
  const final = readJson(path.join(ROOT, 'final.json'));
  const payload = {
    schema: 'ghc.family.owner-canonical-payload.v15', state: 'VALID_EXACT_FINAL_OWNER_SCOPED_CANONICAL', owner: 'Auren Lark', phase: 'v701-v7', branch: BRANCH, source: SOURCE,
    head: validation.head, check_count: validation.checks.length, checks: validation.checks, owner_files: validation.changed.length, tracked_files: validation.tracked,
    strict_json_parses: 4, manifest_replays: final.manifest.entries.length, privacy_hits: validation.privacy_hits, security_findings: validation.security_findings,
    successful_invocations: 1, successful_replays: 0, predecessor_canonical_replay: false, predecessor_composite_replay: false, x1_session_replay: false, x2_session_replay: false,
    same_owner_only: true, independent_reproduction: false, full_repository_suite: false, repository_mutated: false, terminal_verdict: 'NOT_READY_FOR_STAGE_20', boundary: BOUNDARY,
  };
  const payloadBytes = canonical(payload);
  const payloadSha = shaBytes(payloadBytes);
  fs.writeFileSync(payloadFile, payloadBytes, {flag: 'wx'});
  const receipt = {schema: 'ghc.family.owner-canonical-receipt.v15', payload, payload_sha256: payloadSha, invoked_utc: new Date().toISOString(), repository_mutated: false};
  writeExclusive(receiptFile, receipt);
  console.log(JSON.stringify({state: payload.state, check_count: payload.check_count, payload_sha256: payloadSha, receipt_sha256: bindExternal(receiptFile).sha256, head: payload.head, tracked_files: payload.tracked_files}));
}

const command = process.argv.at(-1);
if (command === 'plan') makePlan();
else if (command === 'x1' || command === 'x2') runStage(command);
else if (command === 'final') makeFinal();
else if (command === 'validate') validateCommand();
else if (command && command.startsWith('canonical=')) canonicalRun(command.slice('canonical='.length));
else throw new Error('usage: plan | x1 | x2 | final | validate | canonical=<exact-final>');
