"""Bounded explicit operation dispatcher for one lattice capability group."""
import json,pathlib,runpy,sys
BASE=pathlib.Path(__file__).resolve().parent.parent/'runtime'
K2=runpy.run_path(str(BASE/'lattice-extensions.txt'));K1=K2['K1']
ALLOWED=['translation-transport', 'trinity-geometry-bridge', 'lattice-observation-gap', 'lattice-authority-reservation']
def main():
 try:
  request=K1['strict'](sys.stdin.read(262145));op=request.get('op') if type(request) is dict else None
  if op not in ALLOWED:out=K1['envelope'](False,error='runner_scope')
  elif op in K1['OPERATIONS']:out=K1['evaluate'](request,ALLOWED)
  else:out=K2['evaluate'](request,ALLOWED)
 except (K1['Refusal'],ValueError,RecursionError) as exc:out=K1['envelope'](False,error=str(exc) if isinstance(exc,K1['Refusal']) else 'invalid_json')
 print(json.dumps(out,sort_keys=True,allow_nan=False));return 0 if out['ok'] else 2
if __name__=='__main__':raise SystemExit(main())
