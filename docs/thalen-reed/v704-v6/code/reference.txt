'use strict';
// Planning-only internal reference: enumerate assignments of distinguishable supply
// units to demand-bin multisets; deduplicate the resulting integer tables.
// No import of runtime, no domain invocation in canonical or later sessions.
const ordered=xs=>xs.slice().sort((a,b)=>JSON.stringify(a).localeCompare(JSON.stringify(b),'en'));
function fraction(n,d=1){let a=Math.abs(n),b=d;while(b){const r=a%b;a=b;b=r;}return `${n/a}/${d/a}`;}
function enumerate(a,b){const origins=[];a.forEach((n,i)=>{for(let k=0;k<n;k++)origins.push(i);});const unique=new Map(),matrix=[[0,0,0],[0,0,0],[0,0,0]];function visit(k,left){if(k===origins.length){unique.set(JSON.stringify(matrix),matrix.map(row=>row.slice()));return;}for(let j=0;j<3;j++)if(left[j]>0){left[j]--;matrix[origins[k]][j]++;visit(k+1,left);matrix[origins[k]][j]--;left[j]++;}}visit(0,b.slice());return ordered([...unique.values()]);}
function compute(a,b,c){const all=enumerate(a,b),cost=all.map(t=>{let v=0;for(let i=0;i<3;i++)for(let j=0;j<3;j++)v+=c[i][j]*t[i][j];return v;}),lo=cost.reduce((x,y)=>x<y?x:y),hi=cost.reduce((x,y)=>x>y?x:y);return {all,cost,lo,hi,optimal:all.filter((_,i)=>cost[i]===lo)};}
function oracle(r){const a=r.supply,b=r.demand,c=r.cost,M=a[0]+a[1]+a[2],t=compute(a,b,c),Q=n=>fraction(n,M);let data;
 switch(r.op){
 case 'transport_record':data={bins:3,mass:M,lattice_denominator:M,couplings:t.all.length};break;
 case 'coupling_count':data={count:t.all.length};break;
 case 'cost_extrema':data={minimum:Q(t.lo),maximum:Q(t.hi)};break;
 case 'optimal_couplings':data={value:Q(t.lo),integer_tables:t.optimal,denominator:M};break;
 case 'independent_coupling':{let num=0,lattice=true;for(let i=0;i<3;i++)for(let j=0;j<3;j++){num+=a[i]*b[j]*c[i][j];if((a[i]*b[j])%M)lattice=false;}data={matrix:a.map(n=>b.map(m=>fraction(n*m,M*M))),expected_cost:fraction(num,M*M),integer_lattice_member:lattice};break;}
 case 'minimum_disagreement':{let overlap=0;for(let i=0;i<3;i++)overlap+=Math.min(a[i],b[i]);data={value:Q(M-overlap),total_variation:Q(M-overlap)};break;}
 case 'line_distance':data={value:Q(Math.abs(a[0]-b[0])+Math.abs(a[0]+a[1]-b[0]-b[1])),cost:'absolute labelled index distance'};break;
 case 'cell_bounds':data=a.map(n=>b.map(m=>({minimum:Q(Math.max(0,n+m-M)),maximum:Q(Math.min(n,m))})));break;
 case 'optimal_support':{const possible=[[false,false,false],[false,false,false],[false,false,false]],forced=[[true,true,true],[true,true,true],[true,true,true]];for(const x of t.optimal)for(let i=0;i<3;i++)for(let j=0;j<3;j++){possible[i][j]||=x[i][j]>0;forced[i][j]&&=x[i][j]>0;}data={possible,forced};break;}
 case 'cost_histogram':data=[...new Set(t.cost)].sort((x,y)=>x-y).map(v=>({cost:Q(v),count:t.cost.filter(x=>x===v).length}));break;
 case 'dual_certificate':data={value:Q(t.lo)};break;
 case 'slackness_certificate':data={value:Q(t.lo),all_optimal_positive_cells_tight:true};break;
 case 'cost_shift':{let offset=0;for(let i=0;i<3;i++)offset+=i*a[i]+(2-i)*b[i];data={original:Q(t.lo),shifted:Q(t.lo+offset),offset:Q(offset),same_optimal_tables:true};break;}
 case 'transpose_covariance':data={original:Q(t.lo),transposed:Q(t.lo),same_optimal_tables:true};break;
 case 'relabel_covariance':data={value:Q(t.lo),same_optimal_tables:true,row_permutation:[2,0,1],column_permutation:[1,2,0]};break;
 case 'coupling_mixture':{const first=t.all[0],last=t.all[t.all.length-1];let value=0;const matrix=[];for(let i=0;i<3;i++){matrix.push([]);for(let j=0;j<3;j++){const n=first[i][j]+last[i][j];matrix[i].push(fraction(n,2*M));value+=n*c[i][j];}}data={matrix,expected_cost:fraction(value,2*M),first_integer_table:first,last_integer_table:last,physical_interpretation:false};break;}
 case 'accessible_summary':data={mass_units:M,feasible_lattice_tables:t.all.length,optimal_lattice_tables:t.optimal.length,minimum_cost:Q(t.lo),maximum_cost:Q(t.hi),statement:'Finite declared transport table; no real allocation.'};break;
 case 'cell_cost_sensitivity':{const changed=t.cost.map((n,i)=>n+t.all[i][0][0]),best=Math.min(...changed);data={original:Q(t.lo),perturbed:Q(best),increase:Q(best-t.lo),upper_bound:Q(Math.min(...t.optimal.map(x=>x[0][0]))),new_optimal_count:changed.filter(n=>n===best).length};break;}
 case 'external_evidence_gap':data={mass_units:M,feasible_lattice_tables:t.all.length,missing:['independent implementation','empirical allocation data and justified costs'],external_credit:0};break;
 case 'allocation_authority_gate':data={mass_units:M,minimum_synthetic_cost:Q(t.lo),gate:'Real allocation, people, resources, identity and competent authority remain outside scope',action_performed:false};break;
 default:throw Error('undefined oracle operation');
 }
 return {outcome:r.op==='coupling_mixture'?'represented':r.op==='external_evidence_gap'?'open_gap':r.op==='allocation_authority_gate'?'exact_gate':'completed',data};
}
module.exports={oracle,enumerate};
