'use strict';
module.exports=function(core){const operations=['ig_sum_nimber','ig_product','ig_sum_outcome','ig_sum_moves','ig_sum_certificate','ig_nim_equivalence','ig_path_counts','ig_trace','ig_evidence_gap','ig_authority_gate'];
function product(a,b){const nodes=[];for(let left=0;left<a.length;left++)for(let right=0;right<b.length;right++)nodes.push({left,right});const moves=nodes.map(({left,right})=>[...a[left].map(u=>({component:0,target:u*b.length+right})),...b[right].map(v=>({component:1,target:left*b.length+v}))]);const graph=moves.map(r=>r.map(x=>x.target));return{nodes,moves,analysis:core.analyze(graph)};}
function evaluate(r){const op=typeof r?.op==='string'?r.op:'unknown';if(!r||typeof r!=='object'||Array.isArray(r)||Object.keys(r).sort().join('|')!=='input|op')return{ok:false,op,error:'request_fields',external_credit:false};const x=r.input,error=core.validate(x);if(error)return{ok:false,op,error,external_credit:false};if(!operations.includes(op))return{ok:false,op,error:'unknown_operation',external_credit:false};const a=x.moves,b=x.other_moves,A=core.analyze(a),B=core.analyze(b),values=A.values;let data;
if(op.startsWith('ig_sum')||op==='ig_product'){const p=product(a,b),idx=x.start*b.length+x.other_start,pv=p.analysis.values;
if(op==='ig_sum_nimber')data={left_value:values[x.start],right_value:B.values[x.other_start],product_value:pv[idx],xor_value:values[x.start]^B.values[x.other_start],equal:pv[idx]===(values[x.start]^B.values[x.other_start])};
if(op==='ig_product')data={nodes:p.nodes,moves:p.moves.map((options,position)=>({position,options})),start:idx};
if(op==='ig_sum_outcome')data={start:[x.start,x.other_start],outcome:p.analysis.wins[idx]?'N':'P',mover_wins:p.analysis.wins[idx]};
if(op==='ig_sum_moves')data={start:[x.start,x.other_start],moves:p.moves[idx].filter(o=>!p.analysis.wins[o.target]).map(o=>({...o,target_coordinates:[p.nodes[o.target].left,p.nodes[o.target].right]}))};
if(op==='ig_sum_certificate'){const positions=p.nodes.map(({left,right},i)=>({left,right,grundy:pv[i],component_xor:values[left]^B.values[right],equal:pv[i]===(values[left]^B.values[right]),option_values:p.moves[i].map(o=>pv[o.target]),mex:core.mex(p.moves[i].map(o=>pv[o.target]))}));data={positions,all_equal:positions.every(z=>z.equal&&z.grundy===z.mex)};}
}
if(op==='ig_nim_equivalence')data={start:x.start,grundy:values[x.start],contexts:Array.from({length:x.heap+1},(_,heap)=>{const h=Array.from({length:heap+1},(_,i)=>Array.from({length:i},(_,j)=>j)),p=product(a,h),mover=p.analysis.wins[x.start*h.length+heap];return{heap,mover_wins:mover,losing_iff_equal:!mover===(values[x.start]===heap)};}),bounded_context_only:true};
if(op==='ig_path_counts'){const counts=a.map(()=>Array(a.length).fill(0));for(const v of A.order.slice().reverse())if(a[v].length===0)counts[v][0]=1;else for(const u of a[v])for(let k=0;k<a.length-1;k++)counts[v][k+1]+=counts[u][k];const h=counts[x.start];data={start:x.start,horizon:x.horizon,terminal_counts:Array.from({length:x.horizon+1},(_,k)=>h[k]||0),beyond_horizon:h.slice(x.horizon+1).reduce((s,v)=>s+v,0),total_complete_plays:h.reduce((s,v)=>s+v,0)};}
if(op==='ig_trace'){let v=x.start,trace=[v];while(a[v].length){const options=a[v].filter(u=>!A.wins[u]);v=(options.length?options:a[v])[0];trace.push(v);}data={trace,moves:trace.length-1,winner:(trace.length-1)%2?'first':'second',model_agents_only:true,real_agency:false};}
if(op==='ig_evidence_gap')data={claim:x.claim,required:core.claims[x.claim],provided:x.provided,missing:core.claims[x.claim].filter(v=>!x.provided.includes(v)),externally_verified:false,claim_established:false,state:'open_gap'};
if(op==='ig_authority_gate')data={label:x.authority,authority_granted:false,rights_allocated:false,production:false,stage20:false,state:'exact_gate'};
return{ok:true,op,data,external_credit:false};}
return{operations,evaluate,product};};
