'use strict';
const operations=['ig_profile','ig_reachable','ig_terminals','ig_topology','ig_rank','ig_nimbers','ig_options','ig_outcomes','ig_winning_moves','ig_policy'];
const fields=['profile_id','moves','other_moves','start','other_start','heap','horizon','play','claim','provided','authority'];
const claims={empirical_gmut:['real_observations','typed_physical_mapping','independent_review'],production_thos:['real_operators','safety_monitoring','governed_trial','independent_review'],freed_id:['real_keys','interoperability','privacy_review','recovery_governance'],accessibility:['assistive_technology','affected_user_review','manual_evaluation']};
const roles=[...new Set(Object.values(claims).flat())];
function topology(g){const degree=g.map(()=>0);g.forEach(r=>r.forEach(v=>degree[v]++));const ready=degree.map((v,i)=>v===0?i:null).filter(v=>v!==null),out=[];while(ready.length){ready.sort((u,v)=>u-v);const v=ready.shift();out.push(v);for(const u of g[v])if(--degree[u]===0)ready.push(u);}return out.length===g.length?out:null;}
function validate(x){if(!x||typeof x!=='object'||Array.isArray(x)||Object.keys(x).sort().join('|')!==fields.slice().sort().join('|'))return'closed_fields';if(typeof x.profile_id!=='string'||!/^synthetic-game-\d+$/.test(x.profile_id))return'profile_id';if(x.play!=='normal')return'normal_play_only';for(const key of ['moves','other_moves']){const g=x[key];if(!Array.isArray(g)||g.length<1||g.length>8)return'graph_bounds';for(const row of g)if(!Array.isArray(row)||row.some(v=>!Number.isInteger(v)||v<0||v>=g.length)||new Set(row).size!==row.length||row.some((v,i)=>i&&v<=row[i-1]))return'move_shape';if(topology(g)===null)return'cyclic_game';}if(!Number.isInteger(x.start)||x.start<0||x.start>=x.moves.length||!Number.isInteger(x.other_start)||x.other_start<0||x.other_start>=x.other_moves.length)return'start_bounds';if(!Number.isInteger(x.heap)||x.heap<0||x.heap>7)return'heap_bounds';if(!Number.isInteger(x.horizon)||x.horizon<0||x.horizon>6)return'horizon_bounds';if(!Object.hasOwn(claims,x.claim)||!Array.isArray(x.provided)||new Set(x.provided).size!==x.provided.length||x.provided.some(v=>!roles.includes(v)))return'evidence_shape';if(!['none','claimed_reviewed','synthetic_only','unreviewed'].includes(x.authority))return'authority_shape';return null;}
function mex(values){const seen=new Uint8Array(values.length+1);for(const k of values)if(k<=values.length)seen[k]=1;for(let k=0;k<seen.length;k++)if(!seen[k])return k;throw Error('mex invariant');}
function analyze(g){const order=topology(g);if(order===null)throw Error('acyclic analysis required');const values=g.map(()=>0),ranks=g.map(()=>0),wins=g.map(()=>false);for(const v of order.slice().reverse()){values[v]=mex(g[v].map(u=>values[u]));ranks[v]=g[v].length?1+Math.max(...g[v].map(u=>ranks[u])):0;wins[v]=g[v].some(u=>!wins[u]);}return{order,values,ranks,wins};}
function evaluate(r){const op=typeof r?.op==='string'?r.op:'unknown';if(!r||typeof r!=='object'||Array.isArray(r)||Object.keys(r).sort().join('|')!=='input|op')return{ok:false,op,error:'request_fields',external_credit:false};const x=r.input,error=validate(x);if(error)return{ok:false,op,error,external_credit:false};if(!operations.includes(op))return{ok:false,op,error:'unknown_operation',external_credit:false};const a=x.moves,n=a.length,A=analyze(a),values=A.values;let data;
if(op==='ig_profile')data={positions:n,other_positions:x.other_moves.length,move_edges:a.reduce((z,r)=>z+r.length,0),play:x.play,start:x.start,finite_acyclic:true};
if(op==='ig_reachable'){const seen=new Uint8Array(n),queue=[x.start];seen[x.start]=1;for(let i=0;i<queue.length;i++)for(const u of a[queue[i]])if(!seen[u]){seen[u]=1;queue.push(u);}data={start:x.start,positions:Array.from(seen).flatMap((v,i)=>v?[i]:[])};}
if(op==='ig_terminals')data={positions:a.flatMap((r,i)=>r.length?[]:[i]),terminal_mover_wins:false};
if(op==='ig_topology')data={order:A.order};
if(op==='ig_rank')data={rank:A.ranks,strict_on_all_moves:a.every((r,v)=>r.every(u=>A.ranks[v]>A.ranks[u]))};
if(op==='ig_nimbers')data={values,start_value:values[x.start]};
if(op==='ig_options')data={positions:a.map((r,v)=>({position:v,option_positions:r.slice(),option_values:r.map(u=>values[u]),distinct_values:[...new Set(r.map(u=>values[u]))].sort((u,v)=>u-v),mex:values[v],lower_values:Array.from({length:values[v]},(_,k)=>k),same_value_option:r.some(u=>values[u]===values[v])}))};
if(op==='ig_outcomes')data={outcomes:A.wins.map(v=>v?'N':'P'),start_outcome:A.wins[x.start]?'N':'P'};
if(op==='ig_winning_moves')data={start:x.start,moves:a[x.start].filter(u=>!A.wins[u]),mover_wins:A.wins[x.start]};
if(op==='ig_policy')data={positions:a.map((r,v)=>{const choices=r.filter(u=>!A.wins[u]);return{position:v,winning_options:choices,chosen:choices.length?choices[0]:null,state:r.length===0?'terminal':choices.length?'winning':'losing'};})};
return{ok:true,op,data,external_credit:false};}
module.exports={operations,claims,roles,validate,topology,mex,analyze,evaluate};
