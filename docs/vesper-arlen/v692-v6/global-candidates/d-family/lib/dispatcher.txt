'use strict';
require.extensions['.txt'] = require.extensions['.js'];
const fs = require('fs');
const core = require('./simplicial-core.txt');
const extended = require('./simplicial-extended.txt');
const request = JSON.parse(fs.readFileSync(0, 'utf8'));
const first = core.safeExecute(request);
const output = first.error && first.error.code === 'unknown_operation' ? extended.safeExecute(request) : first;
process.stdout.write(JSON.stringify(output));
