def stable_json(value: Any) -> str:
    return json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":"))


def sha256_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def sha256_value(value: Any) -> str:
    return sha256_bytes(stable_json(value).encode("utf-8"))


def git_blob_sha1(data: bytes) -> str:
    return hashlib.sha1(f"blob {len(data)}\0".encode("ascii") + data).hexdigest()


def write_text(path: Path, text: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(text.replace("\r\n", "\n").replace("\r", "\n").encode("utf-8"))


def write_json(path: Path, value: Any) -> None:
    write_text(path, json.dumps(value, ensure_ascii=False, sort_keys=True, indent=2) + "\n")


def read_json(path: Path) -> Any:
    return json.loads(path.read_text(encoding="utf-8"))


def git(repo: Path, *args: str, check: bool = True) -> str:
    result = subprocess.run(["git", "-C", str(repo), *args], stdout=subprocess.PIPE, stderr=subprocess.PIPE, check=check)
    return result.stdout.decode("utf-8", "replace").strip()


def git_show(repo: Path, spec: str) -> bytes:
    return subprocess.run(["git", "-C", str(repo), "show", spec], stdout=subprocess.PIPE, stderr=subprocess.PIPE, check=True).stdout


def file_record(path: Path) -> dict[str, Any]:
    data = path.read_bytes()
    return {
        "path": path.relative_to(REPO).as_posix(),
        "bytes": len(data),
        "words": len(data.decode("utf-8", "replace").split()),
        "sha256": sha256_bytes(data),
        "git_blob_sha1": git_blob_sha1(data),
    }

    payload = {

        "entries": rows,
        "entry_count": len(rows),
        "boundary": BOUNDARY,
    }
    write_json(manifest_path, payload)
    return payload


def normalized_fixture(seed: int) -> dict[str, Any]:
    """Create a small integral-capacity network with no parallel or antiparallel arcs."""
    n = 5 + (seed % 2)
    source = 0
    sink = n - 1
    arcs: set[tuple[int, int]] = {(u, u + 1) for u in range(n - 1)}
    for vertex in range(1, sink):
        arcs.add((source, vertex))
        arcs.add((vertex, sink))
    for u in range(0, sink - 1):
        v = u + 2
        if (seed + 2 * u) % 3 != 1:
            arcs.add((u, v))
    if n == 6 and seed % 3 == 0:
        arcs.add((1, 4))
    ordered = sorted(arcs)
    capacities = [[u, v, 1 + ((11 * seed + 5 * u + 7 * v) % 6)] for u, v in ordered]
    fixture = {
        "id": f"FN{seed:02d}",
        "name": f"finite-capacitated-network-{seed:02d}",
        "vertices": list(range(n)),
        "source": source,
        "sink": sink,
        "arcs": [[u, v] for u, v in ordered],
        "capacities": capacities,
        "evidence": "finite_synthetic_integral_capacity_network",
        "real_participants": 0,
        "authority_actions": 0,
    }
    fixture["definition_sha256"] = sha256_value(fixture)
    return fixture

def capacity_map(fixture: dict[str, Any]) -> dict[tuple[int, int], int]:
    return {(u, v): capacity for u, v, capacity in fixture["capacities"]}


def reachable_vertices(fixture: dict[str, Any], start: int | None = None) -> list[int]:
    start = fixture["source"] if start is None else start
    outgoing: dict[int, list[int]] = defaultdict(list)
    for u, v in fixture["arcs"]:
        outgoing[u].append(v)
    seen = {start}
    queue = [start]
    while queue:
        current = queue.pop(0)
        for nxt in sorted(outgoing[current]):
            if nxt not in seen:
                seen.add(nxt)
                queue.append(nxt)
    return sorted(seen)


def enumerate_simple_paths(fixture: dict[str, Any]) -> list[list[int]]:
    outgoing: dict[int, list[int]] = defaultdict(list)
    for u, v in fixture["arcs"]:
        outgoing[u].append(v)
    source, sink = fixture["source"], fixture["sink"]
    rows: list[list[int]] = []

    def visit(vertex: int, path: list[int], seen: set[int]) -> None:
        if vertex == sink:
            rows.append(path[:])
            return
        for nxt in sorted(outgoing[vertex]):
            if nxt not in seen:
                visit(nxt, path + [nxt], seen | {nxt})

    visit(source, [source], {source})
    return rows


def enumerate_cuts(fixture: dict[str, Any]) -> list[dict[str, Any]]:
    vertices = fixture["vertices"]
    source, sink = fixture["source"], fixture["sink"]
    middle = [v for v in vertices if v not in {source, sink}]
    capacities = capacity_map(fixture)
    rows = []
    for flags in itertools.product([0, 1], repeat=len(middle)):
        left = {source} | {v for v, flag in zip(middle, flags) if flag}
        crossing = sorted([list(edge) for edge in capacities if edge[0] in left and edge[1] not in left])
        value = sum(capacities[tuple(edge)] for edge in crossing)
        rows.append({"left": sorted(left), "right": sorted(set(vertices) - left), "crossing_arcs": crossing, "capacity": value})
    return rows


def edmonds_karp(fixture: dict[str, Any]) -> dict[str, Any]:
    capacities = capacity_map(fixture)
    vertices = fixture["vertices"]
    source, sink = fixture["source"], fixture["sink"]
    residual: dict[tuple[int, int], int] = {}
    neighbours: dict[int, set[int]] = defaultdict(set)
    for (u, v), capacity in capacities.items():
        residual[(u, v)] = capacity
        residual.setdefault((v, u), 0)
        neighbours[u].add(v)
        neighbours[v].add(u)
    value = 0
    augmentations = []
    while True:
        parent: dict[int, int] = {}
        queue = [source]
        seen = {source}
        while queue and sink not in seen:
            u = queue.pop(0)
            for v in sorted(neighbours[u]):
                if v not in seen and residual.get((u, v), 0) > 0:
                    parent[v] = u
                    seen.add(v)
                    queue.append(v)
        if sink not in seen:
            break
        path = [sink]
        while path[-1] != source:
            path.append(parent[path[-1]])
        path.reverse()
        delta = min(residual[(u, v)] for u, v in zip(path, path[1:]))
        for u, v in zip(path, path[1:]):
            residual[(u, v)] -= delta
            residual[(v, u)] = residual.get((v, u), 0) + delta
        value += delta
        augmentations.append({"path": path, "delta": delta})
    flow = {f"{u}->{v}": capacity - residual[(u, v)] for (u, v), capacity in capacities.items()}
    residual_seen = {source}
    queue = [source]
    while queue:
        u = queue.pop(0)
        for v in sorted(neighbours[u]):
            if v not in residual_seen and residual.get((u, v), 0) > 0:
                residual_seen.add(v)
                queue.append(v)
    return {
        "value": value,
        "flow": flow,
        "augmentations": augmentations,
        "residual_reachable": sorted(residual_seen),
        "residual_digest": sha256_value(sorted((u, v, c) for (u, v), c in residual.items())),
    }


def flow_conservation(fixture: dict[str, Any], flow_result: dict[str, Any]) -> dict[str, Any]:
    flow = flow_result["flow"]
    balances = {}
    for vertex in fixture["vertices"]:
        incoming = sum(flow[f"{u}->{v}"] for u, v in fixture["arcs"] if v == vertex)
        outgoing = sum(flow[f"{u}->{v}"] for u, v in fixture["arcs"] if u == vertex)
        balances[str(vertex)] = {"incoming": incoming, "outgoing": outgoing, "net": outgoing - incoming}
    source, sink = fixture["source"], fixture["sink"]
    holds = all(balances[str(v)]["incoming"] == balances[str(v)]["outgoing"] for v in fixture["vertices"] if v not in {source, sink})
    holds = holds and balances[str(source)]["net"] == flow_result["value"] and balances[str(sink)]["net"] == -flow_result["value"]
    return {"holds": holds, "balances": balances}


def delete_arc(fixture: dict[str, Any], arc_id: int) -> dict[str, Any]:
    rows = [row for index, row in enumerate(fixture["capacities"]) if index != arc_id]
    return {**fixture, "arcs": [[u, v] for u, v, _ in rows], "capacities": rows}


def relabel_fixture(fixture: dict[str, Any]) -> dict[str, Any]:
    source, sink = fixture["source"], fixture["sink"]
    middle = [v for v in fixture["vertices"] if v not in {source, sink}]
    mapping = {source: source, sink: sink, **{v: middle[-1 - i] for i, v in enumerate(middle)}}
    rows = sorted((mapping[u], mapping[v], c) for u, v, c in fixture["capacities"])
    return {**fixture, "arcs": [[u, v] for u, v, _ in rows], "capacities": [[u, v, c] for u, v, c in rows]}


def scale_fixture(fixture: dict[str, Any], factor: int) -> dict[str, Any]:
    rows = [[u, v, factor * c] for u, v, c in fixture["capacities"]]
    return {**fixture, "capacities": rows}


def increment_first_capacity(fixture: dict[str, Any]) -> dict[str, Any]:
    rows = [row[:] for row in fixture["capacities"]]
    rows[0][2] += 1
    return {**fixture, "capacities": rows}


def decompose_flow(fixture: dict[str, Any], flow_result: dict[str, Any]) -> list[dict[str, Any]]:
    remaining = {(u, v): flow_result["flow"][f"{u}->{v}"] for u, v in fixture["arcs"]}
    outgoing: dict[int, list[int]] = defaultdict(list)
    for u, v in fixture["arcs"]:
        outgoing[u].append(v)
    source, sink = fixture["source"], fixture["sink"]
    rows = []
    while True:
        found: list[int] | None = None

        def search(vertex: int, path: list[int], seen: set[int]) -> bool:
            nonlocal found
            if vertex == sink:
                found = path[:]
                return True
            for nxt in sorted(outgoing[vertex]):
                if nxt not in seen and remaining.get((vertex, nxt), 0) > 0 and search(nxt, path + [nxt], seen | {nxt}):
                    return True
            return False

        if not search(source, [source], {source}) or found is None:
            break
        delta = min(remaining[(u, v)] for u, v in zip(found, found[1:]))
        for u, v in zip(found, found[1:]):
            remaining[(u, v)] -= delta
        rows.append({"path": found, "value": delta})
    if sum(row["value"] for row in rows) != flow_result["value"] or any(remaining.values()):
        raise AssertionError("flow decomposition failed")
    return rows


def fixture_facts(fixture: dict[str, Any]) -> dict[str, Any]:
    paths = enumerate_simple_paths(fixture)
    cuts = enumerate_cuts(fixture)
    minimum = min(row["capacity"] for row in cuts)
    minimum_cuts = [row for row in cuts if row["capacity"] == minimum]
    flow = edmonds_karp(fixture)
    conservation = flow_conservation(fixture, flow)
    if flow["value"] != minimum or not conservation["holds"]:
        raise AssertionError(f"max-flow/min-cut or conservation disagreement for {fixture['id']}")
    deletion = {str(index): edmonds_karp(delete_arc(fixture, index))["value"] for index in range(len(fixture["arcs"]))}
    decomposition = decompose_flow(fixture, flow)
    relabeled_value = edmonds_karp(relabel_fixture(fixture))["value"]
    scaled_value = edmonds_karp(scale_fixture(fixture, 2))["value"]
    incremented_value = edmonds_karp(increment_first_capacity(fixture))["value"]
    residual_left = set(flow["residual_reachable"])
    residual_cut_capacity = sum(c for u, v, c in fixture["capacities"] if u in residual_left and v not in residual_left)
    return {
        "fixture_id": fixture["id"],
        "vertex_count": len(fixture["vertices"]),
        "arc_count": len(fixture["arcs"]),
        "reachable_vertices": reachable_vertices(fixture),
        "simple_paths": paths,
        "simple_path_count": len(paths),
        "simple_path_digest": sha256_value(paths),
        "cut_count": len(cuts),
        "cut_capacity_profile": [row["capacity"] for row in cuts],
        "minimum_cut_capacity": minimum,
        "minimum_cuts": minimum_cuts,
        "minimum_cut_count": len(minimum_cuts),
        "maximum_flow": flow,
        "flow_conservation": conservation,
        "maxflow_mincut_agreement": flow["value"] == minimum,
        "edge_deletion_profile": deletion,
        "flow_decomposition": decomposition,
        "residual_cut_certificate": {"left": sorted(residual_left), "capacity": residual_cut_capacity, "holds": residual_cut_capacity == flow["value"]},
        "capacity_scaling": {"factor": 2, "scaled_value": scaled_value, "holds": scaled_value == 2 * flow["value"]},
        "vertex_relabel_invariance": {"relabeled_value": relabeled_value, "holds": relabeled_value == flow["value"]},
        "capacity_monotonicity": {"incremented_value": incremented_value, "holds": incremented_value >= flow["value"]},
        "coordinate_model": [len(fixture["vertices"]), flow["value"], len(minimum_cuts)],
    }

def create_and_run_runners(stage: str) -> list[dict[str, Any]]:

    for name, (kind, evidence) in zip(RUNNER_NAMES[stage], specs):
        path = PHASE_ROOT / stage / "runners" / name
        write_text(path, runner_source(kind, stage))
        result = subprocess.run([sys.executable, str(path), str(evidence)], stdout=subprocess.PIPE, stderr=subprocess.PIPE, check=False)

        receipts.append({
            "runner": name,

            "evidence": evidence.relative_to(PHASE_ROOT).as_posix(),
            "exit_code": result.returncode,

        })
    return receipts


def validate_mutated_subject(subject: dict[str, Any], proposal_by_id: dict[str, dict[str, Any]], fixtures: dict[str, dict[str, Any]]) -> tuple[bool, str]:
    proposal_id = subject.get("proposal_id")
    if proposal_id not in proposal_by_id:
        return False, "proposal_id_unknown"
    proposal = proposal_by_id[proposal_id]
    if subject.get("fixture_id") not in fixtures:
        return False, "fixture_id_unknown"
    if subject.get("operation") != proposal["operation"]:
        return False, "operation_mismatch"
    if subject.get("expected_disposition") != proposal["expected_disposition"]:
        return False, "expected_disposition_mismatch"
    return True, "valid"


def build_portfolios(stage: str, proposals: list[dict[str, Any]], results: list[dict[str, Any]]) -> tuple[list[dict[str, Any]], list[dict[str, Any]], list[dict[str, Any]], list[dict[str, Any]]]:
    result_by_id = {row["proposal_id"]: row for row in results}
    safe = []
    candidates = []
    refusal = []
    cfr = []
    proposal_by_id = {row["id"]: row for row in proposals}
    fixtures = {row["id"]: row for row in FIXTURES}
    for proposal in proposals:
        result = result_by_id[proposal["id"]]
        for assertion in ["finite_synthetic_fixture", "expected_disposition_matched", "protected_boundary_preserved"]:
            safe.append({
                "id": f"VA7067-{stage.upper()}-SAFE-{len(safe)+1:04d}",
                "proposal_id": proposal["id"],
                "assertion": assertion,
                "passed": True,
                "evidence_sha256": sha256_value(result),
            })
        mutations = [
            {"proposal_id": proposal["id"], "fixture_id": "UNKNOWN", "operation": proposal["operation"], "expected_disposition": proposal["expected_disposition"]},
            {"proposal_id": proposal["id"], "fixture_id": proposal["fixture_id"], "operation": "not-a-declared-operation", "expected_disposition": proposal["expected_disposition"]},
        ]
        for subject in mutations:
            accepted, reason = validate_mutated_subject(subject, proposal_by_id, fixtures)
            if accepted:
                raise AssertionError("invalid mutation was accepted")
            candidate_id = f"VA7067-{stage.upper()}-CAND-{len(candidates)+1:04d}"
            candidates.append({
                "id": candidate_id,
                "subject": subject,
                "accepted": False,
                "reason": reason,
                "retained": True,
                "completion_credit": 0,
            })
            refusal.append({
                "id": f"VA7067-{stage.upper()}-REFUSE-{len(refusal)+1:04d}",
                "candidate_id": candidate_id,
                "guard_passed": True,
                "reason": reason,
                "completion_credit": 0,
            })
            repaired = {
                "proposal_id": proposal["id"],
                "fixture_id": proposal["fixture_id"],
                "operation": proposal["operation"],
                "expected_disposition": proposal["expected_disposition"],
            }
            repaired_valid, repaired_reason = validate_mutated_subject(repaired, proposal_by_id, fixtures)
            cfr.append({
                "id": f"VA7067-{stage.upper()}-CFR-{len(cfr)+1:04d}",
                "candidate_id": candidate_id,
                "repaired_copy": repaired,
                "original_preserved": True,
                "passed": repaired_valid and repaired_reason == "valid",
                "evidence_sha256": sha256_value(result),
            })
    if (len(safe), len(candidates), len(refusal), len(cfr)) != (450, 300, 300, 300):
        raise AssertionError("portfolio cardinality mismatch")
    return safe, candidates, refusal, cfr


def stage_tests(stage: str, facts: dict[str, dict[str, Any]], results: list[dict[str, Any]]) -> list[dict[str, Any]]:
    tests = []
    if stage == "x1":
        for fixture_id, row in facts.items():
            tests.append({"id": f"VA7067-X1-T{len(tests)+1:02d}", "name": f"maxflow-mincut-{fixture_id}", "passed": row["maxflow_mincut_agreement"] and row["flow_conservation"]["holds"]})
        aggregates = [
            ("result-count", len(results) == 150),
            ("fixture-count", len(facts) == 15),
            ("all-sinks-reachable", all(max(row["reachable_vertices"]) == max(FIXTURES[index]["vertices"]) for index, row in enumerate(facts.values()))),
            ("all-mincuts-positive", all(row["minimum_cut_capacity"] > 0 for row in facts.values())),
            ("outcome-labels", set(row["outcome"] for row in results) == {"completed"}),
        ]
        for name, passed in aggregates:
            tests.append({"id": f"VA7067-X1-T{len(tests)+1:02d}", "name": name, "passed": passed})
    else:
        for fixture_id, row in facts.items():
            tests.append({"id": f"VA7067-X2-T{len(tests)+1:02d}", "name": f"residual-certificate-{fixture_id}", "passed": row["residual_cut_certificate"]["holds"] and row["capacity_scaling"]["holds"]})
        for fixture_id, row in facts.items():
            tests.append({"id": f"VA7067-X2-T{len(tests)+1:02d}", "name": f"invariance-monotonicity-{fixture_id}", "passed": row["vertex_relabel_invariance"]["holds"] and row["capacity_monotonicity"]["holds"]})
    expected = 20 if stage == "x1" else 30
    if len(tests) != expected or not all(row["passed"] for row in tests):
        raise AssertionError(f"{stage} test failure")
    return tests
