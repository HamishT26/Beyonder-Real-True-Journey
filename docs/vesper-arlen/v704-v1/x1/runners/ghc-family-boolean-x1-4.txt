"use strict";
const fs=require("node:fs");
const path=require("node:path");
const cp=require("node:child_process");
const allowed=["bf_weight","bf_essential_witnesses"];
let request;
try{request=JSON.parse(fs.readFileSync(0,"utf8"));}catch(error){process.stdout.write(JSON.stringify({disposition:"refused",code:"INVALID_JSON"})+"\n");process.exit(2);}
if(!allowed.includes(request.operation)){process.stdout.write(JSON.stringify({disposition:"refused",code:"OPERATION_NOT_ADMITTED"})+"\n");process.exit(2);}
const query=path.join(__dirname,"..","..","..","..","..","tools","query-vesper-v704-v1.txt");
const child=cp.spawnSync(process.execPath,[query],{input:JSON.stringify(request),encoding:"utf8"});
process.stdout.write(child.stdout);
process.stderr.write(child.stderr);
process.exit(child.status===null?3:child.status);
