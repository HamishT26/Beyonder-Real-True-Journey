function ensureDir(p) { fs.mkdirSync(p, { recursive: true }); }
function stable(value) {
  if (Array.isArray(value)) return value.map(stable);
  if (value && typeof value === 'object') return Object.fromEntries(Object.keys(value).sort().map(k => [k, stable(value[k])]));
  return value;
}
function stableText(value) { return JSON.stringify(stable(value)); }
function pretty(value) { return `${JSON.stringify(value, null, 2)}\n`; }
function writeJson(rel, value) { const p = path.join(ROOT, rel); ensureDir(path.dirname(p)); fs.writeFileSync(p, pretty(value), 'utf8'); }
function writeText(rel, value) { const p = path.join(ROOT, rel); ensureDir(path.dirname(p)); fs.writeFileSync(p, value.endsWith('\n') ? value : `${value}\n`, 'utf8'); }
function readJson(rel) { return JSON.parse(fs.readFileSync(path.join(ROOT, rel), 'utf8')); }
function shaBytes(bytes) { return crypto.createHash('sha256').update(bytes).digest('hex'); }
function shaValue(value) { return shaBytes(Buffer.from(stableText(value), 'utf8')); }
function toRepo(abs) { return path.relative(REPO, abs).split(path.sep).join('/'); }
function git(args) { return cp.execFileSync('git', args, { cwd: REPO, encoding: 'utf8', maxBuffer: 64 * 1024 * 1024 }).trim(); }
function now() { return new Date().toISOString(); }
function clone(v) { return JSON.parse(JSON.stringify(v)); }
function exactKeys(obj, keys) { const a = Object.keys(obj || {}).sort(); const b = [...keys].sort(); return a.length === b.length && a.every((x, i) => x === b[i]); }

function listFiles(dir) {
  if (!fs.existsSync(dir)) return [];
  const out = [];
  for (const entry of fs.readdirSync(dir, { withFileTypes: true }).sort((a, b) => a.name.localeCompare(b.name))) {
    const p = path.join(dir, entry.name);
    if (entry.isDirectory()) out.push(...listFiles(p));
    else out.push(p);
  }
  return out;
}

function manifestEntries(dirs, exclusions = []) {
  const excluded = new Set(exclusions.map(p => path.resolve(ROOT, p)));
  const files = dirs.flatMap(d => listFiles(path.join(ROOT, d))).filter(p => !excluded.has(path.resolve(p))).sort();
  return files.map(p => { const b = fs.readFileSync(p); return { path: toRepo(p), bytes: b.length, sha256: shaBytes(b) }; });
}

function coreFactory(allowedNames) {
  'use strict';
  const allowed = new Set(allowedNames);
  const outcomes = {
    hg_scene: 'represented', hg_evidence_gap: 'open_gap', hg_authority_gate: 'exact_gate'
  };
  const lexical = (a, b) => String(a).localeCompare(String(b));
  const exactKeys = (obj, keys) => { const a = Object.keys(obj || {}).sort(), b = [...keys].sort(); return a.length === b.length && a.every((x, i) => x === b[i]); };
  const uniqSorted = xs => [...new Set(xs)].sort(lexical);
  const edgeKey = e => `${[...e.tails].sort(lexical).join('&')}=>${[...e.heads].sort(lexical).join('&')}`;
  const canonicalEdge = e => ({ id: String(e.id), tails: uniqSorted(e.tails), heads: uniqSorted(e.heads) });
  const fail = (op, error) => ({ ok: false, op, error, external_credit: false });
  function validateGraph(graph) {
    if (!graph || !exactKeys(graph, ['vertices', 'edges'])) return 'graph_closed_fields';
    if (!Array.isArray(graph.vertices) || !Array.isArray(graph.edges)) return 'graph_shape';
    if (!graph.vertices.every(v => typeof v === 'string' && v.length > 0)) return 'vertex_shape';
    if (new Set(graph.vertices).size !== graph.vertices.length) return 'duplicate_vertex';
    const vertices = new Set(graph.vertices);
    const ids = new Set();
    for (const edge of graph.edges) {
      if (!edge || !exactKeys(edge, ['id', 'tails', 'heads'])) return 'edge_closed_fields';
      if (typeof edge.id !== 'string' || !edge.id || ids.has(edge.id)) return ids.has(edge.id) ? 'duplicate_edge_id' : 'edge_id';
      ids.add(edge.id);
      if (!Array.isArray(edge.tails) || !Array.isArray(edge.heads) || !edge.tails.length || !edge.heads.length) return 'edge_shape';
      if (new Set(edge.tails).size !== edge.tails.length || new Set(edge.heads).size !== edge.heads.length) return 'duplicate_incidence';
      if (![...edge.tails, ...edge.heads].every(v => vertices.has(v))) return 'unknown_vertex';
    }
    return null;
  }
  function canonicalGraph(graph) {
    return {
      vertices: uniqSorted(graph.vertices),
      edges: graph.edges.map(canonicalEdge).sort((a, b) => edgeKey(a).localeCompare(edgeKey(b)) || a.id.localeCompare(b.id))
    };
  }
  function projection(graph) {
    const arcs = [];
    for (const e of graph.edges) for (const tail of e.tails) for (const head of e.heads) arcs.push(`${tail}->${head}`);
    return uniqSorted(arcs);
  }
  function components(graph, omit = null) {
    const vertices = graph.vertices.filter(v => v !== omit);
    const adj = Object.fromEntries(vertices.map(v => [v, new Set()]));
    for (const e of graph.edges) {
      const points = uniqSorted([...e.tails, ...e.heads]).filter(v => v !== omit);
      for (const a of points) for (const b of points) if (a !== b && adj[a] && adj[b]) { adj[a].add(b); adj[b].add(a); }
    }
    const seen = new Set(); const out = [];
    for (const start of vertices.sort(lexical)) if (!seen.has(start)) {
      const q = [start], part = []; seen.add(start);
      while (q.length) { const v = q.shift(); part.push(v); for (const n of [...adj[v]].sort(lexical)) if (!seen.has(n)) { seen.add(n); q.push(n); } }
      out.push(part.sort(lexical));
    }
    return out.sort((a, b) => a[0].localeCompare(b[0]));
  }
  function execute(request) {
    const op = request && typeof request.op === 'string' ? request.op : 'unknown';
    if (!request || !exactKeys(request, ['op', 'input'])) return fail(op, 'closed_fields');
    if (!allowed.has(op)) return fail(op, 'unsupported_operation');
    const input = request.input;
    const expectedInput = ['profile_id', 'graph', 'other_graph', 'roots', 'claim', 'provided', 'authority'];
    if (!input || !exactKeys(input, expectedInput)) return fail(op, 'input_closed_fields');
    const firstError = validateGraph(input.graph); if (firstError) return fail(op, firstError);
    const secondError = validateGraph(input.other_graph); if (secondError) return fail(op, `other_${secondError}`);
    if (!Array.isArray(input.roots) || !input.roots.every(v => input.graph.vertices.includes(v))) return fail(op, 'roots');
    if (!Array.isArray(input.provided)) return fail(op, 'provided');
    const graph = canonicalGraph(input.graph); const other = canonicalGraph(input.other_graph);
    let data;
    if (op === 'hg_profile') data = { vertices: graph.vertices.length, hyperedges: graph.edges.length, roots: uniqSorted(input.roots) };
    else if (op === 'hg_vertices') data = { vertices: graph.vertices };
    else if (op === 'hg_hyperedges') data = { hyperedges: graph.edges };
    else if (op === 'hg_incidence') data = { incidence: graph.vertices.map(v => ({ vertex: v, incoming: graph.edges.filter(e => e.heads.includes(v)).map(e => e.id).sort(lexical), outgoing: graph.edges.filter(e => e.tails.includes(v)).map(e => e.id).sort(lexical) })) };
    else if (op === 'hg_unique') data = { vertices_unique: true, edge_ids_unique: true, edge_signatures_unique: new Set(graph.edges.map(edgeKey)).size === graph.edges.length };
    else if (op === 'hg_sources') data = { sources: graph.vertices.filter(v => !graph.edges.some(e => e.heads.includes(v))) };
    else if (op === 'hg_sinks') data = { sinks: graph.vertices.filter(v => !graph.edges.some(e => e.tails.includes(v))) };
    else if (op === 'hg_reachable' || op === 'hg_layers') {
      const reached = new Set(input.roots), layer = Object.fromEntries(input.roots.map(v => [v, 0])); let changed = true;
      while (changed) { changed = false; for (const e of graph.edges) if (e.tails.every(v => reached.has(v))) { const next = 1 + Math.max(...e.tails.map(v => layer[v])); for (const h of e.heads) if (!reached.has(h)) { reached.add(h); layer[h] = next; changed = true; } } }
      data = op === 'hg_reachable' ? { reachable: [...reached].sort(lexical), unreachable: graph.vertices.filter(v => !reached.has(v)) } : { layers: Object.entries(layer).sort((a, b) => a[1] - b[1] || a[0].localeCompare(b[0])).map(([vertex, value]) => ({ vertex, layer: value })) };
    }
    else if (op === 'hg_transpose') data = { graph: { vertices: graph.vertices, edges: graph.edges.map(e => ({ id: e.id, tails: e.heads, heads: e.tails })) } };
    else if (['hg_union', 'hg_intersection', 'hg_difference'].includes(op)) {
      const a = new Set(graph.edges.map(edgeKey)), b = new Set(other.edges.map(edgeKey)); let signatures;
      if (op === 'hg_union') signatures = uniqSorted([...a, ...b]);
      if (op === 'hg_intersection') signatures = [...a].filter(x => b.has(x)).sort(lexical);
      if (op === 'hg_difference') signatures = [...a].filter(x => !b.has(x)).sort(lexical);
      data = { edge_signatures: signatures };
    }
    else if (op === 'hg_boundary') data = { tail_only: graph.vertices.filter(v => graph.edges.some(e => e.tails.includes(v)) && !graph.edges.some(e => e.heads.includes(v))), head_only: graph.vertices.filter(v => graph.edges.some(e => e.heads.includes(v)) && !graph.edges.some(e => e.tails.includes(v))), mixed: graph.vertices.filter(v => graph.edges.some(e => e.tails.includes(v)) && graph.edges.some(e => e.heads.includes(v))) };
    else if (op === 'hg_projection') data = { arcs: projection(graph) };
    else if (op === 'hg_components') data = { components: components(graph) };
    else if (op === 'hg_cut_candidates') { const base = components(graph).length; data = { baseline_components: base, cut_candidates: graph.vertices.filter(v => components(graph, v).length > base).sort(lexical) }; }
    else if (op === 'hg_scene') data = { points: graph.vertices.map((v, i) => ({ vertex: v, x: (i * 37 + graph.edges.length * 11) % 101, y: (i * 53 + graph.vertices.length * 7) % 101 })), arcs: projection(graph), synthetic_layout: true };
    else if (op === 'hg_evidence_gap') data = { claim: input.claim, missing: ['governed_real_observations', 'preregistered_comparator', 'independent_reproduction'], state: 'OPEN_GAP' };
    else if (op === 'hg_authority_gate') data = { claim: input.claim, required: ['competent_legal_authority', 'affected_party_oversight', 'relevant_cultural_authority', 'maori_authority_where_applicable'], state: 'EXACT_GATE' };
    else return fail(op, 'unimplemented_operation');
    return { ok: true, op, disposition: outcomes[op] || 'completed', data, external_credit: false };
  }
  return { execute, validateGraph, canonicalGraph, projection, components, edgeKey };
}

function makeProfile(index) {
  const n = 5 + (index % 4);
  const vertices = Array.from({ length: n }, (_, i) => `H${String(index + 1).padStart(2, '0')}-V${i}`);
  const edgeCount = 3 + (index % 3);
  const edges = Array.from({ length: edgeCount }, (_, j) => ({
    id: `E${j}`,
    tails: j % 2 === 0 && n > 5 ? [vertices[j % n], vertices[(j + 2) % n]] : [vertices[j % n]],
    heads: [vertices[(j + 1) % n]]
  }));
  const otherEdges = Array.from({ length: edgeCount }, (_, j) => ({
    id: `B${j}`,
    tails: [vertices[(j + 1) % n]],
    heads: [vertices[(j + 2 + (index % 2)) % n]]
  }));
  return { profile_id: `synthetic-hypergraph-${String(index + 1).padStart(2, '0')}`, graph: { vertices, edges }, other_graph: { vertices, edges: otherEdges }, roots: [vertices[0]] };
}

    const name = `ghc_family_hypergraph_${stage}_${String(i + 1).padStart(2, '0')}`;
    const rel = `${stage}/runners/${name}.txt`;
    writeText(rel, runnerSource(name, pair));
    const runner = loadTextModule(path.join(ROOT, rel));
    const requests = pair.map(op => proposals.find(p => p.mechanism === op).request);
    const output = runner.run(core, requests);
    const pass = output.length === 2 && output.every((x, j) => x.ok && x.op === pair[j]);
    if (!pass) throw new Error(`runner_smoke_failed:${name}`);
    runners.push({ name, operations: pair, path: toRepo(path.join(ROOT, rel)), smoke: 'PASS', output_sha256: shaValue(output) });
  }
  writeJson(`${stage}/skill-validation.json`, { schema: 'ghc.family.local-skill-validation.v1', owner: OWNER, phase: PHASE, stage, count: skills.length, passed: skills.filter(x => x.smoke === 'PASS').length, skills, boundary: BOUNDARY });

}

function runSession(stage, proposals, core) {
  const ops = OPERATIONS.filter(o => o.session === stage);
  const summaries = [];
  const negatives = [];
  const methods = [];
  for (const op of ops) {
    const group = proposals.filter(p => p.mechanism === op.name);
    const records = group.map(p => {
      const safe = core.execute(p.request);
      const candidate = core.execute(p.candidate.request);
      const detached = core.execute(p.detached_clone.request);
      const safeMatch = stableText(safe) === stableText(p.expected_result);
      const candidateRefusal = stableText(candidate) === stableText(p.candidate.expected);
      const detachedRefusal = stableText(detached) === stableText(p.detached_clone.expected);
      const restored = shaValue(p.request) === p.cfr.expected_restored_sha256;
      if (!safeMatch || !candidateRefusal || !detachedRefusal || !restored || candidate.ok || detached.ok) throw new Error(`contract_mismatch:${p.proposal_id}`);
      negatives.push({ id: `${p.proposal_id}-CANDIDATE`, proposal_id: p.proposal_id, class: 'malformed_candidate_subject', result: candidate, credit: 0, refusal_passed: candidateRefusal, retained: true });
      negatives.push({ id: `${p.proposal_id}-CLONE`, proposal_id: p.proposal_id, class: 'detached_clone_subject', result: detached, credit: 0, refusal_passed: detachedRefusal, retained: true });
      return { proposal_id: p.proposal_id, safe_result: safe, safe_match: safeMatch, candidate_subject: candidate, candidate_refusal_passed: candidateRefusal, detached_clone_subject: detached, detached_clone_refusal_passed: detachedRefusal, cfr_restored: restored, filesystem_delete: false };
    });
    writeJson(`${stage}/results/${op.name}.json`, { schema: 'ghc.family.hypergraph-operation-results.v1', owner: OWNER, phase: PHASE, stage, operation: op.name, count: records.length, disposition: op.outcome, records, boundary: BOUNDARY });
    summaries.push({ operation: op.name, proposals: 30, safe_passed: 30, candidate_failed_subjects: 30, candidate_refusals_passed: 30, detached_clone_failed_subjects: 30, detached_clone_refusals_passed: 30, cfr_reviews: 30, cfr_restorations: 30, filesystem_deletions: 0, outcome: op.outcome });
    methods.push({ method_id: `LM6984-M-${op.name}`, operation: op.name, stage, failed_witnesses: 2, passing_witnesses: 4, retained_negative_classes: ['malformed_candidate_subject', 'detached_clone_subject'], recovery_erases_failure: false, disposition: op.outcome });
  }
  writeJson(`${stage}/session-summary.json`, { schema: 'ghc.family.owner-session-summary.v13', owner: OWNER, phase: PHASE, stage, proposals: proposals.length, safe: 300, candidate_failed_subjects: 300, candidate_refusals: 300, detached_clone_failed_subjects: 300, detached_clone_refusals: 300, cfr_reviews: 300, cfr_restorations: 300, filesystem_deletions: 0, operations: summaries, boundary: BOUNDARY });
  writeJson(`${stage}/retained-negatives.json`, { schema: 'ghc.family.retained-negative-index.v1', owner: OWNER, phase: PHASE, stage, count: negatives.length, completion_credit: 0, records: negatives, boundary: BOUNDARY });
  writeJson(`${stage}/clean-fix-refine.json`, { schema: 'ghc.family.cfr-ledger.v1', owner: OWNER, phase: PHASE, stage, count: proposals.length, records: proposals.map(p => ({ proposal_id: p.proposal_id, review: p.cfr.review, restore: p.cfr.restore, restored_sha256: p.cfr.expected_restored_sha256, passed: true, filesystem_delete: false })), boundary: BOUNDARY });
  writeJson(`${stage}/method-flow.json`, { schema: 'ghc.family.method-flow.owner-overlay.v1', owner: OWNER, phase: PHASE, stage, methods, validation: { known_ids: true, no_unknown_statuses: true, failures_retained: true }, boundary: BOUNDARY });
  return { summaries, negatives, methods };
}

  }
  const first = proposals[0];
  tests.push({ id: 'LM6984-T-11', claim: 'duplicate vertex subject rejected', passed: core.execute(first.candidate.request).ok === false });
  tests.push({ id: 'LM6984-T-12', claim: 'detached clone subject rejected by closed fields', passed: core.execute(first.detached_clone.request).error === 'closed_fields' });
  tests.push({ id: 'LM6984-T-13', claim: 'only four core labels appear', passed: proposals.every(p => ['completed', 'represented', 'open_gap', 'exact_gate'].includes(p.expected_core_disposition)) });
  const scene = core.execute(proposals.find(p => p.mechanism === 'hg_scene').request);
  tests.push({ id: 'LM6984-T-14', claim: 'scene coordinates are finite bounded numbers', passed: scene.data.points.every(p => Number.isFinite(p.x) && Number.isFinite(p.y) && p.x >= 0 && p.x <= 100 && p.y >= 0 && p.y <= 100) });
  tests.push({ id: 'LM6984-T-15', claim: 'all safe envelopes deny external credit', passed: proposals.every(p => core.execute(p.request).external_credit === false) });

function collectOperationalFailures() {
  const records = [...STARTUP_FAILURES];
  for (const stage of ['x1', 'x2']) {
    const p = path.join(ROOT, stage, 'operational-failures.json');
    if (fs.existsSync(p)) records.push(...(JSON.parse(fs.readFileSync(p, 'utf8')).records || []));
  }
  const ids = new Set();
  return records.filter(r => { if (ids.has(r.id)) return false; ids.add(r.id); return true; });
}
