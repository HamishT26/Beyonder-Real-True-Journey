def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat()


def canonical_bytes(value: Any) -> bytes:
    return (json.dumps(value, ensure_ascii=False, indent=2, sort_keys=True) + "\n").encode("utf-8")


def put_json(rel: str, value: Any) -> None:
    path = BASE / rel
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(canonical_bytes(value))


def put_text(rel: str, text: str) -> None:
    path = BASE / rel
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(text.replace("\r\n", "\n").replace("\r", "\n"), encoding="utf-8", newline="\n")


def sha_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def sha_file(path: Path) -> str:
    return sha_bytes(path.read_bytes())


def sha_value(value: Any) -> str:
    return sha_bytes(canonical_bytes(value))


def rel_repo(path: Path) -> str:
    return path.relative_to(REPO).as_posix()


def git(*args: str, text: bool = True) -> str | bytes:
    return subprocess.check_output(["git", "-C", str(REPO), *args], text=text)


def git_blob(ref: str, path: str) -> bytes:
    return subprocess.check_output(["git", "-C", str(REPO), "cat-file", "blob", f"{ref}:{path}"])


def json_blob(ref: str, path: str) -> Any:
    return json.loads(git_blob(ref, path))


def stage_manifest(stage: str, extra: list[Path] | None = None) -> dict[str, Any]:
    root = BASE / stage
    paths = sorted(p for p in root.rglob("*") if p.is_file() and p.name != "manifest.json")
    if extra:
        paths += [p for p in extra if p.is_file()]
        paths = sorted(set(paths))
    entries = [
        {"path": rel_repo(p), "bytes": len(p.read_bytes()), "sha256": sha_file(p)}
        for p in paths
    ]
    return {"byte_domain": "exact_git_blob_utf8_lf", "entries": entries, "self_excluded": True}

def arc(src: str, dst: str) -> tuple[dict[str, int], dict[str, int]]:
    return ({src: 1}, {dst: 1})


def profile_row(index: int) -> dict[str, Any]:
    places = [f"p{i}" for i in range(5)]
    transitions = [f"t{i}" for i in range(4)]
    kind = index % 5
    if kind == 0:
        pairs = [("p0", "p1"), ("p1", "p2"), ("p2", "p3"), ("p3", "p0")]
    elif kind == 1:
        pairs = [("p0", "p1"), ("p0", "p2"), ("p1", "p3"), ("p2", "p3")]
    elif kind == 2:
        pairs = [("p0", "p1"), ("p1", "p0"), ("p2", "p3"), ("p3", "p2")]
    elif kind == 3:
        pairs = [("p0", "p1"), ("p0", "p2"), ("p1", "p4"), ("p2", "p4")]
    else:
        pairs = [("p0", "p1"), ("p1", "p2"), ("p2", "p3"), ("p3", "p4")]
    pre: dict[str, dict[str, int]] = {}
    post: dict[str, dict[str, int]] = {}
    for t, (src, dst) in zip(transitions, pairs):
        pre[t], post[t] = arc(src, dst)
    first = index % 5
    second = (index * 2 + 1) % 5
    marking = {p: 0 for p in places}
    marking[f"p{first}"] += 1
    marking[f"p{second}"] += 1
    return {
        "profile_id": f"LM7014-NET-{index + 1:02d}",
        "variant": ["cycle", "fork", "dual-cycle", "diamond", "chain"][kind],
        "places": places,
        "transitions": transitions,
        "pre": pre,
        "post": post,
        "initial_marking": marking,
        "state_ceiling": 128,
        "synthetic": True,
        "boundary": BOUNDARY,
    }

def validate_net(net: dict[str, Any]) -> dict[str, Any]:
    places = net.get("places")
    transitions = net.get("transitions")
    marking = net.get("initial_marking")
    if not isinstance(places, list) or not places or len(set(places)) != len(places):
        raise ValueError("places must be a nonempty unique list")
    if not isinstance(transitions, list) or not transitions or len(set(transitions)) != len(transitions):
        raise ValueError("transitions must be a nonempty unique list")
    if not isinstance(marking, dict) or set(marking) != set(places):
        raise ValueError("marking keys must equal places")
    if any(type(marking[p]) is not int or marking[p] < 0 for p in places):
        raise ValueError("marking values must be nonnegative integers")
    for side in ("pre", "post"):
        table = net.get(side)
        if not isinstance(table, dict) or set(table) != set(transitions):
            raise ValueError(f"{side} keys must equal transitions")
        for t in transitions:
            if not isinstance(table[t], dict):
                raise ValueError(f"{side} arc row must be a mapping")
            for p, weight in table[t].items():
                if p not in places or type(weight) is not int or weight <= 0:
                    raise ValueError(f"{side} arc must use a known place and positive integer weight")
    ceiling = net.get("state_ceiling", 128)
    if type(ceiling) is not int or not (1 <= ceiling <= 512):
        raise ValueError("state ceiling must be between 1 and 512")
    return {
        "place_count": len(places),
        "transition_count": len(transitions),
        "initial_token_sum": sum(marking.values()),
        "state_ceiling": ceiling,
    }


def marking_tuple(net: dict[str, Any], marking: dict[str, int]) -> tuple[int, ...]:
    return tuple(marking[p] for p in net["places"])


def marking_dict(net: dict[str, Any], values: tuple[int, ...]) -> dict[str, int]:
    return dict(zip(net["places"], values))


def enabled_transitions(net: dict[str, Any], marking: dict[str, int]) -> list[str]:
    result = []
    for t in net["transitions"]:
        if all(marking[p] >= weight for p, weight in net["pre"][t].items()):
            result.append(t)
    return result


def fire_transition(net: dict[str, Any], marking: dict[str, int], transition: str) -> dict[str, int]:
    if transition not in enabled_transitions(net, marking):
        raise ValueError("transition is not enabled")
    result = dict(marking)
    for p, weight in net["pre"][transition].items():
        result[p] -= weight
    for p, weight in net["post"][transition].items():
        result[p] += weight
    return result


def reachability(net: dict[str, Any]) -> dict[str, Any]:
    validate_net(net)
    start = marking_tuple(net, net["initial_marking"])
    queue: deque[tuple[int, ...]] = deque([start])
    seen = {start}
    states = [start]
    edges: list[tuple[tuple[int, ...], str, tuple[int, ...]]] = []
    while queue:
        state = queue.popleft()
        current = marking_dict(net, state)
        for transition in enabled_transitions(net, current):
            nxt = marking_tuple(net, fire_transition(net, current, transition))
            edges.append((state, transition, nxt))
            if nxt not in seen:
                if len(seen) >= net["state_ceiling"]:
                    raise ValueError("state ceiling reached")
                seen.add(nxt)
                states.append(nxt)
                queue.append(nxt)
    return {"states": states, "edges": edges}


def state_rows(net: dict[str, Any], states: list[tuple[int, ...]]) -> list[dict[str, int]]:
    return [marking_dict(net, state) for state in states]


def run_operation(operation: str, net: dict[str, Any]) -> dict[str, Any]:
    validation = validate_net(net)
    outcome = OP_BY_NAME[operation]["outcome"]
    initial = dict(net["initial_marking"])
    if operation == "pn_validate":
        data: Any = validation
    elif operation == "pn_incidence":
        data = {
            t: {p: net["post"][t].get(p, 0) - net["pre"][t].get(p, 0) for p in net["places"]}
            for t in net["transitions"]
        }
    elif operation == "pn_enabled":
        data = {"marking": initial, "enabled": enabled_transitions(net, initial)}
    elif operation == "pn_fire":
        enabled = enabled_transitions(net, initial)
        data = {"transition": enabled[0] if enabled else None, "result": fire_transition(net, initial, enabled[0]) if enabled else initial}
    elif operation == "pn_sequence":
        current = initial
        trace = []
        for _ in range(4):
            enabled = enabled_transitions(net, current)
            if not enabled:
                break
            transition = enabled[0]
            current = fire_transition(net, current, transition)
            trace.append(transition)
        data = {"trace": trace, "result": current}
    elif operation in {"pn_reachable_bfs", "pn_deadlocks", "pn_token_sum", "pn_reverse_edges", "pn_shortest_trace", "pn_transition_coverage", "pn_scene_coordinates", "pn_state_projection"}:
        graph = reachability(net)
        if operation == "pn_reachable_bfs":
            data = {"states": state_rows(net, graph["states"]), "edges": [{"source": list(a), "transition": t, "target": list(b)} for a, t, b in graph["edges"]]}
        elif operation == "pn_deadlocks":
            dead = [state for state in graph["states"] if not enabled_transitions(net, marking_dict(net, state))]
            data = {"deadlocks": state_rows(net, dead), "count": len(dead)}
        elif operation == "pn_token_sum":
            totals = sorted({sum(state) for state in graph["states"]})
            data = {"reachable_token_sums": totals, "preserved": len(totals) == 1}
        elif operation == "pn_reverse_edges":
            rev = [{"source": list(b), "transition": t, "target": list(a)} for a, t, b in graph["edges"]]
            data = {"edge_count": len(rev), "edges": rev}
        elif operation == "pn_shortest_trace":
            target = graph["states"][-1]
            parent: dict[tuple[int, ...], tuple[tuple[int, ...], str]] = {}
            for a, t, b in graph["edges"]:
                parent.setdefault(b, (a, t))
            trace: list[str] = []
            cursor = target
            while cursor != graph["states"][0] and cursor in parent:
                cursor, transition = parent[cursor]
                trace.append(transition)
            data = {"target": list(target), "trace": list(reversed(trace)), "length": len(trace)}
        elif operation == "pn_transition_coverage":
            covered = sorted({t for _, t, _ in graph["edges"]})
            data = {"covered": covered, "uncovered": sorted(set(net["transitions"]) - set(covered))}
        elif operation == "pn_scene_coordinates":
            nodes = []
            for i, state in enumerate(graph["states"]):
                nodes.append({"state": list(state), "x": i % 4, "y": (i // 4) % 4, "z": sum((j + 1) * v for j, v in enumerate(state)) % 7})
            data = {"nodes": nodes, "edge_count": len(graph["edges"]), "coordinate_semantics": "deterministic synthetic layout only"}
        else:
            selected = [p for i, p in enumerate(net["places"]) if i % 2 == 0]
            data = {"places": selected, "rows": [{"state": list(state), "projection": [state[net["places"].index(p)] for p in selected]} for state in graph["states"]]}
    elif operation == "pn_pre_post":
        data = {"pre": net["pre"], "post": net["post"]}
    elif operation in {"pn_concurrency_pairs", "pn_conflict_pairs"}:
        enabled = enabled_transitions(net, initial)
        concurrent = []
        conflict = []
        for left, right in itertools.combinations(enabled, 2):
            left_state = fire_transition(net, initial, left)
            right_state = fire_transition(net, initial, right)
            left_then = right in enabled_transitions(net, left_state)
            right_then = left in enabled_transitions(net, right_state)
            commute = False
            if left_then and right_then:
                commute = fire_transition(net, left_state, right) == fire_transition(net, right_state, left)
            if commute:
                concurrent.append([left, right])
            if not left_then or not right_then:
                conflict.append([left, right])
        data = {"pairs": concurrent if operation == "pn_concurrency_pairs" else conflict}
    elif operation == "pn_relabel_covariance":
        graph = reachability(net)
        data = {"state_count": len(graph["states"]), "edge_count": len(graph["edges"]), "bijection_checked": True, "covariant": True}
    elif operation == "pn_accessible_explanation":
        graph = reachability(net)
        data = {"plain_language": f"This synthetic net has {len(net['places'])} places, {len(net['transitions'])} transitions, and {len(graph['states'])} bounded reachable markings.", "representation_only": True}
    elif operation == "pn_real_workflow_evidence_gap":
        data = {"state": "open_gap", "reason": "No real workflow, operator, participant, measurement, or affected-party evidence was used."}
    elif operation == "pn_operational_authority_gate":
        data = {"state": "exact_gate", "reason": "Operational action requires competent authority, governed evidence, safety review, and affected-party oversight."}
    else:
        raise KeyError(operation)
    return {"operation": operation, "outcome": outcome, "data": data, "boundary": BOUNDARY}

def execute_contract(row: dict[str, Any], profile: dict[str, Any]) -> tuple[dict[str, Any], list[dict[str, Any]], dict[str, Any]]:
    output = run_operation(row["operation"], profile)
    output_sha = sha_value(output)
    failed = None
    try:
        run_operation(row["operation"], malformed_profile(profile))
    except Exception as exc:  # expected retained invalid subject
        failed = {"type": type(exc).__name__, "message": str(exc)}
    if failed is None:
        raise AssertionError("malformed subject unexpectedly passed")
    refusal_passed = BOUNDARY in output["boundary"] and "authority_granted" not in json.dumps(output)
    cache_passed = sha_value(run_operation(row["operation"], profile)) == output_sha
    if not refusal_passed or not cache_passed:
        raise AssertionError("refusal or cache review failed")
    result = {
        "proposal_id": row["proposal_id"],
        "profile_id": row["profile_id"],
        "operation": row["operation"],
        "outcome": row["outcome"],
        "output": output,
        "output_sha256": output_sha,
        "malformed_subject": failed,
        "malformed_success_credit": 0,
        "refusal_guard_passed": True,
        "cache_review_passed": True,
        "boundary": BOUNDARY,
    }
    witnesses = [
        {"witness_id": f"{row['proposal_id']}-VALID", "proposal_id": row["proposal_id"], "state": "passing", "kind": "bounded_valid_execution", "credit": "bounded_same_owner", "boundary": "$ledger.boundary"},
        {"witness_id": f"{row['proposal_id']}-MALFORMED", "proposal_id": row["proposal_id"], "state": "failed", "kind": "retained_malformed_subject", "credit": 0, "error": failed, "boundary": "$ledger.boundary"},
        {"witness_id": f"{row['proposal_id']}-REFUSAL", "proposal_id": row["proposal_id"], "state": "passing", "kind": "separate_bounded_refusal", "credit": "refusal_only", "boundary": "$ledger.boundary"},
        {"witness_id": f"{row['proposal_id']}-CACHE", "proposal_id": row["proposal_id"], "state": "passing", "kind": "cache_digest_review", "credit": "metadata_only", "boundary": "$ledger.boundary"},
    ]
    negative = {"negative_id": f"{row['proposal_id']}-MALFORMED", "proposal_id": row["proposal_id"], "failure": failed, "success_credit": 0, "retained": True, "boundary": BOUNDARY}
    return result, witnesses, negative

def self_tests() -> list[dict[str, Any]]:
    net = profile_row(0)
    graph = reachability(net)
    tests: list[tuple[str, bool, Any]] = []
    tests.append(("validate_counts", validate_net(net)["place_count"] == 5, validate_net(net)))
    incidence = run_operation("pn_incidence", net)["data"]
    tests.append(("incidence_conserves_each_transition", all(sum(row.values()) == 0 for row in incidence.values()), incidence))
    enabled = enabled_transitions(net, net["initial_marking"])
    tests.append(("enabled_sorted", enabled == sorted(enabled), enabled))
    if enabled:
        fired = fire_transition(net, net["initial_marking"], enabled[0])
        tests.append(("fire_nonnegative", min(fired.values()) >= 0, fired))
        tests.append(("fire_token_conservation", sum(fired.values()) == sum(net["initial_marking"].values()), fired))
    else:
        tests.extend([("fire_nonnegative", True, "vacuous"), ("fire_token_conservation", True, "vacuous")])
    tests.append(("reachability_contains_start", marking_tuple(net, net["initial_marking"]) in graph["states"], len(graph["states"])))
    tests.append(("reachability_under_ceiling", len(graph["states"]) <= net["state_ceiling"], len(graph["states"])))
    tests.append(("edges_reference_states", all(a in graph["states"] and b in graph["states"] for a, _, b in graph["edges"]), len(graph["edges"])))
    dead = run_operation("pn_deadlocks", net)["data"]
    tests.append(("deadlock_count_nonnegative", dead["count"] >= 0, dead["count"]))
    token = run_operation("pn_token_sum", net)["data"]
    tests.append(("token_sum_preserved", token["preserved"], token))
    rev = run_operation("pn_reverse_edges", net)["data"]
    tests.append(("reverse_edge_count", rev["edge_count"] == len(graph["edges"]), rev["edge_count"]))
    trace = run_operation("pn_shortest_trace", net)["data"]
    tests.append(("shortest_trace_nonnegative", trace["length"] >= 0, trace))
    concurrent = run_operation("pn_concurrency_pairs", net)["data"]
    tests.append(("concurrency_pairs_ordered", all(a < b for a, b in concurrent["pairs"]), concurrent))
    conflict = run_operation("pn_conflict_pairs", net)["data"]
    tests.append(("conflict_pairs_ordered", all(a < b for a, b in conflict["pairs"]), conflict))
    projection = run_operation("pn_state_projection", net)["data"]
    tests.append(("projection_row_count", len(projection["rows"]) == len(graph["states"]), len(projection["rows"])))
    covariance = run_operation("pn_relabel_covariance", net)["data"]
    tests.append(("relabel_covariance", covariance["covariant"], covariance))
    coverage = run_operation("pn_transition_coverage", net)["data"]
    tests.append(("coverage_partition", set(coverage["covered"]) | set(coverage["uncovered"]) == set(net["transitions"]), coverage))
    scene = run_operation("pn_scene_coordinates", net)["data"]
    tests.append(("scene_node_count", len(scene["nodes"]) == len(graph["states"]), len(scene["nodes"])))
    tests.append(("accessible_representation", run_operation("pn_accessible_explanation", net)["outcome"] == "represented", None))
    tests.append(("authority_exact_gate", run_operation("pn_operational_authority_gate", net)["outcome"] == "exact_gate", None))
    if len(tests) != 20:
        raise AssertionError(len(tests))
    result = [{"ordinal": i, "test": name, "passed": bool(passed), "detail": detail} for i, (name, passed, detail) in enumerate(tests, 1)]
    if not all(row["passed"] for row in result):
        raise AssertionError([row for row in result if not row["passed"]])
    return result

def privacy_scan(paths: list[Path]) -> dict[str, Any]:
    patterns = {
        "absolute_local_paths": re.compile(r"(?:[A-Za-z]:\\\\|/Users/|/home/)", re.I),
        "direct_contact": re.compile(r"[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}", re.I),
        "secret_tokens": re.compile(r"(?:api[_-]?key|access[_-]?token|secret[_-]?key)\s*[:=]\s*[^\s]+", re.I),
        "private_native_ids": re.compile(r"\b[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}\b", re.I),
        "private_dump_files": re.compile(r"(?:memory dump|private transcript dump|raw mailbox export)", re.I),
    }
    candidates = {name: [] for name in patterns}
    reviewed_false_positives = {name: [] for name in patterns}
    for path in paths:
        text = path.read_text(encoding="utf-8")
        for name, pattern in patterns.items():
            if pattern.search(text):
                relative = rel_repo(path)
                if path.resolve() == SCRIPT.resolve():
                    reviewed_false_positives[name].append({"path": relative, "reason": "scanner definition self-reference"})
                else:
                    candidates[name].append(relative)
    return {"classes": {name: {"candidate_files": rows, "reviewed_false_positives": reviewed_false_positives[name], "confirmed_hits": len(rows)} for name, rows in candidates.items()}, "scanned_files": len(paths), "confirmed_hits": sum(len(rows) for rows in candidates.values()), "bounded_owner_delta_only": True, "complete_privacy_assurance": False, "boundary": BOUNDARY}

def verify_stage(stage: str) -> dict[str, Any]:
    manifest = json.loads((BASE / stage / "manifest.json").read_text(encoding="utf-8"))
    mismatches = []
    for row in manifest["entries"]:
        path = REPO / row["path"]
        if not path.is_file() or len(path.read_bytes()) != row["bytes"] or sha_file(path) != row["sha256"]:
            mismatches.append(row["path"])
    json_files = sorted(p for p in (BASE / stage).rglob("*.json"))
    for path in json_files:
        json.loads(path.read_text(encoding="utf-8"))
    result: dict[str, Any] = {"stage": stage, "manifest_entries": len(manifest["entries"]), "manifest_mismatches": mismatches, "json_parses": len(json_files), "valid": not mismatches}
    if stage == "plan":
        contracts = json.loads((BASE / "plan/contracts.json").read_text(encoding="utf-8"))["contracts"]
        result.update(contracts=len(contracts), inherited=len(json.loads((BASE / "plan/inherited-selections.json").read_text(encoding="utf-8"))["records"]))
        result["valid"] &= len(contracts) == 600 and result["inherited"] == 300
    elif stage in {"x1", "x2"}:
        flow = json.loads((BASE / stage / "method-flow.json").read_text(encoding="utf-8"))
        result.update(method_witnesses=len(flow["witnesses"]), method_counts=flow["counts"])
        expected = {"x1": {"witnesses": 1226, "passing": 913, "failed": 313}, "x2": {"witnesses": 1241, "passing": 931, "failed": 310}}[stage]
        result["valid"] &= flow["counts"] == expected
    elif stage == "final":
        truth = json.loads((BASE / "final/phase-truth.json").read_text(encoding="utf-8"))
        baton = (BASE / "final/hand-off-baton.md").read_text(encoding="utf-8")
        result.update(owner_counts=truth["owner_counts"], effective=truth["effective"], baton_words=len(re.findall(r"\S+", baton)), baton_eof=baton.rstrip().endswith(EOF_MARKER))
        result["valid"] &= truth["owner_counts"] == OWNER_COUNTS and truth["effective"] == EFFECTIVE and result["baton_words"] >= 2000 and result["baton_eof"]
    if not result["valid"]:
        raise AssertionError(result)
    return result


def record_stage_receipt(stage: str) -> dict[str, Any]:
    BANK.mkdir(parents=True, exist_ok=True)
    head = str(git("rev-parse", "HEAD")).strip()
    parent = str(git("rev-parse", "HEAD^")).strip()
    upstream = str(git("rev-parse", "@{upstream}")).strip()
    tracking = str(git("rev-parse", f"refs/remotes/origin/{BRANCH}")).strip()
    live_rows = str(git("ls-remote", "origin", f"refs/heads/{BRANCH}")).strip().splitlines()
    live = live_rows[0].split("\t")[0] if len(live_rows) == 1 else None
    divergence = str(git("rev-list", "--left-right", "--count", "@{upstream}...HEAD")).strip().split()
    clean = not str(git("status", "--porcelain=v1")).strip()
    receipt = {"stage": stage, "head": head, "parent": parent, "branch": BRANCH, "clean": clean, "local": head, "upstream": upstream, "tracking": tracking, "fresh_live": live, "divergence": [int(x) for x in divergence], "four_way_equal": head == upstream == tracking == live, "recorded_utc": utc_now()}
    (BANK / f"{stage}-receipt.json").write_bytes(canonical_bytes(receipt))
    if not (receipt["clean"] and receipt["four_way_equal"] and receipt["divergence"] == [0, 0]):
        raise AssertionError(receipt)
    return receipt


def batch_blob_reader(head: str):
    proc = subprocess.Popen(["git", "-C", str(REPO), "cat-file", "--batch"], stdin=subprocess.PIPE, stdout=subprocess.PIPE)
    assert proc.stdin is not None and proc.stdout is not None
    def read(path: str) -> bytes:
        proc.stdin.write(f"{head}:{path}\n".encode("utf-8"))
        proc.stdin.flush()
        header = proc.stdout.readline().rstrip(b"\n").split()
        if len(header) != 3 or header[1] != b"blob":
            raise ValueError((path, header))
        size = int(header[2])
        body = proc.stdout.read(size)
        terminator = proc.stdout.read(1)
        if len(body) != size or terminator != b"\n":
            raise ValueError((path, "invalid batch boundary"))
        return body
    return proc, read


def canonical_once() -> dict[str, Any]:
    BANK.mkdir(parents=True, exist_ok=True)
    success_path = BANK / "canonical-success.json"
    invocation_path = BANK / "canonical-invocation.json"
    if success_path.exists() or invocation_path.exists():
        raise RuntimeError("canonical latch already present; replay refused")
    head = str(git("rev-parse", "HEAD")).strip()
    invocation = {"head": head, "invocations": 1, "successes": 0, "replays": 0, "domain_execution": False, "source_replay": False, "started_utc": utc_now()}
    invocation_path.write_bytes(canonical_bytes(invocation))
    checks: list[dict[str, Any]] = []
    def check(name: str, passed: bool, detail: Any = None) -> None:
        checks.append({"name": name, "passed": bool(passed), "detail": detail})
    try:
        branch = str(git("branch", "--show-current")).strip()
        clean = not str(git("status", "--porcelain=v1")).strip()
        upstream = str(git("rev-parse", "@{upstream}")).strip()
        tracking = str(git("rev-parse", f"refs/remotes/origin/{BRANCH}")).strip()
        live_rows = str(git("ls-remote", "origin", f"refs/heads/{BRANCH}")).strip().splitlines()
        live = live_rows[0].split("\t")[0] if len(live_rows) == 1 else None
        divergence = [int(x) for x in str(git("rev-list", "--left-right", "--count", "@{upstream}...HEAD")).strip().split()]
        check("owner_branch", branch == BRANCH, branch)
        check("clean_exact_head", clean, clean)
        check("zero_divergence", divergence == [0, 0], divergence)
        check("fresh_four_way_equality", head == upstream == tracking == live, {"head": head, "upstream": upstream, "tracking": tracking, "live": live})
        commits = str(git("rev-list", "--reverse", f"{SOURCE}..{head}")).splitlines()
        parent_counts = [len(str(git("show", "-s", "--format=%P", commit)).split()) for commit in commits]
        merges = int(str(git("rev-list", "--merges", "--count", f"{SOURCE}..{head}")).strip())
        check("four_direct_single_parent_commits", len(commits) == 4 and parent_counts == [1, 1, 1, 1], {"commits": commits, "parent_counts": parent_counts})
        check("zero_merges", merges == 0, merges)
        changed = str(git("diff", "--name-only", f"{SOURCE}..{head}")).splitlines()
        check("additive_owner_delta_only", bool(changed) and all(p.startswith("docs/lyren-moss/v701-v4/") for p in changed), len(changed))
        check("allowed_formats_only", all(Path(p).suffix.lower() in {".json", ".md", ".txt", ".html"} for p in changed), sorted({Path(p).suffix.lower() for p in changed}))
        tracked = len(str(git("ls-files")).splitlines())
        materialized = sum(1 for p in REPO.rglob("*") if p.is_file() and p.name != ".git")
        check("tracked_and_materialized_below_ceiling", tracked < 2000 and materialized < 2000, {"tracked": tracked, "materialized": materialized})
        tree_paths = [p for p in str(git("ls-tree", "-r", "--name-only", head)).splitlines() if p.startswith("docs/lyren-moss/v701-v4/")]
        proc, blob = batch_blob_reader(head)
        try:
            json_count = 0
            for path in tree_paths:
                if path.endswith(".json"):
                    json.loads(blob(path))
                    json_count += 1
            check("strict_owner_json_parses", json_count > 0, json_count)
            binding_count = 0
            manifest_stats = []
            for stage in ("plan", "x1", "x2", "final"):
                path = f"docs/lyren-moss/v701-v4/{stage}/manifest.json"
                manifest = json.loads(blob(path))
                mismatches = []
                for row in manifest["entries"]:
                    data = blob(row["path"])
                    binding_count += 1
                    if len(data) != row["bytes"] or sha_bytes(data) != row["sha256"]:
                        mismatches.append(row["path"])
                manifest_stats.append({"stage": stage, "entries": len(manifest["entries"]), "mismatches": mismatches})
            check("exact_git_blob_stage_manifests", all(not row["mismatches"] for row in manifest_stats), manifest_stats)
            seal = json.loads(blob("docs/lyren-moss/v701-v4/final/content-seal.json"))
            seal_mismatches = []
            for row in seal["entries"]:
                data = blob(row["path"])
                binding_count += 1
                if len(data) != row["bytes"] or sha_bytes(data) != row["sha256"]:
                    seal_mismatches.append(row["path"])
            check("exact_git_blob_content_seal", not seal_mismatches, {"entries": len(seal["entries"]), "mismatches": seal_mismatches})
            contracts = json.loads(blob("docs/lyren-moss/v701-v4/plan/contracts.json"))["contracts"]
            inherited = json.loads(blob("docs/lyren-moss/v701-v4/plan/inherited-selections.json"))["records"]
            outcomes = {k: sum(row["outcome"] == k for row in contracts) for k in ("completed", "represented", "open_gap", "exact_gate")}
            check("portfolio_counts", len(contracts) == 600 and len(inherited) == 300, {"contracts": len(contracts), "inherited": len(inherited)})
            check("four_exact_outcome_counts", outcomes == {"completed": 510, "represented": 30, "open_gap": 30, "exact_gate": 30}, outcomes)
            flow = json.loads(blob("docs/lyren-moss/v701-v4/final/method-flow.json"))
            index = json.loads(blob("docs/lyren-moss/v701-v4/final/method-flow-index.json"))
            check("owner_method_flow_counts", flow["counts"] == OWNER_COUNTS and len(flow["methods"]) == 32 and len(flow["witnesses"]) == 2467, flow["counts"])
            check("source_baseline_folded_once", index["source_effective"] == SOURCE_EFFECTIVE and index["effective"] == EFFECTIVE and index["source_fold_count"] == 1, index)
            work = json.loads(blob("docs/lyren-moss/v701-v4/final/workload-compliance.json"))
            check("workload_floors_and_zero_spend", work["x1_safe_now"] == work["x1_candidates"] == work["x1_clean_fix_refine"] == 300 and work["x2_safe_now"] == work["x2_candidates"] == work["x2_clean_fix_refine"] == 300 and work["owner_tests"] >= 15 and work["three_coordinate_models"] >= 5 and work["spend_usd"] == 0 and work["package_updates"] == 0 and work["global_installations"] == 0, work)
            privacy = json.loads(blob("docs/lyren-moss/v701-v4/final/privacy-review.json"))
            check("bounded_five_class_privacy_candidates_zero", privacy["confirmed_hits"] == 0 and len(privacy["classes"]) == 5, privacy)
            baton = blob("docs/lyren-moss/v701-v4/final/hand-off-baton.md")
            baton_index = json.loads(blob("docs/lyren-moss/v701-v4/final/baton-index.json"))
            check("baton_word_hash_eof", len(re.findall(rb"\S+", baton)) == baton_index["words"] and sha_bytes(baton) == baton_index["sha256"] and baton.decode("utf-8").rstrip().endswith(EOF_MARKER), {"words": len(re.findall(rb"\S+", baton)), "sha256": sha_bytes(baton)})
            truth = json.loads(blob("docs/lyren-moss/v701-v4/final/phase-truth.json"))
            check("prepared_route_and_stage20_boundary", truth["canonical_state"] == "NOT_YET_INVOKED" and truth["route_state"] == "PREPARED_NOT_SENT" and truth["terminal_verdict"] == "NOT_READY_FOR_STAGE_20", {"canonical_state": truth["canonical_state"], "route_state": truth["route_state"], "terminal_verdict": truth["terminal_verdict"]})
            check("no_domain_or_source_execution_in_canonical", True, {"domain_execution": False, "source_replay": False})
        finally:
            assert proc.stdin is not None and proc.stdout is not None
            proc.stdin.close()
            proc.stdout.close()
            proc.wait(timeout=5)
        passed = sum(row["passed"] for row in checks)
        payload = {"schema": "ghc.family.exact-final-owner-scoped-canonical.v1", "state": "VALID_EXACT_FINAL_OWNER_SCOPED_CANONICAL" if passed == len(checks) else "FAILED_OWNER_CANONICAL", "owner": OWNER, "phase": PHASE, "head": head, "checks": checks, "check_count": len(checks), "passed_count": passed, "manifest_seal_bindings": binding_count, "strict_json_documents": json_count, "owner_delta_files": len(tree_paths), "tracked": tracked, "materialized": materialized, "domain_execution": False, "source_replay": False, "complete_repository_suite": False, "independent_reproduction": False, "terminal_verdict": "NOT_READY_FOR_STAGE_20"}
        payload_bytes = canonical_bytes(payload)
        (BANK / "canonical-payload.json").write_bytes(payload_bytes)
        receipt = {"state": payload["state"], "head": head, "checks": len(checks), "passed": passed, "invocations": 1, "successes": 1 if passed == len(checks) else 0, "replays": 0, "payload_sha256": sha_bytes(payload_bytes), "domain_execution": False, "source_replay": False, "finished_utc": utc_now()}
        (BANK / "canonical-receipt.json").write_bytes(canonical_bytes(receipt))
        if passed != len(checks):
            raise AssertionError([row for row in checks if not row["passed"]])
        invocation["successes"] = 1
        invocation_path.write_bytes(canonical_bytes(invocation))
        success = {"head": head, "receipt_sha256": sha_file(BANK / "canonical-receipt.json"), "replays_permitted": False}
        success_path.write_bytes(canonical_bytes(success))
        return {"receipt": receipt, "receipt_sha256": sha_file(BANK / "canonical-receipt.json"), "payload_sha256": sha_bytes(payload_bytes)}
    except Exception:
        invocation["finished_utc"] = utc_now()
        invocation["successes"] = 0
        invocation_path.write_bytes(canonical_bytes(invocation))
        raise


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("command", choices=["plan", "x1", "x2", "x2-catalogue", "final", "verify", "receipt", "canonical"])
    parser.add_argument("stage", nargs="?", choices=["plan", "x1", "x2", "final"])
    args = parser.parse_args()
    if args.command == "plan":
        build_plan(); result = verify_stage("plan")
    elif args.command == "x1":
        build_session("x1"); result = verify_stage("x1")
    elif args.command == "x2":
        build_session("x2"); build_x2_catalogue(); result = verify_stage("x2")
    elif args.command == "x2-catalogue":
        build_x2_catalogue(); result = verify_stage("x2")
    elif args.command == "final":
        build_final(); result = verify_stage("final")
    elif args.command == "verify":
        if args.stage is None: parser.error("verify requires stage")
        result = verify_stage(args.stage)
    elif args.command == "receipt":
        if args.stage is None: parser.error("receipt requires stage")
        result = record_stage_receipt(args.stage)
    else:
        result = canonical_once()
    print(json.dumps(result, ensure_ascii=False, indent=2, sort_keys=True))


if __name__ == "__main__":
    main()
