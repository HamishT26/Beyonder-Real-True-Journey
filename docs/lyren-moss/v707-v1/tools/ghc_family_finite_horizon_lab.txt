def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat().replace("+00:00", "Z")


def canonical_bytes(value: Any) -> bytes:
    return (json.dumps(value, sort_keys=True, ensure_ascii=False, separators=(",", ":")) + "\n").encode("utf-8")


def pretty_bytes(value: Any) -> bytes:
    return (json.dumps(value, sort_keys=True, ensure_ascii=False, indent=2) + "\n").encode("utf-8")


def sha_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def sha_value(value: Any) -> str:
    return sha_bytes(canonical_bytes(value))


def sha_file(path: Path) -> str:
    return sha_bytes(path.read_bytes())


def git_blob_sha1(data: bytes) -> str:
    return hashlib.sha1(b"blob " + str(len(data)).encode("ascii") + b"\0" + data).hexdigest()


def write_bytes(path: Path, data: bytes) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(data)


def put_json(rel: str, value: Any) -> None:
    write_bytes(BASE / rel, pretty_bytes(value))


def put_text(rel: str, value: str) -> None:
    write_bytes(BASE / rel, (value.rstrip() + "\n").replace("\r\n", "\n").encode("utf-8"))


def git(*args: str, cwd: Path = REPO, text: bool = True) -> str | bytes:
    result = subprocess.run(["git", *args], cwd=cwd, capture_output=True, check=False)
    if result.returncode != 0:
        raise RuntimeError(result.stderr.decode("utf-8", errors="replace"))
    return result.stdout.decode("utf-8").strip() if text else result.stdout


def git_blob(path: str, revision: str = "HEAD") -> bytes:
    return git("show", f"{revision}:{path}", text=False)


def git_index_blob(path: str) -> bytes:
    return git("show", f":{path}", text=False)


def f(value: int, denominator: int = 1) -> Fraction:
    return Fraction(value, denominator)


def fjson(value: Fraction) -> dict[str, int]:
    return {"n": value.numerator, "d": value.denominator}


def ftext(value: Fraction) -> str:
    return str(value.numerator) if value.denominator == 1 else f"{value.numerator}/{value.denominator}"


def encode(value: Any) -> Any:
    if isinstance(value, Fraction):
        return fjson(value)
    if isinstance(value, list):
        return [encode(item) for item in value]
    if isinstance(value, tuple):
        return [encode(item) for item in value]
    if isinstance(value, dict):
        return {key: encode(item) for key, item in value.items()}
    return value

def validate_fixture(row: dict[str, Any]) -> None:
    if isinstance(row.get("horizon"), bool) or not isinstance(row.get("horizon"), int) or not 1 <= row["horizon"] <= 6:
        raise ValueError("horizon must be an integer from one through six")
    if row.get("states") != 2 or row.get("actions") not in {2, 3}:
        raise ValueError("bounded state/action shape required")
    gamma = row.get("gamma")
    if not isinstance(gamma, Fraction) or isinstance(gamma.numerator, bool) or not 0 <= gamma < 1:
        raise ValueError("gamma must be a declared rational in [0,1)")
    if len(row.get("initial", [])) != row["states"] or any(value < 0 for value in row["initial"]) or sum(row["initial"], Fraction()) != 1:
        raise ValueError("initial distribution must be a nonnegative simplex")
    if len(row.get("terminal", [])) != row["states"]:
        raise ValueError("terminal vector shape mismatch")
    for value in row["terminal"]:
        if not isinstance(value, Fraction) or isinstance(value.numerator, bool):
            raise ValueError("terminal values must be strict rationals")
    for state in range(row["states"]):
        if len(row["rewards"][state]) != row["actions"] or len(row["transitions"][state]) != row["actions"]:
            raise ValueError("action rows must match declared action count")
        for action in range(row["actions"]):
            reward = row["rewards"][state][action]
            probs = row["transitions"][state][action]
            if not isinstance(reward, Fraction) or isinstance(reward.numerator, bool):
                raise ValueError("rewards must be strict rationals")
            if len(probs) != row["states"] or any(not isinstance(value, Fraction) or value < 0 for value in probs) or sum(probs, Fraction()) != 1:
                raise ValueError("transition row must be a nonnegative simplex")


def backward(row: dict[str, Any]) -> dict[str, Any]:
    validate_fixture(row)
    values: list[list[Fraction]] = [[Fraction() for _ in range(row["states"])] for _ in range(row["horizon"] + 1)]
    values[row["horizon"]] = list(row["terminal"])
    policies: list[list[int]] = [[0 for _ in range(row["states"])] for _ in range(row["horizon"])]
    action_values: list[list[list[Fraction]]] = [[[] for _ in range(row["states"])] for _ in range(row["horizon"])]
    for time in range(row["horizon"] - 1, -1, -1):
        for state in range(row["states"]):
            qs = []
            for action in range(row["actions"]):
                expected = sum(row["transitions"][state][action][nxt] * values[time + 1][nxt] for nxt in range(row["states"]))
                qs.append(row["rewards"][state][action] + row["gamma"] * expected)
            best = max(qs)
            policies[time][state] = min(action for action, value in enumerate(qs) if value == best)
            values[time][state] = best
            action_values[time][state] = qs
    initial_return = sum(row["initial"][state] * values[0][state] for state in range(row["states"]))
    return {"values": values, "policy": policies, "action_values": action_values, "initial_return": initial_return}


def evaluate_policy(row: dict[str, Any], policy: list[list[int]]) -> Fraction:
    values = list(row["terminal"])
    for time in range(row["horizon"] - 1, -1, -1):
        prior = []
        for state in range(row["states"]):
            action = policy[time][state]
            expected = sum(row["transitions"][state][action][nxt] * values[nxt] for nxt in range(row["states"]))
            prior.append(row["rewards"][state][action] + row["gamma"] * expected)
        values = prior
    return sum(row["initial"][state] * values[state] for state in range(row["states"]))


def exhaustive_oracle(row: dict[str, Any]) -> dict[str, Any]:
    coordinates = [(time, state) for time in range(row["horizon"]) for state in range(row["states"])]
    best_value: Fraction | None = None
    best_policy: list[list[int]] | None = None
    count = 0
    for choices in itertools.product(range(row["actions"]), repeat=len(coordinates)):
        policy = [[0 for _ in range(row["states"])] for _ in range(row["horizon"])]
        for (time, state), action in zip(coordinates, choices):
            policy[time][state] = action
        value = evaluate_policy(row, policy)
        count += 1
        if best_value is None or value > best_value or (value == best_value and policy < best_policy):
            best_value = value
            best_policy = policy
    return {"best_value": best_value, "best_policy": best_policy, "policy_count": count}


def occupancy(row: dict[str, Any], policy: list[list[int]]) -> dict[str, Any]:
    distributions = [list(row["initial"])]
    discounted_reward = Fraction()
    current = list(row["initial"])
    factor = Fraction(1)
    for time in range(row["horizon"]):
        discounted_reward += factor * sum(current[state] * row["rewards"][state][policy[time][state]] for state in range(row["states"]))
        nxt = [Fraction() for _ in range(row["states"])]
        for state in range(row["states"]):
            action = policy[time][state]
            for target in range(row["states"]):
                nxt[target] += current[state] * row["transitions"][state][action][target]
        distributions.append(nxt)
        current = nxt
        factor *= row["gamma"]
    discounted_reward += factor * sum(current[state] * row["terminal"][state] for state in range(row["states"]))
    return {"distributions": distributions, "return": discounted_reward}


def transform_scale(row: dict[str, Any], factor: Fraction) -> dict[str, Any]:
    changed = deepcopy(row)
    changed["rewards"] = [[value * factor for value in state] for state in row["rewards"]]
    changed["terminal"] = [value * factor for value in row["terminal"]]
    return changed


def transform_translate(row: dict[str, Any], delta: Fraction) -> dict[str, Any]:
    changed = deepcopy(row)
    changed["rewards"] = [[value + delta for value in state] for state in row["rewards"]]
    changed["terminal"] = [value + delta for value in row["terminal"]]
    return changed


def transform_swap(row: dict[str, Any]) -> dict[str, Any]:
    changed = deepcopy(row)
    changed["initial"] = list(reversed(row["initial"]))
    changed["terminal"] = list(reversed(row["terminal"]))
    changed["rewards"] = list(reversed(row["rewards"]))
    changed["transitions"] = []
    for state in reversed(range(row["states"])):
        changed["transitions"].append([list(reversed(probs)) for probs in row["transitions"][state]])
    return changed


def transform_duplicate_action(row: dict[str, Any]) -> dict[str, Any]:
    changed = deepcopy(row)
    changed["actions"] = 3
    for state in range(row["states"]):
        changed["rewards"][state].append(row["rewards"][state][0])
        changed["transitions"][state].append(list(row["transitions"][state][0]))
    return changed

def result_for(operation: dict[str, Any], row: dict[str, Any]) -> dict[str, Any]:
    solved = backward(row)
    policy = solved["policy"]
    op = operation["operation"]
    if op == 1:
        result = {"valid": True, "states": row["states"], "actions": row["actions"], "horizon": row["horizon"]}
    elif op == 2:
        result = {"admitted": 1 <= row["horizon"] <= 6, "horizon": row["horizon"]}
    elif op == 3:
        result = {"initial_return": solved["initial_return"], "policy": policy}
    elif op == 4:
        result = {"values": solved["values"], "table_sha256": sha_value(encode(solved["values"]))}
    elif op == 5:
        result = {"policy": policy, "policy_sha256": sha_value(policy)}
    elif op == 6:
        residuals = []
        for time in range(row["horizon"]):
            for state in range(row["states"]):
                residuals.append(solved["values"][time][state] - max(solved["action_values"][time][state]))
        result = {"residuals": residuals, "all_zero": all(value == 0 for value in residuals)}
    elif op == 7:
        result = {"terminal": solved["values"][-1], "matches": solved["values"][-1] == row["terminal"]}
    elif op == 8:
        occ = occupancy(row, policy)
        result = {"distributions": occ["distributions"], "return": occ["return"], "matches_value": occ["return"] == solved["initial_return"]}
    elif op == 9:
        result = {"fixture_sha256": row["fixture_sha256"], "policy_sha256": sha_value(policy), "result_sha256": sha_value(encode(solved["initial_return"]))}
    elif op == 10:
        ties = []
        for time in range(row["horizon"]):
            for state in range(row["states"]):
                best = max(solved["action_values"][time][state])
                tied = [index for index, value in enumerate(solved["action_values"][time][state]) if value == best]
                if len(tied) > 1:
                    ties.append({"time": time, "state": state, "actions": tied, "selected": min(tied)})
        result = {"ties": ties, "deterministic_rule": "smallest action index"}
    elif op == 11:
        result = {"initial": row["initial"], "return": solved["initial_return"], "recomputed": sum(row["initial"][s] * solved["values"][0][s] for s in range(row["states"]))}
    elif op == 12:
        changed = transform_swap(row)
        swapped = backward(changed)
        result = {"original": solved["initial_return"], "swapped": swapped["initial_return"], "equal": solved["initial_return"] == swapped["initial_return"]}
    elif op == 13:
        delta = Fraction(1, 3)
        translated = backward(transform_translate(row, delta))
        result = {"delta": delta, "original": solved["initial_return"], "translated": translated["initial_return"], "difference": translated["initial_return"] - solved["initial_return"]}
    elif op == 14:
        factor = Fraction(3, 2)
        scaled = backward(transform_scale(row, factor))
        result = {"factor": factor, "original": solved["initial_return"], "scaled": scaled["initial_return"], "value_covariant": scaled["initial_return"] == factor * solved["initial_return"], "policy_equal": scaled["policy"] == policy}
    elif op == 15:
        duplicated = backward(transform_duplicate_action(row))
        result = {"original": solved["initial_return"], "duplicated": duplicated["initial_return"], "value_equal": solved["initial_return"] == duplicated["initial_return"]}
    elif op == 16:
        changed = deepcopy(row)
        changed["horizon"] += 1
        extended = backward(changed)
        result = {"original_horizon": row["horizon"], "extended_horizon": changed["horizon"], "original": solved["initial_return"], "extended": extended["initial_return"], "monotonicity_claimed": False}
    elif op == 17:
        low = deepcopy(row)
        high = deepcopy(row)
        low["gamma"] = max(Fraction(), row["gamma"] - Fraction(1, 20))
        high["gamma"] = min(Fraction(19, 20), row["gamma"] + Fraction(1, 20))
        low_result = backward(low)["initial_return"]
        high_result = backward(high)["initial_return"]
        result = {"gamma_low": low["gamma"], "gamma_high": high["gamma"], "return_low": low_result, "return_high": high_result, "sample_min": min(low_result, high_result), "sample_max": max(low_result, high_result), "global_interval_claimed": False}
    elif op == 18:
        oracle = exhaustive_oracle(row)
        result = {"coordinates": [row["horizon"], oracle["policy_count"], solved["initial_return"]], "physical_geometry_claimed": False}
    elif op == 19:
        result = {"state": "open_gap", "missing": ["real observations", "measurement model", "uncertainty calibration", "independent review"]}
    elif op == 20:
        result = {"state": "exact_gate", "missing": ["competent operational authority", "affected-party review", "production safety case", "rollback governance"]}
    else:
        raise AssertionError(op)
    return encode(result)

def x1_tests() -> list[dict[str, Any]]:
    tests = []
    for row in fixture_rows():
        solved = backward(row)
        oracle = exhaustive_oracle(row)
        tests.append({"test_id": "LY7071-X1-ORACLE-" + row["fixture_id"], "passed": solved["initial_return"] == oracle["best_value"], "runtime": ftext(solved["initial_return"]), "oracle": ftext(oracle["best_value"]), "policy_count": oracle["policy_count"]})
    invalids = []
    base = fixture_rows()[0]
    a = deepcopy(base); a["horizon"] = 0; invalids.append(("zero-horizon", a))
    b = deepcopy(base); b["gamma"] = Fraction(1); invalids.append(("unit-gamma", b))
    c = deepcopy(base); c["initial"] = [Fraction(1), Fraction(1)]; invalids.append(("bad-initial-simplex", c))
    d = deepcopy(base); d["transitions"][0][0] = [Fraction(1), Fraction(1)]; invalids.append(("bad-transition-simplex", d))
    e = deepcopy(base); e["rewards"][0][0] = True; invalids.append(("boolean-reward", e))
    for name, subject in invalids:
        rejected = False
        reason = None
        try:
            validate_fixture(subject)
        except Exception as error:
            rejected = True
            reason = str(error)
        tests.append({"test_id": "LY7071-X1-INVALID-" + name, "passed": rejected, "subject_state": "failed", "original_success_credit": 0, "refusal_reason": reason})
    return tests


def x2_tests() -> list[dict[str, Any]]:
    tests = []
    for row in fixture_rows():
        solved = backward(row)
        scaled = backward(transform_scale(row, Fraction(3, 2)))
        tests.append({"test_id": "LY7071-X2-SCALE-" + row["fixture_id"], "passed": scaled["initial_return"] == Fraction(3, 2) * solved["initial_return"] and scaled["policy"] == solved["policy"]})
    for row in fixture_rows():
        solved = backward(row)
        swapped = backward(transform_swap(row))
        tests.append({"test_id": "LY7071-X2-SWAP-" + row["fixture_id"], "passed": swapped["initial_return"] == solved["initial_return"]})
    return tests

        valid = subprocess.run([sys.executable, "-I", str(path), "--check", row["check"]], input=json.dumps(valid_payloads[row["check"]]), capture_output=True, text=True)
        invalid = subprocess.run([sys.executable, "-I", str(path), "--check", row["check"]], input=json.dumps(invalid_payloads[row["check"]]), capture_output=True, text=True)
        receipts.append({"hook_id": row["hook_id"], "path": rel, "check": row["check"], "valid_returncode": valid.returncode, "invalid_returncode": invalid.returncode, "valid_passed": valid.returncode == 0, "invalid_refused": invalid.returncode == 2, "valid_stdout_sha256": sha_bytes(valid.stdout.encode("utf-8")), "invalid_stdout_sha256": sha_bytes(invalid.stdout.encode("utf-8")), "installed": False})
    if not all(row["valid_passed"] and row["invalid_refused"] for row in receipts):
        raise RuntimeError("manual hook envelope check failed")
    return receipts

def privacy_review() -> dict[str, Any]:
    patterns = {
        "credential_assignment": re.compile("api" + r"[_-]?key\s*=", re.IGNORECASE),
        "delegation_markup": re.compile("<" + "codex_delegation", re.IGNORECASE),
        "private_local_path": re.compile(r"[A-Za-z]:[\\/]+Users[\\/]+[^\\/]+", re.IGNORECASE),
        "private_uri": re.compile("plugin" + r"://|" + "app" + r"://", re.IGNORECASE),
        "raw_uuid": re.compile(r"\b[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}\b", re.IGNORECASE),
    }
    candidates = []
    files = [path for path in BASE.rglob("*") if path.is_file() and path.suffix.lower() in {".json", ".md", ".txt", ".html"}]
    for path in files:
        text = path.read_text(encoding="utf-8", errors="replace")
        for name, pattern in patterns.items():
            for match in pattern.finditer(text):
                candidates.append({"class": name, "path": path.relative_to(REPO).as_posix(), "token_sha256": sha_bytes(match.group(0).encode("utf-8"))})
    return {"classes": list(patterns), "owner_files_scanned": len(files), "candidates": candidates, "confirmed_owner_hits": len(candidates), "complete_privacy_assurance": False, "boundary": BOUNDARY}

def metadata_checks(head: str) -> dict[str, Any]:
    checks = []
    def add(name: str, passed: bool, detail: Any = None) -> None:
        checks.append({"name": name, "passed": bool(passed), "detail": detail})
    add("exact_head", git("rev-parse", "HEAD") == head)
    add("branch", git("branch", "--show-current") == BRANCH)
    add("clean", not bool(git("status", "--porcelain")))
    upstream = git("rev-parse", "@{upstream}")
    add("upstream_equal", upstream == head, upstream)
    divergence = git("rev-list", "--left-right", "--count", "HEAD...@{upstream}")
    add("zero_divergence", divergence.replace("\t", " ").strip() == "0 0", divergence)
    live = git("ls-remote", "--heads", "origin", "refs/heads/" + BRANCH).split()
    add("fresh_live_equal", bool(live) and live[0] == head, live[0] if live else None)
    commits = git("rev-list", "--reverse", SOURCE_HEAD + ".." + head).splitlines()
    add("four_new_commits", len(commits) == 4, commits)
    parents = [{"commit": commit, "parent_count": len(git("rev-list", "--parents", "-n", "1", commit).split()) - 1} for commit in commits]
    add("single_parent_zero_merge", all(row["parent_count"] == 1 for row in parents), parents)
    stage_manifests = [("planning", commits[0]), ("x1", commits[1]), ("x2", commits[2])]
    manifest_errors = []
    for stage, revision in stage_manifests:
        path = f"docs/lyren-moss/v707-v1/{stage}/manifest.json"
        manifest = json.loads(git_blob(path, revision).decode("utf-8"))
        for entry in manifest["entries"]:
            data = git_blob(entry["path"], revision)
            if len(data) != entry["bytes"] or sha_bytes(data) != entry["sha256"]:
                manifest_errors.append({"stage": stage, "path": entry["path"]})
    add("stage_manifests", not manifest_errors, manifest_errors)
    seal = json.loads(git_blob("docs/lyren-moss/v707-v1/final/content-seal.json", head).decode("utf-8"))
    seal_errors = []
    for entry in seal["entries"]:
        data = git_blob(entry["path"], head)
        if len(data) != entry["bytes"] or git_blob_sha1(data) != entry["git_blob_sha1"] or sha_bytes(data) != entry["sha256"]:
            seal_errors.append(entry["path"])
    add("content_seal", not seal_errors, {"entries": len(seal["entries"]), "errors": seal_errors})
    owner_paths = [path for path in git("ls-tree", "-r", "--name-only", head, "docs/lyren-moss/v707-v1").splitlines() if path]
    json_errors = []
    for path in owner_paths:
        if path.endswith(".json"):

                json_errors.append({"path": path, "error": str(error)})
    add("strict_json", not json_errors, {"count": sum(path.endswith(".json") for path in owner_paths), "errors": json_errors})
    invalid_ext = [path for path in owner_paths if Path(path).suffix.lower() not in {".json", ".md", ".txt", ".html"}]
    add("extension_allowlist", not invalid_ext, invalid_ext)
    add("tracked_file_ceiling", len(git("ls-files").splitlines()) <= 2000, len(git("ls-files").splitlines()))
    baton = git_blob("docs/lyren-moss/v707-v1/final/baton.md", head).decode("utf-8")
    add("baton_words", 2000 <= len(re.findall(r"\S+", baton)) <= 100000, len(re.findall(r"\S+", baton)))
    add("baton_eof", baton.rstrip().endswith(EOF_MARKER))

    add("four_outcomes", set(truth["outcomes"]) == ALLOWED_OUTCOMES)
    add("terminal_verdict", truth["terminal_verdict"] == "NOT_READY_FOR_STAGE_20")
    add("route_prepared", truth["route_state"] == "PREPARED_NOT_SENT")
    add("recipient_unclaimed", truth["recipient_completion"] == "UNCLAIMED")
    add("canonical_not_replayed_in_repo", truth["canonical_state"] == "NOT_YET_INVOKED")

    add("x1_tests", json.loads(git_blob("docs/lyren-moss/v707-v1/x1/tests.json", head).decode("utf-8"))["passed"] == 20)

    add("models", json.loads(git_blob("docs/lyren-moss/v707-v1/x2/models.json", head).decode("utf-8"))["count"] == 15)
    add("hooks", json.loads(git_blob("docs/lyren-moss/v707-v1/x2/hook-receipts.json", head).decode("utf-8"))["count"] == 5)
    add("workload", json.loads(git_blob("docs/lyren-moss/v707-v1/final/workload.json", head).decode("utf-8"))["x2"]["candidate"] == 300)
    return {"head": head, "checks": checks, "passed": sum(row["passed"] for row in checks), "total": len(checks), "all_passed": all(row["passed"] for row in checks)}


def preflight() -> dict[str, Any]:
    payload = metadata_checks(git("rev-parse", "HEAD"))
    payload["state"] = "VALID_EXACT_FINAL_OWNER_SCOPED_METADATA_PREFLIGHT" if payload["all_passed"] else "FAILED_METADATA_PREFLIGHT"
    payload["recorded_utc"] = utc_now()
    payload["domain_engines_executed"] = 0
    BANK.mkdir(parents=True, exist_ok=True)
    write_bytes(BANK / "metadata-preflight.json", pretty_bytes(payload))
    return payload


def canonical_once() -> dict[str, Any]:
    latch_path = BANK / "canonical-latch.json"
    if latch_path.exists():
        raise RuntimeError("canonical latch already exists; successful canonical may not be replayed")
    head = git("rev-parse", "HEAD")
    invocation = {"owner": OWNER, "phase": PHASE, "head": head, "invocation": 1, "recorded_utc": utc_now()}
    BANK.mkdir(parents=True, exist_ok=True)
    write_bytes(BANK / "canonical-invocation.json", pretty_bytes(invocation))
    payload = metadata_checks(head)
    payload["state"] = "VALID_EXACT_FINAL_OWNER_SCOPED_METADATA_CANONICAL" if payload["all_passed"] else "FAILED_CANONICAL_NO_SUCCESS_CREDIT"
    payload["domain_engines_executed"] = 0
    payload["invocations"] = 1
    payload["successes"] = 1 if payload["all_passed"] else 0
    payload["replays"] = 0
    write_bytes(BANK / "canonical-payload.json", pretty_bytes(payload))
    receipt = {"state": payload["state"], "head": head, "checks": f"{payload['passed']}/{payload['total']}", "payload_sha256": sha_file(BANK / "canonical-payload.json"), "invocations": 1, "successes": payload["successes"], "replays": 0, "recorded_utc": utc_now(), "same_owner_only": True, "complete_repository_suite": False, "independent_reproduction": False, "boundary": BOUNDARY}
    write_bytes(BANK / "canonical-receipt.json", pretty_bytes(receipt))
    if payload["all_passed"]:
        latch = {"closed": True, "head": head, "payload_sha256": receipt["payload_sha256"], "receipt_sha256": sha_file(BANK / "canonical-receipt.json"), "invocations": 1, "successes": 1, "replays": 0}
        write_bytes(latch_path, pretty_bytes(latch))
    return receipt


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("command", choices=["planning", "x1", "x2", "final", "seal", "receipt", "preflight", "canonical"])
    parser.add_argument("--stage")
    args = parser.parse_args()
    if args.command == "planning":
        build_planning()
    elif args.command == "x1":
        build_stage("x1")
    elif args.command == "x2":
        build_stage("x2")
    elif args.command == "final":
        build_final()
    elif args.command == "seal":
        build_seal_from_index()
    elif args.command == "receipt":
        if not args.stage:
            raise ValueError("--stage is required")
        print(json.dumps(stage_receipt(args.stage), sort_keys=True))
    elif args.command == "preflight":
        print(json.dumps(preflight(), sort_keys=True))
    elif args.command == "canonical":
        print(json.dumps(canonical_once(), sort_keys=True))


if __name__ == "__main__":
    main()
