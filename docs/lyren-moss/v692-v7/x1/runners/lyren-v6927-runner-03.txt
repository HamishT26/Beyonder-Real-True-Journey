"use strict";

const fs = require("fs");
const Module = require("module");
const path = require("path");

function loadTextModule(file) {
  const instance = new Module(file, module);
  instance.filename = file;
  instance.paths = Module._nodeModulePaths(path.dirname(file));
  instance._compile(fs.readFileSync(file, "utf8"), file);
  return instance.exports;
}

const core = loadTextModule(path.join(__dirname, "../lib/fca-core.txt"));
const ALLOWED = ["attribute_closure","formal_concept_check"];

function run(request) {
  if (!request || !ALLOWED.includes(request.op)) return { ok: false, error: "operation_outside_runner", external_credit: false, value: null };
  return core.safeExecute(request);
}

module.exports = { ALLOWED, run, runner: "lyren-v6927-runner-03" };
