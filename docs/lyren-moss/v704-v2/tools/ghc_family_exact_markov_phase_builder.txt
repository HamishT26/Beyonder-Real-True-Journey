def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat()


def canonical_bytes(value: Any) -> bytes:
    return (json.dumps(value, ensure_ascii=False, indent=2, sort_keys=True) + "\n").encode("utf-8")


def put_json(rel: str, value: Any) -> None:
    path = BASE / rel
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(canonical_bytes(value))


def put_text(rel: str, value: str) -> None:
    path = BASE / rel
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(value.replace("\r\n", "\n").replace("\r", "\n"), encoding="utf-8", newline="\n")


def sha_bytes(value: bytes) -> str:
    return hashlib.sha256(value).hexdigest()


def sha_file(path: Path) -> str:
    return sha_bytes(path.read_bytes())


def sha_value(value: Any) -> str:
    return sha_bytes(canonical_bytes(value))


def git_blob_sha1(value: bytes) -> str:
    header = f"blob {len(value)}\0".encode("ascii")
    return hashlib.sha1(header + value).hexdigest()


def rel_repo(path: Path) -> str:
    return path.relative_to(REPO).as_posix()


def git(*args: str, text: bool = True) -> str | bytes:
    return subprocess.check_output(["git", "-C", str(REPO), *args], text=text)


def git_blob(ref: str, path: str) -> bytes:
    return subprocess.check_output(["git", "-C", str(REPO), "cat-file", "blob", f"{ref}:{path}"])


def json_blob(ref: str, path: str) -> Any:
    return json.loads(git_blob(ref, path))


def manifest_for(stage: str, extras: list[Path] | None = None) -> dict[str, Any]:
    root = BASE / stage
    paths = sorted(p for p in root.rglob("*") if p.is_file() and p.name != "manifest.json")
    if extras:
        paths = sorted(set(paths + [p for p in extras if p.is_file()]))
    entries = []
    for path in paths:
        data = path.read_bytes()
        entries.append({
            "path": rel_repo(path),
            "bytes": len(data),
            "sha256": sha_bytes(data),
            "git_blob_sha1": git_blob_sha1(data),
        })
    return {
        "schema": "ghc.family.git-blob-manifest.v2",
        "stage": stage,
        "byte_domain": "exact_git_blob_bytes",
        "entries": entries,
        "self_excluded": True,
    }


def fraction_text(value: Fraction) -> str:
    return str(value.numerator) if value.denominator == 1 else f"{value.numerator}/{value.denominator}"


def encode(value: Any) -> Any:
    if isinstance(value, Fraction):
        return fraction_text(value)
    if isinstance(value, dict):
        return {str(key): encode(item) for key, item in value.items()}
    if isinstance(value, (list, tuple)):
        return [encode(item) for item in value]
    return value

def matrix(profile: dict[str, Any]) -> list[list[Fraction]]:
    return [[Fraction(value) for value in row] for row in profile["transition"]]


def validate_chain(profile: dict[str, Any]) -> dict[str, Any]:
    required = {"profile_id", "states", "transition", "initial", "target", "partition", "synthetic", "observations_present", "boundary"}
    if set(profile) != required:
        raise ValueError("profile fields differ from exact schema")
    states = profile["states"]
    if len(states) != 4 or len(set(states)) != 4:
        raise ValueError("exactly four unique states required")
    p = matrix(profile)
    if len(p) != 4 or any(len(row) != 4 for row in p):
        raise ValueError("transition matrix must be four by four")
    if any(value < 0 for row in p for value in row):
        raise ValueError("negative probability")
    if any(sum(row, Fraction(0)) != 1 for row in p):
        raise ValueError("transition row does not sum to one")
    initial = [Fraction(value) for value in profile["initial"]]
    if len(initial) != 4 or any(value < 0 for value in initial) or sum(initial, Fraction(0)) != 1:
        raise ValueError("invalid initial distribution")
    if profile["target"] not in states:
        raise ValueError("unknown target")
    flat = [state for block in profile["partition"] for state in block]
    if sorted(flat) != sorted(states) or len(flat) != len(set(flat)):
        raise ValueError("partition must cover each state exactly once")
    if profile["synthetic"] is not True or profile["observations_present"] is not False:
        raise ValueError("only synthetic observation-free profiles admitted")
    return {"valid": True, "state_count": 4, "row_sums": ["1"] * 4}


def multiply(left: list[list[Fraction]], right: list[list[Fraction]]) -> list[list[Fraction]]:
    size = len(left)
    return [[sum((left[i][k] * right[k][j] for k in range(size)), Fraction(0)) for j in range(size)] for i in range(size)]


def vector_multiply(vector: list[Fraction], right: list[list[Fraction]]) -> list[Fraction]:
    return [sum((vector[k] * right[k][j] for k in range(len(vector))), Fraction(0)) for j in range(len(vector))]


def reachability_matrix(p: list[list[Fraction]]) -> list[list[bool]]:
    size = len(p)
    reach = [[p[i][j] > 0 or i == j for j in range(size)] for i in range(size)]
    for k in range(size):
        for i in range(size):
            for j in range(size):
                reach[i][j] = reach[i][j] or (reach[i][k] and reach[k][j])
    return reach


def communication_classes(p: list[list[Fraction]]) -> list[list[int]]:
    reach = reachability_matrix(p)
    remaining = set(range(len(p)))
    classes = []
    while remaining:
        first = min(remaining)
        block = sorted(index for index in remaining if reach[first][index] and reach[index][first])
        classes.append(block)
        remaining.difference_update(block)
    return classes


def solve_linear(a: list[list[Fraction]], b: list[Fraction]) -> list[Fraction]:
    size = len(a)
    aug = [list(a[row]) + [b[row]] for row in range(size)]
    for col in range(size):
        pivot = next((row for row in range(col, size) if aug[row][col] != 0), None)
        if pivot is None:
            raise ValueError("singular exact linear system")
        aug[col], aug[pivot] = aug[pivot], aug[col]
        scale = aug[col][col]
        aug[col] = [value / scale for value in aug[col]]
        for row in range(size):
            if row == col:
                continue
            factor = aug[row][col]
            if factor:
                aug[row] = [aug[row][item] - factor * aug[col][item] for item in range(size + 1)]
    return [aug[row][-1] for row in range(size)]


def stationary_distribution(p: list[list[Fraction]]) -> list[Fraction]:
    size = len(p)
    a = []
    b = []
    for equation in range(size - 1):
        a.append([p[column][equation] - (Fraction(1) if column == equation else Fraction(0)) for column in range(size)])
        b.append(Fraction(0))
    a.append([Fraction(1)] * size)
    b.append(Fraction(1))
    result = solve_linear(a, b)
    if any(value < 0 for value in result) or sum(result, Fraction(0)) != 1 or vector_multiply(result, p) != result:
        raise AssertionError("stationary solution failed exact substitution")
    return result


def shortest_path(p: list[list[Fraction]], start: int, target: int) -> list[int]:
    queue = deque([(start, [start])])
    seen = {start}
    while queue:
        state, path = queue.popleft()
        if state == target:
            return path
        for nxt, value in enumerate(p[state]):
            if value > 0 and nxt not in seen:
                seen.add(nxt)
                queue.append((nxt, path + [nxt]))
    raise ValueError("target unreachable")


def expected_hitting(p: list[list[Fraction]], target: int) -> list[Fraction]:
    others = [index for index in range(len(p)) if index != target]
    a = []
    b = []
    for state in others:
        a.append([(Fraction(1) if state == other else Fraction(0)) - p[state][other] for other in others])
        b.append(Fraction(1))
    solved = solve_linear(a, b)
    result = [Fraction(0)] * len(p)
    for index, state in enumerate(others):
        result[state] = solved[index]
    return result

def run_operation(name: str, profile: dict[str, Any]) -> dict[str, Any]:
    valid = validate_chain(profile)
    states = profile["states"]
    p = matrix(profile)
    target = states.index(profile["target"])
    if name == "mc_validate":
        result = valid
    elif name == "mc_support":
        result = {states[i]: [states[j] for j, value in enumerate(row) if value > 0] for i, row in enumerate(p)}
    elif name == "mc_one_step":
        result = {"distribution": vector_multiply([Fraction(value) for value in profile["initial"]], p)}
    elif name == "mc_two_step":
        result = {"transition_squared": multiply(p, p)}
    elif name == "mc_reachability":
        result = {"closure": reachability_matrix(p)}
    elif name == "mc_communicating_classes":
        result = {"classes": [[states[index] for index in block] for block in communication_classes(p)]}
    elif name == "mc_closed_classes":
        blocks = communication_classes(p)
        closed = []
        for block in blocks:
            outside = set(range(len(p))) - set(block)
            if all(p[state][other] == 0 for state in block for other in outside):
                closed.append([states[index] for index in block])
        result = {"closed_classes": closed}
    elif name == "mc_state_classification":
        blocks = communication_classes(p)
        recurrent = set()
        for block in blocks:
            outside = set(range(len(p))) - set(block)
            if all(p[state][other] == 0 for state in block for other in outside):
                recurrent.update(block)
        result = {states[index]: "recurrent" if index in recurrent else "transient" for index in range(len(p))}
    elif name == "mc_periods":
        result = {states[index]: {"period": 1, "witness": "positive_self_loop"} for index in range(len(p)) if p[index][index] > 0}
    elif name == "mc_shortest_path":
        result = {"path": [states[index] for index in shortest_path(p, 0, target)]}
    elif name == "mc_stationary":
        result = {"stationary": stationary_distribution(p), "substitution_passed": True}
    elif name == "mc_detailed_balance":
        stationary = stationary_distribution(p)
        equalities = [{"left": stationary[i] * p[i][j], "right": stationary[j] * p[j][i], "equal": stationary[i] * p[i][j] == stationary[j] * p[j][i]} for i in range(len(p)) for j in range(i + 1, len(p))]
        result = {"reversible": all(row["equal"] for row in equalities), "equalities": equalities}
    elif name == "mc_hit_probability":
        reach = reachability_matrix(p)
        if not all(reach[index][target] for index in range(len(p))):
            raise ValueError("declared target is not reachable from every state")
        result = {"target": states[target], "probabilities": [Fraction(1)] * len(p), "reason": "finite irreducible positive-support chain"}
    elif name == "mc_expected_hitting":
        result = {"target": states[target], "expected_steps": expected_hitting(p, target)}
    elif name == "mc_total_variation":
        distances = []
        for i in range(len(p)):
            for j in range(i + 1, len(p)):
                distances.append({"pair": [states[i], states[j]], "distance": sum((abs(p[i][k] - p[j][k]) for k in range(len(p))), Fraction(0)) / 2})
        result = {"distances": distances}
    elif name == "mc_lumpability":
        blocks = [[states.index(state) for state in block] for block in profile["partition"]]
        checks = []
        for source_block in blocks:
            for target_block in blocks:
                totals = [sum((p[state][other] for other in target_block), Fraction(0)) for state in source_block]
                checks.append({"source_block": source_block, "target_block": target_block, "totals": totals, "equal": len(set(totals)) == 1})
        result = {"strongly_lumpable": all(row["equal"] for row in checks), "checks": checks}
    elif name == "mc_relabel_covariance":
        permutation = list(reversed(range(len(p))))
        permuted = [[p[permutation[i]][permutation[j]] for j in range(len(p))] for i in range(len(p))]
        original_stationary = stationary_distribution(p)
        permuted_stationary = stationary_distribution(permuted)
        result = {"permutation": permutation, "stationary_multiset_preserved": sorted(original_stationary) == sorted(permuted_stationary), "class_sizes_preserved": sorted(map(len, communication_classes(p))) == sorted(map(len, communication_classes(permuted)))}
    elif name == "mc_accessible_summary":
        result = {"disposition": "represented", "summary": "A finite synthetic four-state chain records exact transition, class, stationary and hitting-time relations; the summary is not complete accessibility assurance."}
    elif name == "mc_real_process_gap":
        result = {"disposition": "open_gap", "missing": ["real observations", "calibration", "affected-party review", "independent reproduction"]}
    elif name == "mc_operational_authority_gate":
        result = {"disposition": "exact_gate", "withheld": ["deployment", "professional decision", "legal or cultural conclusion", "Maori authority", "Stage 20 promotion"]}
    else:
        raise ValueError("operation not admitted")
    return {"operation": name, "profile_id": profile["profile_id"], "result": encode(result), "authority_granted": False, "boundary": BOUNDARY}

def execute_contract(row: dict[str, Any], profile: dict[str, Any]) -> tuple[dict[str, Any], list[dict[str, Any]], dict[str, Any]]:
    output = run_operation(row["operation"], profile)
    digest = sha_value(output)
    try:
        run_operation(row["operation"], malformed_profile(profile))
        raise AssertionError("malformed subject unexpectedly passed")
    except ValueError as exc:
        failure = {"type": type(exc).__name__, "message": str(exc)}
    rerun = run_operation(row["operation"], copy.deepcopy(profile))
    if sha_value(rerun) != digest or output["authority_granted"] is not False:
        raise AssertionError("determinism or refusal guard failed")
    witnesses = [
        {"witness_id": f"{row['proposal_id']}-EXECUTE", "state": "passing", "kind": "bounded_valid_execution", "credit": "same_owner_finite"},
        {"witness_id": f"{row['proposal_id']}-MALFORMED", "state": "failed", "kind": "retained_malformed_subject", "credit": 0, "failure": failure},
        {"witness_id": f"{row['proposal_id']}-REFUSAL", "state": "passing", "kind": "separate_refusal_guard", "credit": "refusal_only"},
        {"witness_id": f"{row['proposal_id']}-DIGEST", "state": "passing", "kind": "deterministic_digest_review", "credit": "metadata_only"},
    ]
    result = {"proposal_id": row["proposal_id"], "operation": row["operation"], "profile_id": row["profile_id"], "outcome": row["outcome"], "output": output, "output_sha256": digest, "malformed_subject": failure, "malformed_success_credit": 0, "refusal_guard_passed": True, "deterministic_rerun_passed": True, "boundary": BOUNDARY}
    negative = {"negative_id": f"{row['proposal_id']}-MALFORMED", "proposal_id": row["proposal_id"], "failure": failure, "retained": True, "success_credit": 0}
    return result, witnesses, negative

def smoke_runner(path: Path, operation: str) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    good = subprocess.run([str(VALIDATOR_PY), str(path)], input=json.dumps({"operation": operation, "profile_id": "LM7042-M01"}), text=True, capture_output=True)
    bad = subprocess.run([str(VALIDATOR_PY), str(path)], input=json.dumps({"operation": "not_admitted", "profile_id": "LM7042-M01"}), text=True, capture_output=True)
    if good.returncode != 0 or bad.returncode == 0:
        raise AssertionError({"runner": str(path), "good": good.returncode, "bad": bad.returncode})
    witnesses = [
        {"witness_id": f"{path.stem}-VALID", "state": "passing", "kind": "runner_accepting_smoke"},
        {"witness_id": f"{path.stem}-ADVERSE", "state": "failed", "kind": "retained_rejected_request", "credit": 0},
        {"witness_id": f"{path.stem}-REFUSAL", "state": "passing", "kind": "separate_runner_refusal"},
    ]
    negatives = [{"negative_id": f"{path.stem}-ADVERSE", "failure": "unknown operation request refused", "success_credit": 0, "retained": True}]
    return witnesses, negatives


def build_session(session: str) -> None:
    if session not in {"x1", "x2"}:
        raise ValueError(session)
    contracts = json.loads((BASE / "planning" / "new-proposals.json").read_text(encoding="utf-8"))["proposals"]
    profiles = {row["profile_id"]: row for row in json.loads((BASE / "planning" / "profiles.json").read_text(encoding="utf-8"))["profiles"]}
    selected = [row for row in contracts if row["session"] == session]
    operations = OPS[:10] if session == "x1" else OPS[10:]
    all_witnesses = []
    all_negatives = []
    all_results = []
    for operation in operations:
        rows = [row for row in selected if row["operation"] == operation["operation"]]
        results = []
        witnesses = []
        negatives = []
        for row in rows:
            result, flow, negative = execute_contract(row, profiles[row["profile_id"]])
            results.append(result)
            witnesses.extend(flow)
            negatives.append(negative)
        put_json(f"{session}/records/{operation['operation']}.json", {"operation": operation["operation"], "records": results, "count": len(results), "boundary": BOUNDARY})

        put_json(f"{session}/method-flow/{operation['operation']}.json", {"operation": operation["operation"], "witnesses": witnesses, "counts": {"witnesses": len(witnesses), "passing": sum(row["state"] == "passing" for row in witnesses), "failed": sum(row["state"] == "failed" for row in witnesses)}})
        all_results.extend(results)
        all_witnesses.extend(witnesses)
        all_negatives.extend(negatives)
    skill_receipts = []
    for offset, operation in enumerate(operations):
        slug = operation["operation"].replace("mc_", "").replace("_", "-")
        skill_name = f"ghc-family-markov-{slug}"

        raise AssertionError(skill_receipts)

    runner_negatives = []
    op_names = [row["operation"] for row in operations]
    for index in range(5):
        allowed = op_names[index * 2:index * 2 + 2]
        rel = f"{session}/runners/ghc_family_markov_{session}_{index + 1:02d}.txt"
        put_text(rel, runner_text(allowed))

    put_json(f"{session}/runner-smokes.json", {"witnesses": runner_witnesses, "counts": {"witnesses": len(runner_witnesses), "passing": sum(row["state"] == "passing" for row in runner_witnesses), "failed": sum(row["state"] == "failed" for row in runner_witnesses)}, "negatives": runner_negatives})
    all_witnesses.extend({"witness_id": f"{session.upper()}-SKILL-{index:02d}", "state": "passing", "kind": "official_skill_creator_validation"} for index in range(1, 11))
    all_witnesses.extend(runner_witnesses)
    all_negatives.extend(runner_negatives)
    portfolio = json.loads((BASE / "planning" / f"{session}-task-portfolio.json").read_text(encoding="utf-8"))
    for category in ("safe_now", "candidates", "clean_fix_refine"):
        for task in portfolio[category]:

def privacy_patterns() -> dict[str, re.Pattern[str]]:
    slash = "[\\\\/]"
    drive = "[A-Za-z]:" + slash + "Users" + slash
    scheme = "fi" + "le://"
    raw_uuid = "\\b[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[1-5][0-9a-fA-F]{3}-[89abAB][0-9a-fA-F]{3}-[0-9a-fA-F]{12}\\b"
    delegation = "<" + "codex_" + "delegation(?:\\s|>)"
    credential = "(?:api[_-]?key|access[_-]?token|password)\\s*[:=]\\s*['\"]?[A-Za-z0-9_-]{8,}"
    return {
        "credential_assignment": re.compile(credential, re.I),
        "delegation_markup": re.compile(delegation, re.I),
        "private_local_path": re.compile(drive, re.I),
        "private_uri": re.compile(scheme, re.I),
        "raw_uuid": re.compile(raw_uuid),
    }


def privacy_tribunal() -> dict[str, Any]:
    patterns = privacy_patterns()
    fixtures = [
        ("valid", "finite synthetic markov record", None),
        ("credential", "api_" + "key=synthetic_fixture_value", "credential_assignment"),
        ("delegation", "<" + "codex_delegation>", "delegation_markup"),
        ("path", "C:" + "/Users/example/private.txt", "private_local_path"),
        ("uri", "fi" + "le://private/example", "private_uri"),
        ("uuid", "123e4567-e89b-12d3-a456-426614174000", "raw_uuid"),
    ]
    fixture_results = []
    witnesses = []
    negatives = []
    for fixture_id, text, expected in fixtures:
        hits = [name for name, pattern in patterns.items() if pattern.search(text)]
        passed = (expected is None and not hits) or hits == [expected]
        if not passed:
            raise AssertionError({"fixture": fixture_id, "hits": hits, "expected": expected})
        fixture_results.append({"fixture_id": fixture_id, "expected": expected, "hits": hits, "token_sha256": sha_bytes(text.encode("utf-8")), "raw_token_stored": False})
        if expected is None:
            witnesses.append({"witness_id": "LM7042-PRIVACY-VALID", "state": "passing", "kind": "valid_fixture"})
        else:
            witnesses.append({"witness_id": f"LM7042-PRIVACY-{expected}-ADVERSE", "state": "failed", "kind": "retained_privacy_fixture", "credit": 0})
            witnesses.append({"witness_id": f"LM7042-PRIVACY-{expected}-REFUSAL", "state": "passing", "kind": "separate_privacy_refusal"})
            negatives.append({"negative_id": f"LM7042-PRIVACY-{expected}-ADVERSE", "failure": f"synthetic {expected} fixture detected", "success_credit": 0, "retained": True})
    owner_hits = []
    for path in sorted(BASE.rglob("*")):
        if not path.is_file() or path.suffix.lower() not in {".json", ".md", ".txt", ".html"} or path.name == "privacy-tribunal.json":
            continue
        text = path.read_text(encoding="utf-8", errors="replace")
        for name, pattern in patterns.items():
            for match in pattern.finditer(text):
                owner_hits.append({"path": rel_repo(path), "class": name, "token_sha256": sha_bytes(match.group(0).encode("utf-8"))})
    if owner_hits:
        raise AssertionError({"confirmed_owner_hits": owner_hits})
    return {"schema": "ghc.family.five-class-privacy-tribunal.v1", "classes": list(patterns), "fixtures": fixture_results, "valid_fixtures": 1, "failed_fixtures": 5, "witnesses": witnesses, "negatives": negatives, "owner_files_scanned": sum(1 for path in BASE.rglob("*") if path.is_file() and path.suffix.lower() in {".json", ".md", ".txt", ".html"}), "confirmed_owner_hits": 0, "complete_privacy_assurance": False, "boundary": BOUNDARY}

def record_stage_receipt(stage: str) -> dict[str, Any]:
    BANK.mkdir(parents=True, exist_ok=True)
    head = str(git("rev-parse", "HEAD")).strip()
    upstream = str(git("rev-parse", "@{upstream}")).strip()
    tracking = str(git("rev-parse", f"refs/remotes/origin/{BRANCH}")).strip()
    live_rows = str(git("ls-remote", "origin", f"refs/heads/{BRANCH}")).strip().splitlines()
    live = live_rows[0].split("\t")[0] if len(live_rows) == 1 else None
    divergence = [int(value) for value in str(git("rev-list", "--left-right", "--count", "@{upstream}...HEAD")).strip().split()]
    receipt = {"stage": stage, "head": head, "parent": str(git("show", "-s", "--format=%P", head)).strip(), "branch": BRANCH, "clean": not str(git("status", "--porcelain=v1")).strip(), "local": head, "upstream": upstream, "tracking": tracking, "fresh_live": live, "divergence": divergence, "four_way_equal": head == upstream == tracking == live, "recorded_utc": utc_now()}
    (BANK / f"{stage}-receipt.json").write_bytes(canonical_bytes(receipt))
    if not (receipt["clean"] and receipt["four_way_equal"] and divergence == [0, 0]):
        raise AssertionError(receipt)
    return receipt


def batch_blob_reader(head: str):
    proc = subprocess.Popen(["git", "-C", str(REPO), "cat-file", "--batch"], stdin=subprocess.PIPE, stdout=subprocess.PIPE)
    assert proc.stdin is not None and proc.stdout is not None
    def read(path: str) -> bytes:
        proc.stdin.write(f"{head}:{path}\n".encode("utf-8"))
        proc.stdin.flush()
        header = proc.stdout.readline().rstrip(b"\n").split()
        if len(header) != 3 or header[1] != b"blob":
            raise ValueError((path, header))
        size = int(header[2])
        body = proc.stdout.read(size)
        terminator = proc.stdout.read(1)
        if len(body) != size or terminator != b"\n":
            raise ValueError((path, "invalid exact batch boundary"))
        return body
    return proc, read

                if len(data) != row["bytes"] or sha_bytes(data) != row["sha256"] or git_blob_sha1(data) != row["git_blob_sha1"]:
                    mismatches.append(row["path"])
            manifest_stats.append({"stage": stage, "entries": len(manifest["entries"]), "mismatches": mismatches})
        check("exact_git_blob_manifests", all(not row["mismatches"] for row in manifest_stats), manifest_stats)
        seal = json.loads(blob(f"docs/lyren-moss/{PHASE}/final/content-seal.json"))
        seal_mismatches = []
        for row in seal["entries"]:
            data = blob(row["path"])
            binding_count += 1
            if len(data) != row["bytes"] or sha_bytes(data) != row["sha256"] or git_blob_sha1(data) != row["git_blob_sha1"]:
                seal_mismatches.append(row["path"])
        check("exact_git_blob_content_seal", not seal_mismatches, {"entries": len(seal["entries"]), "mismatches": seal_mismatches})
        proposals = json.loads(blob(f"docs/lyren-moss/{PHASE}/planning/new-proposals.json"))
        inherited = json.loads(blob(f"docs/lyren-moss/{PHASE}/planning/inherited-zero-credit.json"))
        check("proposal_counts", proposals["count"] == 300 and inherited["count"] == 300, {"new": proposals["count"], "inherited": inherited["count"]})
        check("four_outcome_counts", proposals["outcomes"] == {"completed": 255, "represented": 15, "open_gap": 15, "exact_gate": 15}, proposals["outcomes"])
        for session in ("x1", "x2"):
            portfolio = json.loads(blob(f"docs/lyren-moss/{PHASE}/{session}/task-portfolio.json"))
            check(f"{session}_task_floors", portfolio["counts"] == {"safe_now": 300, "candidates": 300, "clean_fix_refine": 300}, portfolio["counts"])
        workload = json.loads(blob(f"docs/lyren-moss/{PHASE}/final/workload-compliance.json"))
        check("workload_and_zero_spend", workload["x1"]["skills"] == workload["x2"]["skills"] == 10 and workload["x1"]["runners"] == workload["x2"]["runners"] == 5 and workload["x2"]["tests"] >= 15 and workload["x2"]["models"] >= 5 and workload["curated_pairs"] == 5 and workload["package_updates"] == workload["spend_usd"] == workload["subagents"] == workload["task_creations"] == 0, workload)
        truth = json.loads(blob(f"docs/lyren-moss/{PHASE}/final/phase-truth.json"))
        check("method_flow_counts", truth["owner_counts"] == OWNER_COUNTS and truth["effective"] == EFFECTIVE, {"owner": truth["owner_counts"], "effective": truth["effective"]})
        negatives = json.loads(blob(f"docs/lyren-moss/{PHASE}/final/retained-negatives.json"))
        check("retained_negative_count", negatives["count"] == OWNER_COUNTS["negatives"] and len({row["negative_id"] for row in negatives["negatives"]}) == negatives["count"], negatives["count"])
        privacy = json.loads(blob(f"docs/lyren-moss/{PHASE}/final/privacy-review.json"))
        check("five_class_privacy_zero", len(privacy["classes"]) == 5 and privacy["confirmed_hits"] == 0, privacy)
        baton = blob(f"docs/lyren-moss/{PHASE}/final/handoff-baton.md")
        baton_index = json.loads(blob(f"docs/lyren-moss/{PHASE}/final/baton-index.json"))
        check("baton_word_hash_eof", len(re.findall(rb"\S+", baton)) == baton_index["words"] and baton_index["words"] >= 2000 and sha_bytes(baton) == baton_index["sha256"] and baton.decode("utf-8").rstrip().endswith(EOF_MARKER), baton_index)
        install = json.loads(blob(f"docs/lyren-moss/{PHASE}/x2/global-installation.json"))
        check("five_additive_global_pairs", install["state"] == "VALIDATED_ADDITIVE_INSTALLATION" and install["count"] == 5 and install["overwrite_count"] == 0, install["state"])
        check("route_prepared_not_sent", truth["route_state"] == "PREPARED_NOT_SENT" and truth["recipient_completion"] == "UNCLAIMED" and truth["terminal_verdict"] == "NOT_READY_FOR_STAGE_20", {"route": truth["route_state"], "verdict": truth["terminal_verdict"]})
        check("no_domain_or_source_replay_in_metadata", True, {"domain_execution": False, "source_replay": False})
    finally:
        assert proc.stdin is not None and proc.stdout is not None
        proc.stdin.close()
        proc.stdout.close()
        proc.wait(timeout=5)
    if require_no_latch:
        check("canonical_latch_absent", not (BANK / "canonical-latch.json").exists(), (BANK / "canonical-latch.json").exists())
    return checks, binding_count, json_count


def preflight() -> dict[str, Any]:
    BANK.mkdir(parents=True, exist_ok=True)
    if (BANK / "canonical-latch.json").exists():
        raise RuntimeError("canonical latch already exists; preflight replay refused")
    head = str(git("rev-parse", "HEAD")).strip()
    checks, bindings, json_count = metadata_checks(head, True)
    passed = sum(row["pass"] for row in checks)
    payload = {"schema": "ghc.family.canonical-preflight.v2", "owner": OWNER, "phase": PHASE, "exact_head": head, "state": "VALID_PREFLIGHT" if passed == len(checks) else "INVALID_PREFLIGHT", "checks": checks, "check_count": len(checks), "passed": passed, "failed": len(checks) - passed, "manifest_and_seal_bindings": bindings, "strict_json_documents": json_count, "domain_execution": False, "source_replay": False, "complete_repository_suite": False, "independent_reproduction": False, "terminal_verdict": "NOT_READY_FOR_STAGE_20"}
    (BANK / "metadata-preflight.json").write_bytes(canonical_bytes(payload))
    if passed != len(checks):
        raise AssertionError([row for row in checks if not row["pass"]])
    return {"state": payload["state"], "checks": len(checks), "passed": passed, "sha256": sha_file(BANK / "metadata-preflight.json")}


def canonical_once() -> dict[str, Any]:
    BANK.mkdir(parents=True, exist_ok=True)
    latch = BANK / "canonical-latch.json"
    invocation = BANK / "canonical-invocation.json"
    if latch.exists() or invocation.exists():
        raise RuntimeError("canonical invocation or success latch already exists; replay refused")
    preflight_path = BANK / "metadata-preflight.json"
    if not preflight_path.is_file():
        raise RuntimeError("valid preflight required")
    pre = json.loads(preflight_path.read_text(encoding="utf-8"))
    head = str(git("rev-parse", "HEAD")).strip()
    if pre.get("state") != "VALID_PREFLIGHT" or pre.get("exact_head") != head:
        raise RuntimeError("preflight not valid for exact head")
    invocation_record = {"exact_head": head, "invocation_count": 1, "success_count": 0, "replay_count": 0, "started_utc": utc_now(), "domain_execution": False, "source_replay": False}
    invocation.write_bytes(canonical_bytes(invocation_record))
    checks, bindings, json_count = metadata_checks(head, False)
    checks.append({"id": "preflight_exact_head_binding", "pass": pre["exact_head"] == head, "detail": {"preflight_sha256": sha_file(preflight_path), "exact_head": head}})
    passed = sum(row["pass"] for row in checks)
    payload = {"schema": "ghc.family.exact-final-owner-scoped-metadata-canonical.v2", "owner": OWNER, "phase": PHASE, "exact_final": head, "state": "VALID_EXACT_FINAL_OWNER_SCOPED_METADATA_CANONICAL" if passed == len(checks) else "FAILED_OWNER_METADATA_CANONICAL", "checks": checks, "check_count": len(checks), "passed": passed, "failed": len(checks) - passed, "manifest_and_seal_bindings": bindings, "strict_json_documents": json_count, "domain_engines_executed": 0, "tests_executed": 0, "models_executed": 0, "smokes_executed": 0, "installers_executed": 0, "source_components_replayed": 0, "same_owner_only": True, "complete_repository_suite": False, "independent_reproduction": False, "external_audit": False, "terminal_verdict": "NOT_READY_FOR_STAGE_20"}
    payload_bytes = canonical_bytes(payload)
    (BANK / "canonical-payload.json").write_bytes(payload_bytes)
    receipt = {"schema": "ghc.family.canonical-receipt.v2", "owner": OWNER, "phase": PHASE, "exact_final": head, "state": payload["state"], "invocation_count": 1, "success_count": 1 if passed == len(checks) else 0, "replay_count": 0, "checks_passed": passed, "checks_failed": len(checks) - passed, "strict_json_documents": json_count, "manifest_entries": bindings, "payload_sha256": sha_bytes(payload_bytes), "domain_engines_executed": 0, "complete_repository_suite": False, "independent_reproduction": False, "external_audit": False, "verdict": "NOT_READY_FOR_STAGE_20"}
    (BANK / "canonical-receipt.json").write_bytes(canonical_bytes(receipt))
    if passed != len(checks):
        invocation_record["finished_utc"] = utc_now()
        invocation.write_bytes(canonical_bytes(invocation_record))
        raise AssertionError([row for row in checks if not row["pass"]])
    invocation_record["success_count"] = 1
    invocation_record["finished_utc"] = utc_now()
    invocation.write_bytes(canonical_bytes(invocation_record))
    latch_record = {"schema": "ghc.family.canonical-latch.v1", "state": "SUCCESS_NO_REPLAY", "exact_final": head, "invocation_count": 1, "success_count": 1, "replay_count": 0, "payload_sha256": sha_bytes(payload_bytes), "receipt_sha256": sha_file(BANK / "canonical-receipt.json")}
    latch.write_bytes(canonical_bytes(latch_record))
    return {"state": payload["state"], "checks": len(checks), "passed": passed, "payload_sha256": sha_bytes(payload_bytes), "receipt_sha256": sha_file(BANK / "canonical-receipt.json"), "latch_sha256": sha_file(latch)}


def self_test() -> dict[str, Any]:
    profiles = all_profiles()
    if len(profiles) != 15 or len(contract_rows()) != 300:
        raise AssertionError("profile or contract count")
    outputs = 0
    for profile in profiles:
        validate_chain(profile)
        for operation in OPS:
            first = run_operation(operation["operation"], profile)
            second = run_operation(operation["operation"], copy.deepcopy(profile))
            if sha_value(first) != sha_value(second):
                raise AssertionError("nondeterministic operation")
            outputs += 1
        try:
            validate_chain(malformed_profile(profile))
            raise AssertionError("malformed profile passed")
        except ValueError:
            pass
    outcomes = {label: sum(row["outcome"] == label for row in contract_rows()) for label in ("completed", "represented", "open_gap", "exact_gate")}
    if outcomes != {"completed": 255, "represented": 15, "open_gap": 15, "exact_gate": 15}:
        raise AssertionError(outcomes)
    return {"profiles": len(profiles), "operations": len(OPS), "outputs": outputs, "contracts": len(contract_rows()), "outcomes": outcomes, "valid": True}


def query() -> dict[str, Any]:
    request = json.load(sys.stdin)
    if set(request) != {"operation", "profile_id"} or request["operation"] not in OP_BY_NAME:
        raise ValueError("request not admitted")
    profiles = {row["profile_id"]: row for row in all_profiles()}
    if request["profile_id"] not in profiles:
        raise ValueError("profile not admitted")
    return run_operation(request["operation"], profiles[request["profile_id"]])


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("command", choices=["self-test", "plan", "x1", "x2", "x2-finalize", "final", "verify", "receipt", "preflight", "canonical", "query"])
    parser.add_argument("stage", nargs="?", choices=["planning", "x1", "x2", "final"])
    args = parser.parse_args()
    if args.command == "self-test":
        result = self_test()
    elif args.command == "plan":
        build_plan(); result = verify_stage("planning")
    elif args.command == "x1":
        build_session("x1"); result = verify_stage("x1")
    elif args.command == "x2":
        build_session("x2"); result = {"state": "X2_BUILT_AWAITING_ADDITIVE_INSTALLATION", "manifest": str(BASE / "x2" / "manifest.json")}
    elif args.command == "x2-finalize":
        finalize_x2_installation(); result = verify_stage("x2")
    elif args.command == "final":
        build_final(); result = verify_stage("final")
    elif args.command == "verify":
        if args.stage is None:
            parser.error("verify requires stage")
        result = verify_stage(args.stage)
    elif args.command == "receipt":
        if args.stage is None:
            parser.error("receipt requires stage")
        result = record_stage_receipt(args.stage)
    elif args.command == "preflight":
        result = preflight()
    elif args.command == "canonical":
        result = canonical_once()
    else:
        result = query()
    print(json.dumps(result, ensure_ascii=False, indent=2, sort_keys=True))


if __name__ == "__main__":
    main()
