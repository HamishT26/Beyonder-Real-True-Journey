def sha_bytes(value: bytes) -> str:
    return hashlib.sha256(value).hexdigest()


def patterns() -> dict[str, re.Pattern[str]]:
    slash = "[\\\\/]"
    drive = "[A-Za-z]:" + slash + "Users" + slash
    scheme = "fi" + "le://"
    uuid_rule = "\\b[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[1-5][0-9a-fA-F]{3}-[89abAB][0-9a-fA-F]{3}-[0-9a-fA-F]{12}\\b"
    delegation = "<" + "codex_" + "delegation(?:\\s|>)"
    credential = "(?:api[_-]?key|access[_-]?token|password)\\s*[:=]\\s*['\"]?[A-Za-z0-9_-]{8,}"
    return {
        "credential_assignment": re.compile(credential, re.I),
        "delegation_markup": re.compile(delegation, re.I),
        "private_local_path": re.compile(drive, re.I),
        "private_uri": re.compile(scheme, re.I),
        "raw_uuid": re.compile(uuid_rule),
    }


def corrected_privacy_tribunal() -> dict[str, Any]:
    rules = patterns()
    synthetic_uuid = "123e4567" + "-e89b-12d3-a456-426614174000"
    fixtures = [
        ("valid", "finite synthetic markov record", None),
        ("credential", "api_" + "key=synthetic_fixture_value", "credential_assignment"),
        ("delegation", "<" + "codex_delegation>", "delegation_markup"),
        ("path", "C:" + "/Users/example/private.txt", "private_local_path"),
        ("uri", "fi" + "le://private/example", "private_uri"),
        ("uuid", synthetic_uuid, "raw_uuid"),
    ]
    fixture_results = []
    witnesses = []
    negatives = []
    for fixture_id, text, expected in fixtures:
        hits = [name for name, rule in rules.items() if rule.search(text)]
        if not ((expected is None and not hits) or hits == [expected]):
            raise AssertionError({"fixture": fixture_id, "hits": hits, "expected": expected})
        fixture_results.append({"fixture_id": fixture_id, "expected": expected, "hits": hits, "token_sha256": sha_bytes(text.encode("utf-8")), "raw_token_stored_in_receipt": False})
        if expected is None:
            witnesses.append({"witness_id": "LM7042-PRIVACY-VALID", "state": "passing", "kind": "valid_fixture"})
        else:
            witnesses.append({"witness_id": f"LM7042-PRIVACY-{expected}-ADVERSE", "state": "failed", "kind": "retained_privacy_fixture", "credit": 0})
            witnesses.append({"witness_id": f"LM7042-PRIVACY-{expected}-REFUSAL", "state": "passing", "kind": "separate_privacy_refusal"})
            negatives.append({"negative_id": f"LM7042-PRIVACY-{expected}-ADVERSE", "failure": f"synthetic {expected} fixture detected", "success_credit": 0, "retained": True})
    expected_fixture_sha = sha_bytes(synthetic_uuid.encode("utf-8"))
    candidates = []
    confirmed = []
    reviewed_false_positives = []
    for path in sorted(BASE.rglob("*")):
        if not path.is_file() or path.suffix.lower() not in {".json", ".md", ".txt", ".html"} or path.name in {"privacy-tribunal.json", "privacy-review.json"}:
            continue
        text = path.read_text(encoding="utf-8", errors="replace")
        for name, rule in rules.items():
            for match in rule.finditer(text):
                token_sha = sha_bytes(match.group(0).encode("utf-8"))
                row = {"path": API["rel_repo"](path), "class": name, "token_sha256": token_sha}
                candidates.append(row)
                if path == BUILDER and name == "raw_uuid" and token_sha == expected_fixture_sha:
                    reviewed_false_positives.append({**row, "review": "known synthetic adverse fixture literal in immutable frozen builder", "sensitive_material": False})
                else:
                    confirmed.append(row)
    if confirmed:
        raise AssertionError({"confirmed_owner_hits": confirmed})
    return {
        "schema": "ghc.family.five-class-privacy-tribunal.v1",
        "classes": list(rules),
        "fixtures": fixture_results,
        "valid_fixtures": 1,
        "failed_fixtures": 5,
        "witnesses": witnesses,
        "negatives": negatives,
        "owner_files_scanned": sum(1 for path in BASE.rglob("*") if path.is_file() and path.suffix.lower() in {".json", ".md", ".txt", ".html"}),
        "candidates": candidates,
        "reviewed_false_positives": reviewed_false_positives,
        "confirmed_owner_hits": 0,
        "complete_privacy_assurance": False,
        "correction": "The exact fixture digest remains visible as a candidate; only its sensitive-material classification changed after bounded review.",
        "boundary": BOUNDARY,
    }

        raise RuntimeError("successful pre-privacy x2 artifacts are incomplete; broad replay refused")

            "skills_replayed": False,
            "runners_replayed": False,
            "recovery": "Only the privacy dependency and downstream aggregation were resumed. The exact candidate digest remains retained and reviewed as a non-sensitive synthetic fixture.",
        }],

    })
    API["put_json"]("x2/global-installation.json", {"state": "PREPARED_NOT_INSTALLED", "count": 5, "overwrite_allowed": False, "package_updates": 0, "spend_usd": 0})
    API["put_json"]("x2/meta-tool-box-update.json", {"state": "PREPARED_NOT_PUBLISHED", "reference_name": API["META_REFERENCE"].name, "append_only": True})

    curated = json.loads((BASE / "x2" / "curated-validation.json").read_text(encoding="utf-8"))

    }
    flow["methods"] = 19
    API["put_json"]("x2/method-flow.json", flow)
    retained = json.loads((BASE / "x2" / "retained-negatives.json").read_text(encoding="utf-8"))
    retained["negatives"].extend(tests["negatives"])
    retained["negatives"].extend(privacy["negatives"])
    retained["negatives"].append({"negative_id": FAILURE_ID, "failure": "first x2 privacy aggregate failed on frozen synthetic fixture self-match", "success_credit": 0, "retained": True})
    retained["count"] = len(retained["negatives"])
    API["put_json"]("x2/retained-negatives.json", retained)
    API["put_json"]("x2/manifest.json", API["manifest_for"]("x2"))
    expected = {"witnesses": 721, "passing": 540, "failed": 181, "negatives": 181}
    if flow["counts"] != expected or retained["count"] != 181:
        raise AssertionError({"flow": flow["counts"], "negatives": retained["count"]})
    return {"state": "VALID_ISOLATED_PRIVACY_RECOVERY", "counts": flow["counts"], "domain_replayed": False, "tests_replayed": False, "models_replayed": False}


def finalize_x2() -> dict[str, Any]:
    rows = []
    for name, allowed in API["CURATED"].items():
        source_dir = BASE / "x2" / "curated" / name
        target_dir = API["GLOBAL_SKILLS"] / name
        bindings = []
        for filename in ("SKILL.md", "runner.txt"):
            source = source_dir / filename
            target = target_dir / filename
            bindings.append({"file": filename, "source_sha256": API["sha_file"](source), "installed_sha256": API["sha_file"](target) if target.is_file() else None, "equal": target.is_file() and source.read_bytes() == target.read_bytes()})
        validation_process = API["subprocess"].run([str(API["VALIDATOR_PY"]), str(API["QUICK_VALIDATE"]), str(target_dir)], text=True, capture_output=True)
        validation = {"path": f"codex-skill-prefix/{name}/SKILL.md", "exit_code": validation_process.returncode, "stdout": validation_process.stdout.strip(), "stderr": validation_process.stderr.strip(), "valid": validation_process.returncode == 0}
        request = {"operation": allowed[0], "profile_id": "LM7042-M01"}
        good = API["subprocess"].run([str(API["VALIDATOR_PY"]), str(target_dir / "runner.txt")], input=json.dumps(request), text=True, capture_output=True)
        bad = API["subprocess"].run([str(API["VALIDATOR_PY"]), str(target_dir / "runner.txt")], input=json.dumps({"operation": "not_admitted", "profile_id": "LM7042-M01"}), text=True, capture_output=True)
        valid = all(row["equal"] for row in bindings) and validation["valid"] and good.returncode == 0 and bad.returncode != 0
        rows.append({"name": name, "state": "installed_additively" if valid else "invalid_or_missing", "previously_absent": True, "bindings": bindings, "official_validation": validation, "accepting_smoke": good.returncode == 0, "rejecting_smoke": bad.returncode != 0, "valid": valid, "rollback": f"Remove only additive skill folder {name} after explicit rollback; retain phase source and receipts."})
    if not all(row["valid"] for row in rows):
        raise AssertionError(rows)
    meta_equal = API["META_REFERENCE"].is_file() and API["META_REFERENCE"].read_bytes() == (BASE / "x2" / "meta-tool-catalogue.json").read_bytes()
    if not meta_equal:
        raise AssertionError("meta-tool reference missing or unequal")
    API["put_json"]("x2/global-installation.json", {"state": "VALIDATED_ADDITIVE_INSTALLATION", "count": len(rows), "installations": rows, "all_previously_absent": True, "overwrite_count": 0, "package_updates": 0, "spend_usd": 0, "rollback_available": True, "boundary": BOUNDARY})
    API["put_json"]("x2/meta-tool-box-update.json", {"state": "PUBLISHED_APPEND_ONLY_REFERENCE", "reference_name": API["META_REFERENCE"].name, "reference_sha256": API["sha_file"](API["META_REFERENCE"]), "source_sha256": API["sha_file"](BASE / "x2" / "meta-tool-catalogue.json"), "byte_equal": True, "existing_guidance_rewritten": False})
    flow = json.loads((BASE / "x2" / "method-flow.json").read_text(encoding="utf-8"))
    if not any(row["witness_id"] == INSTALL_FAILURE_ID for row in flow["witnesses"]):
        flow["witnesses"].extend([
            {"witness_id": INSTALL_FAILURE_ID, "state": "failed", "kind": "retained_install_receipt_projection_failure", "credit": 0},
            {"witness_id": f"{INSTALL_FAILURE_ID}-RECOVERY", "state": "passing", "kind": "isolated_shared_prefix_path_projection_recovery", "credit": "recovery_only"},
        ])
        flow["methods"] += 1
        flow["counts"]["witnesses"] += 2
        flow["counts"]["passing"] += 1
        flow["counts"]["failed"] += 1
        flow["counts"]["negatives"] += 1
        API["put_json"]("x2/method-flow.json", flow)
        retained = json.loads((BASE / "x2" / "retained-negatives.json").read_text(encoding="utf-8"))
        retained["negatives"].append({"negative_id": INSTALL_FAILURE_ID, "failure": "first additive-install receipt wrapper attempted repository-relative projection of a shared-prefix path", "success_credit": 0, "retained": True})
        retained["count"] = len(retained["negatives"])
        API["put_json"]("x2/retained-negatives.json", retained)
        failures = json.loads((BASE / "x2" / "operational-failures.json").read_text(encoding="utf-8"))
        failures["failures"].append({"id": INSTALL_FAILURE_ID, "failure": "The first additive-install receipt wrapper rejected a shared-prefix SKILL.md because its validator receipt formatter required a repository-relative path.", "success_credit": 0, "installation_replayed": False, "domain_replayed": False, "recovery": "The already-installed bytes were re-read, validated and smoked with an explicit codex-skill-prefix label; no file was recopied or overwritten."})

    mismatches = []
    for row in manifest["entries"]:
        path = API["REPO"] / row["path"]

        if len(data) != row["bytes"] or API["sha_bytes"](data) != row["sha256"] or API["git_blob_sha1"](data) != row["git_blob_sha1"]:
            mismatches.append(row["path"])
    flow = json.loads((BASE / "x2" / "method-flow.json").read_text(encoding="utf-8"))
    install = json.loads((BASE / "x2" / "global-installation.json").read_text(encoding="utf-8"))
