function replaceOne(text, search, replacement, label) {
  const index = text.indexOf(search);
  if (index < 0) throw new Error(`missing literal transform: ${label}`);
  if (text.indexOf(search, index + search.length) >= 0) throw new Error(`non-unique literal transform: ${label}`);
  return text.slice(0, index) + replacement + text.slice(index + search.length);
}

function replacePattern(text, pattern, replacement, label) {
  let count = 0;
  const out = text.replace(pattern, (...args) => {
    count += 1;
    return typeof replacement === 'function' ? replacement(...args) : replacement;
  });
  if (count !== 1) throw new Error(`expected one pattern transform for ${label}, got ${count}`);
  return out;
}
