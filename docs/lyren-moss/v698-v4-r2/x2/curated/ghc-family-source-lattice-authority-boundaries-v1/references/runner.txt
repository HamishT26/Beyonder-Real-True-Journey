'use strict';
module.exports = {
  name: "ghc_family_source_lattice_authority_boundaries_v1",
  operations: ["sl_cut_candidates","sl_scene","sl_evidence_gap","sl_authority_gate"],
  run(core, requests) {
    if (!core || typeof core.execute !== 'function') throw new Error('missing_core');
    if (!Array.isArray(requests)) throw new Error('requests_shape');
    return requests.map(request => core.execute(request));
  }
};
