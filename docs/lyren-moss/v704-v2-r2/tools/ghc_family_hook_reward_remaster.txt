def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat().replace("+00:00", "Z")


def nz_now() -> str:
    try:
        from zoneinfo import ZoneInfo

        return datetime.now(ZoneInfo("Pacific/Auckland")).isoformat()
    except Exception:
        return "unavailable"


def canonical_bytes(value: Any) -> bytes:
    return (json.dumps(value, sort_keys=True, ensure_ascii=False, separators=(",", ":")) + "\n").encode("utf-8")


def pretty_bytes(value: Any) -> bytes:
    return (json.dumps(value, sort_keys=True, ensure_ascii=False, indent=2) + "\n").encode("utf-8")


def write_bytes(path: Path, data: bytes) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_bytes(data)


def put_json(rel: str, value: Any) -> None:
    write_bytes(BASE / rel, pretty_bytes(value))


def put_text(rel: str, value: str) -> None:
    write_bytes(BASE / rel, value.replace("\r\n", "\n").encode("utf-8"))


def sha_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def sha_file(path: Path) -> str:
    return sha_bytes(path.read_bytes())


def sha_value(value: Any) -> str:
    return sha_bytes(canonical_bytes(value))


def git_blob_sha1(data: bytes) -> str:
    return hashlib.sha1(b"blob " + str(len(data)).encode("ascii") + b"\0" + data).hexdigest()

def fraction(value: Any) -> Fraction:
    return value if isinstance(value, Fraction) else Fraction(str(value))


def ftext(value: Fraction) -> str:
    return str(value.numerator) if value.denominator == 1 else f"{value.numerator}/{value.denominator}"


def encode(value: Any) -> Any:
    if isinstance(value, Fraction):
        return ftext(value)
    if isinstance(value, dict):
        return {str(key): encode(item) for key, item in value.items()}
    if isinstance(value, (list, tuple)):
        return [encode(item) for item in value]
    return value


def matrix(profile: dict[str, Any]) -> list[list[Fraction]]:
    return [[fraction(cell) for cell in row] for row in profile["transition"]]


def vector(profile: dict[str, Any], key: str) -> list[Fraction]:
    return [fraction(cell) for cell in profile[key]]


def multiply(left: list[list[Fraction]], right: list[list[Fraction]]) -> list[list[Fraction]]:
    return [
        [sum((left[i][k] * right[k][j] for k in range(len(right))), Fraction()) for j in range(len(right[0]))]
        for i in range(len(left))
    ]


def vector_multiply(left: list[Fraction], right: list[list[Fraction]]) -> list[Fraction]:
    return [sum((left[k] * right[k][j] for k in range(len(left))), Fraction()) for j in range(len(right[0]))]


def dot(left: list[Fraction], right: list[Fraction]) -> Fraction:
    return sum((a * b for a, b in zip(left, right)), Fraction())


def solve_linear(a: list[list[Fraction]], b: list[Fraction]) -> list[Fraction]:
    rows = [list(row) + [rhs] for row, rhs in zip(a, b)]
    n = len(rows)
    for col in range(n):
        pivot = next((row for row in range(col, n) if rows[row][col] != 0), None)
        if pivot is None:
            raise ValueError("singular exact system")
        rows[col], rows[pivot] = rows[pivot], rows[col]
        scale = rows[col][col]
        rows[col] = [value / scale for value in rows[col]]
        for row in range(n):
            if row == col:
                continue
            factor = rows[row][col]
            rows[row] = [value - factor * pivot_value for value, pivot_value in zip(rows[row], rows[col])]
    return [rows[index][-1] for index in range(n)]


def stationary_distribution(p: list[list[Fraction]]) -> list[Fraction]:
    n = len(p)
    a = [[p[j][i] - (Fraction(1) if i == j else Fraction()) for j in range(n)] for i in range(n)]
    a[-1] = [Fraction(1) for _ in range(n)]
    b = [Fraction() for _ in range(n)]
    b[-1] = Fraction(1)
    answer = solve_linear(a, b)
    residual = vector_multiply(answer, p)
    if residual != answer:
        raise ValueError("stationary substitution failed")
    return answer

def validate_profile(profile: dict[str, Any]) -> dict[str, Any]:
    states = profile["states"]
    p = matrix(profile)
    rewards = vector(profile, "rewards")
    if len(states) != 4 or len(p) != 4 or any(len(row) != 4 for row in p):
        raise ValueError("expected exact four-state transition matrix")
    if any(value < 0 for row in p for value in row):
        raise ValueError("negative transition probability")
    if any(sum(row, Fraction()) != 1 for row in p):
        raise ValueError("transition row does not sum to one")
    if len(rewards) != 4:
        raise ValueError("reward vector length mismatch")
    if not profile["hook_events"] or any(event not in ALLOWED_EVENTS for event in profile["hook_events"]):
        raise ValueError("unsupported hook event")
    if not 1 <= int(profile["timeout_seconds"]) <= 5:
        raise ValueError("timeout outside bounded policy")
    plugin_path = PurePosixPath(profile["plugin_path"])
    if not str(plugin_path).startswith("./") or ".." in plugin_path.parts:
        raise ValueError("plugin path escapes package")
    return {
        "states": 4,
        "rows_normalized": 4,
        "reward_count": 4,
        "hook_event_count": len(profile["hook_events"]),
    }

def run_operation(name: str, profile: dict[str, Any]) -> dict[str, Any]:
    validation = validate_profile(profile)
    p = matrix(profile)
    initial = vector(profile, "initial")
    rewards = vector(profile, "rewards")
    if name == "reward_validate":
        return validation
    if name == "one_step_reward":
        distribution = vector_multiply(initial, p)
        return {"distribution": encode(distribution), "expected_reward": ftext(dot(distribution, rewards))}
    if name == "two_step_reward":
        distribution = vector_multiply(initial, multiply(p, p))
        return {"distribution": encode(distribution), "expected_reward": ftext(dot(distribution, rewards))}
    if name == "stationary_reward":
        stationary = stationary_distribution(p)
        return {"stationary": encode(stationary), "expected_reward": ftext(dot(stationary, rewards))}
    if name == "discounted_value":
        gamma = fraction(profile["discount"])
        a = [
            [(Fraction(1) if i == j else Fraction()) - gamma * p[i][j] for j in range(4)]
            for i in range(4)
        ]
        value = solve_linear(a, rewards)
        residual = [rewards[i] + gamma * sum((p[i][j] * value[j] for j in range(4)), Fraction()) for i in range(4)]
        return {"value": encode(value), "residual_exact": residual == value}
    if name == "finite_horizon_reward":
        current = initial
        total = Fraction()
        steps = []
        for step in range(int(profile["finite_horizon"])):
            reward = dot(current, rewards)
            total += reward
            steps.append({"step": step, "reward": ftext(reward)})
            current = vector_multiply(current, p)
        return {"steps": steps, "total": ftext(total)}
    if name == "reward_range":
        return {"minimum": ftext(min(rewards)), "maximum": ftext(max(rewards))}
    if name == "relabel_reward_covariance":
        permutation = [2, 0, 3, 1]
        relabeled = [rewards[index] for index in permutation]
        return {
            "permutation": permutation,
            "multiset_equal": sorted(relabeled) == sorted(rewards),
            "original_digest": sha_value(encode(rewards)),
            "relabeled_digest": sha_value(encode(relabeled)),
        }
    if name == "event_sequence_validate":
        return {"events": profile["hook_events"], "all_supported": all(event in ALLOWED_EVENTS for event in profile["hook_events"])}
    if name == "matcher_acceptance":
        expression = re.compile(profile["hook_matcher"])
        candidates = ["startup", "resume", "Bash", "apply_patch", "WebSearch"]
        return {"matcher": profile["hook_matcher"], "accepted": [item for item in candidates if expression.search(item)]}
    if name == "timeout_budget":
        return {"timeout_seconds": profile["timeout_seconds"], "within_five_second_budget": int(profile["timeout_seconds"]) <= 5}
    if name == "plugin_path_boundary":
        path = PurePosixPath(profile["plugin_path"])
        return {"path": str(path), "inside_plugin": str(path).startswith("./") and ".." not in path.parts}
    if name == "nonblocking_guard":
        return {"continue": True, "blocking_decision_emitted": False}
    if name == "duplicate_send_guard":
        states = {"not_sent": "one_attempt_permitted", "accepted": "stop", "opaque": "stop", "unresolved": "stop", "rejected_preacceptance": "corrected_only"}
        return {"states": states, "accepted_resend_permitted": False}
    if name == "route_retry_budget":
        return {"discovery_recoveries": 5, "corrected_preacceptance_rejections": 3, "blind_resend": False}
    if name == "authored_extension_policy":
        return {"allowed": [".json", ".md", ".txt", ".html"], "pdf_allowed": False, "authored_image_allowed": False}
    if name == "hook_concurrency_reservation":
        return {"matching_hooks_may_start_concurrently": True, "ordering_claimed": False, "background_hooks_used": False}
    if name == "accessible_summary":
        return {"text": "This synthetic profile checks exact rewards and bounded lifecycle rules without acting on a live service.", "accessibility_complete": False}
    if name == "live_hook_trust_gap":
        return {"state": "open_gap", "missing": ["user hook trust review", "new-session live observation"]}
    if name == "operational_authority_gate":
        return {"state": "exact_gate", "missing": ["competent operational authority", "affected-party acceptance", "production evidence"]}
    raise ValueError("unsupported operation")

def validate_skill(path: Path) -> dict[str, Any]:

        "stdout_sha256": sha_bytes(completed.stdout.encode("utf-8")),

def query_one(operation: str, profile_index: int) -> dict[str, Any]:

def execute_session(session: str) -> None:
    profiles = {row["profile_id"]: row for row in load_profiles()}
    contracts = [row for row in contract_rows() if row["session"] == session]
    portfolio = task_portfolio(session, contracts)
    executions = []
    witnesses = []
    negatives = []
    result_by_id = {}
    for contract in contracts:
        profile = profiles[contract["profile_id"]]
        result = encode(run_operation(contract["operation"], profile))
        result_digest = sha_value(result)
        result_by_id[contract["proposal_id"]] = {"result": result, "digest": result_digest}
        executions.append({**contract, "result": result, "result_sha256": result_digest, "status": contract["outcome"]})
        witnesses.append({"witness_id": contract["proposal_id"] + "-VALID", "state": "passing", "kind": "valid_execution"})
    for task in portfolio["safe_now"]:
        cached = result_by_id[task["proposal_id"]]
        passed = bool(cached["result"]) and len(cached["digest"]) == 64
        if not passed:
            raise AssertionError("safe task failed")
        witnesses.append({"witness_id": task["task_id"], "state": "passing", "kind": task["check"]})
    for task in portfolio["candidate"]:
        contract = next(row for row in contracts if row["proposal_id"] == task["proposal_id"])
        profile = profiles[contract["profile_id"]]
        failure = None
        try:
            run_operation(contract["operation"], malformed(profile, task["invalid_variant"]))
        except Exception as error:
            failure = {"type": type(error).__name__, "message": str(error)}
        if failure is None:
            raise AssertionError("invalid candidate was accepted")
        negative_id = task["task_id"] + "-INVALID"
        negatives.append({"negative_id": negative_id, "proposal_id": task["proposal_id"], "failure": failure, "success_credit": 0, "retained": True})
        witnesses.append({"witness_id": negative_id, "state": "failed", "credit": 0, "kind": "invalid_subject"})
        witnesses.append({"witness_id": task["task_id"] + "-REFUSAL", "state": "passing", "kind": "separate_refusal"})

        )

import argparse
import json
import re
import subprocess
import sys
from pathlib import Path

ALLOWED_MODELS = {"gpt-5.6-sol", "gpt-6-astra"}
ALLOWED_EXTENSIONS = {".json", ".md", ".txt", ".html"}

def read_payload():
    raw = sys.stdin.buffer.read(131073)
    if len(raw) > 131072:
        return {}
    try:
        value = json.loads(raw.decode("utf-8"))
    except Exception:
        return {}
    return value if isinstance(value, dict) else {}

def git(cwd, *args):
    completed = subprocess.run(["git", *args], cwd=cwd, capture_output=True, text=True)
    return completed.returncode, completed.stdout.strip()

def context(event, message):
    if event == "SessionStart":
        print(json.dumps({"continue": True, "hookSpecificOutput": {"hookEventName": "SessionStart", "additionalContext": message}}))
    elif message:
        print(json.dumps({"systemMessage": message}))

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--check", required=True)
    args = parser.parse_args()
    payload = read_payload()
    event = payload.get("hook_event_name", "")
    cwd = Path(payload.get("cwd", ".")).resolve()
    command = str((payload.get("tool_input") or {}).get("command", ""))
    check = args.check
    message = ""
    if check == "model_allowlist":
        model = payload.get("model")
        message = "GHC model allowlist observed." if model in ALLOWED_MODELS else "Model is outside the local GHC allowlist; do not begin a phase side effect."
    elif check == "d_first_storage":
        message = "D-first owner storage is active." if cwd.drive.upper() == "D:" else "C drive is an essential-control exception; keep bulk phase data on D."
    elif check == "branch_identity":
        code, branch = git(cwd, "branch", "--show-current")
        message = "Owner branch: " + branch if code == 0 and branch else "No Git branch resolved; avoid repository mutation."
    elif check == "file_ceiling":
        code, files = git(cwd, "ls-files")
        count = len(files.splitlines()) if code == 0 and files else 0
        message = "Tracked file count is " + str(count) + " of 2000." if count <= 2000 else "Tracked file ceiling exceeded; stop additions and rotate safely."
    elif check == "coordinator_marker":
        marker = None
        for root in (cwd, *cwd.parents):
            candidate = root / ".codex" / "coordination" / "project.yaml"
            if candidate.is_file():
                marker = candidate
                break
        message = "Coordinator marker present; load the board before substantial writes." if marker else "Coordinator board is not enabled; remain single-task and do not infer claims."
    elif check == "destructive_root_advisory":
        broad = re.search(r"(?i)(git\\s+reset\\s+--hard|rm\\s+-rf\\s+[~/]|Remove-Item.+-Recurse.+(?:HOME|CODEX_HOME|[A-Z]:\\\\?$))", command)

    elif check == "authored_extension_advisory":
        forbidden = re.search(r"(?i)\\.(pdf|png|jpg|jpeg|gif|tex|py|ps1|js|mjs|cjs)(?:\\s|$)", command)
        if forbidden:

if __name__ == "__main__":

def privacy_review() -> dict[str, Any]:
    patterns = {
        "credential_assignment": re.compile("api" + r"[_-]?key\s*=", re.IGNORECASE),
        "delegation_markup": re.compile("<" + "codex_delegation", re.IGNORECASE),
        "private_local_path": re.compile(r"[A-Za-z]:[\\/]+Users[\\/]+[^\\/]+", re.IGNORECASE),
        "private_uri": re.compile("plugin" + r"://|app://", re.IGNORECASE),
        "raw_uuid": re.compile(r"\b[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}\b", re.IGNORECASE),
    }
    candidates = []
    files = [path for path in BASE.rglob("*") if path.is_file()]
    for path in files:
        if path.suffix.lower() not in {".json", ".md", ".txt", ".html"}:
            continue
        text = path.read_text(encoding="utf-8", errors="replace")
        for name, pattern in patterns.items():
            for match in pattern.finditer(text):
                candidates.append({
                    "class": name,
                    "path": path.relative_to(REPO).as_posix(),
                    "token_sha256": sha_bytes(match.group(0).encode("utf-8")),
                })
    return {
        "classes": list(patterns),
        "owner_files_scanned": len(files),
        "candidates": candidates,
        "confirmed_owner_hits": len(candidates),
        "complete_privacy_assurance": False,
        "boundary": BOUNDARY,
    }

def build_seal_from_index() -> None:
    prefix = "docs/lyren-moss/v704-v2-r2/"
    paths = [path for path in git("ls-files", prefix).splitlines() if path]
    excluded = {prefix + "final/content-seal.json", prefix + "final/manifest.json"}
    entries = []
    for path in sorted(paths):
        if path in excluded:
            continue
        data = git_index_blob(path)
        entries.append({"path": path, "bytes": len(data), "git_blob_sha1": git_blob_sha1(data), "sha256": sha_bytes(data)})
    payload = {"schema": "ghc.family.content-seal.v3", "entries": entries, "count": len(entries), "excluded": sorted(excluded), "boundary": BOUNDARY}
    put_json("final/content-seal.json", payload)
    put_json("final/manifest.json", {"schema": "ghc.family.final-manifest.v3", "entries": entries, "count": len(entries), "self_exclusions": sorted(excluded), "boundary": BOUNDARY})


def stage_receipt(stage: str) -> dict[str, Any]:
    head = git("rev-parse", "HEAD")
    receipt = {
        "owner": OWNER,
        "phase": PHASE,
        "stage": stage,
        "head": head,
        "branch": git("branch", "--show-current"),
        "parent": git("rev-parse", "HEAD^"),
        "clean": not bool(git("status", "--porcelain")),
        "recorded_utc": utc_now(),
        "recorded_nz": nz_now(),
    }
    BANK.mkdir(parents=True, exist_ok=True)
    write_bytes(BANK / f"{stage}-receipt.json", pretty_bytes(receipt))

def metadata_checks(head: str) -> dict[str, Any]:
    checks = []
    def add(name: str, passed: bool, detail: Any = None) -> None:
        checks.append({"name": name, "passed": bool(passed), "detail": detail})
    add("exact_head", git("rev-parse", "HEAD") == head)
    add("branch", git("branch", "--show-current") == BRANCH)
    add("clean", not bool(git("status", "--porcelain")))
    upstream = git("rev-parse", "@{upstream}")
    add("upstream_equal", upstream == head, upstream)
    divergence = git("rev-list", "--left-right", "--count", "HEAD...@{upstream}")
    add("zero_divergence", divergence.replace("\t", " ").strip() == "0 0", divergence)
    live = git("ls-remote", "--heads", "origin", "refs/heads/" + BRANCH).split()
    add("fresh_live_equal", bool(live) and live[0] == head, live[0] if live else None)
    commits = git("rev-list", "--reverse", PARENT + ".." + head).splitlines()
    add("four_new_commits", len(commits) == 4, commits)
    parents = []
    for commit in commits:
        tokens = git("rev-list", "--parents", "-n", "1", commit).split()
        parents.append({"commit": commit, "parent_count": len(tokens) - 1})
    add("single_parent_zero_merge", all(row["parent_count"] == 1 for row in parents), parents)
    seal_path = "docs/lyren-moss/v704-v2-r2/final/content-seal.json"
    seal = json.loads(git_head_blob(seal_path, head).decode("utf-8"))
    mismatches = []
    for entry in seal["entries"]:
        data = git_head_blob(entry["path"], head)
        if len(data) != entry["bytes"] or git_blob_sha1(data) != entry["git_blob_sha1"] or sha_bytes(data) != entry["sha256"]:
            mismatches.append(entry["path"])
    add("content_seal", not mismatches, {"entries": len(seal["entries"]), "mismatches": mismatches})
    owner_paths = [path for path in git("ls-tree", "-r", "--name-only", head, "docs/lyren-moss/v704-v2-r2").splitlines() if path]
    json_errors = []
    for path in owner_paths:
        if path.endswith(".json"):
            try:
                json.loads(git_head_blob(path, head).decode("utf-8"))
            except Exception as error:
                json_errors.append({"path": path, "error": str(error)})
    add("strict_json", not json_errors, {"count": sum(path.endswith(".json") for path in owner_paths), "errors": json_errors})
    invalid_ext = [path for path in owner_paths if Path(path).suffix.lower() not in {".json", ".md", ".txt", ".html"}]
    add("extension_allowlist", not invalid_ext, invalid_ext)
    add("file_ceiling", int(git("ls-files").count("\n") + (1 if git("ls-files") else 0)) <= 2000)
    baton = git_head_blob("docs/lyren-moss/v704-v2-r2/final/handoff-baton.md", head).decode("utf-8")
    add("baton_words", len(re.findall(r"\S+", baton)) >= 2000, len(re.findall(r"\S+", baton)))
    add("baton_eof", baton.rstrip().endswith(EOF_MARKER))
    phase_truth = json.loads(git_head_blob("docs/lyren-moss/v704-v2-r2/final/phase-truth.json", head).decode("utf-8"))
    add("four_outcomes", set(phase_truth["outcomes"]) == ALLOWED_OUTCOMES)
    add("terminal_verdict", phase_truth["terminal_verdict"] == "NOT_READY_FOR_STAGE_20")
    add("route_prepared_not_sent", phase_truth["route_state"] == "PREPARED_NOT_SENT")
    add("recipient_unclaimed", phase_truth["recipient_completion"] == "UNCLAIMED")
    add("canonical_not_replayed_in_repo", phase_truth["canonical_state"] == "NOT_YET_INVOKED")
    add("models_15", json.loads(git_head_blob("docs/lyren-moss/v704-v2-r2/x2/models.json", head).decode("utf-8"))["count"] == 15)
    add("hooks_10", json.loads(git_head_blob("docs/lyren-moss/v704-v2-r2/planning/hook-plan.json", head).decode("utf-8"))["count"] == 10)
    add("x1_tests_15", json.loads(git_head_blob("docs/lyren-moss/v704-v2-r2/x1/tests.json", head).decode("utf-8"))["count"] == 15)
    add("x2_tests_30", json.loads(git_head_blob("docs/lyren-moss/v704-v2-r2/x2/tests.json", head).decode("utf-8"))["count"] == 30)
    add("plugin_install_parity", json.loads(git_head_blob("docs/lyren-moss/v704-v2-r2/x2/plugin-installation.json", head).decode("utf-8"))["state"] == "INSTALLED_PENDING_USER_TRUST_AND_NEW_SESSION")
    add("global_install_parity", json.loads(git_head_blob("docs/lyren-moss/v704-v2-r2/x2/global-installation.json", head).decode("utf-8"))["state"] == "INSTALLED_EXACT_PARITY")
    return {"head": head, "checks": checks, "passed": sum(row["passed"] for row in checks), "total": len(checks), "all_passed": all(row["passed"] for row in checks)}


def preflight() -> dict[str, Any]:
    head = git("rev-parse", "HEAD")
    payload = metadata_checks(head)
    payload["state"] = "VALID_EXACT_FINAL_OWNER_SCOPED_METADATA_PREFLIGHT" if payload["all_passed"] else "FAILED_METADATA_PREFLIGHT"
    payload["recorded_utc"] = utc_now()
    payload["domain_engines_executed"] = 0
    BANK.mkdir(parents=True, exist_ok=True)
    write_bytes(BANK / "metadata-preflight.json", pretty_bytes(payload))
    return payload


def canonical_once() -> dict[str, Any]:
    latch_path = BANK / "canonical-latch.json"
    if latch_path.exists():
        raise RuntimeError("canonical latch already exists; successful canonical may not be replayed")
    invocation = {"owner": OWNER, "phase": PHASE, "head": git("rev-parse", "HEAD"), "invocation": 1, "recorded_utc": utc_now()}
    BANK.mkdir(parents=True, exist_ok=True)
    write_bytes(BANK / "canonical-invocation.json", pretty_bytes(invocation))
    payload = metadata_checks(invocation["head"])
    payload["state"] = "VALID_EXACT_FINAL_OWNER_SCOPED_METADATA_CANONICAL" if payload["all_passed"] else "FAILED_CANONICAL_NO_SUCCESS_CREDIT"
    payload["domain_engines_executed"] = 0
    payload["invocations"] = 1
    payload["successes"] = 1 if payload["all_passed"] else 0
    payload["replays"] = 0
    write_bytes(BANK / "canonical-payload.json", pretty_bytes(payload))
    receipt = {
        "state": payload["state"],
        "head": invocation["head"],
        "checks": f"{payload['passed']}/{payload['total']}",
        "payload_sha256": sha_file(BANK / "canonical-payload.json"),
        "invocations": 1,
        "successes": payload["successes"],
        "replays": 0,
        "recorded_utc": utc_now(),
        "same_owner_only": True,
        "complete_repository_suite": False,
        "independent_reproduction": False,
        "boundary": BOUNDARY,
    }
    write_bytes(BANK / "canonical-receipt.json", pretty_bytes(receipt))
    if payload["all_passed"]:
        latch = {
            "closed": True,
            "head": invocation["head"],
            "payload_sha256": receipt["payload_sha256"],
            "receipt_sha256": sha_file(BANK / "canonical-receipt.json"),
            "invocations": 1,
            "successes": 1,
            "replays": 0,
        }
        write_bytes(latch_path, pretty_bytes(latch))
    return receipt


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("command", choices=["planning", "x1", "x2", "finalize-installation", "final", "seal", "receipt", "preflight", "canonical", "query"])
    parser.add_argument("--stage")
    parser.add_argument("--operation")
    parser.add_argument("--profile-index", type=int, default=1)
    args = parser.parse_args()
    if args.command == "planning":
        build_planning()
    elif args.command == "x1":
        execute_session("x1")
    elif args.command == "x2":
        build_x2()
    elif args.command == "finalize-installation":
        finalize_installation()
    elif args.command == "final":
        build_final()
    elif args.command == "seal":
        build_seal_from_index()
    elif args.command == "receipt":
        if not args.stage:
            raise ValueError("--stage is required")
        print(json.dumps(stage_receipt(args.stage), sort_keys=True))
    elif args.command == "preflight":
        print(json.dumps(preflight(), sort_keys=True))
    elif args.command == "canonical":
        print(json.dumps(canonical_once(), sort_keys=True))
    elif args.command == "query":
        if not args.operation:
            raise ValueError("--operation is required")
        print(json.dumps(query_one(args.operation, args.profile_index), sort_keys=True))


if __name__ == "__main__":
    main()
