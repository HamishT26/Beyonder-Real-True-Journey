"use strict";
const crypto=require("crypto");
function stable(v){if(Array.isArray(v))return v.map(stable);if(v&&typeof v==="object")return Object.fromEntries(Object.keys(v).sort().map(k=>[k,stable(v[k])]));return v;}
function sha(v){return crypto.createHash("sha256").update(Buffer.from(JSON.stringify(stable(v)))).digest("hex");}

 if(p.schema!=="ghc.family.finite-graph-proposal.v1"||!/^SA6945-P[0-9]{3}$/.test(p.proposal_id)||!Number.isInteger(p.case_id)||p.case_id<1||p.case_id>5)return {accepted:false,error:"proposal_identity",numerical_execution:false,planning_only:true,proposal_completion_credit:0};
 if(p.numerical_or_graph_execution_in_x1!==false)return {accepted:false,error:"x1_execution_boundary",numerical_execution:false,planning_only:true,proposal_completion_credit:0};if(p.planning_lane!=="x1_planning_only"||p.execution_lane!=="x2_owner_synthetic")return {accepted:false,error:"lane_boundary",numerical_execution:false,planning_only:true,proposal_completion_credit:0};

}
