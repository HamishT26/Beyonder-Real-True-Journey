function adjacency(g){const a=Object.fromEntries(g.nodes.map(n=>[n,[]]));for(const [u,v] of g.edges)a[u].push(v);for(const n of g.nodes)a[n].sort();return a;}
function reverseAdj(g){const a=Object.fromEntries(g.nodes.map(n=>[n,[]]));for(const [u,v] of g.edges)a[v].push(u);for(const n of g.nodes)a[n].sort();return a;}
function reachable(g,start){if(!g.nodes.includes(start))fail("unknown_start");const a=adjacency(g),seen=new Set([start]),queue=[start];while(queue.length){const u=queue.shift();for(const v of a[u])if(!seen.has(v)){seen.add(v);queue.push(v);}}return [...seen].sort();}
function shortest(g,start,target,trace=false){if(!g.nodes.includes(start)||!g.nodes.includes(target))fail("unknown_endpoint");const a=adjacency(g),seen=new Set([start]),pred={},queue=[start],frontiers=[[start]];while(queue.length&&!seen.has(target)){const next=[];for(const u of queue)for(const v of a[u])if(!seen.has(v)){seen.add(v);pred[v]=u;next.push(v);}next.sort();queue.splice(0,queue.length,...next);if(next.length)frontiers.push([...next]);}if(!seen.has(target))return {found:false,path:[],distance:null,...(trace?{frontiers,predecessors:pred}:{})};const path=[target];while(path[0]!==start)path.unshift(pred[path[0]]);return {found:true,path,distance:path.length-1,...(trace?{frontiers,predecessors:pred}:{})};}
function topo(g){const a=adjacency(g),degree=Object.fromEntries(g.nodes.map(n=>[n,0]));for(const [,v] of g.edges)degree[v]++;const queue=g.nodes.filter(n=>degree[n]===0).sort(),order=[];while(queue.length){const u=queue.shift();order.push(u);for(const v of a[u]){degree[v]--;if(degree[v]===0){queue.push(v);queue.sort();}}}if(order.length!==g.nodes.length)fail("cycle");return order;}

function componentsWeak(g){const a=Object.fromEntries(g.nodes.map(n=>[n,[]]));for(const [u,v] of g.edges){a[u].push(v);a[v].push(u);}for(const n of g.nodes)a[n].sort();const seen=new Set(),out=[];for(const n of g.nodes)if(!seen.has(n)){const q=[n],c=[];seen.add(n);while(q.length){const u=q.shift();c.push(u);for(const v of a[u])if(!seen.has(v)){seen.add(v);q.push(v);}}out.push(c.sort());}return out.sort((x,y)=>x.join("\u0000").localeCompare(y.join("\u0000")));}
function closurePairs(g){const pairs=[];for(const u of g.nodes)for(const v of reachable(g,u))if(u!==v)pairs.push([u,v]);return pairs.sort((a,b)=>edgeKey(a).localeCompare(edgeKey(b)));}
