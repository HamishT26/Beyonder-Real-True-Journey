function g(args) {
  return cp.execFileSync("git", args, { cwd: ROOT, encoding: "utf8", maxBuffer: 64 * 1024 * 1024 }).trim();
}

function read(relative) {
  return fs.readFileSync(path.join(ROOT, relative), "utf8");
}

function readJson(relative) {
  return JSON.parse(read(relative));
}

function write(relative, content) {
  const absolute = path.join(ROOT, relative);
  fs.mkdirSync(path.dirname(absolute), { recursive: true });
  fs.writeFileSync(absolute, content, { encoding: "utf8", flag: "wx" });
}

function writeJson(relative, value) {
  write(relative, `${JSON.stringify(value, null, 2)}\n`);
}

function sha(value) {
  return crypto.createHash("sha256").update(value).digest("hex");
}

function normalizedBytes(absolute) {
  return Buffer.from(fs.readFileSync(absolute, "utf8").replace(/\r\n/g, "\n"), "utf8");
}

function collect(directory, output = []) {
  if (!fs.existsSync(directory)) return output;
  for (const name of fs.readdirSync(directory).sort()) {
    const absolute = path.join(directory, name);
    const stat = fs.statSync(absolute);
    if (stat.isDirectory()) collect(absolute, output);
    else output.push(absolute);
  }
  return output;
}

function words(text) {
  return (text.match(/\S+/g) || []).length;
}

function canonicalCard(card) {
  const content = { ...card };
  delete content.content_sha256;
  return JSON.stringify(content);
}

function finalizeCard(card) {
  return { ...card, content_sha256: sha(Buffer.from(canonicalCard(card), "utf8")) };
}

function escapeHtml(value) {
  return String(value).replaceAll("&", "&amp;").replaceAll("<", "&lt;").replaceAll(">", "&gt;").replaceAll('"', "&quot;");
}
