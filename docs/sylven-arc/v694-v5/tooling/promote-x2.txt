"use strict";
const fs=require("fs"),path=require("path"),crypto=require("crypto"),cp=require("child_process");const ROOT=path.resolve(__dirname,"../../../.."),BASE="docs/sylven-arc/v694-v5",X2=path.join(ROOT,BASE,"x2");const names=["ghc-family-finite-graph-structure-v2","ghc-family-finite-graph-reachability-v2","ghc-family-finite-graph-partition-v2","ghc-family-finite-graph-reversible-patch-v2","ghc-family-finite-graph-accessible-evidence-v2"];
function sha(v){return crypto.createHash("sha256").update(v).digest("hex");}function encode(v){return JSON.stringify(v,null,2)+"\n";}function collect(d,o=[]){for(const n of fs.readdirSync(d)){const a=path.join(d,n),s=fs.statSync(a);if(s.isDirectory())collect(a,o);else o.push(a);}return o;}function copyTree(source,target){fs.mkdirSync(target,{recursive:false});for(const file of collect(source)){const rel=path.relative(source,file),dest=path.join(target,rel);fs.mkdirSync(path.dirname(dest),{recursive:true});fs.copyFileSync(file,dest,fs.constants.COPYFILE_EXCL);}}
function main(){
 const quick=process.argv[2],skillRoot=process.argv[3],runnerRoot=process.argv[4];if(!quick||!skillRoot||!runnerRoot||!fs.existsSync(quick)||!fs.existsSync(skillRoot))throw new Error("arguments");const receiptPath=path.join(X2,"global-installation-receipt.json");if(fs.existsSync(receiptPath))throw new Error("receipt_exists");const plans=names.map((name,i)=>({name,source:path.join(X2,"global-skills",name),target:path.join(skillRoot,name),runnerName:`ghc_family_finite_graph_group_${i+1}.txt`,contract:path.join(X2,"contracts",`${String(i*20+1).padStart(3,"0")}.json`)}));for(const p of plans){if(fs.existsSync(p.target)||!fs.existsSync(p.source))throw new Error(`skill_collision_or_missing_${p.name}`);}for(const n of ["graph-plan-kernel.txt","graph-kernel.txt",...plans.map(p=>p.runnerName)])if(fs.existsSync(path.join(runnerRoot,n)))throw new Error(`runner_collision_${n}`);
 const sourceSmokes=[];for(const p of plans){const bindings=JSON.parse(fs.readFileSync(path.join(p.source,"source-bindings.json"),"utf8"));if(bindings.sources.length!==4)throw new Error(`bindings_${p.name}`);for(const b of bindings.sources){const file=path.join(p.source,b.reference);if(sha(fs.readFileSync(file))!==b.sha256)throw new Error(`source_hash_${p.name}`);}const runner=path.join(p.source,"scripts",p.runnerName),a=cp.spawnSync("node",[runner,p.contract],{encoding:"utf8"}),b=cp.spawnSync("node",[runner,p.contract,"--invalid"],{encoding:"utf8"});if(a.status!==0||b.status!==2||JSON.parse(a.stdout).accepted!==true||JSON.parse(b.stdout).error!=="input_shape")throw new Error(`source_smoke_${p.name}`);sourceSmokes.push({surface:"owner_merged_source",skill:p.name,runner:p.runnerName,accept_exit:0,invalid_subject_exit:2,invalid_subject_result:"fail",invalid_subject_success_credit:0,refusal_check:"pass"});}
 fs.mkdirSync(runnerRoot,{recursive:true});for(const p of plans)copyTree(p.source,p.target);fs.copyFileSync(path.join(ROOT,BASE,"tooling","graph-plan-kernel.txt"),path.join(runnerRoot,"graph-plan-kernel.txt"),fs.constants.COPYFILE_EXCL);fs.copyFileSync(path.join(ROOT,BASE,"tooling","graph-kernel.txt"),path.join(runnerRoot,"graph-kernel.txt"),fs.constants.COPYFILE_EXCL);for(const p of plans)fs.copyFileSync(path.join(X2,"public-runners",p.runnerName),path.join(runnerRoot,p.runnerName),fs.constants.COPYFILE_EXCL);
 const installedSmokes=[],publicSmokes=[],validations=[];for(const p of plans){const q=cp.spawnSync("python",[quick,p.target],{encoding:"utf8",env:{...process.env,PYTHONDONTWRITEBYTECODE:"1",PYTHONUTF8:"1"}});if(q.status!==0)throw new Error(`installed_skill_${p.name}`);validations.push({skill:p.name,exit_code:q.status,output:q.stdout.trim()});for(const [surface,runner,out] of [["installed_skill",path.join(p.target,"scripts",p.runnerName),installedSmokes],["public_runner",path.join(runnerRoot,p.runnerName),publicSmokes]]){const a=cp.spawnSync("node",[runner,p.contract],{encoding:"utf8"}),b=cp.spawnSync("node",[runner,p.contract,"--invalid"],{encoding:"utf8"});if(a.status!==0||b.status!==2||JSON.parse(a.stdout).accepted!==true||JSON.parse(b.stdout).error!=="input_shape")throw new Error(`${surface}_${p.name}`);out.push({surface,skill:p.name,runner:p.runnerName,accept_exit:0,invalid_subject_exit:2,invalid_subject_result:"fail",invalid_subject_success_credit:0,refusal_check:"pass"});}}
 const parity=[];for(const p of plans)for(const file of collect(p.source)){const rel=path.relative(p.source,file),dest=path.join(p.target,rel),sourceBytes=fs.readFileSync(file),destBytes=fs.readFileSync(dest);if(!sourceBytes.equals(destBytes))throw new Error(`parity_${p.name}_${rel}`);parity.push({kind:"global_skill",skill:p.name,relative:rel.replaceAll("\\","/"),bytes:sourceBytes.length,sha256:sha(sourceBytes),equal:true});}for(const n of ["graph-plan-kernel.txt","graph-kernel.txt"]){const s=fs.readFileSync(path.join(ROOT,BASE,"tooling",n)),d=fs.readFileSync(path.join(runnerRoot,n));if(!s.equals(d))throw new Error(`support_${n}`);parity.push({kind:"runtime_support",name:n,bytes:s.length,sha256:sha(s),equal:true});}for(const p of plans){const s=fs.readFileSync(path.join(X2,"public-runners",p.runnerName)),d=fs.readFileSync(path.join(runnerRoot,p.runnerName));if(!s.equals(d))throw new Error(`public_${p.runnerName}`);parity.push({kind:"public_runner",name:p.runnerName,bytes:s.length,sha256:sha(s),equal:true});}if(parity.length!==52)throw new Error(`parity_count_${parity.length}`);
 const receipt={schema:"ghc.family.sylven-additive-global-installation.v1",status:"VALIDATED_ADDITIVE_GLOBAL_INSTALLATION",owner:"Sylven Arc",phase:"v694-v5",destination_absence_before_write:true,overwrites:0,deletions:0,new_packages:0,spend_usd:0,merged_skills:5,public_runners:5,runtime_support_files:2,source_complete_reads:30,source_official_validations:15,installed_skill_validations:validations,source_smokes:sourceSmokes,installed_smokes:installedSmokes,public_smokes:publicSmokes,parity_bindings:parity,parity_count:parity.length,rollback:"Stop selecting an installed capability and retain its bytes and receipts; no deletion or overwrite is authorized by this receipt.",same_owner_only:true,independent_reproduction:false,terminal_verdict:"NOT_READY_FOR_STAGE_20"};fs.writeFileSync(receiptPath,encode(receipt),{encoding:"utf8",flag:"wx"});process.stdout.write(JSON.stringify({status:receipt.status,skills:5,public_runners:5,support_files:2,source_smokes:sourceSmokes.length,installed_smokes:installedSmokes.length,public_smokes:publicSmokes.length,parity:parity.length,overwrites:0}));
}
main();
