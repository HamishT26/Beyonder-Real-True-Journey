"use strict";
const fs=require("fs"),path=require("path"),crypto=require("crypto"),cp=require("child_process");
const ROOT=path.resolve(__dirname,"../../../.."),BASE="docs/sylven-arc/v694-v5",PLAN=path.join(ROOT,BASE,"plan"),PARENT="cce7c0dbbc44044b1e648be731663def37c76db9";
function stable(v){if(Array.isArray(v))return v.map(stable);if(v&&typeof v==="object")return Object.fromEntries(Object.keys(v).sort().map(k=>[k,stable(v[k])]));return v;}
function sha(v){return crypto.createHash("sha256").update(Buffer.isBuffer(v)?v:Buffer.from(JSON.stringify(stable(v)))).digest("hex");}
function load(n){return JSON.parse(fs.readFileSync(path.join(PLAN,n),"utf8"));}
function collect(d,o=[]){for(const n of fs.readdirSync(d)){const a=path.join(d,n),s=fs.statSync(a);if(s.isDirectory())collect(a,o);else o.push(a);}return o;}
function tokens(s){return new Set(String(s).toLowerCase().match(/[a-z0-9]+/g)||[]);}
function jac(a,b){const x=tokens(a),y=tokens(b),u=new Set([...x,...y]);let n=0;for(const t of x)if(y.has(t))n++;return u.size?n/u.size:1;}
function git(a){const r=cp.spawnSync("git",["-C",ROOT,...a],{encoding:"utf8"});if(r.status!==0)throw new Error(`git_${a[0]}`);return r.stdout.trim();}
function main(){
 const sourcePath=process.argv[2];if(!sourcePath)throw new Error("source_proposals_argument");if(git(["rev-parse","HEAD"])!==PARENT||fs.existsSync(path.join(ROOT,BASE,"x1"))||fs.existsSync(path.join(ROOT,BASE,"x2")))throw new Error("lifecycle");
 const p=load("proposal-freeze.json"),source=JSON.parse(fs.readFileSync(path.resolve(sourcePath),"utf8")),novelty=load("novelty-audit.json"),portfolio=load("portfolios.json"),cap=load("capability-plan.json"),flow=load("method-flow-startup.json"),neg=load("retained-negatives-startup.json"),truth=load("phase-truth.json"),route=load("route-plan.json"),selection=load("tool-selection.json"),manifest=load("manifest.json");
 if(selection.current_plan_validator!==`${BASE}/tooling/validate-plan-current.txt`||selection.historical_unselected.length!==1)throw new Error("tool_selection");if(p.count!==100||p.proposals.length!==100||new Set(p.proposals.map(x=>x.proposal_id)).size!==100||new Set(p.proposals.map(x=>x.title)).size!==100||new Set(p.proposals.map(x=>x.definition_sha256)).size!==100)throw new Error("proposal_identity");
 const dispositions={completed:0,represented:0,open_gap:0,exact_gate:0};for(const x of p.proposals){dispositions[x.expected_disposition]++;if(sha({operation:x.operation,input:x.input,hypothesis:x.hypothesis,disposition:x.expected_disposition})!==x.definition_sha256||x.numerical_or_graph_execution_in_x1!==false)throw new Error(`proposal_${x.proposal_id}`);}if(JSON.stringify(dispositions)!==JSON.stringify({completed:80,represented:10,open_gap:5,exact_gate:5}))throw new Error("dispositions");
 const sourceRows=source.proposals||source.entries,sourceTitles=new Set(sourceRows.map(x=>x.title));if(sourceRows.length!==100||p.proposals.some(x=>sourceTitles.has(x.title)))throw new Error("source_collision");let max=0;for(const x of p.proposals)for(const y of sourceRows)max=Math.max(max,jac(x.title,y.title));if(max>0.72||Math.abs(max-novelty.maximum_token_jaccard.score)>1e-12)throw new Error("similarity");
 for(const lane of [portfolio.x1,portfolio.x2])if(lane.safe.length!==100||lane.candidate.length!==100||lane.clean_fix_refine.length!==100)throw new Error("portfolio");if(portfolio.exact_packets.length!==50||portfolio.blocked_packets.length!==30||portfolio.exact_packets.some(x=>x.executed)||portfolio.blocked_packets.some(x=>x.executed))throw new Error("gated_packets");if(cap.x1_skills.length!==10||cap.x2_skills.length!==10||cap.x1_runners.length!==5||cap.x2_runners.length!==5||cap.global_skills.length!==5||cap.global_runners.length!==5||flow.methods.length!==5||flow.witnesses.length!==10||neg.count!==5||truth.x2_exists!==false||route.early_contact!==false)throw new Error("counts");
 for(const e of manifest.entries){const b=fs.readFileSync(path.join(ROOT,e.path));if(b.length!==e.bytes||sha(b)!==e.sha256)throw new Error(`manifest_${e.path}`);}const files=collect(path.join(ROOT,BASE));let json=0,maxWords=0;for(const file of files){const b=fs.readFileSync(file),text=b.toString("utf8"),words=(text.match(/\S+/g)||[]).length;maxWords=Math.max(maxWords,words);if(b.includes(13)||words>100000||/\b[a-z]:[\\/](?:users|ghc-archives)[\\/]/i.test(text)||/\b[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}\b/i.test(text))throw new Error(`document_${file}`);if(path.extname(file)===".json"){JSON.parse(text);json++;}}const status=git(["status","--porcelain=v1","--untracked-files=all"]).split(/\r?\n/).filter(Boolean);if(status.some(x=>!x.slice(3).replaceAll("\\","/").startsWith(`${BASE}/`)))throw new Error("scope");process.stdout.write(JSON.stringify({status:"VALID_PLANNING_ONLY_CANDIDATE",proposals:100,dispositions,maximum_similarity:max,portfolio_each_session:{safe:100,candidate:100,clean_fix_refine:100},exact:50,blocked:30,methods:5,witnesses:10,json_documents:json,owner_phase_files:files.length,maximum_words:maxWords,manifest_entries:manifest.entry_count,x1_exists:false,x2_exists:false,privacy_hits:0}));
}
main();
