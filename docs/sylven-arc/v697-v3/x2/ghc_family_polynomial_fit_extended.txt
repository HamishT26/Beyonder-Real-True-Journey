'use strict';
const C=require('../x1/ghc_family_polynomial_fit_core.txt');
const extra=['fit_derivative_predict','fit_integral_interval','fit_leave_one_out','fit_degree_compare','fit_functional_identifiability','fit_node_affine_pullback','fit_parameter_box_envelope','fit_dependency_criticism','fit_observation_gap','fit_authority_gate'],operations=[...C.operations,...extra];
function falling(k,n){let result=1;for(let j=0;j<n;j++)result*=k-j;return result;}
function binomial(k,n){let result=1;for(let j=1;j<=n;j++)result=result*(k-j+1)/j;return result;}
function pullback(coefficients,scale,offset){const out=Array.from({length:coefficients.length},C.zero);coefficients.forEach((coefficient,k)=>{for(let j=0;j<=k;j++)out[j]=C.add(out[j],C.mul(coefficient,C.mul(C.fromInt(binomial(k,j)),C.mul(C.pow(scale,j),C.pow(offset,k-j)))));});return out;}
function interval(value){C.need(Array.isArray(value)&&value.length===2,'interval');const out=value.map(C.parse);C.need(C.compare(out[0],out[1])<=0,'interval_order');return out;}
function evaluate(request,allowed=operations){C.keys(request,['op','input']);C.need(typeof request.op==='string'&&allowed.includes(request.op)&&operations.includes(request.op),'operation');if(C.operations.includes(request.op))return C.evaluate(request,allowed);
 const op=request.op,i=request.input,fields={fit_derivative_predict:['point','order'],fit_integral_interval:['interval'],fit_leave_one_out:[],fit_degree_compare:[],fit_functional_identifiability:['weights'],fit_node_affine_pullback:['scale','offset'],fit_parameter_box_envelope:['point','parameter_budget'],fit_dependency_criticism:[],fit_observation_gap:['real_rows'],fit_authority_gate:['authority','release']},a=C.admit(i,fields[op]),s=C.solve(a.degree,a.pairs);let value,outcome='completed';
 switch(op){
 case 'fit_derivative_predict':{const point=C.parse(i.point),order=C.integer(i.order,0,4),weights=Array.from({length:s.columns},(_,k)=>k<order?C.zero():C.mul(C.fromInt(falling(k,order)),C.pow(point,k-order)));value=C.functional(s,weights,'derivative_order_'+order);break;}
 case 'fit_integral_interval':{const [lower,upper]=interval(i.interval),weights=Array.from({length:s.columns},(_,k)=>C.div(C.sub(C.pow(upper,k+1),C.pow(lower,k+1)),C.fromInt(k+1)));value=C.functional(s,weights,'declared_coordinate_interval_integral');break;}
 case 'fit_leave_one_out':{const rows=s.pairs.map((held,index)=>{const other=s.pairs.filter((row,k)=>k!==index),sub=C.solve(a.degree,other),prediction=C.functional(sub,Array.from({length:s.columns},(_,k)=>C.pow(held.x,k)),'held_constraint_value');return {held_x:C.format(held.x),held_y:C.format(held.y),prediction,matches_if_identified:prediction.identified?prediction.value===C.format(held.y):null};});value={rows,empirical_validation:false,empty_holdout_is_not_validation:rows.length===0};break;}
 case 'fit_degree_compare':{const profiles=[Math.max(0,a.degree-1),a.degree].map(degree=>{const z=C.solve(degree,a.pairs);return {degree,rank:z.rank,nullity:z.basis.length,consistent:z.consistent,unique:z.unique};});value={profiles,true_model_selected:false,likelihood:null};break;}
 case 'fit_functional_identifiability':value=C.functional(s,C.coefficients(i.weights,s.columns),'declared_coefficient_linear_functional');break;
 case 'fit_node_affine_pullback':{const scale=C.parse(i.scale),offset=C.parse(i.offset);C.need(scale.n!==0n,'noninvertible_scale');value={scale:C.format(scale),offset:C.format(offset),samples_in_new_coordinate:s.pairs.map(row=>({x:C.format(C.div(C.sub(row.x,offset),scale)),y:C.format(row.y)})),particular_pullback:s.consistent?C.vectorFormat(pullback(s.particular,scale,offset)):null,nullspace_pullbacks:s.basis.map(row=>C.vectorFormat(pullback(row,scale,offset))),consistent:s.consistent,invertible_coordinate_map:true};break;}
 case 'fit_parameter_box_envelope':{outcome='represented';const point=C.parse(i.point),budget=C.parse(i.parameter_budget);C.need(C.compare(budget,C.zero())>=0,'parameter_budget');const weights=Array.from({length:s.columns},(_,k)=>C.pow(point,k)),effects=s.basis.map(row=>C.dot(weights,row)),center=s.consistent?C.dot(weights,s.particular):null,radius=C.mul(budget,effects.reduce((sum,x)=>C.add(sum,C.abs(x)),C.zero()));value={consistent:s.consistent,parameter_budget:C.format(budget),lower:center===null?null:C.format(C.sub(center,radius)),upper:center===null?null:C.format(C.add(center,radius)),particular_value:center===null?null:C.format(center),nullspace_effects:C.vectorFormat(effects),coordinate_box_not_prior_or_covariance:true,physical_model_error_known:false};break;}
 case 'fit_dependency_criticism':outcome='represented';value={rank:s.rank,nullity:s.basis.length,consistent:s.consistent,arithmetic_rounding_used:false,condition_number_computed:false,data_uncertainty_known:false,physical_model_error_known:false,exact_fit_is_not_empirical_validation:true,units_are_declared_coordinates_only:true};break;
 case 'fit_observation_gap':C.integer(i.real_rows,0,0);outcome='open_gap';value={real_rows:0,likelihood:null,calibration:null,declared_constraint_count:s.pairs.length,missing:['specified physical observable','governed observations','instrument and calibration model','uncertainty and model discrepancy','independent review']};break;
 case 'fit_authority_gate':C.need(i.authority===false&&i.release===false,'authority_gate');outcome='exact_gate';value={authority:false,release:false,declared_constraint_count:s.pairs.length,reserved:['professional','legal','cultural','affected-party','Maori authority']};break;
 }
 return C.envelope(outcome,value);
}
function invoke(request,allowed=operations){try{return {accepted:true,result:evaluate(request,allowed)};}catch(e){if(!(e instanceof C.Refusal))throw e;return {accepted:false,refusal:e.code,subject_credit:0,authority:false};}}
