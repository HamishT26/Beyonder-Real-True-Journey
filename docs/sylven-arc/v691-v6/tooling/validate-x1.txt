"use strict";

const fs=require("fs");const path=require("path");const crypto=require("crypto");const cp=require("child_process");require.extensions[".txt"]=require.extensions[".js"];
const ROOT=path.resolve(__dirname,"../../../.."),BASE="docs/sylven-arc/v691-v6",X1=path.join(ROOT,BASE,"x1"),PLAN_ROOT="98591e4a91d3eb5f6a9b98b552f08063f8c2e39c",engine=require("./simulation-engine.txt");
function sha(value){return crypto.createHash("sha256").update(Buffer.isBuffer(value)?value:Buffer.from(value)).digest("hex");}function stable(v){if(Array.isArray(v))return v.map(stable);if(v&&typeof v==="object")return Object.fromEntries(Object.keys(v).sort().map(k=>[k,stable(v[k])]));return v;}function same(a,b){return JSON.stringify(stable(a))===JSON.stringify(stable(b));}function load(p){return JSON.parse(fs.readFileSync(path.join(X1,p),"utf8"));}function collect(d,o=[]){for(const n of fs.readdirSync(d)){const a=path.join(d,n),s=fs.statSync(a);if(s.isDirectory())collect(a,o);else o.push(a);}return o;}function git(a){const r=cp.spawnSync("git",["-C",ROOT,...a],{encoding:"utf8"});if(r.status!==0)throw new Error(`git_${a[0]}`);return r.stdout.trim();}

  const safe=load("execution/safe-now.json"),candidate=load("execution/candidate.json"),refine=load("execution/clean-fix-refine.json"),flow=load("evidence/method-flow.json"),truth=load("phase-truth.json"),observed=load("validation/observed-results.json"),quick=load("validation/quick-skill-validation.json"),receipt=load("validation/x1-validation-receipt.json"),manifest=load("manifest.json");
  const operational=load("operational-failures.json"),negativesLedger=load("evidence/retained-negatives.json");

}
main();
