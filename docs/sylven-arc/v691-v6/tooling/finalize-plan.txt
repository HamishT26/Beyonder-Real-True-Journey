function stable(value) {
  if (Array.isArray(value)) return value.map(stable);
  if (value && typeof value === "object") return Object.fromEntries(Object.keys(value).sort().map(key => [key, stable(value[key])]));
  return value;
}
function sha(value) { return crypto.createHash("sha256").update(Buffer.isBuffer(value) ? value : Buffer.from(JSON.stringify(stable(value)))).digest("hex"); }
function load(name) { return JSON.parse(fs.readFileSync(path.join(PLAN, name), "utf8")); }
function encode(value) { return JSON.stringify(value, null, 2) + "\n"; }
function tokens(value) { return new Set(String(value).toLowerCase().match(/[a-z0-9]+/g) || []); }
function jaccard(left, right) {
  const a=tokens(left), b=tokens(right), union=new Set([...a,...b]); let intersection=0;
  for (const token of a) if (b.has(token)) intersection += 1;
  return union.size ? intersection / union.size : 1;
}
function collect(directory, output=[]) {
  for (const name of fs.readdirSync(directory)) {
    const absolute=path.join(directory,name), stat=fs.statSync(absolute);
    if (stat.isDirectory()) collect(absolute,output); else output.push(absolute);
  }
  return output;
}
