function stable(value) {
  if (Array.isArray(value)) return value.map(stable);
  if (value && typeof value === "object") return Object.fromEntries(Object.keys(value).sort().map(k => [k, stable(value[k])]));
  return value;
}

function jaccard(a, b) {
  const tokens = s => new Set(s.toLowerCase().match(/[a-z0-9]+/g) || []);
  const x = tokens(a), y = tokens(b), union = new Set([...x, ...y]);
  let intersect = 0; for (const token of x) if (y.has(token)) intersect++;
  return union.size ? intersect / union.size : 1;
}
function lcg(seed, steps, modulus) {
  const values = []; let x = seed;
  for (let i = 0; i < steps; i++) { x = (17 * x + 43) % modulus; values.push(x); }
  return values;
}
function topo(events) {
  const map = new Map(events.map(e => [e.id, e])); const indegree = new Map(events.map(e => [e.id, 0]));
  const children = new Map(events.map(e => [e.id, []]));
  for (const e of events) for (const p of e.parents) { if (!map.has(p)) throw new Error("parent"); indegree.set(e.id, indegree.get(e.id) + 1); children.get(p).push(e.id); }
  const queue = [...indegree].filter(([,n]) => n === 0).map(([id]) => id).sort(); const order = [];
  while (queue.length) { const id = queue.shift(); order.push(id); for (const child of children.get(id).sort()) { indegree.set(child, indegree.get(child) - 1); if (indegree.get(child) === 0) queue.push(child), queue.sort(); } }
  if (order.length !== events.length) throw new Error("cycle"); return order;
}
