"use strict";
const fs=require("fs"),path=require("path"),crypto=require("crypto"),cp=require("child_process");
const ROOT=path.resolve(__dirname,"../../../.."),BASE="docs/sylven-arc/v691-v6",X2=path.join(ROOT,BASE,"x2");
const groups=[{name:"ghc-family-simulation-state-contracts",ops:["scenario_schema","agent_registry","deterministic_step","state_digest"]},{name:"ghc-family-message-clock-contracts",ops:["message_envelope","logical_clock","schedule_order","causal_trace"]},{name:"ghc-family-queue-resource-contracts",ops:["priority_queue","resource_budget","seed_replay","retry_budget"]},{name:"ghc-family-fault-delivery-contracts",ops:["message_delivery","drop_fault","duplicate_fault","delay_fault"]},{name:"ghc-family-partition-rights-contracts",ops:["partition_fault","evidence_ceiling","participant_gap","authority_gate"]}];
function sha(value){return crypto.createHash("sha256").update(value).digest("hex");}function encode(value){return JSON.stringify(value,null,2)+"\n";}function same(a,b){const stable=v=>Array.isArray(v)?v.map(stable):v&&typeof v==="object"?Object.fromEntries(Object.keys(v).sort().map(k=>[k,stable(v[k])])):v;return JSON.stringify(stable(a))===JSON.stringify(stable(b));}
function main(){
 const quick=process.argv[2],skillRoot=process.argv[3],runnerRoot=process.argv[4];if(!quick||!skillRoot||!runnerRoot||!fs.existsSync(quick)||!fs.existsSync(skillRoot))throw new Error("promotion_arguments");const receiptPath=path.join(X2,"validation","global-promotion-receipt.json");if(fs.existsSync(receiptPath))throw new Error("promotion_replay");
 const plans=groups.map((g,i)=>({group:g,skillSource:path.join(X2,"curated","skills",g.name,"SKILL.md"),skillTarget:path.join(skillRoot,g.name,"SKILL.md"),runnerSource:path.join(X2,"curated","runners",`simulation-curated-group-${i+1}.txt`),runnerTarget:path.join(runnerRoot,`simulation-curated-group-${i+1}.txt`)}));
 for(const p of plans){if(!fs.existsSync(p.skillSource)||!fs.existsSync(p.runnerSource))throw new Error(`missing_source_${p.group.name}`);if(fs.existsSync(path.dirname(p.skillTarget))||fs.existsSync(p.runnerTarget))throw new Error(`collision_${p.group.name}`);const q=cp.spawnSync("python",[quick,path.dirname(p.skillSource)],{encoding:"utf8",env:{...process.env,PYTHONDONTWRITEBYTECODE:"1",PYTHONUTF8:"1"}});if(q.status!==0)throw new Error(`source_skill_${p.group.name}`);}
 fs.mkdirSync(runnerRoot,{recursive:true});for(const p of plans){fs.mkdirSync(path.dirname(p.skillTarget),{recursive:false});fs.copyFileSync(p.skillSource,p.skillTarget,fs.constants.COPYFILE_EXCL);fs.copyFileSync(p.runnerSource,p.runnerTarget,fs.constants.COPYFILE_EXCL);}
 const contracts=fs.readdirSync(path.join(X2,"contracts")).map(name=>path.join(X2,"contracts",name)).map(file=>({file,data:JSON.parse(fs.readFileSync(file,"utf8"))})),bindings=[],smokes=[];
 for(const p of plans){const sourceSkill=fs.readFileSync(p.skillSource),targetSkill=fs.readFileSync(p.skillTarget),sourceRunner=fs.readFileSync(p.runnerSource),targetRunner=fs.readFileSync(p.runnerTarget);if(!sourceSkill.equals(targetSkill)||!sourceRunner.equals(targetRunner))throw new Error(`byte_mismatch_${p.group.name}`);const q=cp.spawnSync("python",[quick,path.dirname(p.skillTarget)],{encoding:"utf8",env:{...process.env,PYTHONDONTWRITEBYTECODE:"1",PYTHONUTF8:"1"}});if(q.status!==0)throw new Error(`global_skill_${p.group.name}`);bindings.push({kind:"skill",name:p.group.name,source:`${BASE}/x2/curated/skills/${p.group.name}/SKILL.md`,target_label:`global-skill:${p.group.name}`,bytes:sourceSkill.length,sha256:sha(sourceSkill),byte_equal:true,quick_validated:true},{kind:"runner",name:path.basename(p.runnerTarget),source:`${BASE}/x2/curated/runners/${path.basename(p.runnerTarget)}`,target_label:`d-first-simulation-common:${path.basename(p.runnerTarget)}`,bytes:sourceRunner.length,sha256:sha(sourceRunner),byte_equal:true});for(const operation of p.group.ops){const fixture=contracts.find(c=>c.data.operation===operation);if(!fixture)throw new Error(`fixture_${operation}`);const r=cp.spawnSync("node",[p.runnerTarget,fixture.file,path.join(ROOT,BASE,"tooling","simulation-engine.txt")],{encoding:"utf8"});if(r.status!==0||!same(JSON.parse(r.stdout),fixture.data.expected))throw new Error(`global_smoke_${operation}`);smokes.push({group:p.group.name,operation,proposal_id:fixture.data.proposal_id,passed:true,external_action:false});}}
 const receipt={schema:"ghc.family.sylven-global-promotion.v1",status:"VALIDATED_GLOBAL_PROMOTION",owner:"Sylven Arc",phase:"v691-v6",collision_preflight:{targets:10,collisions:0},skills_installed:5,runners_installed:5,merged_source_operations:20,skill_quick_validations:5,global_runner_smokes:smokes.length,bindings,smokes,packages_installed:0,incremental_spend_usd:0,overwrites:0,deletions:0,external_messages:0,same_owner_validation:true,independent_reproduction:false,terminal_verdict:"NOT_READY_FOR_STAGE_20"};fs.writeFileSync(receiptPath,encode(receipt),{encoding:"utf8",flag:"wx"});process.stdout.write(JSON.stringify({status:receipt.status,skills:5,runners:5,bindings:bindings.length,global_runner_smokes:smokes.length,collisions:0,overwrites:0}));
}
main();
