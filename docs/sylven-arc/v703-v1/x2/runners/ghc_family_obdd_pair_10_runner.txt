'use strict';
const fs=require('fs');
const p=process.argv[2];
if(!p){console.error(JSON.stringify({ok:false,error:'fixture path required'}));process.exit(2);}
let x;try{x=JSON.parse(fs.readFileSync(p,'utf8'));}catch(e){console.error(JSON.stringify({ok:false,error:'invalid json'}));process.exit(2);}
const allowed=new Set(['external_observation_gap','authority_reservation']);
const vars=Array.isArray(x.variables)?x.variables:[];
const valid=x.kind==='obdd-fixture'&&x.valid===true&&allowed.has(x.operation)&&vars.length>=2&&vars.length<=3&&new Set(vars).size===vars.length&&Number.isInteger(x.mask)&&x.mask>=0&&x.mask<(1<<(1<<vars.length));
if(!valid){console.error(JSON.stringify({ok:false,pair:10,rejected:true}));process.exit(2);}
console.log(JSON.stringify({ok:true,pair:10,operation:x.operation,accepted:true}));
