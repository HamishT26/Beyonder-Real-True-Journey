// Exact finite set-function core for Sylven v700-v3. No real beliefs, people, identities, observations or authority.
const BOUNDARY='Finite synthetic same-owner software evidence only; no human-belief measurement, empirical confirmation, identity event, professional/production/legal/cultural/Maori authority, independent reproduction, proof, canon or Stage 20 claim.';
function gcd(a,b){a=a<0n?-a:a;b=b<0n?-b:b;while(b)[a,b]=[b,a%b];return a||1n}
function Q(n,d=1n){n=BigInt(n);d=BigInt(d);if(d===0n)throw Error('zero_denominator');if(d<0n){n=-n;d=-d}const g=gcd(n,d);return [n/g,d/g]}
function parseQ(s){if(typeof s!=='string'||!/^-?\d+\/[1-9]\d*$/.test(s))throw Error('rational_string');const [n,d]=s.split('/');return Q(n,d)}
const add=(a,b)=>Q(a[0]*b[1]+b[0]*a[1],a[1]*b[1]),sub=(a,b)=>Q(a[0]*b[1]-b[0]*a[1],a[1]*b[1]),mul=(a,b)=>Q(a[0]*b[0],a[1]*b[1]),div=(a,b)=>{if(b[0]===0n)throw Error('division_by_zero');return Q(a[0]*b[1],a[1]*b[0])},neg=a=>[-a[0],a[1]],str=a=>`${a[0]}/${a[1]}`,cmp=(a,b)=>a[0]*b[1]<b[0]*a[1]?-1:a[0]*b[1]>b[0]*a[1]?1:0;
function validate(r){if(!r||typeof r!=='object'||Array.isArray(r))throw Error('request_object');const allowed=new Set(['op','profile_id','frame','mass','mass2','conditioning_mask','marginal_keep','values','target_mask','extension_label','execute','action']);for(const k of Object.keys(r))if(!allowed.has(k))throw Error('unexpected_field');if(typeof r.op!=='string')throw Error('operation');if(!Array.isArray(r.frame)||r.frame.length<2||r.frame.length>5||new Set(r.frame).size!==r.frame.length||r.frame.some(x=>typeof x!=='string'||!/^[a-z]$/.test(x)))throw Error('frame');const all=(1<<r.frame.length)-1;function mass(rows){if(!Array.isArray(rows)||!rows.length||rows.length>16)throw Error('mass_rows');const map=new Map();let total=Q(0);for(const row of rows){if(!row||!Number.isInteger(row.mask)||row.mask<0||row.mask>all||map.has(row.mask))throw Error('mass_mask');const q=parseQ(row.value);if(q[0]<0n)throw Error('negative_mass');map.set(row.mask,q);total=add(total,q)}if(cmp(total,Q(1))!==0)throw Error('mass_total');if(map.has(0)&&map.get(0)[0]!==0n)throw Error('empty_mass');return map}const m=mass(r.mass),m2=mass(r.mass2);if(!Number.isInteger(r.target_mask)||r.target_mask<0||r.target_mask>all)throw Error('target_mask');if(!Number.isInteger(r.conditioning_mask)||r.conditioning_mask<1||r.conditioning_mask>all)throw Error('conditioning_mask');if(!Array.isArray(r.marginal_keep)||!r.marginal_keep.length||r.marginal_keep.some(i=>!Number.isInteger(i)||i<0||i>=r.frame.length)||new Set(r.marginal_keep).size!==r.marginal_keep.length)throw Error('marginal_keep');if(!Array.isArray(r.values)||r.values.length!==r.frame.length||r.values.some(v=>!Number.isInteger(v)||Math.abs(v)>20))throw Error('values');return {all,m,m2}}
function sumMap(m,pred){let q=Q(0);for(const[k,v]of m)if(pred(k))q=add(q,v);return q}function table(all,fn){return Array.from({length:all+1},(_,mask)=>({mask,value:str(fn(mask))}))}
function belief(m,A){return sumMap(m,B=>(B&~A)===0)}function plausibility(m,A){return sumMap(m,B=>(B&A)!==0)}function commonality(m,A){return sumMap(m,B=>(A&~B)===0)}
function mobiusFromBelief(m,all){const bel=Array.from({length:all+1},(_,A)=>belief(m,A)),out=[];for(let A=0;A<=all;A++){let q=Q(0);for(let B=A;;B=(B-1)&A){const parity=((A^B).toString(2).match(/1/g)||[]).length;q=add(q,parity%2?neg(bel[B]):bel[B]);if(B===0)break}out.push({mask:A,value:str(q)})}return out}
function envelope(op,value,disposition='completed'){return {ok:true,value,error:null,disposition,boundary:BOUNDARY,operation:op}}
function evaluate(r){try{const{all,m}=validate(r),op=r.op,A=r.target_mask;if(op==='frame-structure')return envelope(op,{labels:[...r.frame],size:r.frame.length,powerset_size:all+1});if(op==='mass-normalize')return envelope(op,{entries:[...m].sort((a,b)=>a[0]-b[0]).map(([mask,q])=>({mask,value:str(q)})),sum:'1/1'});if(op==='focal-elements')return envelope(op,[...m].filter(([,q])=>q[0]!==0n).map(([mask,q])=>({mask,value:str(q)})));if(op==='belief-transform')return envelope(op,table(all,x=>belief(m,x)));if(op==='plausibility-transform')return envelope(op,table(all,x=>plausibility(m,x)));if(op==='commonality-transform')return envelope(op,table(all,x=>commonality(m,x)));if(op==='mobius-recover')return envelope(op,mobiusFromBelief(m,all));if(op==='interval-width')return envelope(op,{target:A,belief:str(belief(m,A)),plausibility:str(plausibility(m,A)),width:str(sub(plausibility(m,A),belief(m,A)))});if(op==='singleton-bounds')return envelope(op,r.frame.map((label,i)=>({label,belief:str(belief(m,1<<i)),plausibility:str(plausibility(m,1<<i))})));if(op==='mass-vacuity')return envelope(op,{frame_mass:str(m.get(all)||Q(0)),is_vacuous:m.size===1&&m.has(all)});throw Error('unsupported_operation')}catch(e){return {ok:false,value:null,error:String(e.message||e),disposition:'rejected',boundary:BOUNDARY,operation:r?.op??null}}}
function malformed(r){const x=JSON.parse(JSON.stringify(r));x.mass[0].value='1/0';return x}
module.exports={BOUNDARY,Q,parseQ,add,sub,mul,div,str,cmp,validate,belief,plausibility,commonality,mobiusFromBelief,evaluate,malformed};
