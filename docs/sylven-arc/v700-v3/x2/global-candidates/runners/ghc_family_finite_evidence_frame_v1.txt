'use strict';
function validate(x){if(!x||!Array.isArray(x.frame)||!x.frame.length||new Set(x.frame).size!==x.frame.length)return{ok:false,error:'invalid_frame'};if(!Array.isArray(x.mass)||!x.mass.length)return{ok:false,error:'invalid_mass'};const max=(2**x.frame.length)-1;let sum=0;for(const r of x.mass){if(!Number.isInteger(r.mask)||r.mask<0||r.mask>max||typeof r.value!=='string'||!/^(0|[1-9]\d*)\/[1-9]\d*$/.test(r.value))return{ok:false,error:'invalid_row'};const[a,b]=r.value.split('/').map(Number);sum+=a/b}return Math.abs(sum-1)<1e-12?{ok:true,frameSize:x.frame.length,rows:x.mass.length}:{ok:false,error:'mass_total'}}module.exports={validate};
