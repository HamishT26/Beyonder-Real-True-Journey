def git(*args):return subprocess.check_output(['git','-C',str(ROOT),*args])
def sha(data):return hashlib.sha256(data).hexdigest()
def compact(data):return json.dumps(data,ensure_ascii=False,sort_keys=True,separators=(',',':'),allow_nan=False).encode()
def snapshot(head):
    args=['diff','--raw','--no-abbrev','--no-renames',SOURCE]
    if head=='INDEX':args.insert(1,'--cached')
    else:args.append(head)
    entries=[]
    for line in git(*args).decode().splitlines():
        meta,path=line.split('\t',1);parts=meta.split();entries.append((path,parts[3],parts[4]))
    assert entries and all(status=='A' and (path.startswith(BASE) or path in SCRIPTS+[TEST]) for path,oid,status in entries),'owner allowlist or additive state'
    request=''.join(oid+'\n' for _,oid,_ in entries).encode();proc=subprocess.run(['git','-C',str(ROOT),'cat-file','--batch'],input=request,capture_output=True,check=True)
    data=proc.stdout;pos=0;blobs={}
    for path,oid,_ in entries:
        end=data.index(b'\n',pos);header=data[pos:end].split();size=int(header[2]);assert header[0].decode()==oid and header[1]==b'blob'
        blobs[path]=data[end+1:end+1+size];pos=end+size+2
    assert pos==len(data)
    return blobs
def load(blobs,path):return json.loads(blobs[BASE+path])
def manifest(blobs,path,expected=None):
    m=load(blobs,path);rows=m.get('entries',m.get('targets'));assert isinstance(rows,list)
    paths=[e['path'] for e in rows];assert len(paths)==len(set(paths))
    for e in rows:
        data=blobs[e['path']];assert len(data)==e['bytes'] and sha(data)==e['sha256'],e['path']
    if expected is not None:assert set(paths)==expected,'manifest coverage mismatch'
    return len(rows)
def equality(head):
    branch=git('branch','--show-current').decode().strip();assert branch=='codex/GHC-Family/mira-fenwick-v686-v2-full-tools'
    values=[git('rev-parse',x).decode().strip() for x in ['HEAD','@{upstream}','refs/remotes/origin/'+branch]]
    live=git('ls-remote','--heads','origin','refs/heads/'+branch).decode().split();assert len(live)==2;values.append(live[0])
    divergent=[int(x) for x in git('rev-list','--left-right','--count','HEAD...@{upstream}').split()]
    clean=not git('status','--porcelain=v1')
    assert values==[head]*4 and divergent==[0,0] and clean,'exact clean equality failed'
    return {'local_upstream_tracking_fresh_live':values,'divergence':divergent,'clean':clean}

def main():
    p=argparse.ArgumentParser(description=__doc__);p.add_argument('--head',required=True);p.add_argument('--phase',choices=['evidence','final'],required=True);p.add_argument('--evidence');p.add_argument('--receipt',type=Path,required=True);p.add_argument('--canonical',action='store_true');p.add_argument('--tool-python')
    args=p.parse_args();assert not args.receipt.exists(),'Receipt already exists; no replay.'
    args.receipt.parent.mkdir(parents=True,exist_ok=True)
    before=None
    if args.canonical:
        assert args.phase=='final' and args.head!='INDEX' and args.evidence
        before=equality(args.head)
        marker=args.receipt.with_suffix(args.receipt.suffix+'.invoked')
        with marker.open('x',encoding='utf-8') as f:json.dump({'head':args.head,'owner':'Mira Fenwick','invocations':1},f)
    payload={'schema':'ghc.family.mira.owner-validation.v1','source':SOURCE,'head':args.head,'evidence':args.evidence,'owner':'Mira Fenwick','phase':'v686-v2','observed_at':datetime.now(timezone.utc).isoformat(),
             'canonical_invocation_count':1 if args.canonical else 0,'canonical_success_count':0,'canonical_replay_count':0,'terminal_verdict':'NOT_READY_FOR_STAGE_20'}
    try:
        payload.update(validate(args))
        if args.canonical:payload['before']=before;payload['after']=equality(args.head)
        payload['status']='VALID_EXACT_FINAL_OWNER_SCOPED_CANONICAL' if args.canonical else 'PASS_OWNER_COMPONENT_PREFLIGHT'
        payload['canonical_success_count']=1 if args.canonical else 0
    except Exception as exc:
        payload.update(status='FAILED_RETAINED_ZERO_CREDIT',failure_type=type(exc).__name__,failure=str(exc)[:500],canonical_success_count=0)
    payload['payload_sha256']=sha(compact(payload))
    with args.receipt.open('x',encoding='utf-8',newline='\n') as f:f.write(json.dumps(payload,ensure_ascii=False,sort_keys=True,indent=2)+'\n')
    print(json.dumps({'status':payload['status'],'counts':payload.get('counts'), 'failure':payload.get('failure')}))
    return 1 if payload['status']=='FAILED_RETAINED_ZERO_CREDIT' else 0

if __name__=='__main__':raise SystemExit(main())
