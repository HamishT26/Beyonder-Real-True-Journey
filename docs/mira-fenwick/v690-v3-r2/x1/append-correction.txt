'use strict';
const fs=require('node:fs');
const file=process.argv[2],id=process.argv[3],title=process.argv[4];
if(!file||!id||!title)throw Error('ledger path, correction id and title required');
const l=JSON.parse(fs.readFileSync(file));if(l.methods.some(m=>m.method_id===id))throw Error('correction already recorded');
const m={method_id:id,title,failure_signature:title,trigger_preconditions:['Observed retained owner correction'],privacy_class:'sanitized_public',approval_class:'safe_now',candidate_workaround:'Apply only the recorded bounded correction and verify the affected invariant.',validation_witness_ids:[id+'-FAIL',id+'-PASS'],recurrence_guard:'Derive references from actual identifiers and check the complete link set before sealing.',rollback:'Retain the failed source and correction receipt; stop selecting the corrected projection if its check fails.',recommendation_state:'validated',supersedes:[],protected_gates:['No source or failed witness erasure','No independent reproduction or authority promotion'],retained_negative_ids:[id+'-FAIL'],scope_boundary:l.boundary};
l.methods.push(m);for(const result of ['fail','pass'])l.witnesses.push({witness_id:id+'-'+result.toUpperCase(),method_id:id,procedure:result==='fail'?'Retained incorrect method negative links':'Verify corrected links against actual witness IDs',scope:'Unsealed current owner ledger only',expected:result==='fail'?'Every retained negative link names an actual witness':'All links resolve without witness loss',observed:result==='fail'?'Two method lists contained aliases with no witness':'All links resolve; all prior witnesses retained',result,same_owner_only:true,independent_reproduction:false,retained_negative_ids:[id+'-FAIL'],boundary:l.boundary});
l.state_events.push({event_index:l.state_events.length+1,method_id:id,before:null,after:'candidate',reason:'Retained correction observed',witness_id:id+'-FAIL'},{event_index:l.state_events.length+2,method_id:id,before:'candidate',after:'validated',reason:'Focused link check passed',witness_id:id+'-PASS'});
l.counts={methods:l.methods.length,witnesses:l.witnesses.length,state_events:l.state_events.length,recommendations:l.recommendations.length,states:Object.fromEntries(['observed','candidate','validated','preferred','superseded','deprecated'].map(s=>[s,l.methods.filter(m=>m.recommendation_state===s).length])),witness_results:{pass:l.witnesses.filter(w=>w.result==='pass').length,fail:l.witnesses.filter(w=>w.result==='fail').length}};
fs.writeFileSync(file,JSON.stringify(l,null,2)+'\n');console.log(JSON.stringify(l.counts));
