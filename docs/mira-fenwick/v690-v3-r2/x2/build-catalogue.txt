'use strict';
const fs=require('node:fs'),path=require('node:path'),crypto=require('node:crypto');
const base=path.resolve(__dirname,'..'),root=path.resolve(base,'../../..'),rel=p=>path.relative(root,p).replaceAll('\\','/'),hash=p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex');
const read=p=>JSON.parse(fs.readFileSync(path.join(base,p)));const plan=read('plan/skills-runners.json'),promotions=read('x2/promotion-plan.json');
const old=JSON.parse(fs.readFileSync(path.join(root,'docs/mira-fenwick/v690-v3/x2/meta-tool-catalogue.json')));
const cards=old.cards.map(c=>({...c,card_id:'inherited:'+c.card_id,status:'compatibility',source_execution_credit:0,source_novelty_credit:0}));
function add(name,kind,p,triggers,callers,scope){cards.push({card_id:'mira-r2:'+kind+':'+name,name,kind,source_path:rel(p),sha256:hash(p),status:'current',evidence_state:'validated',owner_scope:scope,triggers,caller_paths:callers,rollback:'Stop selecting the additive package; preserve prior callers and both evidence sets.',protected_gates:['No independent reproduction, real authority or identity promotion','No canonical replay or accepted-message resend'],execution_authority:'owner_self_scoped_delta',repository_scan:false,module_scan:false,cross_lane_scan:false,unchanged_history_scan:false,sibling_lane_mutation:false,supported_endpoint_kinds:['not_applicable'],route_controller_scope:'not_applicable',delivery_ack_required:false});}
for(const s of plan.local_skills)add(s.name,'skill',path.join(base,s.session,'skills',s.name,'SKILL.md'),[s.operation],[rel(path.join(base,s.session,'skill-validation.json'))],'Mira remaster local operation');
for(const r of plan.local_runners)add(r.name,'runner',path.join(base,r.session,'runners',r.name),r.operations,[rel(path.join(base,r.session,'runner-validation.json'))],'Mira remaster paired operations');
for(const g of promotions.groups){add(g.name,'skill',path.join(base,'x2/promotions/skills',g.name,'SKILL.md'),g.operations,[rel(path.join(base,'x2/installed-tool-validation.json'))],'Family finite settlement contracts');add(g.runner,'runner',path.join(base,'x2/promotions/public/runners',g.runner),g.operations,[rel(path.join(base,'x2/installed-tool-validation.json'))],'Family finite settlement contracts');}
const overlaps=[];for(let i=0;i<cards.length;i++)for(let j=i+1;j<cards.length;j++){const shared=cards[i].triggers.filter(x=>cards[j].triggers.includes(x));if(shared.length)overlaps.push({first:cards[i].card_id,second:cards[j].card_id,triggers:shared,resolution:'Intentional guide, paired runner or merged package overlap. Match the operation and caller scope; prefer the installed merged runner for its declared group. No silent overwrite.'});}
fs.writeFileSync(path.join(__dirname,'meta-tool-catalogue.json'),JSON.stringify({schema:old.schema,owner:'Mira Fenwick',phase:'v690-v3-r2',card_count:cards.length,cards,boundary:'Source-faithful capability selection. Inherited records have zero new execution or novelty credit.'},null,2)+'\n',{flag:'wx'});
fs.writeFileSync(path.join(__dirname,'catalogue-collision-review.json'),JSON.stringify({cards:cards.length,reviewed_overlaps:overlaps.length,records:overlaps,silent_selection:false,old_namespace_overwrite:false},null,2)+'\n',{flag:'wx'});
console.log(JSON.stringify({cards:cards.length,inherited:old.cards.length,new:cards.length-old.cards.length,reviewed_overlaps:overlaps.length}));
