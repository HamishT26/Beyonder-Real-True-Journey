'use strict';
const first=require('../x1/core.txt');
const {fields,integer,number,boolean,text,array,unique,fail,order,graphStep}=first;
const additions={
 idempotent_replay(p){fields(p,['initial','events']);integer(p.initial,-100000,100000);array(p.events,0,100);const seen=new Map(),applied=[],skipped=[];let value=p.initial;for(const e of p.events){fields(e,['id','delta']);text(e.id);integer(e.delta,-1000,1000);if(seen.has(e.id)){if(seen.get(e.id)!==e.delta)fail('conflicting_event');skipped.push(e.id);}else{seen.set(e.id,e.delta);value+=e.delta;applied.push(e.id);}}return{value,applied,skipped};},
 rollback_projection(p){fields(p,['history','index']);array(p.history,1,100).forEach(x=>integer(x,-100000,100000));integer(p.index,0,p.history.length-1);return{active:p.history[p.index],retained:p.history.slice(),inactive_later_indices:p.history.map((_,i)=>i).filter(i=>i>p.index)};},
 consent_intersection(p){fields(p,['now','records']);integer(p.now);array(p.records,1,30);for(const r of p.records){fields(r,['scopes','expires']);array(r.scopes,0,20).forEach(x=>text(x));unique(r.scopes);integer(r.expires);}const active=p.records.filter(r=>p.now<r.expires);const scopes=active.length===p.records.length?p.records[0].scopes.filter(s=>p.records.every(r=>r.scopes.includes(s))).sort():[];return{scopes,active_records:active.length,external_action:false};},
 revocation_order(p){fields(p,['granted','expires','revoked','now']);['granted','expires','now'].forEach(k=>integer(p[k]));if(p.revoked!==null)integer(p.revoked);if(p.expires<p.granted||(p.revoked!==null&&p.revoked<p.granted))fail('chronology');const reason=p.now>=p.expires?'expired':p.revoked!==null&&p.now>=p.revoked?'revoked':p.now<p.granted?'not_yet_valid':'current';return{active:reason==='current',reason,external_action:false};},
 task_scheduler(p){fields(p,['tasks']);const sorted=order(p.tasks,true),start=new Map(),finish=new Map();for(const id of sorted){const t=p.tasks.find(t=>t.id===id);const at=t.parents.length?Math.max(...t.parents.map(p=>finish.get(p))):0;start.set(id,at);finish.set(id,at+t.duration);}return{starts:Object.fromEntries(start),finishes:Object.fromEntries(finish),makespan:Math.max(...finish.values()),resource_limits_modeled:false};},
 queue_service(p){fields(p,['arrivals','capacity']);array(p.arrivals,0,100).forEach(x=>integer(x,0,1000));integer(p.capacity,0,1000);let q=0;const served=[],backlog=[];for(const a of p.arrivals){q+=a;const s=Math.min(q,p.capacity);q-=s;served.push(s);backlog.push(q);}return{served,backlog,total_arrivals:p.arrivals.reduce((a,b)=>a+b,0),total_served:served.reduce((a,b)=>a+b,0),residual:q};},
 entropy_mix(p){fields(p,['counts']);array(p.counts,1,100).forEach(x=>integer(x,0,10000));const total=p.counts.reduce((a,b)=>a+b,0);if(total===0)fail('empty_distribution');let bits=0;for(const n of p.counts)if(n){const q=n/total;bits-=q*Math.log2(q);}return{total,bits,effective_states:2**bits,physical_entropy:false};},
 energy_step:graphStep,
 html_escape(p){fields(p,['text']);if(typeof p.text!=='string')fail('invalid_type');if(p.text.length>10000)fail('bounds');const map={'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'};return{text:p.text.replace(/[&<>"']/g,c=>map[c]),context:'html_text_only'};},
 simulation_step(p){fields(p,['stock','production','reserve','agents']);['stock','production','reserve'].forEach(k=>integer(p[k],0,10000));array(p.agents,1,30);for(const a of p.agents){fields(a,['id','need','consent']);text(a.id);integer(a.need,0,1000);boolean(a.consent);}unique(p.agents.map(a=>a.id));let stock=p.stock+p.production,consumed=0;const allocations=[],refused=[];for(const a of p.agents){if(!a.consent){allocations.push({id:a.id,amount:0});refused.push(a.id);continue;}const amount=Math.min(a.need,Math.max(0,stock-p.reserve));stock-=amount;consumed+=amount;allocations.push({id:a.id,amount});}return{stock,production:p.production,consumed,allocations,refused,residual:p.stock+p.production-stock-consumed,external_action:false};}
};
const operations={...first.operations,...additions};
const evaluate=r=>first.evaluate(r,operations);
module.exports={...first,operations,evaluate,cli:table=>first.cli(table||operations)};
if(require.main===module)first.cli(operations);
