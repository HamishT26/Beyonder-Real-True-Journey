'use strict';
const allowed=new Set(['initialize','account/rateLimits/read','thread/list','thread/read','thread/turns/list','thread/items/list']);
async function connect(endpoint){
 if(!/^ws:\/\/127\.0\.0\.1:[0-9]+$/.test(endpoint))throw Error('loopback_endpoint_required');
 const ws=new WebSocket(endpoint);await new Promise((resolve,reject)=>{const t=setTimeout(()=>reject(Error('connect_timeout')),10000);ws.addEventListener('open',()=>{clearTimeout(t);resolve();},{once:true});ws.addEventListener('error',()=>{clearTimeout(t);reject(Error('connect_error'));},{once:true});});
 let seq=0;const pending=new Map();ws.addEventListener('message',e=>{let m;try{m=JSON.parse(e.data);}catch{return;}if(pending.has(m.id)){const p=pending.get(m.id);pending.delete(m.id);clearTimeout(p.timer);m.error?p.reject(Object.assign(Error('rpc_error'),{rpcError:m.error})):p.resolve(m.result);}});
 const rpc=(method,params={})=>{if(!allowed.has(method))throw Error('read_only_method_guard');return new Promise((resolve,reject)=>{const id=++seq,timer=setTimeout(()=>{pending.delete(id);reject(Error('rpc_timeout'));},20000);pending.set(id,{resolve,reject,timer});ws.send(JSON.stringify({id,method,params}));});};
 const init=await rpc('initialize',{clientInfo:{name:'mira_durable_recovery',title:'Mira documented local task reader',version:'1.0.0'},capabilities:{experimentalApi:true}});ws.send(JSON.stringify({method:'initialized',params:{}}));
 return{rpc,init,close(){for(const p of pending.values())clearTimeout(p.timer);pending.clear();ws.close();}};
}
module.exports={connect};
