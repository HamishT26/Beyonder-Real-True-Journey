'use strict';
const fs=require('node:fs'),path=require('node:path'),cp=require('node:child_process'),assert=require('node:assert/strict'),crypto=require('node:crypto');
const base=path.resolve(__dirname,'..'),root=path.resolve(base,'../../..'),plan=JSON.parse(fs.readFileSync(path.join(__dirname,'promotion-plan.json'))),cases=JSON.parse(fs.readFileSync(path.join(base,'plan/new-proposals.json'))).proposals;
const mode=process.argv[2];if(!['prepared','installed'].includes(mode))throw Error('mode');
const targetRoot=mode==='prepared'?path.join(__dirname,'promotions/public'):'D:/GHC-Archives/global-tools/family-settlement-lab';
const rows=[];for(const g of plan.groups){const folder=mode==='prepared'?path.join(__dirname,'promotions/skills',g.name):'C:/Users/hamis/.codex/skills/'+g.name;const v=cp.spawnSync('D:/GHC-Archives/global-tools/python/ghc-family-tools/Scripts/python.exe',['C:/Users/hamis/.codex/skills/.system/skill-creator/scripts/quick_validate.py',folder],{encoding:'utf8',timeout:10000,env:{...process.env,PYTHONDONTWRITEBYTECODE:'1'}});assert.equal(v.status,0,v.stderr||v.stdout);rows.push({kind:'skill',name:g.name,result:'pass'});
 const runner=path.join(targetRoot,'runners',g.runner);for(const op of g.operations){const c=cases.find(x=>x.operation===op),out=JSON.parse(cp.execFileSync(process.execPath,[runner],{input:JSON.stringify(c.request),encoding:'utf8',timeout:10000}));assert.deepEqual(out,c.expected_envelope);rows.push({kind:'runner',name:g.runner,operation:op,result:'pass'});}const foreign=cases.find(x=>!g.operations.includes(x.operation));const bad=JSON.parse(cp.execFileSync(process.execPath,[runner],{input:JSON.stringify(foreign.request),encoding:'utf8',timeout:10000}));assert.equal(bad.error,'unknown_operation');rows.push({kind:'runner',name:g.runner,operation:'outside_group_refusal',result:'pass',subject_result:'fail',original_success_credit:0});}
const review=JSON.parse(cp.execFileSync(process.execPath,[path.join(targetRoot,'runners/ghc_family_settlement_recovery.txt'),'--profile',path.join(base,'plan/workflow-v5.json'),'--route',path.join(base,'plan/roster-v5.json')],{encoding:'utf8',timeout:10000}));assert.equal(review.valid,true);rows.push({kind:'workflow',name:'v5-profile-review',result:'pass',checks:review.check_count});
const hashes=plan.targets.map(t=>{const p=mode==='prepared'?path.join(root,t.source):t.destination;const actual=crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex');assert.equal(actual,t.sha256);return{source:t.source,sha256:actual,match:true};});
const result={schema:'ghc.family.promotion-checks.v1',mode,valid:true,global_skills:5,public_runners:5,checks:rows.length,records:rows,byte_bindings:hashes,independent_reproduction:false};
fs.writeFileSync(path.join(__dirname,mode==='prepared'?'promotion-preflight.json':'installed-tool-validation.json'),JSON.stringify(result,null,2)+'\n',{flag:'wx'});console.log(JSON.stringify({mode,checks:rows.length,byte_bindings:hashes.length,valid:true}));
