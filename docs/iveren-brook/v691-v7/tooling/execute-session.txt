"use strict";
const fs=require('fs'),path=require('path'),crypto=require('crypto'),cp=require('child_process'),assert=require('assert/strict');
const BASE=path.resolve(__dirname,'..'),session=process.argv[2],BANK=process.argv[3];assert.ok(['x1','x2'].includes(session));assert.ok(BANK);
const engine=require(BASE+'/'+session+'/engine.txt'),freeze=JSON.parse(fs.readFileSync(BASE+'/plan/proposal-freeze.json','utf8')),plan=JSON.parse(fs.readFileSync(BASE+'/plan/skills-runners.json','utf8'));
const selected=freeze.proposals.filter(p=>p.session===session),rows=[];
const digest=b=>crypto.createHash('sha256').update(b).digest('hex');
for(const p of selected){const skill=plan.local_skills.find(s=>s.operation===p.operation),runner=plan.local_runners.find(r=>r.operations.includes(p.operation)),skillBytes=fs.readFileSync(BASE+'/'+session+'/skills/'+skill.name+'/SKILL.md');assert.ok(skillBytes.toString('utf8').includes('Source operation: '+p.operation));const before=JSON.stringify(p.input),adverseBefore=JSON.stringify(p.adverse_input);const observed=JSON.parse(cp.execFileSync(process.execPath,[BASE+'/'+session+'/runners/'+runner.name,JSON.stringify({operation:p.operation,input:p.input})],{encoding:'utf8',maxBuffer:2e6})),rejected=engine.run(p.operation,p.adverse_input),replayed=engine.run(p.operation,p.input);const failures=[];for(const [name,actual,expected] of [['positive',observed,p.expected],['adverse',rejected,p.adverse_expected],['replay',replayed,p.expected]])try{assert.deepEqual(actual,expected);}catch{failures.push({name,actual,expected});}if(JSON.stringify(p.input)!==before||JSON.stringify(p.adverse_input)!==adverseBefore)failures.push({name:'input_nonmutation'});
 if(observed.ok&&observed.value&&observed.value.segments){const segments=observed.value.segments;for(const j of p.input.jobs){const allocated=segments.filter(s=>s.id===j.id).reduce((n,s)=>n+s.end-s.start,0);if(allocated!==j.service||segments.some(s=>s.id===j.id&&s.start<j.arrival))failures.push({name:'independent_service_conservation',id:j.id});}for(let i=0;i<segments.length;i++){if(segments[i].end<=segments[i].start||(i&&segments[i].start<segments[i-1].end))failures.push({name:'timeline_partition'});}}
 rows.push({proposal_id:p.proposal_id,operation:p.operation,disposition:p.expected_disposition,runner:runner.name,skill:skill.name,skill_sha256:digest(skillBytes),observed,rejected,replayed,input_unchanged:JSON.stringify(p.input)===before&&JSON.stringify(p.adverse_input)===adverseBefore,failures,validation_result:failures.length?'fail':'pass',adverse_subject_original_success_credit:0,adverse_validator_pass:failures.every(f=>f.name!=='adverse')});}
const result={schema:'ghc.family.finite-queue-session-execution.v1',session,owner:'Iveren Brook',phase:'v691-v7',plan_commit:process.argv[4],recorded_at_utc:new Date().toISOString(),rows,total:rows.length,passing:rows.filter(r=>r.validation_result==='pass').length,failing:rows.filter(r=>r.validation_result==='fail').length,portfolios:{safe:100,candidate:100,clean_fix_refine:100},scope:'same-owner synthetic fixed contracts',independent_reproduction:false,source_tests_executed:0,external_actions:0};
fs.writeFileSync(BANK+'/'+session+'-execution-attempt-001.json',JSON.stringify(result,null,2)+'\n',{flag:'wx'});console.log(JSON.stringify({session,passing:result.passing,failing:result.failing,failures:rows.filter(r=>r.validation_result==='fail').map(r=>({proposal_id:r.proposal_id,failures:r.failures}))}));if(result.failing)process.exitCode=1;
