"use strict";
const fs=require('fs'),path=require('path'),assert=require('assert/strict');
const BASE=path.resolve(__dirname,'..'),session=process.argv[2];assert.ok(['x1','x2'].includes(session));
const freeze=JSON.parse(fs.readFileSync(BASE+'/plan/proposal-freeze.json','utf8')),plan=JSON.parse(fs.readFileSync(BASE+'/plan/skills-runners.json','utf8'));
const write=(p,v)=>{const out=BASE+'/'+session+'/'+p;fs.mkdirSync(path.dirname(out),{recursive:true});fs.writeFileSync(out,typeof v==='string'?v:JSON.stringify(v,null,2)+'\n',{flag:'wx'});};
for(const p of freeze.proposals.filter(p=>p.session===session))write('contracts/'+p.proposal_id.slice(-3)+'.json',p);
for(const s of plan.local_skills.filter(s=>s.session===session)){
 const p=freeze.proposals.find(p=>p.operation===s.operation),runner=plan.local_runners.find(r=>r.operations.includes(s.operation));
 const lines=['---','name: '+s.name,'description: "Apply the bounded '+s.operation+' contract to synthetic queue evidence, with closed input fields and retained refusal cases."','---','','# '+p.title.split(':')[0],'','Use this skill when a caller needs the declared finite '+s.operation+' operation. Select '+runner.name+' and the matching frozen operation/input envelope. This is a local software practice lens, not a professional qualification.','','1. Read the matching contract in ../../contracts and preserve its complete input, expected output, adverse input and expected refusal.','2. Read ../../../plan/contract-specifications.md for integer limits, timing conventions, tie rules and evidence boundaries.','3. Pass a JSON object containing operation and input to ../../runners/'+runner.name+'. The runner writes one JSON envelope to standard output. It performs no external action.','4. Compare all envelope fields and confirm that the input is unchanged. Retain a mismatch at zero original success credit. Verify the preregistered adverse request separately.','5. Replay the immutable fixture only within the declared deterministic review task. Keep represented, open_gap and exact_gate consequences at their original ceilings.','','Source operation: '+s.operation+'. Source: '+p.source_refs.join(', ')+'. The case identifiers are '+s.source_case_ids.join(', ')+'.','','Rollback: stop selecting this owner-local skill or point back to the exact retained owner commit. Never erase a failed input, change the oracle to follow the implementation, or overwrite a predecessor skill.','',''+freeze.boundary,'','Same-owner checks are not independent reproduction. Manual accessibility and governed real-world evaluation remain open. NOT_READY_FOR_STAGE_20.',''];
 write('skills/'+s.name+'/SKILL.md',lines.join('\n'));
}
for(const r of plan.local_runners.filter(r=>r.session===session)){
 write('runners/'+r.name,'"use strict";\nconst engine=require("../engine.txt");\nconst allowed='+JSON.stringify(r.operations)+';\nlet result;try{if(process.argv.length!==3)throw Error("INVALID_REQUEST");const request=JSON.parse(process.argv[2]);engine.helpers.closed(request,["operation","input"]);if(!allowed.includes(request.operation))throw Error("UNKNOWN_OPERATION");result=engine.run(request.operation,request.input);}catch(e){result={ok:false,value:null,error:e.message,disposition:"open_gap"};}\nprocess.stdout.write(JSON.stringify(result)+"\\n");\n');
}
console.log(JSON.stringify({session,contracts:100,skills:10,runners:5}));
