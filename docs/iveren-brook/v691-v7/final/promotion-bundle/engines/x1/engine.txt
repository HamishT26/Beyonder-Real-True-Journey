"use strict";
// Finite owner-local queue contracts. Pure functions; no network, files, clocks or randomness.
const fail=code=>{throw new Error(code);};
const object=v=>v!==null&&typeof v==='object'&&!Array.isArray(v)&&(Object.getPrototypeOf(v)===Object.prototype||Object.getPrototypeOf(v)===null);
function closed(v,keys,code='INVALID_REQUEST'){if(!object(v)||Object.keys(v).length!==keys.length||keys.some(k=>!Object.hasOwn(v,k)))fail(code);}
const integer=(v,min=0,max=10000)=>Number.isSafeInteger(v)&&!Object.is(v,-0)&&v>=min&&v<=max;
const label=v=>typeof v==='string'&&/^[A-Za-z][A-Za-z0-9_-]{0,39}$/.test(v);
function list(v,code,max=32){if(!Array.isArray(v)||v.length>max)fail(code);}
function unique(rows,code){if(new Set(rows.map(r=>r.id)).size!==rows.length)fail(code);}
function fraction(n,d){if(d===0||d===0n)return null;let a=BigInt(n),b=BigInt(d);if(b<0n){a=-a;b=-b;}let x=a<0n?-a:a,y=b;while(y){const t=x%y;x=y;y=t;}return String(a/x)+'/'+String(b/x);}
function jobs(v){list(v,'INVALID_WORKLOAD',8);for(const j of v){closed(j,['id','arrival','service','priority','weight','deadline'],'INVALID_WORKLOAD');if(!label(j.id)||!integer(j.arrival,0,1000)||!integer(j.service,1,100)||!integer(j.priority,0,100)||!integer(j.weight,1,100)||!integer(j.deadline,0,1000))fail('INVALID_WORKLOAD');}unique(v,'INVALID_WORKLOAD');return v.map((j,index)=>({...j,index,remaining:j.service}));}
function append(segments,id,start,end){if(segments.length&&segments.at(-1).id===id&&segments.at(-1).end===start)segments.at(-1).end=end;else segments.push({id,start,end});}
function scheduleMetrics(input,segments){const rows=input.map(j=>{const used=segments.filter(s=>s.id===j.id);const start=used[0].start,finish=used.at(-1).end;return {id:j.id,start,finish,wait:finish-j.arrival-j.service,turnaround:finish-j.arrival,response:start-j.arrival};});const makespan=segments.length?segments.at(-1).end:0,busy=input.reduce((n,j)=>n+j.service,0);return {segments,jobs:rows,makespan,busy,idle:makespan-busy,mean_wait:fraction(rows.reduce((n,j)=>n+j.wait,0),rows.length),mean_turnaround:fraction(rows.reduce((n,j)=>n+j.turnaround,0),rows.length),mean_response:fraction(rows.reduce((n,j)=>n+j.response,0),rows.length)};}
function scheduler(input,mode){closed(input,mode==='round_robin'?['jobs','quantum']:mode==='aging_priority'?['jobs','aging_interval']:['jobs']);if(mode==='round_robin'&&!integer(input.quantum,1,100))fail('INVALID_QUANTUM');if(mode==='aging_priority'&&!integer(input.aging_interval,1,100))fail('INVALID_AGING');const pending=jobs(input.jobs),segments=[];let time=0,steps=0;
 if(mode==='round_robin'){const arrivals=[...pending].sort((a,b)=>a.arrival-b.arrival||a.index-b.index),queue=[];let next=0;const admit=()=>{while(next<arrivals.length&&arrivals[next].arrival<=time)queue.push(arrivals[next++]);};while(next<arrivals.length||queue.length){if(++steps>10000)fail('STEP_BUDGET');if(!queue.length)time=Math.max(time,arrivals[next].arrival);admit();const j=queue.shift(),duration=Math.min(input.quantum,j.remaining);append(segments,j.id,time,time+duration);time+=duration;j.remaining-=duration;admit();if(j.remaining)queue.push(j);}return scheduleMetrics(input.jobs,segments);}
 while(pending.some(j=>j.remaining>0)){if(++steps>10000)fail('STEP_BUDGET');let ready=pending.filter(j=>j.remaining>0&&j.arrival<=time);if(!ready.length){time=Math.min(...pending.filter(j=>j.remaining>0).map(j=>j.arrival));ready=pending.filter(j=>j.remaining>0&&j.arrival<=time);}const rank=j=>mode==='sjf'?j.service:mode==='srtf'?j.remaining:mode==='static_priority'?j.priority:mode==='aging_priority'?j.priority-Math.floor((time-j.arrival)/input.aging_interval):0;ready.sort((a,b)=>rank(a)-rank(b)||a.arrival-b.arrival||a.index-b.index);const j=ready[0],duration=mode==='srtf'?1:j.remaining;append(segments,j.id,time,time+duration);time+=duration;j.remaining-=duration;}
 return scheduleMetrics(input.jobs,segments);
}
const operations={
 workload_schema(input){closed(input,['jobs']);const q=jobs(input.jobs);return {job_count:q.length,total_service:q.reduce((n,j)=>n+j.service,0),arrival_span:q.length?[Math.min(...q.map(j=>j.arrival)),Math.max(...q.map(j=>j.arrival))]:null};},
 fcfs:i=>scheduler(i,'fcfs'),sjf:i=>scheduler(i,'sjf'),round_robin:i=>scheduler(i,'round_robin'),srtf:i=>scheduler(i,'srtf'),static_priority:i=>scheduler(i,'static_priority'),aging_priority:i=>scheduler(i,'aging_priority'),
 deadline_metrics(input){closed(input,['records']);list(input.records,'INVALID_RECORDS');for(const r of input.records){closed(r,['id','completion','deadline'],'INVALID_RECORDS');if(!label(r.id)||!integer(r.completion)||!integer(r.deadline))fail('INVALID_RECORDS');}unique(input.records,'INVALID_RECORDS');const records=input.records.map(r=>({id:r.id,lateness:r.completion-r.deadline,tardiness:Math.max(0,r.completion-r.deadline),earliness:Math.max(0,r.deadline-r.completion)}));return {records,missed:records.filter(r=>r.lateness>0).map(r=>r.id),maximum_lateness:records.length?Math.max(...records.map(r=>r.lateness)):null,total_tardiness:records.reduce((n,r)=>n+r.tardiness,0)};},
 occupancy_area(input){closed(input,['spans','window']);const w=input.window;if(!Array.isArray(w)||w.length!==2||!w.every(x=>integer(x))||w[1]<=w[0])fail('INVALID_WINDOW');list(input.spans,'INVALID_SPANS');for(const r of input.spans){closed(r,['id','arrival','finish'],'INVALID_SPANS');if(!label(r.id)||!integer(r.arrival)||!integer(r.finish)||r.finish<r.arrival)fail('INVALID_SPANS');}unique(input.spans,'INVALID_SPANS');const area=input.spans.reduce((n,r)=>n+Math.max(0,Math.min(r.finish,w[1])-Math.max(r.arrival,w[0])),0),complete=input.spans.length>0&&input.spans.every(r=>r.arrival>=w[0]&&r.finish<=w[1]);return {area,mean_occupancy:fraction(area,w[1]-w[0]),complete_cohort:complete,arrival_rate:complete?fraction(input.spans.length,w[1]-w[0]):null,mean_sojourn:complete?fraction(input.spans.reduce((n,r)=>n+r.finish-r.arrival,0),input.spans.length):null,stationarity_claim:false};},
 jain_fairness(input){closed(input,['allocations']);list(input.allocations,'INVALID_ALLOCATION');if(!input.allocations.every(x=>integer(x)))fail('INVALID_ALLOCATION');const total=input.allocations.reduce((n,x)=>n+x,0),squares=input.allocations.reduce((n,x)=>n+x*x,0);return {index:fraction(BigInt(total)*BigInt(total),BigInt(input.allocations.length)*BigInt(squares)),allocation_count:input.allocations.length,total,sum_squares:squares,normative_fairness:false};}
};
function run(operation,input){try{if(!Object.hasOwn(operations,operation))fail('UNKNOWN_OPERATION');return {ok:true,value:operations[operation](input),error:null,disposition:'completed'};}catch(error){return {ok:false,value:null,error:error.message,disposition:'open_gap'};}}
module.exports={run,operations,helpers:{fail,closed,integer,label,list,unique,fraction,jobs,append,scheduleMetrics}};
