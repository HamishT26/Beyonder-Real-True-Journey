"use strict";
// The immutable x1 module supplies validation and metric helpers; no x1 suite is replayed here.
const helpers=require('../x1/engine.txt').helpers;
const {fail,closed,integer,label,list,unique,fraction,jobs,append,scheduleMetrics}=helpers;
function bucket(input,token){closed(input,['capacity','rate','initial','events']);if(!integer(input.capacity,1)||!integer(input.rate)||!integer(input.initial,0,input.capacity))fail('INVALID_BUCKET');list(input.events,'INVALID_EVENTS');let previous=0;for(const e of input.events){closed(e,['id','time','cost'],'INVALID_EVENTS');if(!label(e.id)||!integer(e.time)||!integer(e.cost)||e.time<previous)fail('INVALID_EVENTS');previous=e.time;}unique(input.events,'INVALID_EVENTS');let state=input.initial,time=0;const field=token?'tokens':'level',records=[];for(const e of input.events){state=token?Math.min(input.capacity,state+(e.time-time)*input.rate):Math.max(0,state-(e.time-time)*input.rate);const accepted=token?e.cost<=state:state+e.cost<=input.capacity;if(accepted)state+=token?-e.cost:e.cost;records.push({id:e.id,accepted,[field]:state});time=e.time;}return {records,[field]:state,last_time:time,network_action:false};}
function rat(n,d=1n){let a=BigInt(n),b=BigInt(d),x=a<0n?-a:a,y=b;while(y){const t=x%y;x=y;y=t;}return {n:a/x,d:b/x};}
const compare=(a,b)=>a.n*b.d<b.n*a.d?-1:a.n*b.d>b.n*a.d?1:0;
const plus=(a,b)=>rat(a.n*b.d+b.n*a.d,a.d*b.d);
const operations={
 capacity_admission(input){closed(input,['requests','capacity']);if(!integer(input.capacity,1,100))fail('INVALID_CAPACITY');list(input.requests,'INVALID_REQUESTS');for(const r of input.requests){closed(r,['id','arrival','service'],'INVALID_REQUESTS');if(!label(r.id)||!integer(r.arrival,0,1000)||!integer(r.service,1,100))fail('INVALID_REQUESTS');}unique(input.requests,'INVALID_REQUESTS');const ordered=input.requests.map((r,index)=>({...r,index})).sort((a,b)=>a.arrival-b.arrival||a.index-b.index),accepted=[],rejected=[];let active=[],peak=0;for(const r of ordered){active=active.filter(t=>t>r.arrival);if(active.length<input.capacity){accepted.push({id:r.id,start:r.arrival,finish:r.arrival+r.service});active.push(r.arrival+r.service);peak=Math.max(peak,active.length);}else rejected.push(r.id);}return {accepted,rejected,peak_busy:peak};},
 token_bucket:i=>bucket(i,true),leaky_bucket:i=>bucket(i,false),
 weighted_finish_tags(input){closed(input,['packets']);list(input.packets,'INVALID_PACKETS');const weights=new Map();for(const p of input.packets){closed(p,['id','flow','length','weight','virtual_time'],'INVALID_PACKETS');if(!label(p.id)||!label(p.flow)||!integer(p.length,1)||!integer(p.weight,1,100)||!integer(p.virtual_time))fail('INVALID_PACKETS');if(weights.has(p.flow)&&weights.get(p.flow)!==p.weight)fail('INVALID_PACKETS');weights.set(p.flow,p.weight);}unique(input.packets,'INVALID_PACKETS');const previous=new Map(),tagged=input.packets.map((p,index)=>{const old=previous.get(p.flow)||rat(0n),virtual=rat(p.virtual_time),start=compare(old,virtual)>=0?old:virtual,finish=plus(start,rat(p.length,p.weight));previous.set(p.flow,finish);return {id:p.id,finish,index};});return {tags:tagged.map(t=>({id:t.id,finish:String(t.finish.n)+'/'+String(t.finish.d)})),order:[...tagged].sort((a,b)=>compare(a.finish,b.finish)||a.index-b.index).map(t=>t.id),fair_service_claim:false};},
 deficit_round_robin(input){closed(input,['flows','max_rounds']);list(input.flows,'INVALID_DRR');if(!integer(input.max_rounds,0,100))fail('INVALID_DRR');let count=0;for(const f of input.flows){closed(f,['id','quantum','packets'],'INVALID_DRR');list(f.packets,'INVALID_DRR');count+=f.packets.length;if(!label(f.id)||!integer(f.quantum,1,100)||!f.packets.every(n=>integer(n,1)))fail('INVALID_DRR');}if(count>32)fail('INVALID_DRR');unique(input.flows,'INVALID_DRR');const flows=input.flows.map(f=>({...f,packets:[...f.packets],deficit:0})),departures=[];let round=0;while(round<input.max_rounds&&flows.some(f=>f.packets.length)){round++;for(const f of flows){if(!f.packets.length){f.deficit=0;continue;}f.deficit+=f.quantum;while(f.packets.length&&f.packets[0]<=f.deficit){const length=f.packets.shift();f.deficit-=length;departures.push({flow:f.id,length,round});}if(!f.packets.length)f.deficit=0;}}return {departures,remaining:Object.fromEntries(flows.map(f=>[f.id,f.packets])),deficits:Object.fromEntries(flows.map(f=>[f.id,f.deficit])),rounds_used:round,real_router:false};},
 vacation_fcfs(input){closed(input,['jobs','vacations']);const pending=jobs(input.jobs);list(input.vacations,'INVALID_VACATIONS');for(const v of input.vacations)if(!Array.isArray(v)||v.length!==2||!v.every(n=>integer(n))||v[1]<=v[0])fail('INVALID_VACATIONS');const vacations=input.vacations.map(v=>[...v]).sort((a,b)=>a[0]-b[0]||a[1]-b[1]),segments=[];pending.sort((a,b)=>a.arrival-b.arrival||a.index-b.index);let time=0;for(const j of pending){time=Math.max(time,j.arrival);for(const [start,end] of vacations)if(time<end&&time+j.service>start)time=end;append(segments,j.id,time,time+j.service);time+=j.service;}return scheduleMetrics(input.jobs,segments);},

 comparison_ceiling(input){closed(input,['metric','left','right','evidence','operational_run','independent_review']);const metrics=['mean_wait','mean_turnaround','mean_response','deadline_misses','occupancy','token_admission','backlog_overflow','weighted_tag_order','deficit_departures','vacation_delay'];if(!metrics.includes(input.metric)||!integer(input.left)||!integer(input.right))fail('INVALID_REQUEST');if(input.evidence!=='synthetic'||input.operational_run!==false||input.independent_review!==false)fail('UNSUPPORTED_CLAIM');return {metric:input.metric,difference:input.right-input.left,state:'represented',operational_claim:false,independent_claim:false};},
 evaluation_gap(input){closed(input,['need','participants','evaluation_performed']);const needs=['operator_validation','assistive_technology','affected_user_review','independent_reproduction','real_workload_sample','context_switch_measurement','service_prediction_error','clock_calibration','censoring_assessment','long_horizon_fairness'];if(!needs.includes(input.need))fail('INVALID_REQUEST');if(input.participants!==0||input.evaluation_performed!==false)fail('UNSUPPORTED_CLAIM');return {state:'open_gap',missing:[input.need],participants:0,executed:false};},
 allocation_gate(input){closed(input,['action','required_role','verified_roles','execute']);const roles=['professional','safety','legal','cultural','maori_authority','affected_party','production_operator','privacy_governance','deployment_owner','independent_reviewer'];if(input.action!=='consequential_allocation'||!roles.includes(input.required_role))fail('INVALID_REQUEST');if(!Array.isArray(input.verified_roles)||input.verified_roles.length||input.execute!==false)fail('AUTHORITY_RESERVED');return {state:'exact_gate',missing:[input.required_role],executed:false};}
};
function run(operation,input){try{if(!Object.hasOwn(operations,operation))fail('UNKNOWN_OPERATION');const value=operations[operation](input),disposition=operation==='comparison_ceiling'?'represented':operation==='evaluation_gap'?'open_gap':operation==='allocation_gate'?'exact_gate':'completed';return {ok:true,value,error:null,disposition};}catch(error){return {ok:false,value:null,error:error.message,disposition:'open_gap'};}}
module.exports={run,operations,helpers};
