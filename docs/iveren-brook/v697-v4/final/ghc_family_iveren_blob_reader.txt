"use strict";
const cp=require('child_process');
// One request is written and its complete binary response consumed before the next request.
async function readBlobs(root,head,paths){const child=cp.spawn('git',['-C',root,'cat-file','--batch'],{stdio:['pipe','pipe','pipe']});let buffer=Buffer.alloc(0),ended=false,wake=null,stderr='';const done=new Promise((resolve,reject)=>{child.on('error',reject);child.on('close',code=>code===0?resolve():reject(Error('git batch exited '+code+': '+stderr)));});child.stdout.on('data',b=>{buffer=Buffer.concat([buffer,b]);if(wake){const w=wake;wake=null;w();}});child.stdout.on('end',()=>{ended=true;if(wake){const w=wake;wake=null;w();}});child.stderr.on('data',b=>{stderr+=b.toString('utf8').slice(0,500);});async function more(){if(ended)throw Error('Unexpected end of Git byte stream');await new Promise(resolve=>wake=resolve);}async function take(n){while(buffer.length<n)await more();const result=Buffer.from(buffer.subarray(0,n));buffer=buffer.subarray(n);return result;}async function line(){let at;while((at=buffer.indexOf(10))<0)await more();return (await take(at+1)).subarray(0,at).toString('ascii');}const result=new Map();try{for(const p of paths){if(!/^[0-9a-f]{40}$/.test(head)||/[\r\n]/.test(p)||p.includes('..'))throw Error('Invalid exact blob request');child.stdin.write(head+':'+p+'\n');const header=(await line()).split(' ');if(header.length!==3||header[1]!=='blob')throw Error('Expected Git blob for '+p);const size=Number(header[2]);if(!Number.isSafeInteger(size)||size<0||size>20e6)throw Error('Owner blob size bound');const bytes=await take(size);if((await take(1))[0]!==10)throw Error('Git blob terminator');result.set(p,bytes);}child.stdin.end();await done;return result;}catch(e){child.kill();await done.catch(()=>{});throw e;}}
module.exports={readBlobs};
