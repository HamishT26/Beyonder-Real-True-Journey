test('quadratic polarization',models,m=>eq(value(m,map(x,y)).add(value(m,map(x,y,-1))),value(m,x).mul(TWO).add(dot(y,mul(m.matrix,y)))));
test('gradient difference equals H times displacement',models,m=>eq(map(gradient(m,map(x,y)),gradient(m,x),-1),mul(m.matrix,y)));
test('line polynomial evaluation',models,m=>{const r=call('qp_line_restriction',m,{point:x,direction:y}),t=rat('3/7'),a=r.coefficients.map(rat);eq(a[0].add(a[1].mul(t)).add(a[2].mul(t.mul(t))),value(m,map(x,y.map(q=>q.mul(t)))));});

test('completed-square identity',models,m=>{const r=call('qp_complete_square',m);if(r.available){const p=r.center.map(rat),d=map(x,p,-1);eq(value(m,x),rat(r.center_value).add(dot(d,mul(m.matrix,d)).div(TWO)));}else assert.equal(stationary(m).consistent,false);});
test('convex optimum KKT signs',models.filter(m=>curvature(m).positive_semidefinite),m=>assert.equal(kkt(m,minimum(m).minimizer_candidates[0]).satisfied,true));
test('every corner admitted as candidate',models,m=>{const r=C.candidates(m).candidates.map(r=>r.point.map(q=>q.toString()).join(','));for(const a of [m.lower[0],m.upper[0]])for(const b of [m.lower[1],m.upper[1]])assert.ok(r.includes([a,b].map(q=>q.toString()).join(',')));});
test('candidate box membership',models,m=>assert.ok(C.candidates(m).candidates.every(r=>feasible(m,r.point).feasible)));

test('affine value pullback',models,m=>{const scale=[rat('-2/1'),rat('3/2')],shift=[rat('1/3'),rat('-1/2')],u=[rat('2/5'),rat('-3/7')],r=call('qp_affine_pullback',m,{scale,shift}),p=u.map((q,i)=>q.mul(scale[i]).add(shift[i]));eq(value(model(r.model),u),value(m,p));});
test('affine box round trip',models,m=>{const a=[rat('-2/1'),rat('3/2')],b=[rat('1/3'),rat('-1/2')],n=model(call('qp_affine_pullback',m,{scale:a,shift:b}).model);for(let j=0;j<2;j++){const back=[n.lower[j].mul(a[j]).add(b[j]),n.upper[j].mul(a[j]).add(b[j])].sort((x,y)=>x.cmp(y));eq(back,[m.lower[j],m.upper[j]]);}});
test('parameter value difference',models,m=>{const d=[rat('1/2'),rat('-2/3')],theta=rat('4/5'),r=call('qp_parameter_shift',m,{point:x,direction:d,theta});eq(value(model(r.model),x).sub(value(m,x)),rat(r.value_delta_at_point));});
test('optimal value concavity in linear parameter',models,m=>{const d=[ONE,TWO],left=minimum(shifted(m,d,ONE.neg())).minimum,center=minimum(m).minimum,right=minimum(shifted(m,d,ONE)).minimum;assert.ok(center.mul(TWO).cmp(left.add(right))>=0);});
test('flat minimizer-set reservation',models.filter(m=>curvature(m).type==='zero'),m=>{const r=minimum(m);assert.equal(r.all_minimizers_enumerated,false);eq(r.minimum,value(m,m.lower));assert.equal(r.unique_candidate,true);});
test('semidefinite stationary freedom',models.filter(m=>curvature(m).type==='positive_semidefinite'),m=>{const s=stationary(m);assert.equal(s.nullspace.length,1);assert.equal(s.consistent,false);assert.equal(s.particular,null);});
test('nonconvex certificate refusal',models.filter(m=>!curvature(m).positive_semidefinite),m=>{const r=call('qp_convex_certificate',m,{point:minimum(m).minimizer_candidates[0]});assert.equal(r.certified,false);assert.ok(r.reasons.includes('not_positive_semidefinite'));});
test('fixed-coordinate first-order semantics',[model(fixtures.fixed_model)],m=>{const r=kkt(m,m.lower);assert.equal(r.satisfied,true);assert.ok(r.coordinate_checks.every(r=>r.position==='fixed'&&r.residual.n===0n));});
