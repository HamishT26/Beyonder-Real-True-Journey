"use strict";
const C=require('../x1/ghc_family_qp_core.txt');
const {Q,Z,ONE,rat,vec,model,encode,value,gradient,curvature,stationary,feasible,dot,request,refuse}=C;
function candidates(m){const points=new Map(),add=p=>{if(feasible(m,p).feasible)points.set(p.map(q=>q.toString()).join(','),p);};for(const x of [m.lower[0],m.upper[0]])for(const y of [m.lower[1],m.upper[1]])add([x,y]);for(let fixed=0;fixed<2;fixed++){const free=1-fixed,h=m.matrix[free][free];if(h.n!==0n)for(const endpoint of [m.lower[fixed],m.upper[fixed]]){const p=[Z,Z];p[fixed]=endpoint;p[free]=m.linear[free].add(m.matrix[free][fixed].mul(endpoint)).neg().div(h);add(p);}}const s=stationary(m);if(s.consistent)add(s.particular);const rows=[...points.values()].sort((a,b)=>a[0].cmp(b[0])||a[1].cmp(b[1])).map(point=>({point,value:value(m,point)}));return {candidates:rows,candidate_count:rows.length,complete_minimizer_set_claim:false};}
function minimum(m){const rows=candidates(m).candidates;let best=rows[0].value;for(const r of rows)if(r.value.cmp(best)<0)best=r.value;const points=rows.filter(r=>r.value.cmp(best)===0).map(r=>r.point);return {minimum:best,minimizer_candidates:points,unique_candidate:points.length===1,all_minimizers_enumerated:false,domain:'declared_closed_box'};}
function kkt(m,x){const f=feasible(m,x),g=gradient(m,x),checks=f.positions.map((position,i)=>{const q=g[i];let residual;if(position==='outside')residual=null;else if(position==='fixed')residual=Z;else if(position==='lower')residual=q.n<0n?q.neg():Z;else if(position==='upper')residual=q.n>0n?q:Z;else residual=q.abs();return {position,residual,satisfied:residual!==null&&residual.n===0n};});return {feasible:f.feasible,gradient:g,coordinate_checks:checks,satisfied:f.feasible&&checks.every(r=>r.satisfied)};}
function shifted(m,d,t){return {...m,linear:m.linear.map((g,i)=>g.add(t.mul(d[i])))};}
const schemas={qp_box_candidates:[],qp_box_minimum:[],qp_kkt_residual:['point'],qp_convex_certificate:['point'],qp_affine_pullback:['scale','shift'],qp_parameter_shift:['point','direction','theta'],qp_parameter_sweep:['direction','parameters','physical_claim'],qp_model_criticism:['physical_claim'],qp_observation_gap:['need','observations','physical_claim'],qp_authority_gate:['action','required_role','verified_roles','execute']};
function execute(r){const {op,i,m}=request(r,schemas);if(Object.hasOwn(i,'physical_claim')&&i.physical_claim!==false)refuse('claim_reserved');if(op==='qp_box_candidates')return candidates(m);if(op==='qp_box_minimum')return minimum(m);if(op==='qp_kkt_residual')return kkt(m,vec(i.point,'vector_shape'));if(op==='qp_convex_certificate'){const h=curvature(m),k=kkt(m,vec(i.point,'vector_shape')),reasons=[];if(!h.positive_semidefinite)reasons.push('not_positive_semidefinite');if(!k.feasible)reasons.push('infeasible');if(!k.satisfied)reasons.push('boundary_or_stationarity_residual');return {certified:reasons.length===0,positive_semidefinite:h.positive_semidefinite,kkt_satisfied:k.satisfied,feasible:k.feasible,reasons,certificate_scope:'declared rational quadratic and closed box only',physical_or_allocation_authority:false};}if(op==='qp_affine_pullback'){const a=vec(i.scale,'vector_shape'),b=vec(i.shift,'vector_shape');if(a.some(q=>q.n===0n))refuse('nonzero_scale');const lo=[],hi=[];for(let j=0;j<2;j++){const u=m.lower[j].sub(b[j]).div(a[j]),v=m.upper[j].sub(b[j]).div(a[j]);lo.push(u.cmp(v)<=0?u:v);hi.push(u.cmp(v)<=0?v:u);}const gb=gradient(m,b),out={matrix:m.matrix.map((row,j)=>row.map((q,k)=>a[j].mul(q).mul(a[k]))),linear:gb.map((q,j)=>a[j].mul(q)),constant:value(m,b),lower:lo,upper:hi};return {model:out,mapping:'x=diag(scale)*u+shift'};}if(op==='qp_parameter_shift'){const d=vec(i.direction,'vector_shape'),x=vec(i.point,'vector_shape'),t=rat(i.theta);return {model:shifted(m,d,t),value_delta_at_point:t.mul(dot(d,x)),parameter_is_assumption:true};}if(op==='qp_parameter_sweep'){const d=vec(i.direction,'vector_shape');if(!Array.isArray(i.parameters)||i.parameters.length>9)refuse('parameter_count');const parameters=i.parameters.map(rat);return {rows:parameters.map(theta=>({theta,...minimum(shifted(m,d,theta))})),parameter_choices_are_assumptions:true,physical_robustness:false};}if(op==='qp_model_criticism')return {declared_minimum:minimum(m).minimum,curvature:curvature(m).type,model_error:'unknown',coefficients_are_observations:false,optimum_confers_authority:false,representation_only:true};if(op==='qp_observation_gap'){if(!['coefficient_measurement','model_adequacy','measurement_uncertainty','independent_reproduction','affected_user_evaluation'].includes(i.need))refuse('need');if(i.observations!==0)refuse('claim_reserved');return {need:i.need,physical_observations:0,executed:false,state:'open_gap'};}if(op==='qp_authority_gate'){if(i.action!=='real_resource_allocation'||!['professional','safety','legal','cultural','Maori_authority','affected_party','production_owner','trust_governance'].includes(i.required_role))refuse('authority_context');if(!Array.isArray(i.verified_roles)||i.verified_roles.length||i.execute!==false)refuse('claim_reserved');return {action:i.action,missing:[i.required_role],executed:false,state:'exact_gate'};}throw Error('internal_unhandled_operation');}
const allOperations=[...Object.keys(C.schemas),...Object.keys(schemas)];
function dispatch(r,allowed=allOperations){try{if(!r||!allowed.includes(r.op))refuse('operation');if(Object.hasOwn(C.schemas,r.op))return C.dispatch(r,allowed);const output=execute(r),outcome=['qp_parameter_sweep','qp_model_criticism'].includes(r.op)?'represented':r.op==='qp_observation_gap'?'open_gap':r.op==='qp_authority_gate'?'exact_gate':'completed';return {ok:true,outcome,value:encode(output),evidence_class:'same_owner_finite_synthetic',authority:false};}catch(e){return C.rejection(e);}}
const cli=allowed=>C.cli(allowed,dispatch),wire=(text,allowed=allOperations)=>C.wire(text,allowed,dispatch);
module.exports={...C,schemas:{...C.schemas,...schemas},execute,dispatch,cli,wire,candidates,minimum,kkt,shifted,allOperations};
