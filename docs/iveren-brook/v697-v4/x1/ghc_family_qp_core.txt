function keys(v,expected,reason){if(v===null||typeof v!=='object'||Array.isArray(v)||Object.keys(v).length!==expected.length||expected.some(k=>!Object.hasOwn(v,k)))refuse(reason);}
const gcd=(a,b)=>{a=a<0n?-a:a;b=b<0n?-b:b;while(b){const t=a%b;a=b;b=t;}return a;};
class Q{constructor(n,d=1n){if(d===0n)throw Error('internal_zero_denominator');if(d<0n){n=-n;d=-d;}const g=gcd(n,d);this.n=n/g;this.d=d/g;}add(b){return new Q(this.n*b.d+b.n*this.d,this.d*b.d);}neg(){return new Q(-this.n,this.d);}sub(b){return this.add(b.neg());}mul(b){return new Q(this.n*b.n,this.d*b.d);}div(b){return new Q(this.n*b.d,this.d*b.n);}cmp(b){const d=this.n*b.d-b.n*this.d;return d<0n?-1:d>0n?1:0;}abs(){return this.n<0n?this.neg():this;}toString(){return this.n+'/'+this.d;}}
const Z=new Q(0n),ONE=new Q(1n),TWO=new Q(2n);
function rat(v){if(typeof v!=='string'||!/^(-?(?:0|[1-9][0-9]{0,6}))\/(-?[1-9][0-9]{0,6})$/.test(v))refuse('rational');const [n,d]=v.split('/').map(BigInt);if(n>1000000n||n< -1000000n||d>1000000n||d< -1000000n||d===0n)refuse('rational');return new Q(n,d);}
function vec(v,reason='model_shape'){if(!Array.isArray(v)||v.length!==2)refuse(reason);return v.map(rat);}
function model(v){keys(v,['matrix','linear','constant','lower','upper'],'model_shape');if(!Array.isArray(v.matrix)||v.matrix.length!==2)refuse('model_shape');const H=v.matrix.map(r=>vec(r)),g=vec(v.linear),c=rat(v.constant),lo=vec(v.lower),hi=vec(v.upper);if(H[0][1].cmp(H[1][0])!==0)refuse('symmetric_matrix');if(lo.some((v,i)=>v.cmp(hi[i])>0))refuse('box_order');return {matrix:H,linear:g,constant:c,lower:lo,upper:hi};}
function encode(v){if(v instanceof Q)return v.toString();if(Array.isArray(v))return v.map(encode);if(v&&typeof v==='object')return Object.fromEntries(Object.entries(v).map(([k,x])=>[k,encode(x)]));return v;}

function curvature(m){const a=m.matrix[0][0],b=m.matrix[0][1],d=m.matrix[1][1],det=a.mul(d).sub(b.mul(b)),trace=a.add(d);const kind=det.n>0n?(trace.n>0n?'positive_definite':'negative_definite'):det.n<0n?'indefinite':trace.n>0n?'positive_semidefinite':trace.n<0n?'negative_semidefinite':'zero';return {type:kind,rank:det.n!==0n?2:trace.n!==0n?1:0,determinant:det,trace,positive_semidefinite:['positive_definite','positive_semidefinite','zero'].includes(kind),negative_semidefinite:['negative_definite','negative_semidefinite','zero'].includes(kind)};}
function stationary(m){const a=m.matrix[0][0],b=m.matrix[0][1],d=m.matrix[1][1],[g,h]=m.linear,det=a.mul(d).sub(b.mul(b));if(det.n!==0n)return {consistent:true,rank:2,particular:[b.mul(h).sub(d.mul(g)).div(det),b.mul(g).sub(a.mul(h)).div(det)],nullspace:[],unique:true};if(a.n!==0n){const consistent=a.mul(h).cmp(b.mul(g))===0;return {consistent,rank:1,particular:consistent?[g.neg().div(a),Z]:null,nullspace:[[b.neg().div(a),ONE]],unique:false};}if(d.n!==0n){const consistent=g.n===0n;return {consistent,rank:1,particular:consistent?[Z,h.neg().div(d)]:null,nullspace:[[ONE,Z]],unique:false};}const consistent=g.n===0n&&h.n===0n;return {consistent,rank:0,particular:consistent?[Z,Z]:null,nullspace:[[ONE,Z],[Z,ONE]],unique:false};}
function feasible(m,x){const positions=x.map((a,i)=>a.cmp(m.lower[i])<0||a.cmp(m.upper[i])>0?'outside':m.lower[i].cmp(m.upper[i])===0?'fixed':a.cmp(m.lower[i])===0?'lower':a.cmp(m.upper[i])===0?'upper':'interior');return {feasible:!positions.includes('outside'),positions,lower_slack:x.map((a,i)=>a.sub(m.lower[i])),upper_slack:x.map((a,i)=>m.upper[i].sub(a))};}
