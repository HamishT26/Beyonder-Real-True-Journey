const fs=require('fs'),path=require('path');
function j(p){return JSON.parse(fs.readFileSync(p,'utf8'));}
function main(root){const p=path.join(root,'docs','elaren-kestrel','v692-v3','plan');const f=j(path.join(p,'proposal-freeze.json')),i=j(path.join(p,'inherited-review.json')),x=j(path.join(p,'exact-packets.json')),b=j(path.join(p,'blocked-packets.json')),q=j(path.join(p,'portfolio-plan.json')),n=j(path.join(p,'novelty-review.json'));const checks={new:f.proposals.length===200,inherited:i.records.length===200,no_execution:f.implementation_ran===false&&f.outcomes_observed===false&&f.proposals.every(r=>r.implementation_ran===false&&r.observed_outcome===null),unique:new Set(f.proposals.map(r=>r.proposal_id)).size===200&&new Set(f.proposals.map(r=>r.title)).size===200,labels:f.proposals.every(r=>['completed','represented','open_gap','exact_gate'].includes(r.expected_disposition)),portfolio:['x1','x2'].every(s=>q.sessions[s].safe_now.length===100&&q.sessions[s].candidate.length===100&&q.sessions[s].clean_fix_refine.length===100),packets:x.packets.length===50&&b.packets.length===30,novelty:n.exact_title_collisions===0&&n.within_slate_title_collisions===0};if(!Object.values(checks).every(Boolean))throw new Error(JSON.stringify(checks));process.stdout.write(JSON.stringify({valid:true,checks})+'\n');}
module.exports={main};
if(require.main===module)main(process.argv[2]||process.cwd());
