const edgeKey=(a,b)=>[String(a),String(b)].sort().join('|');
function validGraph(g){return !!g&&Array.isArray(g.nodes)&&Array.isArray(g.edges)&&g.edges.every(e=>Array.isArray(e)&&e.length===2);}
function norm(g){const nodes=[...new Set(g.nodes.map(String))].sort(),ns=new Set(nodes),seen=new Set(),edges=[];for(const e of g.edges){const a=String(e[0]),b=String(e[1]);if(a===b||!ns.has(a)||!ns.has(b))continue;const k=edgeKey(a,b);if(!seen.has(k)){seen.add(k);edges.push(k.split('|'));}}edges.sort((a,b)=>JSON.stringify(a).localeCompare(JSON.stringify(b)));return{nodes,edges};}
function adj(g){const x=norm(g),a=Object.fromEntries(x.nodes.map(n=>[n,new Set()]));for(const[e1,e2]of x.edges){a[e1].add(e2);a[e2].add(e1);}return{g:x,adj:a};}
function proper(g,c){const x=norm(g),conflicts=[],missing=[];for(const n of x.nodes)if(!(n in c))missing.push(n);for(const[a,b]of x.edges)if(c[a]===c[b])conflicts.push([a,b]);return{proper:!conflicts.length&&!missing.length,conflicts,missing};}
function degreeOrder(g){const x=adj(g);return [...x.g.nodes].sort((a,b)=>x.adj[b].size-x.adj[a].size||a.localeCompare(b));}
function greedy(g,order){const x=adj(g),colors={},trace=[];for(const n of order){const used=new Set([...x.adj[n]].filter(v=>v in colors).map(v=>colors[v]));let c=0;while(used.has(c))c++;colors[n]=c;trace.push({node:n,color:c,forbidden:[...used].sort((a,b)=>a-b)});}return{colors,trace,color_count:x.g.nodes.length?Math.max(...Object.values(colors))+1:0};}
function classes(c){const out={};for(const n of Object.keys(c).sort()){const k=String(c[n]);(out[k]||=[]).push(n);}return out;}
function combos(a,r){const out=[],p=[];function f(s){if(p.length===r){out.push([...p]);return;}for(let i=s;i<a.length;i++){p.push(a[i]);f(i+1);p.pop();}}f(0);return out;}
function clique(g){const x=norm(g),es=new Set(x.edges.map(e=>edgeKey(...e)));let w=[];for(let r=1;r<=x.nodes.length;r++)for(const s of combos(x.nodes,r)){let ok=true;for(let i=0;i<s.length;i++)for(let j=i+1;j<s.length;j++)if(!es.has(edgeKey(s[i],s[j])))ok=false;if(ok&&(s.length>w.length||(s.length===w.length&&s.join('|')<w.join('|'))))w=s;}return{lower_bound:w.length,witness:w,exact_global_claim:false};}
function degeneracy(g){const x=adj(g),remain=new Set(x.g.nodes),removal=[];let d=0;while(remain.size){const n=[...remain].sort((a,b)=>[...x.adj[a]].filter(y=>remain.has(y)).length-[...x.adj[b]].filter(y=>remain.has(y)).length||a.localeCompare(b))[0],degree=[...x.adj[n]].filter(y=>remain.has(y)).length;d=Math.max(d,degree);removal.push({node:n,degree});remain.delete(n);}return{degeneracy:d,removal,smallest_last:[...removal].reverse().map(x=>x.node)};}
function bip(g){const x=adj(g),colors={};let conflict=null;for(const s of x.g.nodes){if(s in colors)continue;colors[s]=0;const q=[s];while(q.length&&!conflict){const u=q.shift();for(const v of [...x.adj[u]].sort()){if(!(v in colors)){colors[v]=1-colors[u];q.push(v);}else if(colors[v]===colors[u]){conflict=[u,v];break;}}}if(conflict)break;}return{bipartite:!conflict,colors:conflict?null:colors,conflict_edge:conflict};}
function evaluate(operation,input){try{if(!input||!validGraph(input.graph))return{ok:false,disposition:'open_gap',value:null,error:'INVALID_INPUT'};let value;if(operation==='graph_record_normalization')value={graph:norm(input.graph),performed:false};else if(operation==='proper_coloring_check'){if(!input.coloring||typeof input.coloring!=='object')throw 0;value=proper(input.graph,input.coloring);}else if(operation==='greedy_coloring_trace')value=greedy(input.graph,degreeOrder(input.graph));else if(operation==='degree_ordering')value={order:degreeOrder(input.graph)};else if(operation==='color_class_partition'){if(!input.coloring||typeof input.coloring!=='object')throw 0;value={classes:classes(input.coloring),proper:proper(input.graph,input.coloring).proper};}else if(operation==='clique_lower_bound')value=clique(input.graph);else if(operation==='degeneracy_ordering')value=degeneracy(input.graph);else if(operation==='bipartite_coloring_witness')value=bip(input.graph);else if(operation==='odd_cycle_witness'){const b=bip(input.graph);value={found:!b.bipartite,witness_edge:b.conflict_edge};}else if(operation==='recoloring_delta'){if(!input.before||!input.after)throw 0;const nodes=norm(input.graph).nodes,changes=nodes.filter(n=>input.before[n]!==input.after[n]).map(n=>({node:n,before:input.before[n]??null,after:input.after[n]??null}));value={changes,before:proper(input.graph,input.before),after:proper(input.graph,input.after)};}else return{ok:false,disposition:'open_gap',value:null,error:'UNSUPPORTED_OPERATION'};return{ok:true,disposition:'completed',value,error:null};}catch{return{ok:false,disposition:'open_gap',value:null,error:'INVALID_INPUT'};}}
module.exports={evaluate};
