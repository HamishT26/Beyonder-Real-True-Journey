const edgeKey=(a,b)=>[String(a),String(b)].sort().join('|');
function validGraph(g){return !!g&&Array.isArray(g.nodes)&&Array.isArray(g.edges)&&g.edges.every(e=>Array.isArray(e)&&e.length===2);}
function norm(g){const nodes=[...new Set(g.nodes.map(String))].sort(),ns=new Set(nodes),seen=new Set(),edges=[];for(const e of g.edges){const a=String(e[0]),b=String(e[1]);if(a===b||!ns.has(a)||!ns.has(b))continue;const k=edgeKey(a,b);if(!seen.has(k)){seen.add(k);edges.push(k.split('|'));}}edges.sort((a,b)=>JSON.stringify(a).localeCompare(JSON.stringify(b)));return{nodes,edges};}
function adj(g){const x=norm(g),a=Object.fromEntries(x.nodes.map(n=>[n,new Set()]));for(const[u,v]of x.edges){a[u].add(v);a[v].add(u);}return{g:x,adj:a};}
function proper(g,c){const x=norm(g),conflicts=[],missing=[];for(const n of x.nodes)if(!(n in c))missing.push(n);for(const[a,b]of x.edges)if(c[a]===c[b])conflicts.push([a,b]);return{proper:!conflicts.length&&!missing.length,conflicts,missing};}
function order(g){const x=adj(g);return [...x.g.nodes].sort((a,b)=>x.adj[b].size-x.adj[a].size||a.localeCompare(b));}
function greedy(g,o){const x=adj(g),colors={},trace=[];for(const n of o){const used=new Set([...x.adj[n]].filter(v=>v in colors).map(v=>colors[v]));let c=0;while(used.has(c))c++;colors[n]=c;trace.push({node:n,color:c,forbidden:[...used].sort((a,b)=>a-b)});}return{colors,trace,color_count:x.g.nodes.length?Math.max(...Object.values(colors))+1:0};}
function dsatur(g){const x=adj(g),colors={},trace=[];while(Object.keys(colors).length<x.g.nodes.length){const u=x.g.nodes.filter(n=>!(n in colors));u.sort((a,b)=>{const sa=new Set([...x.adj[a]].filter(v=>v in colors).map(v=>colors[v])).size,sb=new Set([...x.adj[b]].filter(v=>v in colors).map(v=>colors[v])).size;return sb-sa||x.adj[b].size-x.adj[a].size||a.localeCompare(b);});const n=u[0],used=new Set([...x.adj[n]].filter(v=>v in colors).map(v=>colors[v]));let c=0;while(used.has(c))c++;colors[n]=c;trace.push({node:n,color:c,saturation:used.size});}return{colors,trace,color_count:x.g.nodes.length?Math.max(...Object.values(colors))+1:0};}
function search(g,k,lists=null,pre={}){const x=adj(g),o=order(g),colors={...pre};function rec(i){if(i===o.length)return true;const n=o[i];if(n in colors)return[...x.adj[n]].some(v=>v in colors&&colors[v]===colors[n])?false:rec(i+1);const allowed=lists&&lists[n]?lists[n]:Array.from({length:k},(_,j)=>j);for(const c of allowed){if([...x.adj[n]].every(v=>colors[v]!==c)){colors[n]=c;if(rec(i+1))return true;delete colors[n];}}return false;}const feasible=rec(0);return{feasible,colors:feasible?colors:null};}
function chromatic(g){const x=norm(g);if(!x.nodes.length)return{chromatic_number:0,colors:{}};for(let k=1;k<=x.nodes.length;k++){const r=search(x,k);if(r.feasible)return{chromatic_number:k,colors:r.colors};}}
function classes(c){const out={};for(const n of Object.keys(c).sort()){const k=String(c[n]);(out[k]||=[]).push(n);}return out;}
function intervals(a){const nodes=a.map(x=>String(x.id)).sort(),edges=[];for(let i=0;i<a.length;i++)for(let j=i+1;j<a.length;j++)if(Math.max(a[i].start,a[j].start)<Math.min(a[i].end,a[j].end))edges.push([String(a[i].id),String(a[j].id)]);const graph={nodes,edges},r=greedy(graph,order(graph));return{graph:norm(graph),colors:r.colors,color_count:r.color_count};}
function poly(g){const x=norm(g),counts={};for(let k=0;k<=4;k++){let count=0,colors={};function rec(i){if(i===x.nodes.length){count++;return;}const n=x.nodes[i];for(let c=0;c<k;c++){colors[n]=c;const prefix=x.nodes.slice(0,i+1),sub={nodes:prefix,edges:x.edges.filter(e=>prefix.includes(e[0])&&prefix.includes(e[1]))};if(proper(sub,colors).proper)rec(i+1);delete colors[n];}}if(!x.nodes.length)count=1;else rec(0);counts[String(k)]=count;}return{samples:counts,symbolic_proof:false,empirical:false};}
function evaluate(op,input){try{let value,disposition='completed';if(op==='interval_conflict_coloring'){if(!input||!Array.isArray(input.intervals))throw 0;value=intervals(input.intervals);}else if(op==='accessible_color_label_projection'){if(!input||!input.coloring||!input.labels)throw 0;const rows=Object.keys(input.coloring).sort().map(n=>({node:n,color:input.coloring[n],text_label:input.labels[String(input.coloring[n])]??null}));value={rows,color_alone:false,manual_review:false};}else if(op==='accessibility_evaluation_gap'){if(!input||!Array.isArray(input.required)||!Array.isArray(input.provided))throw 0;disposition='open_gap';value={missing:input.required.filter(x=>!input.provided.includes(x)),declared:input.provided,assurance_established:false};}else if(op==='assignment_authority_reservation'){if(!input||!Array.isArray(input.requested)||!Array.isArray(input.declared_grants)||!Array.isArray(input.authority_records))throw 0;disposition='exact_gate';value={candidates:input.requested.filter(x=>input.declared_grants.includes(x)),outside:input.requested.filter(x=>!input.declared_grants.includes(x)),executed:false,authority_established:false};}else{if(!input||!validGraph(input.graph))throw 0;if(op==='dsatur_coloring_trace')value=dsatur(input.graph);else if(op==='exact_chromatic_search')value={...chromatic(input.graph),bounded_nodes:norm(input.graph).nodes.length};else if(op==='list_coloring_feasibility'){if(!input.lists)throw 0;const k=Math.max(0,...Object.values(input.lists).flat())+1;value=search(input.graph,k,input.lists,{});}else if(op==='precoloring_extension'){if(!input.precolored||!Number.isInteger(input.color_limit))throw 0;value=search(input.graph,input.color_limit,null,input.precolored);}else if(op==='equitable_balance_review'){if(!input.coloring)throw 0;const p=proper(input.graph,input.coloring),sizes=Object.values(classes(input.coloring)).map(x=>x.length);value={proper:p.proper,class_sizes:sizes,equitable:!sizes.length||Math.max(...sizes)-Math.min(...sizes)<=1};}else if(op==='chromatic_polynomial_sample'){disposition='represented';value=poly(input.graph);}else return{ok:false,disposition:'open_gap',value:null,error:'UNSUPPORTED_OPERATION'};}return{ok:true,disposition,value,error:null};}catch{return{ok:false,disposition:'open_gap',value:null,error:'INVALID_INPUT'};}}
module.exports={evaluate};
