const ALLOWED=["requirement_trace_matrix","change_impact_closure"];
function stable(value){if(Array.isArray(value))return value.map(stable);if(value&&typeof value==='object')return Object.fromEntries(Object.keys(value).sort().map(function(key){return [key,stable(value[key])];}));return value;};
function compact(value){return JSON.stringify(stable(value));};
function shaBytes(value){return crypto.createHash('sha256').update(value).digest('hex');};
function shaValue(value){return shaBytes(Buffer.from(compact(value),'utf8'));};
function clone(value){return JSON.parse(JSON.stringify(value));};
function assert(ok,message){if(!ok)throw new Error(message);};
function validateLts(lts){
  assert(lts&&typeof lts==='object'&&!Array.isArray(lts),'E_LTS_OBJECT');
  assert(compact(Object.keys(lts).sort())===compact(['initial','states','transitions']),'E_LTS_FIELDS');
  assert(Array.isArray(lts.states)&&lts.states.length>0&&lts.states.length<=12,'E_STATES');
  assert(lts.states.every(function(s){return typeof s==='string'&&/^[a-z][a-z0-9-]{0,39}$/.test(s);})&&new Set(lts.states).size===lts.states.length,'E_STATE_NAMES');
  const states=new Set(lts.states);assert(states.has(lts.initial),'E_INITIAL');
  assert(Array.isArray(lts.transitions)&&lts.transitions.length<=64,'E_TRANSITIONS');
  assert(lts.transitions.every(function(edge){return Array.isArray(edge)&&edge.length===3&&states.has(edge[0])&&states.has(edge[2])&&typeof edge[1]==='string'&&/^[a-z][a-z0-9-]{0,39}$/.test(edge[1]);}),'E_EDGE');
  assert(new Set(lts.transitions.map(function(edge){return edge.join('\u0000');})).size===lts.transitions.length,'E_EDGE_DUPLICATE');
};
function validatePair(input){
  assert(input&&typeof input==='object'&&!Array.isArray(input),'E_PAIR_OBJECT');
  assert(compact(Object.keys(input).sort())===compact(['abstract','case_id','concrete','relation','requirements','synthetic']),'E_PAIR_FIELDS');
  assert(input.synthetic===true&&/^[a-z][a-z0-9-]{0,39}$/.test(input.case_id),'E_SYNTHETIC_CASE');
  validateLts(input.abstract);validateLts(input.concrete);
  assert(Array.isArray(input.relation)&&input.relation.length>0,'E_RELATION');
  const cs=new Set(input.concrete.states),as=new Set(input.abstract.states);
  assert(input.relation.every(function(pair){return Array.isArray(pair)&&pair.length===2&&cs.has(pair[0])&&as.has(pair[1]);}),'E_RELATION_MEMBER');
  assert(new Set(input.relation.map(function(pair){return pair.join('\u0000');})).size===input.relation.length,'E_RELATION_DUPLICATE');
  assert(input.concrete.states.every(function(c){return input.relation.some(function(pair){return pair[0]===c;});}),'E_RELATION_TOTAL');
  assert(input.relation.some(function(pair){return pair[0]===input.concrete.initial&&pair[1]===input.abstract.initial;}),'E_INITIAL_RELATION');
  assert(Array.isArray(input.requirements)&&input.requirements.length>0,'E_REQUIREMENTS');
  const ids=new Set(input.requirements.map(function(r){return r.id;}));
  assert(ids.size===input.requirements.length,'E_REQUIREMENT_DUPLICATE');
  assert(input.requirements.every(function(r){return r&&typeof r.id==='string'&&/^R-[A-Z0-9-]+$/.test(r.id)&&typeof r.action==='string'&&(r.parent===null||ids.has(r.parent));}),'E_REQUIREMENT');
};
function successors(lts){const out=Object.fromEntries(lts.states.slice().sort().map(function(state){return [state,[]];}));for(const edge of lts.transitions)out[edge[0]].push({action:edge[1],to:edge[2]});for(const state of Object.keys(out))out[state].sort(function(a,b){return a.action.localeCompare(b.action)||a.to.localeCompare(b.to);});return out;};
function reachableStates(lts){const out=successors(lts),seen=new Set([lts.initial]),queue=[lts.initial];while(queue.length){const state=queue.shift();for(const edge of out[state])if(!seen.has(edge.to)){seen.add(edge.to);queue.push(edge.to);}}return Array.from(seen).sort();};
function visibleActions(lts){return Array.from(new Set(lts.transitions.map(function(edge){return edge[1];}).filter(function(action){return action!=='tau';}))).sort();};
function relationMap(input){const byConcrete=Object.fromEntries(input.concrete.states.map(function(c){return [c,[]];}));for(const pair of input.relation)byConcrete[pair[0]].push(pair[1]);for(const c of Object.keys(byConcrete))byConcrete[c].sort();return byConcrete;};
function related(input,c,a){return input.relation.some(function(pair){return pair[0]===c&&pair[1]===a;});};
function relationTotality(input){return {total:input.concrete.states.every(function(c){return input.relation.some(function(pair){return pair[0]===c;});}),mapping:relationMap(input)};};
function actionCompatibility(input){const abstract=visibleActions(input.abstract),concrete=visibleActions(input.concrete);return {abstract:abstract,concrete:concrete,equal:compact(abstract)===compact(concrete),tau_is_internal:true};};
function forwardSimulation(input){const failures=[];for(const pair of input.relation){const c=pair[0],a=pair[1];for(const edge of input.concrete.transitions.filter(function(e){return e[0]===c;})){if(edge[1]==='tau'){if(!related(input,edge[2],a))failures.push({concrete:edge,related_abstract:a,reason:'tau_changes_abstract_relation'});continue;}const matches=input.abstract.transitions.filter(function(e){return e[0]===a&&e[1]===edge[1]&&related(input,edge[2],e[2]);});if(!matches.length)failures.push({concrete:edge,related_abstract:a,reason:'no_abstract_match'});}}return {holds:failures.length===0,failures:failures,relation_pairs:input.relation.length,convention:'single tau steps preserve their related abstract state'};};
function backwardCoverage(input){const failures=[];for(const edge of input.abstract.transitions){const sources=input.relation.filter(function(pair){return pair[1]===edge[0];}).map(function(pair){return pair[0];});let found=false;for(const c of sources){for(const ce of input.concrete.transitions.filter(function(e){return e[0]===c&&e[1]===edge[1];})){if(related(input,ce[2],edge[2]))found=true;}}if(!found)failures.push({abstract:edge,reason:'no_related_concrete_witness'});}return {holds:failures.length===0,failures:failures};};
function tracePrefixes(lts,depth){const out=successors(lts),traces=new Set();function visit(state,visible,steps){traces.add(visible.join('>'));if(visible.length>=depth||steps>=depth*3+3)return;for(const edge of out[state])visit(edge.to,edge.action==='tau'?visible:visible.concat([edge.action]),steps+1);}visit(lts.initial,[],0);return Array.from(traces).sort();};
function deadlocks(lts){const out=successors(lts);return reachableStates(lts).filter(function(state){return out[state].length===0;});};
function requirementMatrix(input){const aa=new Set(visibleActions(input.abstract)),ca=new Set(visibleActions(input.concrete));return input.requirements.map(function(r){return {requirement_id:r.id,parent:r.parent,action:r.action,abstract_trace:aa.has(r.action),concrete_trace:ca.has(r.action),bidirectional:r.parent===null||input.requirements.some(function(p){return p.id===r.parent;})};});};
function partitionRefinement(input){const out=successors(input.concrete),map=relationMap(input),groups={};for(const c of input.concrete.states.slice().sort()){const signature=map[c].join('+')+'|'+Array.from(new Set(out[c].filter(function(e){return e.action!=='tau';}).map(function(e){return e.action;}))).sort().join('+');if(!groups[signature])groups[signature]=[];groups[signature].push(c);}return Object.keys(groups).sort().map(function(signature,index){return {block:'B'+String(index+1).padStart(2,'0'),signature:signature,members:groups[signature]};});};
function quotient(input){const blocks=partitionRefinement(input),owner={};for(const block of blocks)for(const state of block.members)owner[state]=block.block;const transitions=[];for(const edge of input.concrete.transitions){const q=[owner[edge[0]],edge[1],owner[edge[2]]];if(!transitions.some(function(existing){return compact(existing)===compact(q);}))transitions.push(q);}return {blocks:blocks,initial:owner[input.concrete.initial],transitions:transitions.sort(function(a,b){return compact(a).localeCompare(compact(b));}),reversible_member_lists:true};};
function minimalCounterexample(input){const f=forwardSimulation(input);if(!f.holds)return {present:true,witness:f.failures[0]};const b=backwardCoverage(input);if(!b.holds)return {present:true,witness:b.failures[0]};const mutated=clone(input);mutated.relation=mutated.relation.slice(0,-1);let rejected=false;try{validatePair(mutated);}catch(error){rejected=true;}return {present:false,valid_pair_holds:true,mutation_probe:{kind:'remove_relation_pair',rejected:rejected,success_credit:0}};};
function changeImpact(input){const root=input.requirements[0].id,seen=new Set([root]);let changed=true;while(changed){changed=false;for(const r of input.requirements)if(r.parent&&seen.has(r.parent)&&!seen.has(r.id)){seen.add(r.id);changed=true;}}return {change_root:root,affected:Array.from(seen).sort(),synthetic:true};};
function compositionProjection(input){const modes=['alpha','beta'],states=[],transitions=[];for(const s of input.abstract.states)for(const mode of modes)states.push(s+'@'+mode);for(const edge of input.abstract.transitions)for(const mode of modes)transitions.push([edge[0]+'@'+mode,edge[1],edge[2]+'@'+mode]);for(const s of input.abstract.states){transitions.push([s+'@alpha','mode',s+'@beta']);transitions.push([s+'@beta','mode',s+'@alpha']);}return {state_count:states.length,transition_count:transitions.length,projection:Object.fromEntries(states.map(function(state){return [state,state.split('@')[0]];})),synthetic:true};};
function accepted(operation,input,value,outcome){return {accepted:true,operation:operation,case_id:input.case_id,outcome:outcome,value:value,input_sha256:shaValue(input),boundary:TERMINAL_BOUNDARY};};
