"use strict";
const fs=require("fs"),D="docs/veylora-quen/v699-v7/x2";
function open(c){const l=JSON.parse(fs.readFileSync(D+"/method-flow.json")),n=JSON.parse(fs.readFileSync(D+"/negative-register.json")),t=JSON.parse(fs.readFileSync(D+"/phase-truth.json"));
 function method(title){const m={...c.clone(l.methods[0]),method_id:"VQ6997-x2-M"+String(l.methods.length+1).padStart(3,"0"),title,validation_witness_ids:[],retained_negative_ids:[],recommendation_state:"candidate",supersedes:[]};l.methods.push(m);return m;}
 function negative(m,id,subject,reason,source){if(n.records.some(x=>x.negative_id===id))throw Error("negative exists");n.records.push({negative_id:id,subject,reason,source,subject_success_credit:0});m.retained_negative_ids.push(id);}
 function witness(m,id,result,procedure,observed,negatives=[]){if(l.witnesses.some(x=>x.witness_id===id))throw Error("witness exists");l.witnesses.push({witness_id:id,method_id:m.method_id,procedure,scope:D,expected:"Exact bounded predicate",observed,result,same_owner_only:true,independent_reproduction:false,retained_negative_ids:negatives,boundary:"Bounded same-owner software only."});m.validation_witness_ids.push(id);}
 function save(){for(const m of l.methods.filter(x=>x.recommendation_state==="candidate")){if(!m.retained_negative_ids.length||!l.witnesses.some(w=>w.method_id===m.method_id&&w.result==="pass"))throw Error("method evidence missing");m.recommendation_state="preferred";l.state_events.push({method_id:m.method_id,from:"candidate",to:"validated",note:"Bounded observations retained."},{method_id:m.method_id,from:"validated",to:"preferred",note:"Use only within the declared scope."});l.recommendations.push({method_id:m.method_id,recommendation:m.recurrence_guard,delivered:false});}l.counts={methods:l.methods.length,witnesses:l.witnesses.length,state_events:l.state_events.length,recommendations:l.recommendations.length,states:{observed:0,candidate:0,validated:0,preferred:l.methods.length,superseded:0,deprecated:0},witness_results:{pass:l.witnesses.filter(w=>w.result==="pass").length,fail:l.witnesses.filter(w=>w.result==="fail").length}};n.count=n.records.length;t.counts=l.counts;t.negatives=n.count;for(const [f,v] of [["method-flow.json",l],["negative-register.json",n],["phase-truth.json",t]])fs.writeFileSync(D+"/"+f,JSON.stringify(v,null,2)+"\n");return l.counts;}
 return {ledger:l,negatives:n,truth:t,method,negative,witness,save};}
module.exports={open};
