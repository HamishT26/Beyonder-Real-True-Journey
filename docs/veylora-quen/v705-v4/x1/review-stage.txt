const fs=require("fs"),path=require("path"),cp=require("child_process"),vm=require("vm"),BASE="docs/veylora-quen/v705-v4",c=require("./core.txt"),walk=d=>fs.readdirSync(d,{withFileTypes:true}).flatMap(x=>x.isDirectory()?walk(d+"/"+x.name):[d+"/"+x.name]);
function inspect(stage,parent,executables){const D=BASE+"/"+stage,read=f=>JSON.parse(fs.readFileSync(D+"/"+f)),checks=[],check=(name,pass,detail=null)=>checks.push({name,pass:!!pass,detail}),ps=JSON.parse(fs.readFileSync(BASE+"/planning/contracts-"+stage+".json")).records,records=read("results/contracts.json").records;
