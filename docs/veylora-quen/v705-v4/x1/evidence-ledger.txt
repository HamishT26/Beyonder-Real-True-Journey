"use strict";
const fs=require("fs"),BASE="docs/veylora-quen/v705-v4";
function open(stage,c){const D=BASE+"/"+stage,ledger=JSON.parse(fs.readFileSync(D+"/method-flow.json")),negativeRegister=JSON.parse(fs.readFileSync(D+"/negative-register.json")),truth=JSON.parse(fs.readFileSync(D+"/phase-truth.json"));
function method(title,module){const m={...c.clone(ledger.methods[0]),method_id:"VQ7054-"+stage+"-M"+String(ledger.methods.length+1).padStart(3,"0"),title,validation_witness_ids:[],retained_negative_ids:[],recommendation_state:"candidate",supersedes:[],changed_file_allowlist:[module],module_allowlist:[module],dependency_allowlist:[BASE+"/x1/core.txt"]};ledger.methods.push(m);return m;}
function negative(m,id,subject,reason,source){if(negativeRegister.records.some(n=>n.negative_id===id))throw Error("Duplicate negative");negativeRegister.records.push({negative_id:id,subject,reason,source,subject_success_credit:0});m.retained_negative_ids.push(id);}
function witness(m,id,result,procedure,observed,negatives=[]){if(ledger.witnesses.some(w=>w.witness_id===id))throw Error("Duplicate witness");m.validation_witness_ids.push(id);ledger.witnesses.push({witness_id:id,method_id:m.method_id,procedure,scope:D,expected:"Declared bounded predicate with no authority promotion",observed,result,same_owner_only:true,independent_reproduction:false,retained_negative_ids:negatives,boundary:"Finite same-owner synthetic software only; NOT_READY_FOR_STAGE_20."});}
function save(){for(const m of ledger.methods.filter(m=>m.recommendation_state==="candidate")){if(!m.retained_negative_ids.length||!ledger.witnesses.some(w=>w.method_id===m.method_id&&w.result==="pass"))throw Error("Method lacks evidence");m.recommendation_state="preferred";ledger.state_events.push({method_id:m.method_id,from:"candidate",to:"validated",note:"Bounded observations recorded."},{method_id:m.method_id,from:"validated",to:"preferred",note:"Select only within the declared scope."});ledger.recommendations.push({method_id:m.method_id,recommendation:m.recurrence_guard,delivered:false});}ledger.counts={methods:ledger.methods.length,witnesses:ledger.witnesses.length,state_events:ledger.state_events.length,recommendations:ledger.recommendations.length,states:{observed:0,candidate:0,validated:0,preferred:ledger.methods.length,superseded:0,deprecated:0},witness_results:{pass:ledger.witnesses.filter(w=>w.result==="pass").length,fail:ledger.witnesses.filter(w=>w.result==="fail").length}};negativeRegister.count=negativeRegister.records.length;truth.counts=ledger.counts;truth.negatives=negativeRegister.count;fs.writeFileSync(D+"/method-flow.json",JSON.stringify(ledger)+"\n");fs.writeFileSync(D+"/negative-register.json",JSON.stringify(negativeRegister,null,2)+"\n");fs.writeFileSync(D+"/phase-truth.json",JSON.stringify(truth,null,2)+"\n");return ledger.counts;}
return {ledger,negativeRegister,truth,method,negative,witness,save};}
module.exports={open};
