"use strict";
function createRuntime(c){const added=["rel_dependency_preservation","rel_join_witness","rel_bcnf_audit","rel_third_normal_form","rel_renaming_covariance","rel_lossless_chase","rel_nonimplication_witness","rel_schema_projection","rel_workload_gap","rel_data_authority_hold"],operations=[...c.operations,...added];
function naturalJoin(rows,parts){let accumulated=[Array(4).fill(null)];for(const mask of parts){const projected=[...new Map(rows.map(row=>{const p=row.map((v,j)=>mask&(1<<j)?v:null);return [JSON.stringify(p),p];})).values()],next=new Map();for(const left of accumulated)for(const right of projected){if(left.some((v,j)=>v!==null&&right[j]!==null&&v!==right[j]))continue;const joined=left.map((v,j)=>v===null?right[j]:v);next.set(JSON.stringify(joined),joined);}accumulated=[...next.values()];}return c.tupleSet(accumulated);}
function chase(f,parts){const table=parts.map((s,i)=>Array.from({length:4},(_,j)=>s&(1<<j)?j:4+4*i+j)),parent=Array.from({length:4+4*parts.length},(_,i)=>i);function find(x){while(parent[x]!==x){parent[x]=parent[parent[x]];x=parent[x];}return x;}let changes=true,merges=0;while(changes){changes=false;for(const [lhs,rhs] of f)for(let i=0;i<table.length;i++)for(let j=i+1;j<table.length;j++){if(!c.bits(lhs).every(bit=>find(table[i][Math.log2(bit)])===find(table[j][Math.log2(bit)])))continue;for(const bit of c.bits(rhs)){const k=Math.log2(bit),a=find(table[i][k]),b=find(table[j][k]);if(a!==b){parent[Math.max(a,b)]=Math.min(a,b);merges++;changes=true;}}}}const tableau=table.map(row=>row.map(find)),distinguished_rows=tableau.map((row,i)=>row.every((v,j)=>v===j)?i:-1).filter(i=>i>=0);return {tableau,distinguished_rows,lossless:distinguished_rows.length>0,equality_merges:merges};}
const permute=(mask,p)=>c.bits(mask).reduce((s,b)=>s|(1<<p[Math.log2(b)]),0);
function evaluate(r){if(c.record(r)&&c.operations.includes(r.operation))return c.evaluate(r);try{c.validate(r,operations);const {operation:op,dependencies:f,determinant:X,consequent:Y,decomposition:parts,permutation:perm}=r,R=c.tupleSet(r.tuples);let value,outcome="completed";switch(op){
case "rel_dependency_preservation":{const p=c.sortFD(parts.flatMap(s=>c.projected(f,s))),lost=c.normalizeFD(f).filter(([l,r])=>(c.closure(l,p)&r)!==r);value={projected_union:p,preserved:lost.length===0,lost_dependencies:lost};break;}
case "rel_join_witness":{const joined=naturalJoin(R,parts),old=new Set(R.map(t=>t.join(","))),extra=joined.filter(t=>!old.has(t.join(",")));value={original:R,joined,spurious:extra,instance_lossless:extra.length===0,semantic_losslessness_inferred:false};break;}
case "rel_bcnf_audit":{const violations=[];for(let s=0;s<16;s++){const cl=c.closure(s,f);if(cl!==15&&(cl&~s))violations.push({determinant:s,implied_nontrivial:cl&~s});}value={bcnf:violations.length===0,violations};break;}
case "rel_third_normal_form":{const keys=c.keys(f),prime=keys.reduce((a,b)=>a|b,0),violations=[];for(let s=0;s<16;s++){const cl=c.closure(s,f),bad=cl&~s&~prime;if(cl!==15&&bad)violations.push({determinant:s,nonprime_rhs:bad});}value={third_normal_form:violations.length===0,prime_mask:prime,keys,violations};break;}
case "rel_renaming_covariance":{const p=f.map(([l,r])=>[permute(l,perm),permute(r,perm)]),rows=Array.from({length:16},(_,s)=>({source:s,target:permute(s,perm),transported_closure:permute(c.closure(s,f),perm),target_closure:c.closure(permute(s,perm),p)}));value={permutation:perm,renamed_dependencies:p,commutes:rows.every(r=>r.transported_closure===r.target_closure),rows};break;}
case "rel_lossless_chase":value=chase(f,parts);break;
case "rel_nonimplication_witness":{const cl=c.closure(X,f),entailed=(cl&Y)===Y,counterexample=entailed?null:[[0,0,0,0],Array.from({length:4},(_,j)=>cl&(1<<j)?0:1)];value={entailed,closure:cl,counterexample,holds_dependencies:counterexample?c.violations(counterexample,f).length===0:null,violates_query:counterexample?c.violations(counterexample,[[X,Y]]).length>0:null};break;}
case "rel_schema_projection":value={coordinates:[c.normalizeFD(f).length,c.keys(f).length,R.length],axes:["nontrivial_singleton_dependencies","candidate_keys","unique_tuples"],projection_only:true};outcome="represented";break;
case "rel_workload_gap":value={observations:0,production_benchmark:false,required:["governed_workload","prespecified_metrics","representative_sampling","independent_review"]};outcome="open_gap";break;
case "rel_data_authority_hold":value={authorized:false,external_actions:0,status:"held",required:["data_steward_authority","affected_parties","privacy_and_accessibility_remedy","Maori_authority_when_applicable"]};outcome="exact_gate";break;
}return c.envelope(op,value,outcome);}catch(e){return c.reject(e);}}
return {...c,operations,evaluate,naturalJoin,chase,permute};}
module.exports={createRuntime};if(require.main===module){const c=require("../x1/core.txt");c.cli(createRuntime(c));}
