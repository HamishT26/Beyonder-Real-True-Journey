def g(*args):return subprocess.check_output(["git","-C",str(ROOT),*args],stderr=subprocess.PIPE).decode("utf-8").strip()
def load(p):return v.strict(p.read_bytes())
def external_write(path,obj):
 path.parent.mkdir(parents=True,exist_ok=True)
 with path.open("x",encoding="utf-8",newline="\n") as f:json.dump(obj,f,indent=2,sort_keys=True);f.write("\n")
def equality(branch):
 heads=[g("rev-parse","HEAD"),g("rev-parse","@{upstream}"),g("rev-parse","refs/remotes/origin/"+branch),
  g("ls-remote","--exit-code","origin","refs/heads/"+branch).split()[0]]
 divergence=[int(x) for x in g("rev-list","--left-right","--count","HEAD...@{upstream}").split()]
 return {"heads":heads,"equal":len(set(heads))==1,"divergence":divergence,"clean":not g("status","--porcelain=v1")}

 checks["owner_file_ceiling"]=len(paths)<2000

 x1paths=[e["path"] for e in x1m["entries"]]+x1m["self_exclusions"]
 checks["x1_immutable"]=not g("diff","--name-only",policy["x1"],head,"--",*x1paths)
 em=load(ROOT/BASE/"validation/evidence-manifest.json")
 epaths=[e["path"] for e in em["entries"]]+em["self_exclusions"]

 scan=v.scan(ROOT,paths)
 checks["strict_json_and_ast"]=scan["strict_json_count"]>0 and scan["python_ast_count"]>0
 checks["configured_privacy"]=scan["privacy"]["valid"]
 checks["configured_security"]=scan["security"]["valid"]
 checks["document_word_ceiling"]=scan["maximum_document_words"]<=100000
 manifest_paths=[BASE+"/x1/x1-manifest.json",BASE+"/x2/deck/card-manifest.json",
  BASE+"/validation/evidence-manifest.json",BASE+"/validation/final-manifest.json"]
 manifests=[v.verify_manifest(ROOT,ROOT/p) for p in manifest_paths]
 checks["all_declared_manifests"]=all(m["valid"] for m in manifests)
 fm=load(ROOT/BASE/"validation/final-manifest.json")
 checks["final_manifest_complete"]={e["path"] for e in fm["entries"]}|set(fm["self_exclusions"])==rels
 # Bind working text, with explicit LF normalization, to the exact committed final blobs.
 ordered=sorted(rels);blobs=v.batch_blobs(ROOT,[head+":"+p for p in ordered])
 checks["working_to_exact_git_blob"]=all(v.normalized((ROOT/p).read_bytes())==b for p,b in zip(ordered,blobs))
 seal=load(ROOT/BASE/"seal/content-seal.json")
 checks["content_seal"]=all(v.sha(v.normalized((ROOT/e["path"]).read_bytes()))==e["sha256_normalized_lf"]
  and len(v.normalized((ROOT/e["path"]).read_bytes()))==e["bytes_normalized_lf"] for e in seal["targets"])
 promotions=v.verify_promotions(ROOT,skills,runners);checks["raw_global_copy_parity"]=promotions["valid"]
 pr=load(ROOT/BASE/"x2/promotion-receipt.json");by_path=dict(zip(ordered,blobs))
 joins=[]
 for e in pr["source_global_parity"]:
  dest=e["global_target"]
  target=skills/dest.removeprefix("skills/") if dest.startswith("skills/") else runners/dest.removeprefix("owner_runner_bank/")
  joins.append(v.normalized(target.read_bytes())==by_path[e["source"]])
 checks["global_to_committed_lf_join"]=all(joins)
 checks["outcomes_exact"]=truth["outcomes"]=={"completed":176,"represented":11,"open_gap":3,"exact_gate":10}
 checks["source_failure_retained"]=truth["source_failed_canonical_credit"]==0 and truth["source_canonical_replayed"] is False
 checks["current_method_schema"]=load(ROOT/BASE/"x2/method-flow/validation.json")["valid"] is True
 ledger=load(ROOT/BASE/"x2/method-flow/ledger.json");w=ledger["witnesses"]
 checks["retained_failure_counts"]=(sum(r["result"]=="fail" for r in w)==531 and sum(r["result"]=="pass" for r in w)==37
  and len({r["witness_id"] for r in w})==len(w))
 checks["retained_failed_definitions"]=(ROOT/BASE/"x2/retained-attempts/firmware-core-v1.py.txt").is_file() and (ROOT/BASE/"x2/retained-attempts/x2-tests-v1.py.txt").is_file()
 checks["release_profile"]=load(ROOT/BASE/"x1/release-profile-validation.json")["status"]=="PASS"
 checks["workflow_dependency_correction"]=load(ROOT/BASE/"x1/workflow-current-profile-overlay.json")["status"]=="VALID_CURRENT_RELEASE_DEPENDENCY_CORRECTED_PLAN"
 packages=load(ROOT/BASE/"x2/package-transaction.json")
 checks["package_contract"]=packages["state"]=="COMPLETE" and packages["direct_count"]==3 and packages["closure_count"]==7 and packages["pip_check_passed"]
 checks["promotion_counts"]=pr["skill_count"]==10 and pr["runner_interface_count"]==5 and len(pr["source_global_parity"])==96
 bi=load(ROOT/BASE/"final/baton-index.json");baton=(ROOT/bi["path"]).read_text(encoding="utf-8")

 route=load(ROOT/BASE/"final/terminal-route-checklist.json")
 checks["prepared_not_sent"]=bi["delivery_state"]==route["state"]=="PREPARED_NOT_SENT" and route["sends_already_made"]==truth["successor_contacts"]==0
 checks["one_existing_sylven_edge"]=route["recipient_title"]=="Sylven Arc" and route["recipient_phase"]=="v688-v7" and route["authorized_sends"]==1
 checks["not_ready_for_stage20"]=truth["terminal_verdict"]=="NOT_READY_FOR_STAGE_20"
 return {"policy":policy,"checks":checks,"equality_before":eq,"scan":scan,"manifests":manifests,"promotions":promotions,
  "seal_targets":len(seal["targets"]),"owner_files":len(paths),"global_lf_bindings":len(joins)}
def main():

  if marker.exists() or receipt.exists():raise SystemExit("canonical_started_preflight_replay_refused")

   print(json.dumps({"canonical_module":entry["stage"],"tests":count,"passed":passed}),flush=True)
