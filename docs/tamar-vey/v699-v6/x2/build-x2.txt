function stable(value) {
  if (Array.isArray(value)) return `[${value.map(stable).join(",")}]`;
  if (value && typeof value === "object") return `{${Object.keys(value).sort().map((key) => `${JSON.stringify(key)}:${stable(value[key])}`).join(",")}}`;
  return JSON.stringify(value);
}

function sha(value) {
  const bytes = Buffer.isBuffer(value) ? value : Buffer.from(typeof value === "string" ? value : stable(value), "utf8");
  return crypto.createHash("sha256").update(bytes).digest("hex");
}

function writeJson(filePath, value) {
  fs.writeFileSync(filePath, `${JSON.stringify(value, null, 2)}\n`, "utf8");
}

function writeText(filePath, value) {
  fs.writeFileSync(filePath, value.endsWith("\n") ? value : `${value}\n`, "utf8");
}

function loadTxtModule(filePath) {
  const code = fs.readFileSync(filePath, "utf8");
  const module = { exports: {} };
  new vm.Script(`(function(exports,module,require,__filename,__dirname){${code}\n})`, { filename: filePath }).runInThisContext()(module.exports, module, require, filePath, path.dirname(filePath));
  return module.exports;
}

function sortSets(sets) {
  return sets.map((set) => [...set].sort()).sort((a, b) => a.length - b.length || a.join("\u0000").localeCompare(b.join("\u0000")));
}

function uniqueEdges(edges) {
  const map = new Map();
  for (const edge of edges) {
    const normalized = [...edge].sort();
    map.set(normalized.join("\u0000"), normalized);
  }
  return [...map.values()].sort((a, b) => a.join("\u0000").localeCompare(b.join("\u0000")));
}

function subsets(values) {
  const output = [];
  for (let mask = 0; mask < 2 ** values.length; mask += 1) {
    const row = [];
    for (let index = 0; index < values.length; index += 1) if (mask & (1 << index)) row.push(values[index]);
    output.push(row);
  }
  return sortSets(output);
}

function subset(left, right) {
  return left.every((value) => right.includes(value));
}

function minimal(sets) {
  return sortSets(sets.filter((candidate) => !sets.some((other) => other.length < candidate.length && subset(other, candidate))));
}

function maximal(sets) {
  return sortSets(sets.filter((candidate) => !sets.some((other) => other.length > candidate.length && subset(candidate, other))));
}

function minimalTransversals(vertices, edges) {
  return minimal(subsets(vertices).filter((set) => edges.every((edge) => edge.some((vertex) => set.includes(vertex)))));
}

function edgeFamilies(edges) {
  return subsets(edges.map((_, index) => index));
}

function disjoint(indices, edges) {
  return new Set(indices.flatMap((index) => edges[index])).size === indices.reduce((sum, index) => sum + edges[index].length, 0);
}

function oracle(raw) {
  const vertices = [...raw.vertices].sort();
  const edges = uniqueEdges(raw.edges);
  const operation = raw.operation;
  let result;
  let outcome = "completed";
  if (operation === "transversal_enumeration") {
    result = { transversals: minimalTransversals(vertices, edges) };
  } else if (operation === "minimum_transversals") {
    const all = minimalTransversals(vertices, edges); const cardinality = Math.min(...all.map((set) => set.length)); result = { cardinality, transversals: all.filter((set) => set.length === cardinality) };
  } else if (operation === "independent_set_enumeration") {
    result = { independent_sets: maximal(subsets(vertices).filter((set) => edges.every((edge) => !edge.every((vertex) => set.includes(vertex))))) };
  } else if (operation === "matching_enumeration") {
    result = { matchings: maximal(edgeFamilies(edges).filter((set) => disjoint(set, edges))).map((set) => set.map((index) => `e${index + 1}`)) };
  } else if (operation === "maximum_matchings") {
    const valid = edgeFamilies(edges).filter((set) => disjoint(set, edges)); const cardinality = Math.max(...valid.map((set) => set.length)); result = { cardinality, matchings: sortSets(valid.filter((set) => set.length === cardinality)).map((set) => set.map((index) => `e${index + 1}`)) };
  } else if (operation === "exact_cover_enumeration") {
    const target = vertices.join("\u0000"); result = { exact_covers: sortSets(edgeFamilies(edges).filter((set) => disjoint(set, edges) && set.flatMap((index) => edges[index]).sort().join("\u0000") === target)).map((set) => set.map((index) => `e${index + 1}`)) };
  } else if (operation === "dual_hypergraph") {
    const ids = edges.map((_, index) => `e${index + 1}`); result = { vertices: ids, edges: vertices.map((vertex) => ({ source_vertex: vertex, members: edges.map((edge, index) => edge.includes(vertex) ? ids[index] : null).filter(Boolean) })) };
  } else if (operation === "blocker_projection") {
    outcome = "represented"; result = { representation: "minimal_transversal_family", edges: minimalTransversals(vertices, edges), complexity_theorem_claimed: false };
  } else if (operation === "hypergraph_external_observation_gap") {
    outcome = "open_gap"; result = { open: true, real_rows: 0, missing: ["real workshop observation", "manual assistive-technology evaluation", "affected-user evaluation", "independent review"] };
  } else if (operation === "hypergraph_governance_authority_gate") {
    outcome = "exact_gate"; result = { executed: false, authority_present: false, reserved: ["professional release", "custody and ownership", "copyright", "remedy", "legal and cultural interpretation", "affected-party acceptance", "Maori wording data governance and authority"] };
  } else throw new Error(`unsupported oracle operation ${operation}`);
  return { ok: true, operation, outcome, result, scope: "finite_synthetic_hypergraph_only", same_owner_only: true, independent_reproduction: false };
}
