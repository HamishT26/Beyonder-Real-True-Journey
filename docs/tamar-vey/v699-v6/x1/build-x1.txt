function stable(value) {
  if (Array.isArray(value)) return `[${value.map(stable).join(",")}]`;
  if (value && typeof value === "object") return `{${Object.keys(value).sort().map((key) => `${JSON.stringify(key)}:${stable(value[key])}`).join(",")}}`;
  return JSON.stringify(value);
}

function sha(value) {
  const bytes = Buffer.isBuffer(value) ? value : Buffer.from(typeof value === "string" ? value : stable(value), "utf8");
  return crypto.createHash("sha256").update(bytes).digest("hex");
}

function writeJson(filePath, value) {
  fs.writeFileSync(filePath, `${JSON.stringify(value, null, 2)}\n`, "utf8");
}

function writeText(filePath, value) {
  fs.writeFileSync(filePath, value.endsWith("\n") ? value : `${value}\n`, "utf8");
}

function loadTxtModule(filePath) {
  const code = fs.readFileSync(filePath, "utf8");
  const module = { exports: {} };
  const wrapper = new vm.Script(`(function(exports,module,require,__filename,__dirname){${code}\n})`, { filename: filePath });
  wrapper.runInThisContext()(module.exports, module, require, filePath, path.dirname(filePath));
  return module.exports;
}

function canonicalEdges(edges) {
  const map = new Map();
  for (const edge of edges) {
    const normalized = [...edge].sort();
    map.set(normalized.join("\u0000"), normalized);
  }
  return [...map.values()].sort((left, right) => left.join("\u0000").localeCompare(right.join("\u0000")));
}

function vertexPairs(vertices) {
  const result = [];
  for (let i = 0; i < vertices.length; i += 1) for (let j = i + 1; j < vertices.length; j += 1) result.push([vertices[i], vertices[j]]);
  return result;
}

function oracle(raw) {
  const vertices = [...raw.vertices].sort();
  const edges = canonicalEdges(raw.edges);
  const focus = [...raw.focus].sort();
  const focusSet = new Set(focus);
  const operation = raw.operation;
  let result;
  if (operation === "vertex_inventory") {
    result = { count: vertices.length, vertices };
  } else if (operation === "edge_normalization") {
    const used = new Set(edges.flat());
    result = { edge_count: edges.length, edges, isolated_vertices: vertices.filter((vertex) => !used.has(vertex)) };
  } else if (operation === "incidence_matrix") {
    result = { vertices, edge_ids: edges.map((_, index) => `e${index + 1}`), matrix: vertices.map((vertex) => edges.map((edge) => edge.indexOf(vertex) >= 0 ? 1 : 0)) };
  } else if (operation === "vertex_degrees") {
    result = { degrees: vertices.map((vertex) => ({ vertex, degree: edges.reduce((count, edge) => count + (edge.indexOf(vertex) >= 0 ? 1 : 0), 0) })) };
  } else if (operation === "pair_codegrees") {
    result = { pairs: vertexPairs(vertices).map(([left, right]) => ({ left, right, codegree: edges.reduce((count, edge) => count + (edge.includes(left) && edge.includes(right) ? 1 : 0), 0) })) };
  } else if (operation === "induced_vertex_subhypergraph") {
    result = { vertices: focus, edges: canonicalEdges(edges.map((edge) => edge.filter((vertex) => focusSet.has(vertex))).filter((edge) => edge.length)) };
  } else if (operation === "edge_restriction") {
    result = { vertices: focus, edges: edges.filter((edge) => edge.every((vertex) => focusSet.has(vertex))) };
  } else if (operation === "primal_graph") {
    result = { vertices, edges: vertexPairs(vertices).filter(([left, right]) => edges.filter((edge) => edge.includes(left) && edge.includes(right)).length > 0) };
  } else if (operation === "line_graph") {
    const ids = edges.map((_, index) => `e${index + 1}`);
    const graphEdges = [];
    for (let i = 0; i < edges.length; i += 1) for (let j = i + 1; j < edges.length; j += 1) if (edges[i].filter((value) => edges[j].includes(value)).length) graphEdges.push([ids[i], ids[j]]);
    result = { vertices: ids, edges: graphEdges };
  } else if (operation === "berge_components") {
    const remaining = new Set(vertices);
    const components = [];
    while (remaining.size) {
      const component = [Array.from(remaining).sort()[0]];
      remaining.delete(component[0]);
      for (let cursor = 0; cursor < component.length; cursor += 1) {
        for (const edge of edges.filter((edge) => edge.includes(component[cursor]))) {
          for (const vertex of edge) if (remaining.delete(vertex)) component.push(vertex);
        }
      }
      components.push(component.sort());
    }
    result = { components };
  } else {
    throw new Error(`oracle does not support ${operation}`);
  }
  return { ok: true, operation, outcome: "completed", result, scope: "finite_synthetic_hypergraph_only", same_owner_only: true, independent_reproduction: false };
}
