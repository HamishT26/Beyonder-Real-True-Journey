"use strict";

const crypto = require("crypto");
const fs = require("fs");
const path = require("path");
const { spawnSync } = require("child_process");

const ROOT = process.cwd();
const BASE_REL = "docs/tamar-vey/v699-v6";
const X1_REL = `${BASE_REL}/x1`;
const X1_DIR = path.join(ROOT, ...X1_REL.split("/"));
const MANIFEST = `${X1_REL}/x1-index-manifest.json`;
const REVIEW = `${X1_REL}/x1-staged-review.json`;
const AUDIT = `${X1_REL}/x1-stage-audit.json`;
const ALLOWED_PREFIXES = [`${X1_REL}/`, `${BASE_REL}/skills/`, `${BASE_REL}/runners/`];
const BOUNDARY = "Same-owner x1 finite synthetic hypergraph Git-index evidence only; not independent reproduction, empirical confirmation, professional or production validation, authority, proof, canon, or Stage 20 evidence.";

function git(args, encoding = null) {
  const result = spawnSync("git", args, { cwd: ROOT, encoding, maxBuffer: 32 * 1024 * 1024 });
  if (result.status !== 0) throw new Error(result.stderr?.toString() || `git ${args.join(" ")} failed`);
  return result.stdout;
}

function sha(bytes) {
  return crypto.createHash("sha256").update(bytes).digest("hex");
}

const staged = git(["diff", "--cached", "--name-only", "--diff-filter=ACMR"], "utf8").trim().split(/\r?\n/).filter(Boolean).sort();
const scoped = staged.filter((name) => ALLOWED_PREFIXES.some((prefix) => name.startsWith(prefix)) && name !== MANIFEST && name !== REVIEW && name !== AUDIT);
if (scoped.length + staged.filter((name) => [MANIFEST, REVIEW, AUDIT].includes(name)).length !== staged.length) throw new Error("staged scope contains a non-x1 owner path");
const entries = scoped.map((name) => {
  const bytes = git(["show", `:${name}`]);
  return { path: name, bytes: bytes.length, sha256: sha(bytes) };
});
const exclusions = [MANIFEST, REVIEW, AUDIT];
fs.writeFileSync(path.join(X1_DIR, "x1-index-manifest.json"), `${JSON.stringify({ schema: "tamar.x1-index-manifest.v1", byte_domain: "git_index_blob", expected_parent: "ead7018c661745bfcc380aa6b91e18c1b8ba0bb7", entries, exclusions, boundary: BOUNDARY }, null, 2)}\n`, "utf8");
fs.writeFileSync(path.join(X1_DIR, "x1-staged-review.json"), `${JSON.stringify({ schema: "tamar.x1-staged-review.v1", expected_parent: "ead7018c661745bfcc380aa6b91e18c1b8ba0bb7", staged_before_receipts: scoped.length, expected_final_staged_count: scoped.length + exclusions.length, expected_final_paths: [...scoped, ...exclusions].sort(), manifest_entries: entries.length, manifest_exclusions: exclusions.length, deletion_issues: [], scope_issues: [], privacy_confirmed_hits: [], x2_implementation_present: false, boundary: BOUNDARY }, null, 2)}\n`, "utf8");
process.stdout.write(`${JSON.stringify({ status: "X1_INDEX_SEALED", entries: entries.length, exclusions: exclusions.length, expected_final_staged_count: scoped.length + exclusions.length })}\n`);
