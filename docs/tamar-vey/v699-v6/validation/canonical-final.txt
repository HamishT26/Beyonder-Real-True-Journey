function arg(name) { const index = process.argv.indexOf(name); return index >= 0 ? process.argv[index + 1] : null; }
const expectedHead = arg("--expected-head");
const receiptDir = arg("--receipt-dir");
if (!expectedHead || !/^[0-9a-f]{40}$/.test(expectedHead)) throw new Error("exact expected head required");
if (!receiptDir) throw new Error("external receipt directory required");

const latchPath = path.join(receiptDir, `canonical-${short}.latch`);

function git(args, encoding = null) { const result = spawnSync("git", args, { cwd: ROOT, encoding, maxBuffer: 256 * 1024 * 1024 }); if (result.status !== 0) throw new Error(result.stderr?.toString() || `git ${args.join(" ")} failed`); return result.stdout; }
const sha = (bytes) => crypto.createHash("sha256").update(bytes).digest("hex");
const stable = (value) => Array.isArray(value) ? `[${value.map(stable).join(",")}]` : value && typeof value === "object" ? `{${Object.keys(value).sort().map((key) => `${JSON.stringify(key)}:${stable(value[key])}`).join(",")}}` : JSON.stringify(value);
const readCommitJson = (commit, name) => JSON.parse(git(["show", `${commit}:${name}`], "utf8"));
const checks = [];
const failures = [];
const add = (name, passed, detail) => { checks.push({ name, passed: Boolean(passed), detail }); if (!passed) failures.push(name); };
function replayManifest(commit, manifestPath) {
  const manifest = readCommitJson(commit, manifestPath); const mismatches = [];
  for (const item of manifest.entries) { const bytes = git(["show", `${commit}:${item.path}`]); const digest = sha(bytes); let mode = item.mode; if (Object.prototype.hasOwnProperty.call(item, "mode")) { const tree = git(["ls-tree", commit, "--", item.path], "utf8").trim(); mode = tree.split(/\s+/)[0]; } if (bytes.length !== item.bytes || digest !== item.sha256 || (Object.prototype.hasOwnProperty.call(item, "mode") && mode !== item.mode)) mismatches.push({ path: item.path, bytes: bytes.length, sha256: digest, mode }); }
  return { manifest_path: manifestPath, entries: manifest.entries.length, exclusions: manifest.exclusions?.length || 0, mismatches };
}

let receipt;
try {
  const head = git(["rev-parse", "HEAD"], "utf8").trim();
  const branch = git(["branch", "--show-current"], "utf8").trim();
  add("exact_head", head === expectedHead, head);

  add("typed_zero_divergence_before", divergence === "0\t0", divergence);
  const upstream = git(["rev-parse", "@{upstream}"], "utf8").trim();
  const tracking = git(["rev-parse", `refs/remotes/origin/${BRANCH}`], "utf8").trim();
  const fresh = git(["ls-remote", "--heads", "origin", `refs/heads/${BRANCH}`], "utf8").trim().split(/\s+/)[0];
  add("four_way_equality_before", [head, upstream, tracking, fresh].every((value) => value === expectedHead), { local: head, upstream, tracking, fresh });

  const edges = [[SOURCE, PLANNING], [PLANNING, X1], [X1, X2], [X2, expectedHead]];
  for (const [parent, child] of edges) add(`direct_parent_${child.slice(0, 8)}`, git(["rev-parse", `${child}^`], "utf8").trim() === parent, { parent, child });
  const commitCount = Number(git(["rev-list", `${SOURCE}..${expectedHead}`, "--count"], "utf8").trim());
  const mergeCount = Number(git(["rev-list", "--merges", `${SOURCE}..${expectedHead}`, "--count"], "utf8").trim());
  const finalParents = git(["show", "-s", "--format=%P", expectedHead], "utf8").trim().split(/\s+/).filter(Boolean).length;
  add("four_phase_commits", commitCount === 4, commitCount);
  add("zero_merges", mergeCount === 0, mergeCount);
  add("one_final_parent", finalParents === 1, finalParents);

  const manifestSpecs = [
    [PLANNING, `${BASE_REL}/planning/planning-index-manifest.json`, "planning_manifest"],
    [X1, `${BASE_REL}/x1/x1-index-manifest.json`, "x1_manifest"],
    [X2, `${BASE_REL}/x2/x2-index-manifest.json`, "x2_manifest"],
    [expectedHead, `${BASE_REL}/validation/final-delta-manifest.json`, "final_delta_manifest"],
    [expectedHead, `${BASE_REL}/validation/final-owner-manifest.json`, "final_owner_manifest"]
  ];
  const manifestResults = [];
  for (const [commit, manifestPath, name] of manifestSpecs) { const result = replayManifest(commit, manifestPath); manifestResults.push({ name, commit, ...result }); add(name, result.mismatches.length === 0, { entries: result.entries, exclusions: result.exclusions, mismatches: result.mismatches.length }); }
  const seal = readCommitJson(expectedHead, `${BASE_REL}/closeout/content-seal.json`);
  const sealMismatches = [];
  for (const item of seal.targets) { const bytes = git(["show", `${expectedHead}:${item.path}`]); const tree = git(["ls-tree", expectedHead, "--", item.path], "utf8").trim(); const mode = tree.split(/\s+/)[0]; if (bytes.length !== item.bytes || sha(bytes) !== item.sha256 || mode !== item.mode) sealMismatches.push(item.path); }
  add("content_seal", sealMismatches.length === 0, { targets: seal.targets.length, mismatches: sealMismatches.length });

  const ownerNames = git(["ls-tree", "-r", "--name-only", expectedHead, "--", BASE_REL], "utf8").trim().split(/\r?\n/).filter(Boolean).sort();
  add("owner_file_ceiling", ownerNames.length < 2000, ownerNames.length);
  let jsonParsed = 0, txtSyntax = 0, markdownChecked = 0, htmlChecked = 0;
  let maximumDocument = { path: "", words: 0, bytes: 0 };
  const privacyCandidates = [], securityCandidates = [];
  const privacyPatterns = [
    ["raw_task_or_thread_identifier", /\b(?:threadId|thread_id|task_id)\b|\b[0-9a-f]{8}-[0-9a-f]{4}-[1-5][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}\b/gi],
    ["private_absolute_path", /\b[A-Za-z]:\\/g],
    ["credential_or_private_key", /\b(?:sk-[A-Za-z0-9_-]{20,}|ghp_[A-Za-z0-9]{20,}|BEGIN (?:RSA |EC )?PRIVATE KEY)\b/g],
    ["private_conversation_or_session_stream", /\b(?:raw transcript|session stream|private screenshot)\b/gi],
    ["private_callable_or_application_state", /\b(?:callable_identifier|private application state|private app state)\b/gi]
  ];
  const securityPatterns = [["dynamic_eval", /\beval\s*\(/g], ["dynamic_function", /\bnew\s+Function\s*\(/g], ["shell_exec", /child_process\.exec\s*\(/g]];
  for (const name of ownerNames) {
    const ext = path.extname(name).toLowerCase(); const bytes = git(["show", `${expectedHead}:${name}`]); const text = bytes.toString("utf8"); const words = text.trim().length ? text.trim().split(/\s+/).length : 0;
    if (![".json", ".md", ".txt", ".html"].includes(ext)) failures.push(`forbidden_extension:${name}`);
    if (words > maximumDocument.words) maximumDocument = { path: name, words, bytes: bytes.length };
    if (words > 100000) failures.push(`document_ceiling:${name}`);

    for (const [kind, pattern] of privacyPatterns) { const count = (text.match(pattern) || []).length; if (count) { const scannerDefinition = ext === ".txt" && /(?:validate|audit|canonical|terminal)/.test(name); privacyCandidates.push({ path: name, kind, count, adjudication: scannerDefinition ? "scanner_definition" : "unresolved" }); } }
