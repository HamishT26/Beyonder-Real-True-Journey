"use strict";
function createRuntime(core){
 const fields={ext_multiplication_matrix:["operation","field","a"],ext_trace_norm_matrix_certificate:["operation","field","a"],ext_minimal_polynomial:["operation","field","a"],ext_minimal_polynomial_vanishes:["operation","field","a","polynomial"],ext_prime_subfield_membership:["operation","field","a"],ext_basis_coordinates:["operation","field","a","beta"],ext_basis_change_matrix:["operation","field","beta"],ext_isomorphism_witness:["operation","field","target_modulus"],ext_independent_implementation_vacancy:["operation","field","requested_check"],ext_release_authority_hold:["operation","field","declarations"]};
 function determinant(matrix,p){const a=matrix.map(row=>row.map(x=>core.mod(x,p)));let det=1;for(let col=0;col<a.length;col++){let pivot=col;while(pivot<a.length&&a[pivot][col]===0)pivot++;if(pivot===a.length)return 0;if(pivot!==col){[a[pivot],a[col]]=[a[col],a[pivot]];det=core.mod(-det,p);}const value=a[col][col];det=core.mod(det*value,p);const inv=core.scalarInverse(value,p);for(let row=col+1;row<a.length;row++){const factor=core.mod(a[row][col]*inv,p);for(let k=col;k<a.length;k++)a[row][k]=core.mod(a[row][k]-factor*a[col][k],p);}}return det;}
 function inverseMatrix(matrix,p){const n=matrix.length,a=matrix.map((row,i)=>[...row.map(x=>core.mod(x,p)),...Array.from({length:n},(_,j)=>i===j?1:0)]);for(let col=0;col<n;col++){let pivot=col;while(pivot<n&&a[pivot][col]===0)pivot++;core.requireThat(pivot<n,"E_BASIS");if(pivot!==col)[a[pivot],a[col]]=[a[col],a[pivot]];const inv=core.scalarInverse(a[col][col],p);for(let k=0;k<2*n;k++)a[col][k]=core.mod(a[col][k]*inv,p);for(let row=0;row<n;row++)if(row!==col){const factor=a[row][col];for(let k=0;k<2*n;k++)a[row][k]=core.mod(a[row][k]-factor*a[col][k],p);}}return a.map(row=>row.slice(n));}
 function matrixVector(matrix,vector,p){return matrix.map(row=>core.mod(row.reduce((sum,x,i)=>sum+x*vector[i],0),p));}
 function multiplicationMatrix(a,spec){const matrix=Array.from({length:spec.n},()=>Array(spec.n).fill(0));for(let col=0;col<spec.n;col++){const basis=Array(spec.n).fill(0);basis[col]=1;const product=core.multiply(a,basis,spec);for(let row=0;row<spec.n;row++)matrix[row][col]=product[row];}return matrix;}
 function matrixTrace(matrix,p){return core.mod(matrix.reduce((sum,row,i)=>sum+row[i],0),p);}
 function extPolyMultiply(a,b,spec){const zero=()=>core.element([0],spec),out=Array.from({length:a.length+b.length-1},zero);for(let i=0;i<a.length;i++)for(let j=0;j<b.length;j++)out[i+j]=core.add(out[i+j],core.multiply(a[i],b[j],spec),spec);return out;}
 function minimalPolynomial(a,spec){const roots=core.orbit(a,spec);let coefficients=[core.element([1],spec)];for(const root of roots)coefficients=extPolyMultiply(coefficients,[core.negate(root,spec),core.element([1],spec)],spec);const result=coefficients.map(value=>{core.requireThat(value.slice(1).every(x=>x===0),"E_MINPOLY");return value[0];});return{polynomial:result,degree:result.length-1,orbit_size:roots.length,monic:result[result.length-1]===1};}
 function evaluatePolynomial(coefficients,a,spec){core.requireThat(Array.isArray(coefficients)&&coefficients.length>=2&&coefficients.length<=spec.n+1,"E_POLYNOMIAL");let result=core.element([0],spec);for(let i=coefficients.length-1;i>=0;i--)result=core.add(core.multiply(result,a,spec),core.element([core.integer(coefficients[i])],spec),spec);return result;}
 function powerBasisMatrix(beta,spec){const matrix=Array.from({length:spec.n},()=>Array(spec.n).fill(0));let current=core.element([1],spec);for(let col=0;col<spec.n;col++){for(let row=0;row<spec.n;row++)matrix[row][col]=current[row];current=core.multiply(current,beta,spec);}return matrix;}
 function enumerateElements(spec){const rows=[];for(let index=0;index<spec.p**spec.n;index++){let x=index,v=[];for(let i=0;i<spec.n;i++){v.push(x%spec.p);x=Math.floor(x/spec.p);}rows.push(v);}return rows;}
 function isomorphism(fieldSpec,targetModulus){const target=core.field({id:fieldSpec.id+"-ALT",p:fieldSpec.p,degree:fieldSpec.n,modulus:targetModulus,basis:"power_ascending"});const root=enumerateElements(target).find(candidate=>candidate.slice(1).some(Boolean)&&evaluatePolynomial(fieldSpec.modulus,candidate,target).every(x=>x===0));core.requireThat(!!root,"E_ISOMORPHISM");const matrix=powerBasisMatrix(root,target),det=determinant(matrix,target.p);core.requireThat(det!==0,"E_ISOMORPHISM");return{target_field:{id:target.id,p:target.p,degree:target.n,modulus:target.modulus,basis:target.basis},generator_image:root,mapping_matrix:matrix,determinant:det,verified:true,universal_identity_claimed:false};}
 function evaluate(request){if(core.record(request)&&Object.hasOwn(core.fields,request.operation))return core.evaluate(request);try{const spec=core.checkRequest(request,fields),op=request.operation;let value,outcome="completed";switch(op){case"ext_multiplication_matrix":{const a=core.element(request.a,spec);value={matrix:multiplicationMatrix(a,spec),dimension:spec.n,basis:"power_ascending"};break;}case"ext_trace_norm_matrix_certificate":{const a=core.element(request.a,spec),matrix=multiplicationMatrix(a,spec),trace=core.trace(a,spec),norm=core.norm(a,spec),mt=matrixTrace(matrix,spec.p),md=determinant(matrix,spec.p);value={matrix_trace:mt,matrix_determinant:md,frobenius_trace:trace.trace,frobenius_norm:norm.norm,agrees:mt===trace.trace&&md===norm.norm};break;}case"ext_minimal_polynomial":value=minimalPolynomial(core.element(request.a,spec),spec);break;case"ext_minimal_polynomial_vanishes":{const result=evaluatePolynomial(request.polynomial,core.element(request.a,spec),spec);value={evaluation:result,vanishes:result.every(x=>x===0)};break;}case"ext_prime_subfield_membership":{const a=core.element(request.a,spec),fixed=core.frobenius(a,spec),member=core.equal(a,fixed);value={member,scalar:member?a[0]:null,frobenius_fixed:member};break;}case"ext_basis_coordinates":{const a=core.element(request.a,spec),beta=core.element(request.beta,spec),matrix=powerBasisMatrix(beta,spec),det=determinant(matrix,spec.p),inverse=inverseMatrix(matrix,spec.p),coordinates=matrixVector(inverse,a,spec.p);value={coordinates,reconstructed:matrixVector(matrix,coordinates,spec.p),basis_determinant:det};break;}case"ext_basis_change_matrix":{const beta=core.element(request.beta,spec),matrix=powerBasisMatrix(beta,spec),det=determinant(matrix,spec.p),inverse=inverseMatrix(matrix,spec.p);value={matrix,inverse,determinant:det,invertible:true};outcome="represented";break;}case"ext_isomorphism_witness":value=isomorphism(spec,request.target_modulus),outcome="represented";break;case"ext_independent_implementation_vacancy":core.requireThat(request.requested_check==="independent trace and norm implementation","E_REQUEST");value={status:"open_gap",independent_implementations:0,external_rows:0,field:{id:spec.id,p:spec.p,degree:spec.n,modulus:spec.modulus,basis:spec.basis}};outcome="open_gap";break;case"ext_release_authority_hold":{const expected=["Maori_authority_reserved","authority_reserved","no_external_action","no_real_operator","synthetic_only"];core.requireThat(core.record(request.declarations)&&core.canon(Object.keys(request.declarations).sort())===core.canon(expected),"E_DECLARATIONS");core.requireThat(expected.every(key=>request.declarations[key]===true),"E_DECLARATIONS");value={status:"exact_gate",release:false,declaration_complete:true,field:{id:spec.id,p:spec.p,degree:spec.n,modulus:spec.modulus,basis:spec.basis},authority_present:false};outcome="exact_gate";break;}default:throw new core.ContractError("E_OPERATION");}return core.envelope(op,value,outcome);}catch(error){return core.rejection(error);}}
 return{...core,fields:{...core.fields,...fields},determinant,inverseMatrix,matrixVector,multiplicationMatrix,matrixTrace,minimalPolynomial,evaluatePolynomial,powerBasisMatrix,enumerateElements,isomorphism,evaluate};
}
module.exports={createRuntime};
