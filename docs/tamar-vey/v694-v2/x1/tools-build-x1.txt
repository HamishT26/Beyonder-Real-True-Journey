function validate(ownerRoot) {
  const x1Root = path.join(ownerRoot, BASE, "x1"), planRoot = path.join(ownerRoot, BASE, "plan"), core = require(path.join(x1Root, "extension-field-core.txt")), checks = [];
  if (fs.existsSync(path.join(x1Root, "validation.json"))) throw new Error("x1 already validated");
  const check = (name, pass, detail = null) => checks.push({ name, pass: !!pass, detail });
  const proposals = load(path.join(planRoot, "proposals.json")).proposals.filter((row) => row.session === "x1"), results = load(path.join(x1Root, "results", "proposals.json")), candidates = load(path.join(x1Root, "results", "candidates.json")), altered = load(path.join(x1Root, "results", "altered-expected.json")), caps = load(path.join(x1Root, "evidence", "local-capabilities.json")), ledger = load(path.join(x1Root, "method-flow.json")), negatives = load(path.join(x1Root, "retained-negative-register.json"));
  check("fifty complete observed contracts", results.rows.length === 50 && results.rows.every((row) => row.matches && row.input_unchanged && core.canon(row.output) === core.canon(proposals.find((proposal) => proposal.proposal_id === row.proposal_id).expected_output)));
  check("one hundred failed candidate subjects", candidates.rows.length === 100 && candidates.rows.every((row) => row.subject_failed && row.subject_success_credit === 0 && row.refusal_check_pass));
  check("fifty altered expectations", altered.rows.length === 50 && altered.rows.every((row) => row.subject_success_credit === 0 && row.refusal_check_pass));
  check("safe and cfr portfolios", load(path.join(x1Root, "results", "safe.json")).passed === 100 && load(path.join(x1Root, "results", "clean-fix-refine.json")).passed === 100);
  check("local capability smokes", caps.skills.length === 10 && caps.skills.every((row) => row.official_validation_pass && row.accepting_pass && row.adverse_refusal_pass) && caps.runners.length === 5 && caps.runners.every((row) => row.refusal_pass));
  check("method counts", ledger.counts.methods === ledger.methods.length && ledger.counts.witnesses === ledger.witnesses.length && ledger.counts.witness_results.fail === ledger.witnesses.filter((row) => row.result === "fail").length && ledger.counts.witness_results.pass === ledger.witnesses.filter((row) => row.result === "pass").length);
  check("negative bijection", negatives.records.length === 158 && new Set(negatives.records.map((row) => row.negative_id)).size === 158 && negatives.records.every((negative) => ledger.witnesses.some((row) => row.result === "fail" && row.retained_negative_ids.includes(negative.negative_id))));
  check("preferred methods", ledger.methods.every((method) => method.recommendation_state === "preferred" && ledger.witnesses.some((row) => row.method_id === method.method_id && row.result === "pass")));
  check("planning immutable", git(ownerRoot, ["diff", "--name-only", PLANNING, "--", BASE + "/plan"]) === "");
  check("x2 absent", !fs.existsSync(path.join(ownerRoot, BASE, "x2")));
  const files = walk(path.join(ownerRoot, BASE)), jsonIssues = [], syntaxIssues = [], privacy = [], extensionIssues = []; let maxWords = 0;
  const patterns = [["uuid",/\b[0-9a-f]{8}(?:-[0-9a-f]{4}){3}-[0-9a-f]{12}\b/ig],["path",/\b[A-Z]:[\\/]/g],["route",/(?:codex|app|thread|session):\/\//g],["delegation",/<(?:codex_delegation|source_thread_id)>/g],["credential",/(?:api_key|password|secret)\s*[:=]\s*["'][A-Za-z0-9_+/=-]{16,}/ig]];
  for (const file of files) { const relative = path.relative(ownerRoot, file).replaceAll(path.sep, "/"), extension = path.extname(file), text = fs.readFileSync(file, "utf8"); if (![".json", ".md", ".txt", ".html"].includes(extension)) extensionIssues.push(relative); maxWords = Math.max(maxWords, (text.match(/\S+/g) || []).length); if (extension === ".json") try { core.parseJSON(text); } catch (error) { jsonIssues.push({ path: relative, error: error.code || error.message }); } if (extension === ".txt") try { new vm.Script(text, { filename: path.basename(file) }); } catch (error) { syntaxIssues.push({ path: relative, error: error.message }); } for (const [kind, regex] of patterns) { regex.lastIndex = 0; if (regex.test(text)) privacy.push({ kind, path: relative }); } }

function stageReview(ownerRoot) {
  const x1Root = path.join(ownerRoot, BASE, "x1"), review = path.join(x1Root, "staged-review.json"); if (fs.existsSync(review)) throw new Error("x1 staged review exists");
  const staged = git(ownerRoot, ["diff", "--cached", "--name-only", "--diff-filter=ACMR"]).split(/\r?\n/).filter(Boolean), deleted = git(ownerRoot, ["diff", "--cached", "--name-only", "--diff-filter=D"]).split(/\r?\n/).filter(Boolean), manifest = load(path.join(x1Root, "manifest.json")), issues = [];
  if (deleted.length) issues.push({ code: "DELETE", paths: deleted }); if (!staged.length || staged.some((file) => !file.startsWith(BASE + "/x1/"))) issues.push({ code: "SCOPE", paths: staged.filter((file) => !file.startsWith(BASE + "/x1/")) });
  for (const binding of manifest.bindings) { const data = git(ownerRoot, ["cat-file", "blob", ":" + binding.path], null); if (data.length !== binding.bytes || sha(data) !== binding.sha256) issues.push({ code: "BINDING", path: binding.path }); }
  const expected = new Set([...manifest.bindings.map((row) => row.path), BASE + "/x1/manifest.json"]), missing = [...expected].filter((file) => !staged.includes(file)), extra = staged.filter((file) => !expected.has(file)); if (missing.length) issues.push({ code: "MISSING", paths: missing }); if (extra.length) issues.push({ code: "EXTRA", paths: extra });
  put(x1Root, "staged-review.json", { schema: "ghc.family.staged-review.v9", owner: OWNER, phase: PHASE, stage: "x1", candidate_paths: staged.length, manifest_bindings: manifest.bindings.length, manifest_exclusions: manifest.exclusions.length, deletions: deleted.length, issues, valid: issues.length === 0, canonical_invocations: 0, successor_contacts: 0, boundary: BOUNDARY });
  process.stdout.write(JSON.stringify({ staged_paths: staged.length, manifest_bindings: manifest.bindings.length, issues: issues.length, valid: issues.length === 0 }) + "\n"); if (issues.length) process.exitCode = 1;
}
function main() {
  const [mode, ownerRoot, skillRoot, bankRoot, builderPath] = process.argv.slice(2);
  if (!["selftest", "execute", "smoke", "validate", "stage-review"].includes(mode)) throw new Error("mode");
  if (!ownerRoot || !skillRoot || !bankRoot || !builderPath) throw new Error("explicit paths required");
  if (mode === "selftest") { const source = coreSource(), core = loadModule(source, "extension-field-core.txt"), proposals = load(path.join(ownerRoot, BASE, "plan", "proposals.json")).proposals.filter((row) => row.session === "x1"), failures = []; for (const proposal of proposals) { const output = core.evaluate(core.clone(proposal.request)); if (core.canon(output) !== core.canon(proposal.expected_output)) failures.push(proposal.proposal_id); for (const subject of proposal.candidate_subjects) { const bad = core.evaluate(core.clone(subject.request)); if (bad.ok !== false || bad.error !== subject.expected_rejection.error) failures.push(subject.negative_id); } } process.stdout.write(JSON.stringify({ status: failures.length ? "FAIL" : "PASS", proposals: proposals.length, candidate_subjects: proposals.length * 2, failures }) + "\n"); if (failures.length) process.exitCode = 1; }
  else if (mode === "execute") execute(ownerRoot, skillRoot, bankRoot, builderPath);
  else if (mode === "smoke") smoke(ownerRoot, skillRoot, bankRoot);
  else if (mode === "validate") validate(ownerRoot);
  else stageReview(ownerRoot);
}
main();
