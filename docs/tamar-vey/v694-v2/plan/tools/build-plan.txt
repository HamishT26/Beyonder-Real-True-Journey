}

function trim(poly, p) {

function scalarInverse(value, p) {
  let a = mod(value, p), t = 0, nextT = 1, r = p, nextR = a;
  if (a === 0) throw new Error("zero inverse");
  while (nextR !== 0) {
    const q = Math.floor(r / nextR);
    [t, nextT] = [nextT, t - q * nextT];
    [r, nextR] = [nextR, r - q * nextR];
  }
  if (r !== 1) throw new Error("nonunit");
  return mod(t, p);
}
function polyAdd(a, b, p) {

function polyMul(a, b, p) {
  const out = Array(a.length + b.length - 1).fill(0);
  for (let i = 0; i < a.length; i++) for (let j = 0; j < b.length; j++) out[i + j] = mod(out[i + j] + a[i] * b[j], p);
  return trim(out, p);
}
function polyDivMod(dividend, divisor, p) {
  const a = trim(dividend, p), b = trim(divisor, p);
  if (b.length === 1 && b[0] === 0) throw new Error("zero divisor");
  const q = Array(Math.max(1, a.length - b.length + 1)).fill(0), r = [...a], inv = scalarInverse(b[b.length - 1], p);
  while (!(r.length === 1 && r[0] === 0) && r.length >= b.length) {
    const shift = r.length - b.length, factor = mod(r[r.length - 1] * inv, p);
    q[shift] = factor;
    for (let i = 0; i < b.length; i++) r[i + shift] = mod(r[i + shift] - factor * b[i], p);
    while (r.length > 1 && r[r.length - 1] === 0) r.pop();
  }
  return { quotient: trim(q, p), remainder: trim(r, p) };
}

function git(root, args, encoding = "utf8") {

function walkFiles(root) {

  return out;
}

function stageReview(ownerRoot) {
  const planRoot = path.join(ownerRoot, BASE, "plan"), reviewPath = path.join(planRoot, "staged-review.json");
  if (fs.existsSync(reviewPath)) throw new Error("staged review already exists");
  const staged = git(ownerRoot, ["diff", "--cached", "--name-only", "--diff-filter=ACMR"]).split(/\r?\n/).filter(Boolean);
  const deleted = git(ownerRoot, ["diff", "--cached", "--name-only", "--diff-filter=D"]).split(/\r?\n/).filter(Boolean);
  const issues = [];
  if (deleted.length) issues.push({ code: "DELETE", paths: deleted });
  if (!staged.length || staged.some((file) => !file.startsWith(BASE + "/plan/"))) issues.push({ code: "SCOPE", paths: staged.filter((file) => !file.startsWith(BASE + "/plan/")) });
  const manifest = JSON.parse(fs.readFileSync(path.join(planRoot, "manifest.json"), "utf8"));
  for (const binding of manifest.bindings) {
    const data = git(ownerRoot, ["cat-file", "blob", ":" + binding.path], null);
    if (data.length !== binding.bytes || sha(data) !== binding.sha256) issues.push({ code: "BINDING", path: binding.path });
  }
  const expected = new Set([...manifest.bindings.map((row) => row.path), BASE + "/plan/manifest.json"]);
  const missing = [...expected].filter((file) => !staged.includes(file));
  const extra = staged.filter((file) => !expected.has(file));
  if (missing.length) issues.push({ code: "MISSING", paths: missing });
  if (extra.length) issues.push({ code: "EXTRA", paths: extra });
  const extensions = new Set(staged.map((file) => path.extname(file)));
  if ([...extensions].some((ext) => ![".json", ".md", ".txt", ".html"].includes(ext))) issues.push({ code: "EXTENSION", extensions: [...extensions] });
  put(planRoot, "staged-review.json", { schema: "ghc.family.staged-review.v9", owner: OWNER, phase: PHASE, stage: "planning", candidate_paths: staged.length, manifest_bindings: manifest.bindings.length, manifest_exclusions: manifest.exclusions.length, deletions: deleted.length, issues, valid: issues.length === 0, planning_only: true, x1_execution_credit: 0, x2_execution_credit: 0, boundary: BOUNDARY });
  process.stdout.write(JSON.stringify({ staged_paths: staged.length, manifest_bindings: manifest.bindings.length, issues: issues.length, valid: issues.length === 0 }) + "\n");
  if (issues.length) process.exitCode = 1;
}
function main() {
  const [mode, ownerRoot, sourceRoot, skillRoot, bankRoot, builderPath] = process.argv.slice(2);
  if (!["selftest", "build", "stage-review"].includes(mode)) throw new Error("mode");
  if (!ownerRoot || !sourceRoot || !skillRoot || !bankRoot || !builderPath) throw new Error("explicit paths required");
  if (mode === "selftest") {
    const definitions = makeDefinitions(sourceRoot);
    const counts = definitions.proposals.reduce((acc, proposal) => { acc[proposal.expected_disposition] = (acc[proposal.expected_disposition] || 0) + 1; return acc; }, {});
    process.stdout.write(JSON.stringify({ status: "PASS", proposals: definitions.proposals.length, inherited: definitions.inherited.length, operations: definitions.groups.length, outcomes: counts, maximum_title_jaccard: definitions.maximumJaccard, fields: FIELDS.map((field) => field.id) }) + "\n");
  } else if (mode === "build") build(ownerRoot, sourceRoot, skillRoot, bankRoot, builderPath);
  else stageReview(ownerRoot);
}
main();
