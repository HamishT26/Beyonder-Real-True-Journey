'use strict';
const fs = require('fs');
const path = require('path');
const core = require('../engine/evidence-core.txt');
const ROOT = path.resolve(__dirname, '..');
const read = rel => JSON.parse(fs.readFileSync(path.join(ROOT, rel), 'utf8'));
const write = (rel, value) => { const p = path.join(ROOT, rel); fs.mkdirSync(path.dirname(p), { recursive: true }); fs.writeFileSync(p, JSON.stringify(value, null, 2) + '\n', 'utf8'); };
const contracts = read('planning/contracts-x1.json').records;
const results = [], negatives = [], refusals = [];
for (const c of contracts) {
  const inputBefore = JSON.stringify(c.request);
  let actual, pass = false, error = null;
  try { actual = core.execute(c.request); pass = core.same(actual, c.expected_output) && inputBefore === JSON.stringify(c.request); } catch (e) { error = e.message; }
  results.push({ contract_id: c.contract_id, operation: c.operation, fixture_id: c.fixture_id, outcome: c.expected_disposition, pass, error, actual });
  for (const m of c.mutations) {
    let observed = null, rejected = false;
    try { core.execute(m.request); } catch (e) { observed = e.message; rejected = observed === m.expected_error; }
    negatives.push({ negative_id: m.negative_id, contract_id: c.contract_id, expected_error: m.expected_error, observed, rejected, original_success_credit: 0, result: 'fail' });
    refusals.push({ witness_id: `${m.negative_id}-REFUSAL`, negative_id: m.negative_id, pass: rejected, subject_success_credit: 0, boundary: 'Passing rejection does not make the invalid subject successful.' });
  }
}
const allPass = results.every(x => x.pass) && negatives.every(x => x.rejected);
write('x1/results.json', { stage: 'x1', count: results.length, pass_count: results.filter(x => x.pass).length, records: results, domain_replayed: false });
write('x1/retained-negatives.json', { stage: 'x1', count: negatives.length, records: negatives, relabelled: 0 });
write('x1/refusal-witnesses.json', { stage: 'x1', count: refusals.length, records: refusals });
for (const [kind, planned] of [['safe', read('planning/portfolios/x1-safe.json')], ['candidate', read('planning/portfolios/x1-candidate.json')], ['cfr', read('planning/portfolios/x1-cfr.json')]]) {
  const receipt = planned.records.map(x => ({ ...x, state: allPass ? 'completed_bounded_check' : 'blocked_by_contract_failure', completion_credit: allPass ? 1 : 0, core_promotion: false }));
  write(`x1/portfolio-${kind}-receipt.json`, { stage: 'x1', class: kind, count: receipt.length, records: receipt, distinct_core_achievements: 0 });
}

  methods: [{ method_id: 'TV7055-M-X1-01', title: 'Exact finite evidence-frame X1 execution', failure_signature: 'complete output mismatch or admitted invalid subject', trigger_preconditions: ['immutable planning commit', 'clean fresh four-way equality'], privacy_class: 'sanitized_public', approval_class: 'authorized_owner_local_synthetic', candidate_workaround: 'retain the failed subject and repair only the narrow operation', validation_witness_ids: results.map(x => `TV7055-W-${x.contract_id}`).concat(refusals.map(x => x.witness_id)), recurrence_guard: 'Compare complete typed envelopes and preserve caller input.', rollback: 'Return to the immutable planning commit; do not erase failures.', recommendation_state: allPass ? 'validated' : 'candidate', supersedes: [], protected_gates: core.GATES, retained_negative_ids: negatives.map(x => x.negative_id), scope_boundary: 'Finite synthetic X1 only.' }],
  witnesses: results.map(x => ({ witness_id: `TV7055-W-${x.contract_id}`, method_id: 'TV7055-M-X1-01', procedure: 'complete frozen-output comparison', scope: x.operation, expected: 'exact envelope equality', observed: x.pass ? 'equal' : x.error || 'mismatch', result: x.pass ? 'pass' : 'fail', same_owner_only: true, independent_reproduction: false, retained_negative_ids: [], boundary: 'Formal finite result only.' })).concat(negatives.map(x => ({ witness_id: `TV7055-W-${x.negative_id}`, method_id: 'TV7055-M-X1-01', procedure: 'retained malformed subject', scope: x.expected_error, expected: 'invalid subject remains zero credit', observed: x.observed, result: 'fail', same_owner_only: true, independent_reproduction: false, retained_negative_ids: [x.negative_id], boundary: 'A passing refusal is separate.' }))).concat(refusals.map(x => ({ witness_id: x.witness_id, method_id: 'TV7055-M-X1-01', procedure: 'rejection guard', scope: x.negative_id, expected: 'reject invalid subject', observed: x.pass ? 'rejected' : 'not rejected', result: x.pass ? 'pass' : 'fail', same_owner_only: true, independent_reproduction: false, retained_negative_ids: [], boundary: x.boundary }))),
  state_events: [{ method_id: 'TV7055-M-X1-01', from: 'candidate', to: allPass ? 'validated' : 'candidate', witness: allPass ? 'all frozen X1 outputs and refusal guards' : 'unresolved failure' }], recommendations: [{ method_id: 'TV7055-M-X1-01', state: allPass ? 'validated' : 'candidate' }],
  counts: { methods: 1, witnesses: results.length + negatives.length + refusals.length, passing: results.filter(x => x.pass).length + refusals.filter(x => x.pass).length, failing: negatives.length + results.filter(x => !x.pass).length },
  boundary: 'Same-owner finite software evidence; NOT_READY_FOR_STAGE_20.'
};
write('x1/method-flow.json', method);
write('x1/summary.json', { stage: 'x1', contracts: results.length, completed: results.filter(x => x.pass).length, malformed_subjects: negatives.length, refusal_witnesses: refusals.filter(x => x.pass).length, safe: 400, candidate: 300, clean_fix_refine: 300, all_pass: allPass, x2_started: false, terminal_verdict: 'NOT_READY_FOR_STAGE_20' });
console.log(JSON.stringify({ ok: allPass, contracts: results.length, malformed: negatives.length, refusals: refusals.filter(x => x.pass).length }));
