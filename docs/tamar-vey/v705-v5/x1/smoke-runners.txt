'use strict';
const fs=require('fs'),path=require('path'),cp=require('child_process'); const ROOT=path.resolve(__dirname,'..');
const contracts=JSON.parse(fs.readFileSync(path.join(ROOT,'planning/contracts-x1.json'),'utf8')).records;
const pairs=[['frame_validate','mass_validate'],['normalize_mass','focal_sets'],['belief_query','plausibility_query'],['commonality_query','doubt_query'],['interval_query','complement_duality']]; const records=[];
for(let i=0;i<pairs.length;i++){const runner=path.join(ROOT,'runners',`ghc_family_tamar_v705_v5_pair_${String(i+1).padStart(2,'0')}_runner.txt`);const c=contracts.find(x=>x.operation===pairs[i][0]);const ok=cp.spawnSync(process.execPath,[runner],{input:JSON.stringify(c.request),encoding:'utf8'});let env={};try{env=JSON.parse(ok.stdout)}catch{}const bad=cp.spawnSync(process.execPath,[runner],{input:JSON.stringify({...c.request,operation:'authority_release'}),encoding:'utf8'});let benv={};try{benv=JSON.parse(bad.stdout)}catch{}records.push({runner:path.basename(runner),accepting:{status:ok.status,pass:ok.status===0&&env.ok===true},adverse:{status:bad.status,code:benv.code,pass:bad.status===2&&benv.code==='E_RUNNER_SCOPE'},invalid_subject_credit:0});}
const receipt={stage:'x1',count:records.length,passed:records.filter(x=>x.accepting.pass&&x.adverse.pass).length,records};fs.writeFileSync(path.join(ROOT,'x1/runner-smokes.json'),JSON.stringify(receipt,null,2)+'\n','utf8');console.log(JSON.stringify(receipt));process.exitCode=receipt.passed===receipt.count?0:1;
