'use strict';
const fs=require('fs'),path=require('path');const ROOT=path.resolve(__dirname,'..');
function validate(x){if(!x||typeof x!=='object'||Array.isArray(x))throw new Error('E_SHAPE');if(!/^[0-9a-f]{40}$/.test(x.head))throw new Error('E_HEAD');if(x.clean!==true)throw new Error('E_DIRTY');if(!Array.isArray(x.divergence)||x.divergence.length!==2||x.divergence.some(n=>n!==0))throw new Error('E_DIVERGENCE');if(x.four_way_equal!==true)throw new Error('E_EQUALITY');if(x.manifests_valid!==true)throw new Error('E_MANIFEST');return{ok:true,scope:'metadata_preflight_only'}}
const valid={head:'0'.repeat(40),clean:true,divergence:[0,0],four_way_equal:true,manifests_valid:true};const mutations=[{...valid,head:'bad'},{...valid,clean:false},{...valid,divergence:[1,0]},{...valid,four_way_equal:false},{...valid,manifests_valid:false}];const records=[{id:'TV7055-CANON-PREFLIGHT-VALID',kind:'valid',pass:validate(valid).ok}];for(let i=0;i<mutations.length;i++){let rejected=false,code=null;try{validate(mutations[i])}catch(e){rejected=true;code=e.message}records.push({id:`TV7055-CANON-PREFLIGHT-N${i+1}`,kind:'invalid_subject',rejected,code,original_success_credit:0})}const receipt={status:records[0].pass&&records.slice(1).every(x=>x.rejected)?'VALID_CANONICAL_PREFLIGHT_FIXTURE':'INVALID_CANONICAL_PREFLIGHT_FIXTURE',valid_fixture_passed:records[0].pass,negative_mutations:5,negative_mutations_rejected:records.slice(1).filter(x=>x.rejected).length,records,domain_replayed:false,canonical_invoked:false,boundary:'Metadata preflight only; NOT_READY_FOR_STAGE_20.'};fs.writeFileSync(path.join(ROOT,'final/canonical-preflight-receipt.json'),JSON.stringify(receipt,null,2)+'\n');console.log(JSON.stringify(receipt));process.exitCode=receipt.status.startsWith('VALID_')?0:1;
