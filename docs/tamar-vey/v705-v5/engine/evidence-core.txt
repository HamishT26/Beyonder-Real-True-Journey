'use strict';

const GATES = [
  'empirical_confirmation', 'real_participants_and_observations', 'production_deployment',
  'professional_or_safety_authority', 'identity_lifecycle_authority',
  'legal_cultural_affected_party_authority', 'Maori_authority_and_data_governance',
  'privacy_and_accessibility_remedy', 'complete_privacy_accessibility_security',
  'independent_reproduction', 'consciousness_personhood', 'AGI_ASI',
  'Theory_of_Everything', 'proof_canon_Stage_20'
];
const FIELDS = ['frame', 'mass', 'operation', 'other_mass', 'query_mask', 'refinement'];
const X1 = new Set(['frame_validate','mass_validate','normalize_mass','focal_sets','belief_query','plausibility_query','commonality_query','doubt_query','interval_query','complement_duality']);

function gcd(a, b) { a = a < 0n ? -a : a; b = b < 0n ? -b : b; while (b) [a, b] = [b, a % b]; return a || 1n; }
function rat(n, d = 1n) { n = BigInt(n); d = BigInt(d); if (d === 0n) throw new Error('E_ZERO_DENOMINATOR'); if (d < 0n) { n = -n; d = -d; } const g = gcd(n, d); return { n: n / g, d: d / g }; }
function parseRat(v) { if (typeof v !== 'string' || !/^-?\d+(?:\/\d+)?$/.test(v)) throw new Error('E_RATIONAL'); const [n, d = '1'] = v.split('/'); return rat(BigInt(n), BigInt(d)); }
function add(a, b) { return rat(a.n * b.d + b.n * a.d, a.d * b.d); }
function sub(a, b) { return rat(a.n * b.d - b.n * a.d, a.d * b.d); }
function rs(a) { return a.d === 1n ? String(a.n) : `${a.n}/${a.d}`; }
function sum(values) { return values.reduce((a, b) => add(a, b), rat(0)); }
function stable(value) { if (Array.isArray(value)) return value.map(stable); if (value && typeof value === 'object') return Object.fromEntries(Object.keys(value).sort().map(k => [k, stable(value[k])])); return value; }
function same(a, b) { return JSON.stringify(stable(a)) === JSON.stringify(stable(b)); }

function validateMass(mass) {
  if (!Array.isArray(mass) || mass.length !== 8) throw new Error('E_MASS_SHAPE');
  const parsed = mass.map(parseRat);
  if (parsed.some(x => x.n < 0n)) throw new Error('E_NEGATIVE_MASS');
  if (rs(sum(parsed)) !== '1') throw new Error('E_MASS_TOTAL');
  if (parsed[0].n !== 0n) throw new Error('E_EMPTY_MASS');
  return parsed;
}
function validateRequest(raw) {
  if (!raw || typeof raw !== 'object' || Array.isArray(raw)) throw new Error('E_REQUEST');
  const keys = Object.keys(raw).sort();
  if (!same(keys, [...FIELDS].sort())) throw new Error('E_FIELDS');
  if (!Array.isArray(raw.frame) || raw.frame.length !== 3 || raw.frame.some(x => typeof x !== 'string' || !x) || new Set(raw.frame).size !== 3) throw new Error('E_FRAME');
  if (!Number.isInteger(raw.query_mask) || raw.query_mask < 1 || raw.query_mask > 7) throw new Error('E_QUERY');
  if (!Array.isArray(raw.refinement) || !same(raw.refinement, [3,4,8])) throw new Error('E_REFINEMENT');
  const mass = validateMass(raw.mass), other = validateMass(raw.other_mass);
  return { operation: raw.operation, frame: [...raw.frame], mass, other, mass_raw: [...raw.mass], other_raw: [...raw.other_mass], query_mask: raw.query_mask, refinement: [...raw.refinement] };
}
function scanJson(raw) {
  if (Buffer.byteLength(raw, 'utf8') > 1024 * 1024) throw new Error('E_BYTES');
  let depth = 0, quoted = false, escape = false, max = 0;
  for (const ch of raw) {
    if (quoted) { if (escape) escape = false; else if (ch === '\\') escape = true; else if (ch === '"') quoted = false; continue; }
    if (ch === '"') quoted = true;
    else if (ch === '{' || ch === '[') { depth += 1; max = Math.max(max, depth); }
    else if (ch === '}' || ch === ']') depth -= 1;
    if (depth < 0 || max > 32) throw new Error('E_DEPTH');
  }
  if (quoted || depth !== 0) throw new Error('E_JSON');
  const keys = [...raw.matchAll(/"((?:[^"\\]|\\.)*)"\s*:/g)].map(m => JSON.parse(`"${m[1]}"`));
  if (new Set(keys).size !== keys.length) throw new Error('E_DUPLICATE_KEY');
  try { return JSON.parse(raw); } catch { throw new Error('E_JSON'); }
}
function belief(m, q) { return sum(m.filter((_, mask) => (mask & ~q) === 0)); }
function plausibility(m, q) { return sum(m.filter((_, mask) => (mask & q) !== 0)); }
function commonality(m, q) { return sum(m.filter((_, mask) => (mask & q) === q)); }

function execute(raw) {
  const req = validateRequest(raw), q = req.query_mask, comp = 7 ^ q, m = req.mass;
  if (!X1.has(req.operation)) throw new Error('E_OPERATION_SCOPE');
  const table = {
    frame_validate: () => ({ frame: req.frame, atom_count: 3, subset_count: 8, unique: true }),
    mass_validate: () => ({ total: rs(sum(m)), empty_mass: rs(m[0]), normalized: true, nonnegative: true }),
    normalize_mass: () => m.map(rs),
    focal_sets: () => m.map((x, mask) => ({ mask, mass: rs(x) })).filter(x => x.mass !== '0'),
    belief_query: () => ({ query_mask: q, belief: rs(belief(m, q)) }),
    plausibility_query: () => ({ query_mask: q, plausibility: rs(plausibility(m, q)) }),
    commonality_query: () => ({ query_mask: q, commonality: rs(commonality(m, q)) }),
    doubt_query: () => ({ query_mask: q, complement_mask: comp, doubt: rs(belief(m, comp)) }),
    interval_query: () => ({ query_mask: q, lower: rs(belief(m, q)), upper: rs(plausibility(m, q)) }),
    complement_duality: () => ({ query_mask: q, belief: rs(belief(m, q)), one_minus_plausibility_complement: rs(sub(rat(1), plausibility(m, comp))) })
  };
  return { ok: true, operation: req.operation, outcome: 'completed', value: table[req.operation](), evidence_class: 'finite_synthetic_evidence_function', authority: false, protected_gates: GATES };
}

module.exports = { GATES, X1, rat, parseRat, add, sub, rs, sum, same, stable, validateMass, validateRequest, scanJson, belief, plausibility, commonality, execute };
