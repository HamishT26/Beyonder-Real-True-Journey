'use strict';
const fs = require('fs');
const path = require('path');
const assert = require('assert');
const core = require('../engine/evidence-core.txt');
const ROOT = path.resolve(__dirname, '..');
const fixtures = JSON.parse(fs.readFileSync(path.join(ROOT, 'planning/fixtures.json'), 'utf8')).records;
const base = fixtures[0];
const req = operation => ({ operation, frame: [...base.frame], mass: [...base.mass], other_mass: [...base.other_mass], query_mask: base.query_mask, refinement: [...base.refinement] });
const tests = [];
function test(id, fn) { try { fn(); tests.push({ id, pass: true }); } catch (e) { tests.push({ id, pass: false, error: e.message }); } }
test('TV7055-X1-T01', () => assert.equal(core.execute(req('frame_validate')).value.subset_count, 8));
test('TV7055-X1-T02', () => assert.equal(core.execute(req('mass_validate')).value.total, '1'));
test('TV7055-X1-T03', () => assert.equal(core.execute(req('normalize_mass')).value.length, 8));
test('TV7055-X1-T04', () => assert(core.execute(req('focal_sets')).value.every(x => x.mass !== '0')));
test('TV7055-X1-T05', () => assert(core.execute(req('belief_query')).value.belief));
test('TV7055-X1-T06', () => assert(core.execute(req('plausibility_query')).value.plausibility));
test('TV7055-X1-T07', () => assert(core.execute(req('commonality_query')).value.commonality));
test('TV7055-X1-T08', () => assert(core.execute(req('doubt_query')).value.doubt));
test('TV7055-X1-T09', () => { const x = core.execute(req('interval_query')).value; assert(core.parseRat(x.lower).n * core.parseRat(x.upper).d <= core.parseRat(x.upper).n * core.parseRat(x.lower).d); });
test('TV7055-X1-T10', () => { const x = core.execute(req('complement_duality')).value; assert.equal(x.belief, x.one_minus_plausibility_complement); });
test('TV7055-X1-T11', () => assert.throws(() => core.execute({ ...req('frame_validate'), extra: true }), /E_FIELDS/));
test('TV7055-X1-T12', () => assert.throws(() => core.execute({ ...req('frame_validate'), mass: ['0','0','0','0','0','0','0','1/2'] }), /E_MASS_TOTAL/));
test('TV7055-X1-T13', () => assert.throws(() => core.execute(req('dempster_rule')), /E_OPERATION_SCOPE/));
test('TV7055-X1-T14', () => assert.throws(() => core.scanJson('{"operation":"x","operation":"y"}'), /E_DUPLICATE_KEY/));
test('TV7055-X1-T15', () => { const before = JSON.stringify(req('belief_query')); const r = req('belief_query'); core.execute(r); assert.equal(JSON.stringify(r), before); });
const receipt = { stage: 'x1', count: tests.length, passed: tests.filter(x => x.pass).length, failed: tests.filter(x => !x.pass).length, records: tests, invocation_count: 1 };
fs.writeFileSync(path.join(ROOT, 'x1/tests-receipt.json'), JSON.stringify(receipt, null, 2) + '\n', 'utf8');
console.log(JSON.stringify(receipt));
process.exitCode = receipt.failed ? 1 : 0;
