function gcd(a, b) { a = a < 0n ? -a : a; b = b < 0n ? -b : b; while (b) [a, b] = [b, a % b]; return a || 1n; }
function rat(n, d = 1n) { n = BigInt(n); d = BigInt(d); if (d === 0n) throw new Error('E_ZERO_DENOMINATOR'); if (d < 0n) { n = -n; d = -d; } const g = gcd(n, d); return { n: n / g, d: d / g }; }
function parseRat(v) { if (typeof v === 'number' && Number.isSafeInteger(v)) return rat(BigInt(v)); if (typeof v !== 'string' || !/^-?\d+(?:\/\d+)?$/.test(v)) throw new Error('E_RATIONAL'); const [n, d = '1'] = v.split('/'); return rat(BigInt(n), BigInt(d)); }
function add(a, b) { return rat(a.n * b.d + b.n * a.d, a.d * b.d); }
function sub(a, b) { return rat(a.n * b.d - b.n * a.d, a.d * b.d); }
function mul(a, b) { return rat(a.n * b.n, a.d * b.d); }
function div(a, b) { if (b.n === 0n) throw new Error('E_DIV_ZERO'); return rat(a.n * b.d, a.d * b.n); }
function rs(a) { return a.d === 1n ? String(a.n) : `${a.n}/${a.d}`; }
function sum(values) { return values.reduce((a, b) => add(a, b), rat(0)); }
function popcount(mask) { let n = 0; for (; mask; mask >>>= 1) n += mask & 1; return n; }
function stable(value) { if (Array.isArray(value)) return value.map(stable); if (value && typeof value === 'object') return Object.fromEntries(Object.keys(value).sort().map(k => [k, stable(value[k])])); return value; }
function sha(value) { return crypto.createHash('sha256').update(typeof value === 'string' || Buffer.isBuffer(value) ? value : JSON.stringify(stable(value))).digest('hex'); }

function validateMass(mass) {
  if (!Array.isArray(mass) || mass.length !== 8) throw new Error('E_MASS_SHAPE');
  const parsed = mass.map(parseRat);
  if (parsed.some(x => x.n < 0n)) throw new Error('E_NEGATIVE_MASS');
  if (rs(sum(parsed)) !== '1') throw new Error('E_MASS_TOTAL');
  if (parsed[0].n !== 0n) throw new Error('E_EMPTY_MASS');
  return parsed;
}
function belief(m, q) { return sum(m.filter((_, mask) => (mask & ~q) === 0)); }
function plausibility(m, q) { return sum(m.filter((_, mask) => (mask & q) !== 0)); }
function commonality(m, q) { return sum(m.filter((_, mask) => (mask & q) === q)); }
function conjunctive(a, b) { const out = Array.from({ length: 8 }, () => rat(0)); for (let i = 0; i < 8; i++) for (let j = 0; j < 8; j++) out[i & j] = add(out[i & j], mul(a[i], b[j])); return out; }
function normalizeConjunctive(c) { const keep = sub(rat(1), c[0]); if (keep.n === 0n) throw new Error('E_TOTAL_CONFLICT'); return c.map((x, i) => i === 0 ? '0' : rs(div(x, keep))); }
function vacuous(m, refinement) { const out = Array.from({ length: 16 }, () => rat(0)); for (let mask = 0; mask < 8; mask++) { let fine = 0; for (let bit = 0; bit < 3; bit++) if (mask & (1 << bit)) fine |= refinement[bit]; out[fine] = add(out[fine], m[mask]); } return out.map(rs); }
function marginalize(fine, refinement) { const out = Array.from({ length: 8 }, () => rat(0)); for (let mask = 0; mask < 16; mask++) { let coarse = 0; for (let bit = 0; bit < 3; bit++) if (mask & refinement[bit]) coarse |= 1 << bit; out[coarse] = add(out[coarse], fine[mask]); } return out.map(rs); }
function pignistic(m) { return [0, 1, 2].map(bit => rs(sum(m.map((x, mask) => mask && (mask & (1 << bit)) ? div(x, rat(popcount(mask))) : rat(0))))); }
