const OPS=["capacity-universe-shape","capacity-boundary","capacity-monotonicity","capacity-dual","mobius-transform","mobius-inversion","capacity-support","choquet-order","choquet-integral","comonotone-additivity"];
function gcd(a,b){a=a<0n?-a:a;b=b<0n?-b:b;while(b)[a,b]=[b,a%b];return a||1n;}
function F(n,d=1n){n=BigInt(n);d=BigInt(d);if(d===0n)throw Error('zero denominator');if(d<0n){n=-n;d=-d;}const g=gcd(n,d);return{n:n/g,d:d/g};}
function P(v){if(v&&typeof v==='object'&&'n'in v)return F(v.n,v.d);if(typeof v==='number'||typeof v==='bigint')return F(v);const[n,d='1']=String(v).split('/');return F(n,d);}
function A(a,b){a=P(a);b=P(b);return F(a.n*b.d+b.n*a.d,a.d*b.d);}function S(a,b){a=P(a);b=P(b);return F(a.n*b.d-b.n*a.d,a.d*b.d);}function M(a,b){a=P(a);b=P(b);return F(a.n*b.n,a.d*b.d);}function C(a,b){const x=S(a,b);return x.n<0n?-1:x.n>0n?1:0;}function R(v){v=P(v);return v.n+'/'+v.d;}function SUM(v){return v.reduce((a,b)=>A(a,b),F(0));}
function pc(m){let c=0;while(m){c+=m&1;m>>>=1;}return c;}function ss(w,m){let s=0;for(let i=0;i<w.length;i++)if(m&(1<<i))s+=w[i];return s;}function norm(w){const t=w.reduce((a,b)=>a+b,0);return w.map(x=>R(F(x,t)));}
function profile(o){const n=3+((o-1)%2),w=Array.from({length:n},(_,i)=>1+((o*3+i*2)%7)),t=w.reduce((a,b)=>a+b,0);const cap=Array.from({length:1<<n},(_,m)=>{const s=ss(w,m);return R(F(s*s,t*t));});const u=Array.from({length:n},(_,i)=>1+((o+i*3)%9)),co=u.map(x=>2*x+1);return{profile_id:'CAP-'+String(o).padStart(2,'0'),ordinal:o,n,atoms:Array.from({length:n},(_,i)=>'a'+(i+1)),weights:w,capacity:cap,utility:u,comonotone:co};}
function mob(cap,n){const out=[];for(let m=0;m<(1<<n);m++){let v=F(0);for(let s=m;;s=(s-1)&m){v=A(v,M(F((pc(m)-pc(s))%2===0?1:-1),P(cap[s])));if(s===0)break;}out.push(R(v));}return out;}function zeta(co,n){const out=[];for(let m=0;m<(1<<n);m++){let v=F(0);for(let s=m;;s=(s-1)&m){v=A(v,P(co[s]));if(s===0)break;}out.push(R(v));}return out;}function dual(cap,n){const f=(1<<n)-1;return Array.from({length:1<<n},(_,m)=>R(S(F(1),P(cap[f^m]))));}
function order(v){const a=v.map((x,i)=>({x,i})).sort((p,q)=>p.x-q.x||p.i-q.i);return{order:a.map(x=>x.i),levels:a.map(x=>x.x),threshold_masks:a.map((_,k)=>a.slice(k).reduce((m,x)=>m|(1<<x.i),0))};}function choquet(v,cap){const o=order(v);let prev=0,r=F(0);for(let k=0;k<v.length;k++){r=A(r,M(F(o.levels[k]-prev),P(cap[o.threshold_masks[k]])));prev=o.levels[k];}return R(r);}
function output(op,p){const full=(1<<p.n)-1;if(op==='capacity-universe-shape')return{n:p.n,atom_count:p.atoms.length,subset_count:p.capacity.length,full_mask:full};if(op==='capacity-boundary')return{empty:p.capacity[0],full:p.capacity[full],normalized:p.capacity[0]==='0/1'&&p.capacity[full]==='1/1'};if(op==='capacity-monotonicity'){let comparable_pairs=0,violations=0;for(let a=0;a<=full;a++)for(let b=0;b<=full;b++)if((a&b)===a){comparable_pairs++;if(C(p.capacity[a],p.capacity[b])>0)violations++;}return{comparable_pairs,violations,monotone:violations===0};}if(op==='capacity-dual'){const d=dual(p.capacity,p.n);return{dual:d,involution:JSON.stringify(dual(d,p.n))===JSON.stringify(p.capacity)};}if(op==='mobius-transform'){const c=mob(p.capacity,p.n);return{coefficients:c,coefficient_sum:R(SUM(c))};}if(op==='mobius-inversion'){const c=mob(p.capacity,p.n),r=zeta(c,p.n);return{recovered:r,exact:JSON.stringify(r)===JSON.stringify(p.capacity)};}if(op==='capacity-support'){const c=mob(p.capacity,p.n),m=c.map((value,mask)=>({value,mask})).filter(x=>x.value!=='0/1').map(x=>x.mask);return{nonzero_count:m.length,masks:m,maximum_order:Math.max(...m.map(pc))};}if(op==='choquet-order')return order(p.utility);if(op==='choquet-integral'){const value=choquet(p.utility,p.capacity);return{value,nonnegative:C(value,'0/1')>=0};}if(op==='comonotone-additivity'){const v=p.utility.map((x,i)=>x+p.comonotone[i]),left=choquet(v,p.capacity),right=R(A(choquet(p.utility,p.capacity),choquet(p.comonotone,p.capacity)));return{left,right,equal:left===right};}throw Error('unsupported op');}
function mutate(op,p){p=JSON.parse(JSON.stringify(p));if(op==='capacity-universe-shape')p.atoms[1]=p.atoms[0];else if(op==='capacity-boundary')p.capacity[0]='1/1';else if(op==='capacity-monotonicity')p.capacity[1]='2/1';else if(op==='capacity-dual')p.capacity.pop();else if(op==='mobius-transform')p.capacity[1]='bad';else if(op==='mobius-inversion')p.capacity[2]='bad';else if(op==='capacity-support')p.capacity[1]='bad';else if(op==='choquet-order')p.utility[0]=NaN;else if(op==='choquet-integral')p.utility[0]=-1;else if(op==='comonotone-additivity')p.comonotone.reverse();return p;}
function guard(op,p){try{if(!OPS.includes(op)||!Number.isInteger(p.n)||p.n<3||p.n>4||p.atoms.length!==p.n||new Set(p.atoms).size!==p.n||p.capacity.length!==(1<<p.n)||p.capacity.some(x=>!/^-?\d+\/[1-9]\d*$/.test(x))||p.utility.length!==p.n||p.comonotone.length!==p.n||p.utility.some(x=>!Number.isFinite(x)||x<0)||p.comonotone.some(x=>!Number.isFinite(x)||x<0))return false;const full=(1<<p.n)-1;if(p.capacity[0]!=='0/1'||p.capacity[full]!=='1/1')return false;for(let a=0;a<=full;a++)for(let b=0;b<=full;b++)if((a&b)===a&&C(p.capacity[a],p.capacity[b])>0)return false;if(op==='comonotone-additivity'){const oa=order(p.utility).order.join(','),ob=order(p.comonotone).order.join(',');if(oa!==ob)return false;}return true;}catch{return false;}}
function evaluate(op,ordinal,invalid=false){let p=profile(ordinal);if(invalid)p=mutate(op,p);const accepted=guard(op,p);let out=null,error=null;if(accepted){try{out=output(op,p);}catch(e){error=e.message;}}const valid=invalid?!accepted:accepted&&out!==null&&!error;return{operation_id:op,profile_id:'CAP-'+String(ordinal).padStart(2,'0'),invalid,accepted,valid,output:out,error,synthetic_only:true,exact_arithmetic:true};}
if(require.main===module){const r=evaluate(process.argv[2]||OPS[0],Number(process.argv[3]||1),process.argv[4]==='invalid');console.log(JSON.stringify(r));process.exit(r.valid?0:1);}module.exports={OPS,profile,evaluate};
