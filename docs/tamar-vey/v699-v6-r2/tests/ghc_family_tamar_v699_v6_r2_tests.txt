const fs=require('fs'),path=require('path');
const root=path.join(__dirname,'..');const x1=require(path.join(root,'engine','ghc_family_finite_sheaf_kernel.txt'));const x2=require(path.join(root,'engine','ghc_family_finite_sheaf_x2_kernel.txt'));
const read=p=>JSON.parse(fs.readFileSync(path.join(root,...p.split('/')),'utf8'));const tests=[];const test=(name,fn)=>{try{tests.push({name,passed:!!fn()});}catch(e){tests.push({name,passed:false,error:e.message});}};
test('x1 operation count',()=>x1.OPS.length===10);
test('x2 operation count',()=>x2.OPS.length===10);
test('all x1 accepting fixtures',()=>x1.OPS.every(op=>Array.from({length:30},(_,i)=>x1.evaluate(op,i+1,false)).every(r=>r.valid&&r.accepted)));
test('all x1 invalid fixtures rejected',()=>x1.OPS.every(op=>Array.from({length:30},(_,i)=>x1.evaluate(op,i+1,true)).every(r=>r.valid&&!r.accepted)));
test('all x2 accepting fixtures',()=>x2.OPS.every(op=>Array.from({length:30},(_,i)=>x2.evaluate(op,i+1,false)).every(r=>r.valid&&r.accepted)));
test('all x2 invalid fixtures rejected',()=>x2.OPS.every(op=>Array.from({length:30},(_,i)=>x2.evaluate(op,i+1,true)).every(r=>r.valid&&!r.accepted)));
test('phase outcome arithmetic',()=>{const o=read('x2/phase-outcome-ledger.json');return o.completed===510&&o.represented===30&&o.open_gap===30&&o.exact_gate===30;});
test('authorized outcome labels only',()=>{const a=new Set(['completed','represented','open_gap','exact_gate']);return read('x2/phase-outcome-ledger.json').records.every(r=>a.has(r.outcome));});
test('five simulation documents',()=>fs.readdirSync(path.join(root,'simulations')).filter(x=>x.endsWith('.html')).length===5);
test('simulation canvas semantics',()=>fs.readdirSync(path.join(root,'simulations')).filter(x=>x.endsWith('.html')).every(f=>fs.readFileSync(path.join(root,'simulations',f),'utf8').includes('role="img"')));
test('simulation table fallback',()=>fs.readdirSync(path.join(root,'simulations')).filter(x=>x.endsWith('.html')).every(f=>fs.readFileSync(path.join(root,'simulations',f),'utf8').includes('Accessible data table')));
test('simulation has no external dependency',()=>fs.readdirSync(path.join(root,'simulations')).filter(x=>x.endsWith('.html')).every(f=>!/<script[^>]+src=|https?:\/\//i.test(fs.readFileSync(path.join(root,'simulations',f),'utf8').replace('This is a wholly synthetic',''))));
test('Journey instruction authority zero',()=>read('planning/journey-source-ledger.json').records.every(r=>r.instruction_authority===false));
test('route projection exact bounds',()=>{const r=read('planning/workflow-v15-candidate.json').assignments;return r.length===210&&r[0].owner==='Veylora Quen'&&r.at(-1).owner==='Eiren Kestrel';});
test('held packet counts',()=>{const h=read('planning/held-packets.json');return h.exact.length===50&&h.blocked.length===30;});
const passed=tests.filter(x=>x.passed).length;const receipt={test_count:tests.length,passed,failed:tests.length-passed,tests,valid:passed===tests.length};console.log(JSON.stringify(receipt));process.exit(receipt.valid?0:1);
