const crypto = require('crypto');
const OPS = ["base-graph","stalk-inventory","restriction-shape","section-shape","local-consistency","coboundary-matrix","sheaf-laplacian","component-partition","source-variant-quarantine","evidence-status-projection"];
function hash(v){return crypto.createHash('sha256').update(JSON.stringify(v)).digest('hex');}
function edges(n, pattern){const e=[];if(pattern==='disconnected'){for(let i=0;i+1<n-1;i++)e.push([i,i+1]);return e;}for(let i=0;i<n-1;i++)e.push([i,i+1]);if(pattern==='cycle'&&n>2)e.push([0,n-1]);if(pattern==='fork'&&n>3)e.push([0,2]);return e.map(x=>x[0]<x[1]?x:[x[1],x[0]]).sort((a,b)=>a[0]-b[0]||a[1]-b[1]);}
function profile(ordinal){const n=3+(ordinal-1)%4;const pattern=['path','cycle','fork','disconnected'][(ordinal-1)%4];const d=1+(ordinal-1)%2;const scale=['1','-1','2/3'][(ordinal-1)%3];const es=edges(n,pattern);return{profile_id:'SHEAF-P'+String(ordinal).padStart(2,'0'),ordinal,vertices:Array.from({length:n},(_,i)=>'v'+i),edges:es.map((x,i)=>({id:'e'+i,endpoints:x})),pattern,dimension:d,scale,vertex_stalks:Array(n).fill(d),edge_stalks:Array(es.length).fill(d),restriction_rows:Array(es.length).fill(d),restriction_cols:Array(es.length).fill(d),section_vectors:Array.from({length:n},()=>Array(d).fill('1')),component_membership:null,source_records:[{id:'A',claim:'bounded-source-a',status:'historical_reference_only'},{id:'B',claim:'bounded-source-b',status:'historical_reference_only'}],projected_claims:[{id:'A',source_status:'historical_reference_only',instruction_authority:false,claim_ceiling:'structural_only'}]};}
function components(p){const adj=new Map(p.vertices.map((v,i)=>[i,[]]));for(const e of p.edges){const[a,b]=e.endpoints;adj.get(a).push(b);adj.get(b).push(a);}const seen=new Set(),out=[];for(let i=0;i<p.vertices.length;i++){if(seen.has(i))continue;const q=[i],c=[];seen.add(i);while(q.length){const x=q.shift();c.push(x);for(const y of adj.get(x)){if(!seen.has(y)){seen.add(y);q.push(y);}}}out.push(c.sort((a,b)=>a-b));}return out;}
function validate(op,p){switch(op){case'base-graph':return new Set(p.vertices).size===p.vertices.length&&p.edges.every(e=>e.endpoints.length===2&&e.endpoints[0]<e.endpoints[1]&&e.endpoints[1]<p.vertices.length);case'stalk-inventory':return p.vertex_stalks.length===p.vertices.length&&p.edge_stalks.length===p.edges.length&&[...p.vertex_stalks,...p.edge_stalks].every(Number.isInteger)&&[...p.vertex_stalks,...p.edge_stalks].every(x=>x>0);case'restriction-shape':return p.restriction_rows.length===p.edges.length&&p.restriction_cols.length===p.edges.length&&p.edges.every((_,i)=>p.restriction_rows[i]===p.edge_stalks[i]&&p.restriction_cols[i]===p.dimension);case'section-shape':return p.section_vectors.length===p.vertices.length&&p.section_vectors.every(v=>v.length===p.dimension);case'local-consistency':return p.edges.every(e=>JSON.stringify(p.section_vectors[e.endpoints[0]])===JSON.stringify(p.section_vectors[e.endpoints[1]]));case'coboundary-matrix':return p.coboundary_rows===undefined?true:p.coboundary_rows===p.edges.length*p.dimension;case'sheaf-laplacian':return p.laplacian_symmetric!==false&&p.laplacian_psd_witness!==false;case'component-partition':{if(!p.edges.every(e=>e.endpoints.length===2&&e.endpoints.every(x=>Number.isInteger(x)&&x>=0&&x<p.vertices.length)))return false;const cs=components(p);const flat=cs.flat();return new Set(flat).size===p.vertices.length&&flat.length===p.vertices.length;}case'source-variant-quarantine':return !p.source_records.some(x=>x.instruction_authority===true)&&p.source_records.every(x=>x.status==='historical_reference_only');case'evidence-status-projection':return p.projected_claims.every(x=>x.source_status&&x.instruction_authority===false&&x.claim_ceiling==='structural_only');default:return false;}}
function mutate(op,p){p=JSON.parse(JSON.stringify(p));switch(op){case'base-graph':p.vertices[1]=p.vertices[0];break;case'stalk-inventory':p.vertex_stalks[0]=0;break;case'restriction-shape':p.restriction_cols[0]=p.dimension+1;break;case'section-shape':p.section_vectors.pop();break;case'local-consistency':p.section_vectors[p.edges[0].endpoints[1]][0]='9';break;case'coboundary-matrix':p.coboundary_rows=p.edges.length*p.dimension+1;break;case'sheaf-laplacian':p.laplacian_symmetric=false;break;case'component-partition':p.edges.push({id:'bad',endpoints:[0,p.vertices.length]});p.component_membership=[0,0];break;case'source-variant-quarantine':p.source_records[0].instruction_authority=true;break;case'evidence-status-projection':p.projected_claims[0].instruction_authority=true;break;}return p;}
function evaluate(op,ordinal,invalid=false){if(!OPS.includes(op))throw new Error('unsupported x1 operation');let p=profile(ordinal);p.coboundary_rows=p.edges.length*p.dimension;p.laplacian_symmetric=true;p.laplacian_psd_witness=true;if(invalid)p=mutate(op,p);const accepted=validate(op,p);const witness={operation_id:op,profile_id:p.profile_id,invalid,accepted,expected_acceptance:!invalid,fixture_sha256:hash(p),component_count:components(profile(ordinal)).length,exact_arithmetic:true,synthetic_only:true};witness.valid=accepted===!invalid;return witness;}
if(require.main===module){const op=process.argv[2]||OPS[0];const ordinal=Number(process.argv[3]||1);const invalid=process.argv[4]==='invalid';const r=evaluate(op,ordinal,invalid);console.log(JSON.stringify(r));process.exit(r.valid?0:1);}
module.exports={OPS,profile,evaluate};
