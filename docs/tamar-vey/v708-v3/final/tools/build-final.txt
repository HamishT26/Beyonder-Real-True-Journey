function readJson(relative) {
  return JSON.parse(fs.readFileSync(path.join(ROOT, ...relative.split("/")), "utf8"));
}
function putJson(relative, value) {
  const target = path.join(ROOT, ...relative.split("/"));
  fs.mkdirSync(path.dirname(target), { recursive: true });
  fs.writeFileSync(target, JSON.stringify(value, null, 2) + "\n", "utf8");
}
function putText(relative, value) {
  const target = path.join(ROOT, ...relative.split("/"));
  fs.mkdirSync(path.dirname(target), { recursive: true });
  fs.writeFileSync(target, value.trim() + "\n", "utf8");
}
function shaFile(relative) {
  return crypto.createHash("sha256").update(fs.readFileSync(path.join(ROOT, ...relative.split("/")))).digest("hex");
}
function words(value) {
  return (value.match(/\S+/g) || []).length;
}
