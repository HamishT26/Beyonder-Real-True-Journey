function sha(data) { return crypto.createHash("sha256").update(data).digest("hex"); }
function stable(value) {
  if (Array.isArray(value)) return value.map(stable);
  if (value && typeof value === "object") return Object.fromEntries(Object.keys(value).sort().map((key) => [key, stable(value[key])]));
  return value;
}
function canon(value) { return JSON.stringify(stable(value)); }
function clone(value) { return JSON.parse(canon(value)); }
function put(root, relative, value) {
  const target = path.join(root, relative);
  fs.mkdirSync(path.dirname(target), { recursive: true });
  const text = typeof value === "string" ? value.replace(/\s+$/, "") + "\n" : JSON.stringify(value, null, 2) + "\n";
  fs.writeFileSync(target, text, { encoding: "utf8", flag: "wx" });
}
function load(file) { return JSON.parse(fs.readFileSync(file, "utf8")); }
function walk(root) {
  const rows = [];
  for (const entry of fs.readdirSync(root, { withFileTypes: true })) {
    const full = path.join(root, entry.name);
    if (entry.isDirectory()) rows.push(...walk(full)); else rows.push(full);
  }
  return rows;
}
function git(root, args, encoding = "utf8") {
  const result = cp.spawnSync("git", ["-C", root, ...args], { encoding, maxBuffer: 128 * 1024 * 1024 });
  if (result.status !== 0) throw new Error(Buffer.from(result.stderr || "").toString("utf8"));
  return encoding === null ? Buffer.from(result.stdout || "") : String(result.stdout || "").trim();
}
function loadTextModule(source, filename) {
  const loaded = new Module(filename, module.parent);
  loaded.filename = filename;
  loaded.paths = Module._nodeModulePaths(path.dirname(filename));
  loaded._compile(source, filename);
  return loaded.exports;
}

function oracleFixture(value) {
  if (!value || !Array.isArray(value.ground_set) || new Set(value.ground_set).size !== value.ground_set.length || value.ground_set.length === 0) throw new Error("oracle ground");
  const ground = [...value.ground_set].sort();
  const adjacency = Object.fromEntries(ground.map((x) => [x, []]));
  for (const [a, b] of value.covers) { if (!adjacency[a] || !adjacency[b] || a === b) throw new Error("oracle cover"); adjacency[a].push(b); }
  function reachable(a, b, seen = new Set()) { if (a === b) return true; if (seen.has(a)) return false; seen.add(a); return adjacency[a].some((next) => reachable(next, b, new Set(seen))); }
  for (const a of ground) for (const b of ground) if (a !== b && reachable(a, b) && reachable(b, a)) throw new Error("oracle cycle");
  return { ground, reachable };
}

  else if (operation === "anti_exchange") { request.fixture.ground_set = []; error = "E_GROUND"; }
  else if (operation === "closed_sets") { const x = request.fixture.ground_set[0]; request.fixture.covers.push([x, x]); error = "E_COVER_SELF"; }
  else if (operation === "union_closure") { delete request.fixture.covers; error = "E_FIXTURE_FIELDS"; }

  else if (operation === "minimal_generators") { request.closed_set = ["z"]; error = "E_CLOSED_SET"; }
  return { request, expected_error: error };

function test(ownerRoot) {
  const x1Root = path.join(ownerRoot, BASE, "x1");
  if (fs.existsSync(path.join(x1Root, "validation", "tests.json"))) throw new Error("tests already run");
  const core = require(path.join(x1Root, "convex-geometry-core.txt")), fixtures = load(path.join(ownerRoot, BASE, "planning", "fixtures.json")).fixtures, contracts = load(path.join(x1Root, "results", "contracts.json")).rows, refusals = load(path.join(x1Root, "results", "refusals.json")).rows;
  const checks = [];
  const check = (name, predicate, detail = null) => checks.push({ name, pass: !!predicate, detail });
  for (const fixture of fixtures) {
    const spec = core.fixture(fixture), contractRows = contracts.filter((row) => row.fixture_id === fixture.fixture_id), refusalRows = refusals.filter((row) => row.fixture_id === fixture.fixture_id);
    check(`${fixture.fixture_id} ten exact contracts`, contractRows.length === 10 && contractRows.every((row) => core.canon(core.evaluate(clone(row.request))) === core.canon(row.expected)));
    check(`${fixture.fixture_id} ten adverse refusals`, refusalRows.length === 10 && refusalRows.every((row) => { const out = core.evaluate(clone(row.subject)); return !out.ok && out.error === row.expected_error; }));
    check(`${fixture.fixture_id} structural duality`, core.antiExchange(spec).valid && core.unionClosure(spec).valid && core.accessibility(spec).valid && core.closedSets(spec).length === core.feasibleSets(spec).length);
  }
  const valid = checks.every((row) => row.pass);
  put(x1Root, "validation/tests.json", { schema: "tamar.x1-tests.v708-v3.v1", tests: checks.length, passed: checks.filter((row) => row.pass).length, failed: checks.filter((row) => !row.pass).length, valid, rows: checks, replayed: false, canonical_invocation: false, boundary: BOUNDARY });
  process.stdout.write(JSON.stringify({ status: valid ? "PASS" : "FAIL", tests: checks.length, passed: checks.filter((row) => row.pass).length, failed: checks.filter((row) => !row.pass).length }) + "\n");
  if (!valid) process.exitCode = 1;
}

function main() {
  const [mode, ownerRoot, builderPath, quickValidate] = process.argv.slice(2);
  if (mode === "execute") execute(ownerRoot, builderPath, quickValidate);
  else if (mode === "test") test(ownerRoot);
  else throw new Error("mode");
}
main();
