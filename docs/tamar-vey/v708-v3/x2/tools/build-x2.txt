function smoke(root,quickValidate){const x2=path.join(root,BASE,"x2");if(fs.existsSync(path.join(x2,"evidence","local-capabilities.json")))throw new Error("smoke already run");const plan=load(path.join(x2,"evidence","smoke-plan.json")),skills=[];for(const name of plan.skills){const folder=path.join(x2,"skills",name),result=cp.spawnSync("python",[quickValidate,folder],{encoding:"utf8"});if(result.status!==0)throw new Error(`skill ${name}: ${result.stdout} ${result.stderr}`);const text=fs.readFileSync(path.join(folder,"SKILL.md"),"utf8");if(!text.includes("## Procedure")||!text.includes("## Boundary"))throw new Error(`skill read ${name}`);skills.push({skill:name,validation:"pass",complete_read_by_main_agent:true,skill_md_bytes:Buffer.byteLength(text),smoke_use:"pass_via_paired_runner"});}const runners=[];for(const row of plan.runners){const runner=path.join(x2,"runners",row.runner),dir=path.join(x2,"smoke",row.runner.replace(/\.txt$/,'')),accepted=cp.spawnSync(process.execPath,[runner,path.join(dir,"valid.json")],{encoding:"utf8"}),rejected=cp.spawnSync(process.execPath,[runner,path.join(dir,"invalid.json")],{encoding:"utf8"}),a=JSON.parse(accepted.stdout),b=JSON.parse(rejected.stdout);if(accepted.status!==0||!a.ok||rejected.status!==2||b.ok!==false)throw new Error(`runner ${row.runner}`);runners.push({runner:row.runner,operations:row.operations,accepting_smoke:"pass",rejecting_smoke:"pass",rejected_subject_credit:0});}put(x2,"evidence/local-capabilities.json",{schema:"tamar.x2-local-capabilities.v708-v3.v1",skill_count:skills.length,runner_count:runners.length,skills,runners,globally_installed:0,ordinary_packages_installed:0,incremental_spend_usd:0,boundary:BOUNDARY});process.stdout.write(JSON.stringify({status:"PASS",skills:skills.length,runners:runners.length})+"\n");}

function test(root){const x2=path.join(root,BASE,"x2");if(fs.existsSync(path.join(x2,"validation","tests.json")))throw new Error("tests already run");const corePath=path.join(x2,"convex-geometry-x2-core.txt"),core=loadTextModule(fs.readFileSync(corePath,"utf8"),corePath),fixtures=load(path.join(root,BASE,"planning","fixtures.json")).fixtures,contracts=load(path.join(x2,"results","contracts.json")).rows,refusals=load(path.join(x2,"results","refusals.json")).rows,checks=[];const check=(name,pass,detail=null)=>checks.push({name,pass:!!pass,detail});for(const fixture of fixtures){const contractRows=contracts.filter(r=>r.fixture_id===fixture.fixture_id),refusalRows=refusals.filter(r=>r.fixture_id===fixture.fixture_id);check(`${fixture.fixture_id} ten exact contracts`,contractRows.length===10&&contractRows.every(r=>equal(core.evaluate(clone(r.request)),r.expected)));check(`${fixture.fixture_id} ten adverse refusals`,refusalRows.length===10&&refusalRows.every(r=>{const out=core.evaluate(clone(r.subject));return !out.ok&&out.error===r.expected_error;}));const s=core.spec(fixture),words=core.basicWords(s);check(`${fixture.fixture_id} structural x2 cross-check`,words.length>0&&words.every(w=>core.shelling(w,s).valid)&&core.rootedCircuits(s).count>=core.implicationBasis(s).count);}const advisory=load(path.join(x2,"consultation","x1-advisory-tests.json"));for(const row of advisory.rows)check(`advisory ${row.label}`,row.pass&&row.mutant_detected&&row.mutant_subject_success_credit===0);const valid=checks.every(r=>r.pass);put(x2,"validation/tests.json",{schema:"tamar.x2-tests.v708-v3.v1",tests:checks.length,passed:checks.filter(r=>r.pass).length,failed:checks.filter(r=>!r.pass).length,valid,rows:checks,replayed:false,canonical_invocation:false,boundary:BOUNDARY});process.stdout.write(JSON.stringify({status:valid?"PASS":"FAIL",tests:checks.length,passed:checks.filter(r=>r.pass).length,failed:checks.filter(r=>!r.pass).length})+"\n");if(!valid)process.exitCode=1;}

function restoreCore(root){const target=path.join(root,BASE,"x2","convex-geometry-x2-core.txt");if(fs.existsSync(target))throw new Error("core already exists");fs.writeFileSync(target,coreSource(),{encoding:"utf8",flag:"wx"});process.stdout.write(JSON.stringify({status:"PASS",restored:"convex-geometry-x2-core.txt",full_execute_replayed:false})+"\n");}
