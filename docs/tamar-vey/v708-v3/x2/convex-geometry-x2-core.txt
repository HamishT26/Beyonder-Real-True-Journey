const crypto=require("node:crypto");
const base=require("../x1/convex-geometry-core.txt");
const OPS=["rooted_circuits","implication_basis","basic_words","shelling_check","feasible_rank","feasible_prefix_trace","deletion_minor","restriction_trace","correction_lineage","coordinate_projection"];
const GATES=["empirical_confirmation","real_participants","production_deployment","professional_authority","legal_cultural_affected_party_authority","Maori_authority","complete_privacy_accessibility_security","independent_reproduction","consciousness_personhood","AGI_ASI","Theory_of_Everything","Stage_20"];
const FIELDS={rooted_circuits:["fixture","operation"],implication_basis:["fixture","operation"],basic_words:["fixture","operation"],shelling_check:["fixture","operation","word"],feasible_rank:["fixture","operation","subset"],feasible_prefix_trace:["fixture","operation","word"],deletion_minor:["element","fixture","operation"],restriction_trace:["fixture","operation","subset"],correction_lineage:["operation","predecessor_sha256","promotion_allowed","query","version0","version0_id","version1","version1_id"],coordinate_projection:["fixture","operation"]};
class ContractError extends Error{constructor(code){super(code);this.code=code;}}
function need(v,c){if(!v)throw new ContractError(c);}
function stable(v){if(Array.isArray(v))return v.map(stable);if(v&&typeof v==="object")return Object.fromEntries(Object.keys(v).sort().map(k=>[k,stable(v[k])]));return v;}
const canon=v=>JSON.stringify(stable(v));const sha=v=>crypto.createHash("sha256").update(canon(v)).digest("hex");const sorted=v=>[...new Set(v)].sort();const equal=(a,b)=>canon(sorted(a))===canon(sorted(b));
function subsets(ground){const out=[];for(let mask=0;mask<(1<<ground.length);mask++)out.push(ground.filter((_,i)=>mask&(1<<i)));return out;}
function exactFields(r){need(r&&typeof r==="object"&&!Array.isArray(r),"E_RECORD");need(typeof r.operation==="string"&&OPS.includes(r.operation),"E_OPERATION");need(canon(Object.keys(r).sort())===canon(FIELDS[r.operation].slice().sort()),"E_FIELDS");}
function spec(value){try{return base.fixture(value);}catch(error){throw new ContractError(error.code||"E_FIXTURE");}}
function sub(value,s,code="E_SUBSET"){try{return base.subset(value,s,code);}catch(error){throw new ContractError(error.code||code);}}
function feasibleSets(s){return base.feasibleSets(s);}
function isFeasible(items,s){return feasibleSets(s).some(row=>equal(row,items));}
function strictPairs(s){const rows=[];for(const root of s.ground)for(const upper of s.ground)if(root!==upper&&s.reach[root][upper])rows.push({stem:[upper],root,support:sorted([root,upper])});return rows.sort((a,b)=>canon(a).localeCompare(canon(b)));}
function rootedCircuits(s){const records=strictPairs(s);return{records,count:records.length,complete_catalogue:true};}
function implicationBasis(s){const records=s.covers.map(([root,upper])=>({premise:[upper],conclusion:root})).sort((a,b)=>canon(a).localeCompare(canon(b)));return{records,count:records.length,complete_rooted_circuit_catalogue:records.length===strictPairs(s).length};}
function basicWords(s){const feasible=feasibleSets(s),out=[];function visit(prefix,remaining){if(!remaining.length){out.push(prefix);return;}for(const x of remaining){const next=[...prefix,x];if(feasible.some(row=>equal(row,next)))visit(next,remaining.filter(y=>y!==x));}}visit([],s.ground);return out.sort((a,b)=>canon(a).localeCompare(canon(b)));}
function word(value,s){need(Array.isArray(value)&&value.length===s.ground.length,"E_WORD");need(equal(value,s.ground),"E_WORD");return value.slice();}
function shelling(value,s){const w=word(value,s),prefixes=[];let valid=true;let first_failure=null;for(let i=0;i<=w.length;i++){const prefix=w.slice(0,i),ok=isFeasible(prefix,s);prefixes.push({index:i,prefix,feasible:ok});if(!ok&&first_failure===null){valid=false;first_failure=i;}}return{word:w,valid,first_failure,prefixes};}
function feasibleRank(value,s){const q=sub(value,s),contained=feasibleSets(s).filter(row=>row.every(x=>q.includes(x))),max=Math.max(...contained.map(row=>row.length));return{subset:q,rank:max,maximizers:contained.filter(row=>row.length===max).sort((a,b)=>canon(a).localeCompare(canon(b))),feasible:isFeasible(q,s)};}
function prefixTrace(value,s){const w=word(value,s);return{word:w,rows:w.map((_,i)=>{const prefix=w.slice(0,i+1),rank=feasibleRank(prefix,s);return{index:i+1,prefix,rank:rank.rank,feasible:rank.feasible};}),valid:shelling(w,s).valid};}
function closureTable(ground,close){return subsets(ground).map(input=>({input,closure:close(input)}));}
function deletion(element,s){need(typeof element==="string"&&s.ground.includes(element),"E_ELEMENT");const ground=s.ground.filter(x=>x!==element);const table=closureTable(ground,input=>base.closureOf(input,s).filter(x=>x!==element));return{deleted:element,ground,table,entry_count:table.length};}
function restriction(value,s){const ground=sub(value,s),table=closureTable(ground,input=>base.closureOf(input,s).filter(x=>ground.includes(x)));const closed=table.filter(row=>equal(row.input,row.closure)).map(row=>row.input);return{ground,table,closed_sets:closed,entry_count:table.length};}
function correction(r){need(r.version0_id==="V0"&&r.version1_id==="V1","E_VERSION");const v0=spec(r.version0),v1=spec(r.version1);need(equal(v0.ground,v1.ground),"E_GROUND");need(r.predecessor_sha256===sha(r.version0),"E_PREDECESSOR");need(r.promotion_allowed===false,"E_AUTHORITY");const query=sub(r.query,v0);return{version0_id:"V0",version1_id:"V1",predecessor_sha256:r.predecessor_sha256,version0_sha256:sha(r.version0),version1_sha256:sha(r.version1),query,version0_closure:base.closureOf(query,v0),version1_closure:base.closureOf(query,v1),original_preserved:true,promotion_allowed:false,protected_effects:0};}
function coordinate(s){const closed=base.closedSets(s).length,circuits=strictPairs(s).length,words=basicWords(s).length;return{coordinate_labels:["ground_size","closed_set_count","rooted_circuit_count"],coordinates:[s.ground.length,closed,circuits],basic_word_count:words,alt_text:"dimensionless finite structure with "+s.ground.length+" elements, "+closed+" closed sets, and "+circuits+" rooted circuits",represented_only:true,physical_measurement:false,accessibility_complete:false,authority:false};}
function evaluate(r){try{exactFields(r);let value;switch(r.operation){case"rooted_circuits":value=rootedCircuits(spec(r.fixture));break;case"implication_basis":value=implicationBasis(spec(r.fixture));break;case"basic_words":{const s=spec(r.fixture),rows=basicWords(s);value={words:rows,count:rows.length};break;}case"shelling_check":value=shelling(r.word,spec(r.fixture));break;case"feasible_rank":value=feasibleRank(r.subset,spec(r.fixture));break;case"feasible_prefix_trace":value=prefixTrace(r.word,spec(r.fixture));break;case"deletion_minor":value=deletion(r.element,spec(r.fixture));break;case"restriction_trace":value=restriction(r.subset,spec(r.fixture));break;case"correction_lineage":value=correction(r);break;case"coordinate_projection":value=coordinate(spec(r.fixture));break;default:throw new ContractError("E_OPERATION");}return{ok:true,operation:r.operation,outcome:r.operation==="coordinate_projection"?"represented":"completed",evidence_class:"finite_synthetic_poset_convex_geometry",value,authority:false,protected_gates:GATES.slice()};}catch(error){if(!(error instanceof ContractError))throw error;return{ok:false,error:error.code,authority:false,subject_success_credit:0};}}
function cli(allowed){let out;try{const file=process.argv[2];need(typeof file==="string","E_INPUT_FILE");const request=JSON.parse(require("node:fs").readFileSync(file,"utf8"));need(!allowed||allowed.includes(request.operation),"E_SCOPE");out=evaluate(request);}catch(error){out=error instanceof ContractError?{ok:false,error:error.code,authority:false,subject_success_credit:0}:{ok:false,error:"E_INPUT_IO",authority:false,subject_success_credit:0};}process.stdout.write(canon(out)+"\n");process.exitCode=out.ok?0:2;}
