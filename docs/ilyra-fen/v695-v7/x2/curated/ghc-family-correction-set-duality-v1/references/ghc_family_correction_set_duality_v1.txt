"use strict";

const fs = require("fs");
const Module = require("module");
const path = require("path");

function loadTextModule(file) {
  const instance = new Module(file, module);
  instance.filename = file;
  instance.paths = Module._nodeModulePaths(path.dirname(file));
  instance._compile(fs.readFileSync(file, "utf8"), file);
  return instance.exports;
}
const core = loadTextModule(path.resolve(__dirname, "inconsistency-extended.txt"));
const ALLOWED = ["free_formula_set", "consistency_partition", "hitting_set_duality", "deletion_distance"];
function run(request) {
  if (!request || !ALLOWED.includes(request.operation)) return { ok: false, error: "E_RUNNER_OPERATION", original_success_credit: 0 };
  return core.safeExecute(request);
}
module.exports = { ALLOWED, run, runner: "ghc_family_correction_set_duality_v1.txt" };
if (require.main === module) {
  const input = JSON.parse(fs.readFileSync(0, "utf8"));
  process.stdout.write(JSON.stringify(run(input)) + "\n");
}
