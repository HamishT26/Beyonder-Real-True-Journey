"use strict";
const OPS=["dual_order_projection","dependency_depth","critical_path_witness","minimal_cutsets","correction_lineage","evidence_frontier","authority_vacancy","accessible_projection","external_evidence_gap","release_gate"];
function fail(code){throw new Error(code);}
function combinations(values,size){const out=[];function visit(start,current){if(current.length===size){out.push([...current]);return;}for(let i=start;i<values.length;i++)visit(i+1,[...current,values[i]]);}visit(0,[]);return out;}
function depths(g){const values=Array(g.nodes.length).fill(0);for(const node of g.order)if(g.inc[node].length)values[node]=Math.max(...g.inc[node].map(parent=>values[parent]+1));return values;}
function critical(g){const best=Array.from({length:g.nodes.length},(_,i)=>[i]);for(const node of g.order)for(const child of g.out[node]){const candidate=[...best[node],child],prior=best[child];if(candidate.length>prior.length||(candidate.length===prior.length&&JSON.stringify(candidate)<JSON.stringify(prior)))best[child]=candidate;}return best.reduce((chosen,row)=>row.length>chosen.length||(row.length===chosen.length&&JSON.stringify(row)<JSON.stringify(chosen))?row:chosen,best[0]);}
function cutsets(g){const roots=Array.from({length:g.nodes.length},(_,i)=>i).filter(i=>!g.inc[i].length),leaves=Array.from({length:g.nodes.length},(_,i)=>i).filter(i=>!g.out[i].length),pairs=[];for(const left of roots)for(const right of leaves)if(left!==right&&base.shortest(g,left,right)!==null)pairs.push([left,right]);const internal=Array.from({length:g.nodes.length},(_,i)=>i).filter(i=>!roots.includes(i)&&!leaves.includes(i)),found=[];for(let size=1;size<=internal.length;size++)for(const candidate of combinations(internal,size)){const removed=new Set(candidate);if(pairs.length&&pairs.every(pair=>base.shortest(g,pair[0],pair[1],removed)===null)&&!found.some(prior=>prior.every(item=>removed.has(item))))found.push(candidate);}return {reachable_root_leaf_pairs:pairs,minimal_internal_cutsets:found};}
function stringList(value,allowed=null){if(!Array.isArray(value)||value.some(item=>typeof item!=="string")||new Set(value).size!==value.length)fail("string_list");if(allowed&&value.some(item=>!allowed.includes(item)))fail("unknown_label");return value;}
function evaluate(input){base.keys(input,["op","graph","args"]);if(!OPS.includes(input.op))fail("operation");const g=base.normalize(input.graph),n=g.nodes.length,a=input.args;let value,disposition="completed";
 if(input.op==="dual_order_projection"){base.keys(a,[]);value={nodes:[...g.nodes].reverse(),edges:g.edges.map(edge=>[n-1-edge[1],n-1-edge[0]]).sort((x,y)=>x[0]-y[0]||x[1]-y[1]),authority_credit:false};}
 else if(input.op==="dependency_depth"){base.keys(a,[]);value=depths(g);}
 else if(input.op==="critical_path_witness"){base.keys(a,[]);value={path:critical(g),bounded_witness_only:true};}
 else if(input.op==="minimal_cutsets"){base.keys(a,[]);value=cutsets(g);}
 else if(input.op==="correction_lineage"){base.keys(a,["events"]);if(!Array.isArray(a.events)||a.events.length<1||a.events.length>16)fail("events");const ids=[],states=[];for(let i=0;i<a.events.length;i++){const event=a.events[i];base.keys(event,["event_id","before","after"]);if(typeof event.event_id!=="string"||typeof event.before!=="string"||typeof event.after!=="string"||ids.includes(event.event_id)||(i&&a.events[i-1].after!==event.before))fail("correction_chain");ids.push(event.event_id);if(i===0)states.push(event.before);states.push(event.after);}value={event_ids:ids,states,current:states[states.length-1],erased:false};}
 else if(input.op==="evidence_frontier"){base.keys(a,["available"]);const available=stringList(a.available,g.nodes),ready=[],missing=[];for(let i=0;i<n;i++){const required=Array.from({length:n},(_,j)=>j).filter(j=>j!==i&&g.reach[j][i]).map(j=>g.nodes[j]);(required.every(item=>available.includes(item))?ready:missing).push(g.nodes[i]);}value={available,satisfiable:ready,missing_frontier:missing,external_evidence_credit:false};}
 else if(input.op==="authority_vacancy"){base.keys(a,["required","authorizations"]);const required=stringList(a.required),present=stringList(a.authorizations);value={required,present,vacant:required.filter(item=>!present.includes(item)),denial_inferred:false,consent_inferred:false};disposition="represented";}
 else if(input.op==="accessible_projection"){base.keys(a,[]);value={rows:g.nodes.map((label,i)=>({obligation:label,depends_on:g.inc[i].map(j=>g.nodes[j])})),manual_accessibility_reviewed:false};disposition="represented";}
 else if(input.op==="external_evidence_gap"){base.keys(a,["required","available"]);const required=stringList(a.required),available=stringList(a.available);value={missing:required.filter(item=>!available.includes(item)),synthetic_fixture:g.id,absence_treated_as_false_data:false};disposition="open_gap";}
 else {base.keys(a,["requested","evidence","authorizations"]);const requested=stringList(a.requested);stringList(a.evidence);stringList(a.authorizations);value={closed:requested,release:false,reason:"exact_evidence_and_authority_absent",synthetic_input:true};disposition="exact_gate";}
 return base.envelope(value,disposition);
}
function dispatch(input){try{return evaluate(input);}catch(error){return base.rejected(error.message);}}
module.exports={OPS,depths,critical,cutsets,dispatch};
