"""Manual owner-local predicates. No installed host hooks and no actions."""
import json,hashlib,sys
def digest(d):return hashlib.sha256((json.dumps(d,sort_keys=True,ensure_ascii=True,indent=2)+'\n').encode()).hexdigest()
def check(req):
 mode=req['mode'];v=req['value']
 if mode=='definition-fixity':ok=digest({k:x for k,x in v.items() if k!='definition_sha256'})==v['definition_sha256']
 elif mode=='result-fixity':ok=digest(v['result'])==v['sha256']
 elif mode=='outcome-ceiling':ok=v['disposition'] in ('completed','represented','open_gap','exact_gate') and v['completion_credit']==int(v['disposition']=='completed')
 elif mode=='resource-bound':ok=type(v['n']) is int and 0<=v['n']<=6 and type(v['materialized']) is int and 0<=v['materialized']<2000
 elif mode=='route-hold':ok=v=={'state':'PREPARED_NOT_SENT','target':'Neris Solane','phase':'v706-v6','send':False}
 else:raise ValueError('Unknown manual hook')
 return dict(mode=mode,passed=ok,manual_only=True,installed=False,host_event_observed=False,actions=0)
if __name__=='__main__':
 data=sys.stdin.buffer.read(65537)
 if len(data)>65536:raise ValueError('Bound exceeded')
 r=check(json.loads(data));print(json.dumps(r,sort_keys=True));sys.exit(0 if r['passed'] else 2)
