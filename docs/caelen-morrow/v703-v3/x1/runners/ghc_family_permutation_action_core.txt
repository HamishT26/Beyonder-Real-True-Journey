"use strict";
const assert = require("assert/strict");

const X1 = new Set([
  "permutation_validate",
  "permutation_compose",
  "permutation_inverse",
  "generated_group_closure",
  "element_order_profile",
  "cycle_decomposition",
  "orbit_partition",
  "point_stabilizer",
  "orbit_stabilizer_certificate",
  "fixed_point_profile"
]);
const X2 = new Set([
  "burnside_orbit_count",
  "cycle_index_signature",
  "conjugacy_class_partition",
  "subgroup_orbit_refinement",
  "setwise_stabilizer",
  "equivariant_map_certificate",
  "action_law_certificate",
  "trinity_nonconversion_bridge",
  "external_observation_gap",
  "authority_reservation"
]);
const CASE_ORDER = ["trivial-3", "c2-fixed-3", "c3-regular", "s3-natural", "c4-regular", "v4-regular", "d4-square", "s4-natural", "c5-regular", "d5-pentagon", "a4-natural", "c6-regular", "s3-diagonal-6", "c2-three-swaps", "s3-fixedpoint-4"];
const identity = n => Array.from({ length: n }, (_, i) => i);
const key = p => p.join(",");
const compare = (a, b) => key(a).localeCompare(key(b), "en");
const validPermutation = (p, n) => Array.isArray(p) && p.length === n && p.every(Number.isInteger) && p.every(x => x >= 0 && x < n) && new Set(p).size === n;
const compose = (left, right) => right.map(x => left[x]);
const inverse = p => {
  const out = Array(p.length);
  p.forEach((x, i) => { out[x] = i; });
  return out;
};
const gcd = (a, b) => b === 0 ? Math.abs(a) : gcd(b, a % b);
const lcm = (a, b) => Math.abs(a * b) / gcd(a, b);
const reduced = (numerator, denominator) => {
  const d = gcd(numerator, denominator);
  const sign = denominator < 0 ? -1 : 1;
  return `${sign * numerator / d}/${Math.abs(denominator) / d}`;
};
const rankPermutation = p => {
  const remaining = identity(p.length);
  let rank = 0;
  for (let i = 0; i < p.length; i++) {
    const at = remaining.indexOf(p[i]);
    rank = rank * (p.length - i) + at;
    remaining.splice(at, 1);
  }
  return rank;
};
const closure = (generators, n) => {
  const byRank = new Map();
  const add = p => byRank.set(rankPermutation(p), p);
  add(identity(n));
  generators.forEach(add);
  generators.map(inverse).forEach(add);
  let changed = true;
  while (changed) {
    changed = false;
    const snapshot = [...byRank.values()];
    for (const left of snapshot) for (const right of snapshot) {
      const product = compose(left, right);
      const rank = rankPermutation(product);
      if (!byRank.has(rank)) {
        byRank.set(rank, product);
        changed = true;
      }
    }
  }
  return [...byRank.values()].sort(compare);
};
const cycles = p => {
  const seen = Array(p.length).fill(false);
  const rows = [];
  for (let start = 0; start < p.length; start++) {
    if (seen[start]) continue;
    const row = [];
    let current = start;
    do {
      seen[current] = true;
      row.push(current);
      current = p[current];
    } while (!seen[current]);
    rows.push(row);
  }
  return rows.sort((a, b) => a[0] - b[0]);
};
const orderOf = p => cycles(p).reduce((out, row) => lcm(out, row.length), 1);
const orbitOf = (group, point) => [...new Set(group.map(g => g[point]))].sort((a, b) => a - b);
const partition = (group, n) => {
  const representative = Array(n).fill(-1);
  const rows = [];
  for (let point = 0; point < n; point++) {
    if (representative[point] >= 0) continue;
    const orbit = orbitOf(group, point);
    const index = rows.length;
    orbit.forEach(x => { representative[x] = index; });
    rows.push(orbit);
  }
  return rows;
};
const typeKey = p => {
  const counts = new Map();
  cycles(p).forEach(row => counts.set(row.length, (counts.get(row.length) || 0) + 1));
  return [...counts.entries()].sort((a, b) => a[0] - b[0]).map(([length, count]) => `${length}^${count}`).join(" ");
};
const conjugacy = group => {
  const byKey = new Map(group.map(g => [key(g), g]));
  const unseen = new Set(byKey.keys());
  const rows = [];
  while (unseen.size) {
    const seedKey = [...unseen].sort()[0];
    const seed = byKey.get(seedKey);
    const members = [...new Set(group.map(h => key(compose(compose(h, seed), inverse(h)))))].sort();
    members.forEach(member => unseen.delete(member));
    rows.push(members.map(member => byKey.get(member)));
  }
  return rows.sort((a, b) => key(a[0]).localeCompare(key(b[0]), "en"));
};
const refusal = reason => ({ ok: false, outcome: "open_gap", error: "ADMISSION_REFUSED", reason, subject_success_credit: 0, authority: false, external_calls: 0 });

function admit(request, allowedOperations) {
  if (!request || typeof request !== "object" || Array.isArray(request)) return refusal("request_shape");
  if (Object.keys(request).sort().join(",") !== "input,operation") return refusal("request_fields");
  if (!allowedOperations.has(request.operation)) return refusal("unknown_operation");
  const input = request.input;
  if (!input || typeof input !== "object" || Array.isArray(input)) return refusal("input_shape");
  const required = ["case_id", "degree", "equivariant_map", "generators", "point", "subset"];
  const allowed = new Set([...required, "fixture_tag"]);
  if (Object.keys(input).some(field => !allowed.has(field)) || required.some(field => !(field in input))) return refusal("input_fields");
  const n = input.degree;
  if (!Number.isInteger(n) || n < 3 || n > 6) return refusal("degree");
  if (!Array.isArray(input.generators) || input.generators.length < 1 || input.generators.length > 2) return refusal("generator_count");
  if (input.generators.some(p => !Array.isArray(p) || p.length !== n)) return refusal("permutation_length");
  if (input.generators.some(p => !validPermutation(p, n))) return refusal("permutation_bijection");
  if (!Number.isInteger(input.point) || input.point < 0 || input.point >= n) return refusal("point");
  if (!Array.isArray(input.subset) || input.subset.some(x => !Number.isInteger(x) || x < 0 || x >= n) || new Set(input.subset).size !== input.subset.length) return refusal("subset");
  if (!validPermutation(input.equivariant_map, n)) return refusal("equivariant_map");
  return { ok: true };
}

function evaluateAdmitted(request) {
  const { operation, input } = request;
  const n = input.degree;
  const group = closure(input.generators, n);
  const subgroup = closure([input.generators[0]], n);
  let value;
  let outcome = "completed";
  if (operation === "permutation_validate") {
    value = { degree: n, generator_count: input.generators.length, valid: input.generators.every(p => validPermutation(p, n)) };
  } else if (operation === "permutation_compose") {
    const left = input.generators[0], right = input.generators[1] || identity(n);
    value = { left, right, composition: compose(left, right) };
  } else if (operation === "permutation_inverse") {
    const selected = input.generators[0], inv = inverse(selected);
    value = { selected, inverse: inv, left_identity: compose(selected, inv), right_identity: compose(inv, selected) };
  } else if (operation === "generated_group_closure") {
    value = { order: group.length, elements: group };
  } else if (operation === "element_order_profile") {
    value = { rows: group.map(element => ({ element, order: orderOf(element) })) };
  } else if (operation === "cycle_decomposition") {
    const selected = input.generators[0];
    value = { selected, cycles: cycles(selected), cycle_type: typeKey(selected) };
  } else if (operation === "orbit_partition") {
    value = { orbits: partition(group, n) };
  } else if (operation === "point_stabilizer") {
    const stabilizer = group.filter(g => g[input.point] === input.point);
    value = { point: input.point, orbit: orbitOf(group, input.point), stabilizer_order: stabilizer.length, stabilizer_elements: stabilizer };
  } else if (operation === "orbit_stabilizer_certificate") {
    const orbit = orbitOf(group, input.point), stabilizer = group.filter(g => g[input.point] === input.point);
    value = { point: input.point, group_order: group.length, orbit_size: orbit.length, stabilizer_order: stabilizer.length, product: orbit.length * stabilizer.length, holds: group.length === orbit.length * stabilizer.length };
  } else if (operation === "fixed_point_profile") {
    value = { rows: group.map(element => ({ element, fixed_points: identity(n).filter(x => element[x] === x) })) };
  } else if (operation === "burnside_orbit_count") {
    const fixedCounts = group.map(element => identity(n).filter(x => element[x] === x).length);
    const sum = fixedCounts.reduce((a, b) => a + b, 0), orbitCount = partition(group, n).length;
    value = { group_order: group.length, fixed_counts: fixedCounts, fixed_sum: sum, average: reduced(sum, group.length), orbit_count: orbitCount, holds: sum / group.length === orbitCount };
  } else if (operation === "cycle_index_signature") {
    const counts = new Map();
    group.forEach(element => counts.set(typeKey(element), (counts.get(typeKey(element)) || 0) + 1));
    value = { group_order: group.length, terms: [...counts.entries()].sort().map(([cycle_type, count]) => ({ cycle_type, count, coefficient: reduced(count, group.length) })) };
  } else if (operation === "conjugacy_class_partition") {
    const classes = conjugacy(group);
    value = { class_count: classes.length, classes, covered_elements: classes.reduce((sum, row) => sum + row.length, 0) };
  } else if (operation === "subgroup_orbit_refinement") {
    const whole = partition(group, n), restricted = partition(subgroup, n);
    value = { group_order: group.length, subgroup_order: subgroup.length, whole_orbits: whole, subgroup_orbits: restricted, refines: restricted.every(row => whole.some(parent => row.every(x => parent.includes(x)))) };
  } else if (operation === "setwise_stabilizer") {
    const selected = [...input.subset].sort((a, b) => a - b);
    const stabilizer = group.filter(g => key(selected.map(x => g[x]).sort((a, b) => a - b)) === key(selected));
    value = { subset: selected, stabilizer_order: stabilizer.length, stabilizer_elements: stabilizer };
  } else if (operation === "equivariant_map_certificate") {
    const failures = [];
    for (const g of group) for (const x of identity(n)) if (input.equivariant_map[g[x]] !== g[input.equivariant_map[x]]) failures.push({ element: g, point: x });
    value = { map: input.equivariant_map, checked_pairs: group.length * n, equivariant: failures.length === 0, failures };
  } else if (operation === "action_law_certificate") {
    let checked = 0;
    const failures = [];
    for (const g of group) for (const h of group) for (const x of identity(n)) {
      checked++;
      if (compose(g, h)[x] !== g[h[x]]) failures.push({ g, h, x });
    }
    value = { checked_triples: checked, identity_holds: identity(n).every(x => identity(n)[x] === x), composition_holds: failures.length === 0, failures };
  } else if (operation === "trinity_nonconversion_bridge") {
    outcome = "represented";
    value = { finite_structure: "permutation_action", representation: "typed finite symmetry and handover analogy", physical_measurement_rows: 0, identity_events: 0, participant_rows: 0, conversion_to_empirical_claim: false };
  } else if (operation === "external_observation_gap") {
    outcome = "open_gap";
    value = { missing: ["governed_observations", "calibration", "independent_comparison"], observation_rows: 0, network_calls: 0, closed: false };
  } else if (operation === "authority_reservation") {
    outcome = "exact_gate";
    const at = CASE_ORDER.indexOf(input.case_id);
    value = { action: ["real_identity_decision", "professional_imposition_decision", "legal_or_cultural_publication"][((at < 0 ? 0 : at) % 3)], executed: false, required: ["exact_governed_target", "competent_affected_authority", "consent_and_safety", "recovery_and_review"] };
  } else {
    throw new Error(`Unexpected admitted operation ${operation}`);
  }
  return { ok: true, operation, outcome, value, authority: false, external_calls: 0 };
}

function evaluate(request, session = "all") {
  const allowed = session === "x1" ? X1 : session === "x2" ? X2 : new Set([...X1, ...X2]);
  const admission = admit(request, allowed);
  return admission.ok ? evaluateAdmitted(request) : admission;
}

function parseAndEvaluate(raw, session = "all") {
  if (typeof raw !== "string" || raw.length > 12000) return refusal("wire_length");
  let request;
  try { request = JSON.parse(raw); } catch { return refusal("wire_syntax"); }
  return evaluate(request, session);
}

module.exports = { X1, X2, CASE_ORDER, evaluate, parseAndEvaluate, refusal, rankPermutation };

if (process.argv[1] === __filename) {
  const session = process.argv[2] || "all";
  let raw = "";
  process.stdin.setEncoding("utf8");
  process.stdin.on("data", chunk => { raw += chunk; });
  process.stdin.on("end", () => process.stdout.write(JSON.stringify(parseAndEvaluate(raw, session))));
}
