"use strict";
const core=require('./ghc_family_permutation_action_extension.txt');
const allowed=["conjugacy_class_partition","subgroup_orbit_refinement"];
function run(payload){if(!payload||Object.keys(payload).sort().join(',')!=='left,right')throw Error('PAIR_PAYLOAD');if(payload.left.operation!==allowed[0]||payload.right.operation!==allowed[1])throw Error('PAIR_OPERATION');return {left:core.evaluate(payload.left),right:core.evaluate(payload.right)};}
module.exports={allowed,run};
