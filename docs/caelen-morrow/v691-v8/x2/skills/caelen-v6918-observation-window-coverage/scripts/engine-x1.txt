function closed(value,keys,code='INVALID_REQUEST'){if(!object(value)||Object.keys(value).length!==keys.length||keys.some(key=>!Object.hasOwn(value,key)))fail(code);}

function list(value,code,min=0,max=16){if(!Array.isArray(value)||value.length<min||value.length>max)fail(code);}
function gcd(a,b){a=a<0n?-a:a;b=b<0n?-b:b;while(b)[a,b]=[b,a%b];return a;}
function fraction(n,d=1n){n=BigInt(n);d=BigInt(d);if(d===0n)return null;if(d<0n){n=-n;d=-d;}const g=gcd(n,d);return String(n/g)+'/'+String(d/g);}

const sum=values=>values.reduce((a,b)=>a+b,0);
const label=value=>typeof value==='string'&&/^[A-Za-z][A-Za-z0-9_-]{0,31}$/.test(value);

function run(operation,input){try{if(!Object.hasOwn(operations,operation))fail('UNKNOWN_OPERATION');return {ok:true,value:operations[operation](input),error:null,disposition:'completed'};}catch(error){return {ok:false,value:null,error:error.message,disposition:'open_gap'};}}
function cli(allowed){if(process.argv.length!==3){process.stdout.write(JSON.stringify({ok:false,value:null,error:'INVALID_ARGUMENTS',disposition:'open_gap'})+'\n');return 2;}let request;try{request=JSON.parse(fs.readFileSync(process.argv[2],'utf8'));closed(request,['operation','input']);}catch(error){process.stdout.write(JSON.stringify({ok:false,value:null,error:'INVALID_REQUEST',disposition:'open_gap'})+'\n');return 2;}if(!allowed.includes(request.operation)){process.stdout.write(JSON.stringify({ok:false,value:null,error:'UNKNOWN_OPERATION',disposition:'open_gap'})+'\n');return 2;}const result=run(request.operation,request.input);process.stdout.write(JSON.stringify(result)+'\n');return result.ok?0:2;}
