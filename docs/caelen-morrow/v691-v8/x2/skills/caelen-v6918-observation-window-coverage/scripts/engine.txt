'use strict';
const fs=require('fs');
const Base=require('./engine-x1.txt');
const {closed,integer,list,fraction}=Base.helpers;
const fail=code=>{throw new Error(code);};
const label=value=>typeof value==='string'&&/^[A-Za-z][A-Za-z0-9_-]{0,39}$/.test(value);
const sum=values=>values.reduce((a,b)=>a+b,0);
const operations={
 risk_difference(input){closed(input,['left_events','left_total','right_events','right_total'],'INVALID_RISK');if(!integer(input.left_total,1,10000)||!integer(input.right_total,1,10000)||!integer(input.left_events,0,input.left_total)||!integer(input.right_events,0,input.right_total))fail('INVALID_RISK');return {left_risk:fraction(input.left_events,input.left_total),right_risk:fraction(input.right_events,input.right_total),difference:fraction(input.right_events*input.left_total-input.left_events*input.right_total,input.right_total*input.left_total),causal_claim:false};},
 matched_paired_effect(input){closed(input,['pairs'],'INVALID_PAIRS');list(input.pairs,'INVALID_PAIRS',1,16);if(!input.pairs.every(pair=>Array.isArray(pair)&&pair.length===2&&pair.every(value=>integer(value,-100,100))))fail('INVALID_PAIRS');const differences=input.pairs.map(([before,after])=>after-before);return {differences,mean_difference:fraction(sum(differences),differences.length),positive:differences.filter(v=>v>0).length,negative:differences.filter(v=>v<0).length,ties:differences.filter(v=>v===0).length,causal_claim:false};},
 missingness_pattern(input){closed(input,['fields','records'],'INVALID_MISSINGNESS');list(input.fields,'INVALID_MISSINGNESS',1,8);list(input.records,'INVALID_MISSINGNESS',0,16);if(!input.fields.every(label)||new Set(input.fields).size!==input.fields.length||!input.records.every(row=>row!==null&&typeof row==='object'&&!Array.isArray(row)&&Object.keys(row).every(key=>input.fields.includes(key))))fail('INVALID_MISSINGNESS');const missing=Object.fromEntries(input.fields.map(field=>[field,input.records.filter(row=>!Object.hasOwn(row,field)||row[field]===null).length]));return {rows:input.records.length,missing,complete_rows:input.records.filter(row=>input.fields.every(field=>Object.hasOwn(row,field)&&row[field]!==null)).length,imputation_performed:false};},
 threshold_sensitivity(input){closed(input,['scores','thresholds'],'INVALID_THRESHOLDS');list(input.scores,'INVALID_THRESHOLDS',0,32);list(input.thresholds,'INVALID_THRESHOLDS',1,16);if(!input.scores.every(value=>integer(value,-10000,10000))||!input.thresholds.every(value=>integer(value,-10000,10000))||new Set(input.thresholds).size!==input.thresholds.length||input.thresholds.some((value,index)=>index&&value<input.thresholds[index-1]))fail('INVALID_THRESHOLDS');const accepted_counts=input.thresholds.map(threshold=>input.scores.filter(score=>score>=threshold).length);return {thresholds:input.thresholds,accepted_counts,monotone_nonincreasing:accepted_counts.every((value,index)=>index===0||value<=accepted_counts[index-1]),decision_authority:false};},
 affine_calibration(input){closed(input,['values','scale','offset'],'INVALID_CALIBRATION');list(input.values,'INVALID_CALIBRATION',1,16);if(!input.values.every(value=>integer(value,-10000,10000))||!Array.isArray(input.scale)||input.scale.length!==2||!Array.isArray(input.offset)||input.offset.length!==2||!input.scale.every(value=>integer(value,-10000,10000))||!input.offset.every(value=>integer(value,-10000,10000))||input.scale[1]===0||input.offset[1]===0)fail('INVALID_CALIBRATION');const [sn,sd]=input.scale,[on,od]=input.offset;return {corrected:input.values.map(value=>fraction(BigInt(value)*BigInt(sn)*BigInt(od)+BigInt(on)*BigInt(sd),BigInt(sd)*BigInt(od))),calibration_observed:false,units_verified:false};},
 expanded_uncertainty(input){closed(input,['standard','coverage'],'INVALID_UNCERTAINTY');if(!Array.isArray(input.standard)||input.standard.length!==2||!Array.isArray(input.coverage)||input.coverage.length!==2||!input.standard.every(value=>integer(value,-10000,10000))||!input.coverage.every(value=>integer(value,-10000,10000))||input.standard[0]<0||input.standard[1]<=0||input.coverage[0]<0||input.coverage[1]<=0)fail('INVALID_UNCERTAINTY');const [un,ud]=input.standard,[kn,kd]=input.coverage;return {standard:fraction(un,ud),coverage_factor:fraction(kn,kd),expanded:fraction(BigInt(un)*BigInt(kn),BigInt(ud)*BigInt(kd)),coverage_probability_claim:false};},
 observation_window_coverage(input){closed(input,['intervals','window'],'INVALID_WINDOW');list(input.intervals,'INVALID_WINDOW',0,32);if(!Array.isArray(input.window)||input.window.length!==2||!input.window.every(value=>integer(value,-10000,10000))||input.window[1]<=input.window[0]||!input.intervals.every(interval=>Array.isArray(interval)&&interval.length===2&&interval.every(value=>integer(value,-10000,10000))&&interval[1]>=interval[0]))fail('INVALID_WINDOW');const [start,end]=input.window,clipped=input.intervals.map(([a,b])=>[Math.max(start,a),Math.min(end,b)]).filter(([a,b])=>b>a).sort((a,b)=>a[0]-b[0]||a[1]-b[1]),merged=[];for(const interval of clipped){if(!merged.length||interval[0]>merged.at(-1)[1])merged.push([...interval]);else merged.at(-1)[1]=Math.max(merged.at(-1)[1],interval[1]);}const gaps=[];let cursor=start;for(const [a,b]of merged){if(a>cursor)gaps.push([cursor,a]);cursor=Math.max(cursor,b);}if(cursor<end)gaps.push([cursor,end]);const covered=merged.reduce((n,[a,b])=>n+b-a,0);return {window:input.window,covered,uncovered:end-start-covered,merged,gaps,measurement_quality_claim:false};},
 inference_evidence_ceiling(input){closed(input,['metric','left','right','evidence','operational','independent_review']);const metrics=['mean','variance','likelihood','posterior','sign_probability','permutation_probability','classification_rate','agreement','risk_difference','uncertainty'];if(!metrics.includes(input.metric)||!integer(input.left,-10000,10000)||!integer(input.right,-10000,10000))fail('INVALID_REQUEST');if(input.evidence!=='synthetic'||input.operational!==false||input.independent_review!==false)fail('UNSUPPORTED_CLAIM');return {metric:input.metric,difference:input.right-input.left,state:'represented',operational_claim:false,independent_claim:false};},
 inference_evaluation_gap(input){closed(input,['need','participants','evaluation_performed']);const needs=['manual_browser','assistive_technology','affected_user_review','independent_reproduction','real_sample','calibration_source','censoring_review','bias_assessment','legal_review','maori_authority_review'];if(!needs.includes(input.need))fail('INVALID_REQUEST');if(input.participants!==0||input.evaluation_performed!==false)fail('UNSUPPORTED_CLAIM');return {state:'open_gap',missing:[input.need],participants:0,executed:false};},
 inference_authority_gate(input){closed(input,['action','required_role','verified_roles','execute']);const roles=['professional','safety','legal','cultural','maori_authority','affected_party','privacy_governance','production_owner','deployment_owner','independent_reviewer'];if(input.action!=='consequential_inference'||!roles.includes(input.required_role))fail('INVALID_REQUEST');if(!Array.isArray(input.verified_roles)||input.verified_roles.length||input.execute!==false)fail('AUTHORITY_RESERVED');return {state:'exact_gate',missing:[input.required_role],executed:false};}
};
function run(operation,input){try{if(!Object.hasOwn(operations,operation))fail('UNKNOWN_OPERATION');const value=operations[operation](input),disposition=operation==='inference_evidence_ceiling'?'represented':operation==='inference_evaluation_gap'?'open_gap':operation==='inference_authority_gate'?'exact_gate':'completed';return {ok:true,value,error:null,disposition};}catch(error){return {ok:false,value:null,error:error.message,disposition:'open_gap'};}}
function cli(allowed){if(process.argv.length!==3){process.stdout.write(JSON.stringify({ok:false,value:null,error:'INVALID_ARGUMENTS',disposition:'open_gap'})+'\n');return 2;}let request;try{request=JSON.parse(fs.readFileSync(process.argv[2],'utf8'));closed(request,['operation','input']);}catch(error){process.stdout.write(JSON.stringify({ok:false,value:null,error:'INVALID_REQUEST',disposition:'open_gap'})+'\n');return 2;}if(!allowed.includes(request.operation)){process.stdout.write(JSON.stringify({ok:false,value:null,error:'UNKNOWN_OPERATION',disposition:'open_gap'})+'\n');return 2;}const result=run(request.operation,request.input);process.stdout.write(JSON.stringify(result)+'\n');return result.ok?0:2;}
module.exports={run,operations,cli};
