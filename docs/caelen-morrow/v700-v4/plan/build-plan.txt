function integer(n) { return String(n); }
function profileFor(i) {
  const id = `TP-${String(i).padStart(2, '0')}`;
  const a = (i % 7) - 3;
  const b = ((i * 2) % 9) - 4;
  const c = ((i * 3) % 11) - 5;
  const d = ((i * 5) % 13) - 6;
  const vector = [integer(a), integer(b), integer(c)];
  const vector2 = [integer(b + 1), integer(c - 1), integer(a + 2)];
  const matrix = [
    ['0', integer(a), '-inf'],
    [integer(b), '0', integer(c)],
    [integer(d), '-inf', '0']
  ];
  const matrix2 = [
    ['0', integer(c), integer(b)],
    ['-inf', '0', integer(a)],
    [integer(c - 1), integer(d + 1), '0']
  ];
  const edges = [
    { from: 0, to: 1, weight: integer(a) }, { from: 1, to: 2, weight: integer(b) },
    { from: 2, to: 0, weight: integer(c) }, { from: 0, to: 2, weight: integer(d) },
    { from: 2, to: 1, weight: integer(a + 1) }, { from: 1, to: 0, weight: integer(c - 1) }
  ];
  const generators = [vector, vector2, [integer(c + 1), integer(a - 1), integer(b + 2)]];
  const scalars = ['0', integer((i % 3) - 1), integer((i % 5) - 2)];
  const point = [0, 1, 2].map(k => integer(Math.max(...generators.map((g, j) => Number(g[k]) + Number(scalars[j])))));
  return {
    profile_id: id,
    scalar_a: integer(a), scalar_b: integer(b), scalar_c: integer(c), scalar_d: integer(d),
    vector, vector2, matrix, matrix2, edges,
    path: [0, 1, 2], power_limit: 4,
    polynomial_terms: [
      { coefficient: integer(a), exponent: 0 }, { coefficient: integer(b), exponent: 1 },
      { coefficient: integer(c), exponent: 2 }, { coefficient: integer(d), exponent: 3 }
    ],
    evaluation_x: integer((i % 5) - 2), generators, scalars, point,
    bound_vector: [integer(a + 3), integer(b + 4), integer(c + 5)],
    eigenvalue_candidate: '0', eigenvector_candidate: vector,
    letterpress_ticket: {

}

function requestFor(operation, p) {
  const common = { op: operation, profile_id: p.profile_id };
  switch (operation) {
    case 'carrier-boundary': return { ...common, carrier: 'Q_max_with_negative_infinity', values: [p.scalar_a, p.scalar_b, '-inf'] };
    case 'max-plus-add': return { ...common, left: p.vector, right: p.vector2 };
    case 'max-plus-multiply': return { ...common, left: p.scalar_a, right: p.scalar_b };
    case 'min-plus-add': return { ...common, left: p.vector, right: p.vector2 };
    case 'min-plus-multiply': return { ...common, left: p.scalar_c, right: p.scalar_d };
    case 'projective-normalize': return { ...common, vector: p.vector };
    case 'max-plus-matvec': return { ...common, matrix: p.matrix, vector: p.vector2 };
    case 'max-plus-matmul': return { ...common, left: p.matrix, right: p.matrix2 };
    case 'path-weight': return { ...common, edges: p.edges, path: p.path };
    case 'kleene-truncate': return { ...common, matrix: p.matrix, power_limit: p.power_limit };
    case 'tropical-polynomial-evaluate': return { ...common, terms: p.polynomial_terms, x: p.evaluation_x };
    case 'bend-locus-candidate': return { ...common, terms: p.polynomial_terms, x: p.evaluation_x };
    case 'residuation': return { ...common, matrix: p.matrix, bound: p.bound_vector };
    case 'tropical-combination': return { ...common, generators: p.generators, scalars: p.scalars };
    case 'generator-membership-certificate': return { ...common, generators: p.generators, scalars: p.scalars, point: p.point };
    case 'cycle-mean': return { ...common, vertices: 3, edges: p.edges };
    case 'eigenpair-candidate-check': return { ...common, matrix: p.matrix, lambda: p.eigenvalue_candidate, vector: p.eigenvector_candidate };
    case 'letterpress-analogy-boundary': return { ...common, ticket: p.letterpress_ticket, execution: 'synthetic_documentation_only' };
    case 'observation-gap': return { ...common, real_rows: 0, observation_status: 'absent' };
    case 'authority-reservation': return { ...common, execute: false, authority_status: 'not_present' };
    default: throw new Error(`unknown operation ${operation}`);
  }
}
