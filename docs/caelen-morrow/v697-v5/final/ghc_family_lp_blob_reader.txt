"use strict";
const cp=require("child_process");
async function readBlobs(root,head,paths){const child=cp.spawn("git",["-C",root,"cat-file","--batch"],{stdio:["pipe","pipe","pipe"]});let buffer=Buffer.alloc(0),ended=false,wake=null,stderr="";const done=new Promise((resolve,reject)=>{child.on("error",reject);child.on("close",code=>code===0?resolve():reject(Error("bounded_git_stream")));});child.stdout.on("data",b=>{buffer=Buffer.concat([buffer,b]);if(wake){const w=wake;wake=null;w();}});child.stdout.on("end",()=>{ended=true;if(wake){const w=wake;wake=null;w();}});child.stderr.on("data",b=>stderr+=b.toString("utf8").slice(0,200));async function more(){if(ended)throw Error("bounded_git_stream_end");await new Promise(resolve=>wake=resolve);}async function take(n){while(buffer.length<n)await more();const b=Buffer.from(buffer.subarray(0,n));buffer=buffer.subarray(n);return b;}async function line(){let at;while((at=buffer.indexOf(10))<0)await more();return(await take(at+1)).subarray(0,at).toString("ascii");}const result=new Map();try{for(const p of paths){if(!/^[0-9a-f]{40}$/.test(head)||/[\r\n]/.test(p)||p.includes(".."))throw Error("blob_scope");child.stdin.write(head+":"+p+"\n");const h=(await line()).split(" ");if(h.length!==3||h[1]!=="blob")throw Error("blob_type");const n=Number(h[2]);if(!Number.isSafeInteger(n)||n<0||n>20e6)throw Error("blob_size");const b=await take(n);if((await take(1))[0]!==10)throw Error("blob_terminator");result.set(p,b);}child.stdin.end();await done;return result;}catch(e){child.kill();await done.catch(()=>{});throw e;}}
module.exports={readBlobs};
