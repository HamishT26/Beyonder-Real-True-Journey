function gcd(a, b) {
  a = Math.abs(a); b = Math.abs(b);
  while (b) [a, b] = [b, a % b];
  return a;
}
function cross(a, b, c) {
  return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0]);
}
function pointKey(p) { return `${p[0]},${p[1]}`; }
function sortPoints(points) {
  return [...points].sort((a, b) => a[0] - b[0] || a[1] - b[1]);
}
function uniquePoints(points) {
  const seen = new Set();
  return sortPoints(points).filter((p) => !seen.has(pointKey(p)) && seen.add(pointKey(p)));
}
function area2(poly) {
  let total = 0;
  for (let i = 0; i < poly.length; i++) {
    const a = poly[i], b = poly[(i + 1) % poly.length];
    total += a[0] * b[1] - b[0] * a[1];
  }
  return total;
}
function onSegment(a, b, p) {
  return cross(a, b, p) === 0 && p[0] >= Math.min(a[0], b[0]) && p[0] <= Math.max(a[0], b[0]) && p[1] >= Math.min(a[1], b[1]) && p[1] <= Math.max(a[1], b[1]);
}
function segmentClass(a, b, c, d) {
  const c1 = cross(a, b, c), c2 = cross(a, b, d), c3 = cross(c, d, a), c4 = cross(c, d, b);
  if (c1 === 0 && c2 === 0 && c3 === 0 && c4 === 0) {
    const overlap = Math.max(Math.min(a[0], b[0]), Math.min(c[0], d[0])) <= Math.min(Math.max(a[0], b[0]), Math.max(c[0], d[0])) && Math.max(Math.min(a[1], b[1]), Math.min(c[1], d[1])) <= Math.min(Math.max(a[1], b[1]), Math.max(c[1], d[1]));
    return overlap ? "collinear_overlap_or_touch" : "none";
  }
  if ((c1 === 0 && onSegment(a, b, c)) || (c2 === 0 && onSegment(a, b, d)) || (c3 === 0 && onSegment(c, d, a)) || (c4 === 0 && onSegment(c, d, b))) return "touch";
  return ((c1 > 0) !== (c2 > 0) && (c3 > 0) !== (c4 > 0)) ? "proper_cross" : "none";
}
function hull(points) {
  const pts = uniquePoints(points);
  if (pts.length <= 1) return pts;
  const lower = [];
  for (const p of pts) {
    while (lower.length >= 2 && cross(lower[lower.length - 2], lower[lower.length - 1], p) <= 0) lower.pop();
    lower.push(p);
  }
  const upper = [];
  for (const p of [...pts].reverse()) {
    while (upper.length >= 2 && cross(upper[upper.length - 2], upper[upper.length - 1], p) <= 0) upper.pop();
    upper.push(p);
  }
  return lower.slice(0, -1).concat(upper.slice(0, -1));
}
function pointInPolygon(poly, p) {
  for (let i = 0; i < poly.length; i++) if (onSegment(poly[i], poly[(i + 1) % poly.length], p)) return "boundary";
  let inside = false;
  for (let i = 0, j = poly.length - 1; i < poly.length; j = i++) {
    const a = poly[i], b = poly[j];
    const intersects = ((a[1] > p[1]) !== (b[1] > p[1])) && p[0] < ((b[0] - a[0]) * (p[1] - a[1])) / (b[1] - a[1]) + a[0];
    if (intersects) inside = !inside;
  }
  return inside ? "inside" : "outside";
}
function latticeBoundary(poly) {
  let total = 0;
  for (let i = 0; i < poly.length; i++) {
    const a = poly[i], b = poly[(i + 1) % poly.length];
    total += gcd(b[0] - a[0], b[1] - a[1]);
  }
  return total;
}
function reversePatch(patch) {
  if (patch.kind === "insert") return { kind: "remove", index: patch.index, point: patch.point };
  if (patch.kind === "remove") return { kind: "insert", index: patch.index, point: patch.point };
  return { kind: "replace", index: patch.index, point: patch.previous, previous: patch.point };
}
function applyPatch(poly, patch) {
  const out = poly.map((p) => [...p]);
  if (patch.kind === "insert") out.splice(patch.index, 0, [...patch.point]);
  else if (patch.kind === "remove") out.splice(patch.index, 1);
  else out[patch.index] = [...patch.point];
  return out;
}
