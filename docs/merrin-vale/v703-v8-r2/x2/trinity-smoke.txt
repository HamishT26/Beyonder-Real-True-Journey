'use strict';
const fs=require('fs'),path=require('path'),crypto=require('crypto');
const names=['trinity-control-tower-latest.json','system-suite-status.json','trinity-api-book-v5.json','v16-trinity-verdict-v1.json','trinity-runtime-model-resolution-v1.json'];
const sha=b=>crypto.createHash('sha256').update(b).digest('hex'),escape=s=>String(s).replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
function outputPath(p){let ancestor=path.resolve(p);const tail=[];while(!fs.existsSync(ancestor)){const parent=path.dirname(ancestor);if(parent===ancestor)throw new SmokeError('OUTPUT_ROOT');tail.unshift(path.basename(ancestor));ancestor=parent;}return path.join(fs.realpathSync(ancestor),...tail);}
class SmokeError extends Error{constructor(code){super(code);this.code=code;}}
function inspect(contract,sourceRoot,maxAgeHours=168,now=new Date()){
 if(!contract||contract.authority_model!=='repo_first'||!Array.isArray(contract.read_surfaces)||contract.read_surfaces.length!==5)throw new SmokeError('CONTRACT_SHAPE');
 const root=fs.realpathSync(sourceRoot),resolved=contract.read_surfaces.map(p=>fs.realpathSync(p));
 if(new Set(resolved).size!==5||resolved.some(p=>path.dirname(p)!==root)||resolved.map(p=>path.basename(p)).sort().join('|')!==names.slice().sort().join('|'))throw new SmokeError('READ_ALLOWLIST');
 if(!Number.isFinite(maxAgeHours)||maxAgeHours<0||!Number.isFinite(now.valueOf()))throw new SmokeError('FRESHNESS_BOUND');
 const rows=resolved.map(p=>{const stat=fs.statSync(p);if(!stat.isFile()||stat.size>2*1024*1024)throw new SmokeError('SURFACE_BOUND');const raw=fs.readFileSync(p),o=JSON.parse(raw.toString('utf8').replace(/^\uFEFF/,''));if(!o||Array.isArray(o)||typeof o!=='object')throw new SmokeError('SURFACE_SHAPE');const name=path.basename(p);if(name==='trinity-api-book-v5.json'&&!Array.isArray(o.apis)&&!(o.apis&&typeof o.apis==='object'))throw new SmokeError('API_BOOK_SHAPE');if(name==='v16-trinity-verdict-v1.json'&&!o.pillars)throw new SmokeError('VERDICT_SHAPE');const time=typeof o.generated_utc==='string'?Date.parse(o.generated_utc):NaN,age=Number.isFinite(time)?Math.max(0,(now.valueOf()-time)/3600000):null,fields={};for(const k of ['overall_status','effective_success','checkpoint_class','runtime_truth_complete','authority_model']){const v=o[k];if(typeof v==='boolean')fields[k]=v;else if(typeof v==='string'&&/^[A-Za-z0-9 _.,:-]{1,120}$/.test(v))fields[k]=v;}return{name,bytes:raw.length,sha256:sha(raw),generated_utc:Number.isFinite(time)?new Date(time).toISOString():null,age_hours:age===null?null:Math.round(age*100)/100,freshness:age===null?'UNKNOWN_TIMESTAMP':time>now.valueOf()?'FUTURE_TIMESTAMP':age>maxAgeHours?'STALE':'WITHIN_DECLARED_WINDOW',fields,structural_read_pass:true};});
 const after=resolved.map(p=>sha(fs.readFileSync(p)));if(after.some((h,i)=>h!==rows[i].sha256))throw new SmokeError('SOURCE_CHANGED_DURING_READ');return{schema:'ghc.trinity.readonly-smoke.v1',status:'PASS_READ_ONLY_STRUCTURE',read_surfaces:5,source_mutations:0,network_requests:0,write_actions_executed:0,rows,freshness_window_hours:maxAgeHours,all_within_freshness_window:rows.every(r=>r.freshness==='WITHIN_DECLARED_WINDOW'),stale_sources_are_currentness_gaps:true,scope:'Read-only structural and fixity checks of the five explicitly declared local surfaces; no old system suite or connector execution.',utc:now.toISOString()};
}
function render(r){return'<!doctype html><html lang="en"><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1"><title>Trinity read-only control tower</title><style>body{max-width:76rem;margin:2rem auto;padding:1.5rem;background:#f6f4eb;color:#203c34;font:18px/1.6 system-ui}h1{font-family:Georgia,serif;font-size:2.8rem}table{border-collapse:collapse;width:100%}th,td{padding:.75rem;border-bottom:1px solid #93aaa0;text-align:left}caption{text-align:left;font-weight:700;margin:1rem 0}.tag{font-weight:700}.boundary{padding:1rem;border:2px solid #557f70}code{overflow-wrap:anywhere}a:focus-visible{outline:3px solid #b25a24}@media(max-width:700px){.table-wrap{overflow:auto}h1{font-size:2rem}}</style><main><h1>Trinity read-only control tower</h1><p class="boundary">This is a snapshot of five declared JSON surfaces. No connector, historic suite or write action was run. Readability does not establish currentness or deployment readiness.</p><p id="status">'+escape(r.status)+'</p><p>Freshness window: '+r.freshness_window_hours+' hours. Captured '+escape(r.utc)+'.</p><div class="table-wrap"><table><caption>Declared surfaces and observed freshness</caption><thead><tr><th scope="col">Surface</th><th scope="col">Generated</th><th scope="col">Freshness</th><th scope="col">Bytes</th></tr></thead><tbody>'+r.rows.map(x=>'<tr><th scope="row">'+escape(x.name)+'</th><td>'+escape(x.generated_utc||'Unknown')+'</td><td class="tag">'+x.freshness+'</td><td>'+x.bytes+'</td></tr>').join('')+'</tbody></table></div><h2>Exact source bindings</h2>'+r.rows.map(x=>'<details><summary>'+escape(x.name)+'</summary><p><code>'+x.sha256+'</code></p><pre>'+escape(JSON.stringify(x.fields,null,2))+'</pre></details>').join('')+'<p>Stale or unknown timestamps remain open currentness gaps. No source record was changed to make this view green.</p></main></html>';}
function main(argv){const a={};for(let i=0;i<argv.length;i+=2){if(!['--contract','--source-root','--output','--max-age-hours'].includes(argv[i])||a[argv[i]]!==undefined||!argv[i+1])throw new SmokeError('ARGUMENTS');a[argv[i]]=argv[i+1];}if(!a['--contract']||!a['--source-root']||!a['--output'])throw new SmokeError('ARGUMENTS');const contract=JSON.parse(fs.readFileSync(a['--contract'],'utf8').replace(/^\uFEFF/,'')),base=outputPath(a['--output']),source=fs.realpathSync(a['--source-root']),b=base.toLowerCase(),s=source.toLowerCase();if(b===s||b.startsWith(s+path.sep)||s.startsWith(b+path.sep))throw new SmokeError('OUTPUT_IS_SOURCE');const r=inspect(contract,source,a['--max-age-hours']===undefined?168:Number(a['--max-age-hours'])),run_name='run-'+Date.now().toString(36)+'-'+crypto.randomBytes(3).toString('hex'),out=path.join(base,run_name);fs.mkdirSync(out,{recursive:true});fs.writeFileSync(path.join(out,'smoke-receipt.json'),JSON.stringify(r,null,2)+'\n',{flag:'wx'});fs.writeFileSync(path.join(out,'summary.html'),render(r),{flag:'wx'});console.log(JSON.stringify({status:r.status,run_name,read_surfaces:r.read_surfaces,source_mutations:0,all_within_freshness_window:r.all_within_freshness_window}));}
module.exports={inspect,render,names,SmokeError,outputPath};if(require.main===module){try{main(process.argv.slice(2));}catch(e){console.error(JSON.stringify({ok:false,error:e.code||e.name||'SMOKE_FAILURE',source_mutations:0}));process.exitCode=2;}}
