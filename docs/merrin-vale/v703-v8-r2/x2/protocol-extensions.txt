const a=require('../x1/protocol-core.txt'),operations=['pg_dominators','pg_edge_cuts','pg_vertex_cuts','pg_longest_dag','pg_trap_components','pg_failure_frontier','pg_reverse_distance','pg_trinity_map','pg_observation_gap','pg_authority_gate'];
const edgeSort=(x,y)=>x[0]-y[0]||x[1]-y[1],cutSort=(x,y)=>x.length-y.length||JSON.stringify(x).localeCompare(JSON.stringify(y));
function can(f){return a.reach(f).some(n=>f.targets.includes(n));}
function withoutNodes(f,deleted){return{...f,nodes:f.nodes.filter(n=>!deleted.includes(n)),edges:f.edges.filter(e=>e.every(n=>!deleted.includes(n))),targets:f.targets.filter(n=>!deleted.includes(n))};}
function minimalCuts(items,blocks){const result=[];for(let bits=0;bits<2**items.length;bits++){const selected=items.filter((_,i)=>bits&(2**i));if(result.some(c=>c.every(x=>selected.includes(x))))continue;if(blocks(selected)&&selected.every((_,i)=>!blocks(selected.filter((_,j)=>i!==j))))result.push(selected);}return result.sort(cutSort);}
function value(op,f){switch(op){case'pg_dominators':{const possible=can(f);return{completion_reachable:possible,mandatory_states:possible?a.sorted(f.nodes).filter(n=>n===f.start||!can(withoutNodes(f,[n]))):[]};}case'pg_edge_cuts':{const edges=f.edges.slice().sort(edgeSort),cuts=minimalCuts(edges,deleted=>!can({...f,edges:edges.filter(e=>!deleted.includes(e))}));return{already_unreachable:!can(f),minimum_size:cuts.length?cuts[0].length:null,minimal_cuts:cuts};}case'pg_vertex_cuts':{const nodes=a.sorted(f.nodes).filter(n=>n!==f.start&&!f.targets.includes(n)),cuts=minimalCuts(nodes,deleted=>!can(withoutNodes(f,deleted)));return{already_unreachable:!can(f),minimum_size:cuts.length?cuts[0].length:null,minimal_cuts:cuts};}case'pg_longest_dag':{const adj=a.adjacency(f),incoming=Object.fromEntries(f.nodes.map(n=>[n,f.edges.filter(e=>e[1]===n).length])),queue=a.sorted(f.nodes.filter(n=>incoming[n]===0)),order=[];while(queue.length){const n=queue.shift();order.push(n);for(const v of adj[n]){incoming[v]--;if(incoming[v]===0){queue.push(v);queue.sort((x,y)=>x-y);}}}if(order.length!==f.nodes.length)return{acyclic:false,longest_distance:null};const dist=Object.fromEntries(f.nodes.map(n=>[n,null]));dist[f.start]=0;for(const n of order)if(dist[n]!==null)for(const v of adj[n])dist[v]=dist[v]===null?dist[n]+1:Math.max(dist[v],dist[n]+1);const values=f.targets.map(n=>dist[n]).filter(x=>x!==null);return{acyclic:true,longest_distance:values.length?Math.max(...values):null};}case'pg_trap_components':{const reachable=a.reach(f);return a.scc(f).filter(g=>g.some(n=>reachable.includes(n))&&!g.some(n=>f.targets.includes(n))&&!f.edges.some(([x,y])=>g.includes(x)&&!g.includes(y)));}case'pg_failure_frontier':{const reachable=a.reach(f);return f.edges.filter(([x,y])=>reachable.includes(x)&&!f.forbidden.includes(x)&&f.forbidden.includes(y)).slice().sort(edgeSort);}case'pg_reverse_distance':return a.distances(f,f.targets,true);case'pg_trinity_map':return{representation:'declared-state-transition-analogy',mind:'finite reachability and obstruction',body:'phase admission and evidence capture',heart:'authority and failure retention',real_system_observed:false};case'pg_observation_gap':return{observed_live_traces:0,required:['real task-service traces','provider availability observations','independent reproduction'],gap_open:true};case'pg_authority_gate':return{granted_authority:false,required:['scope-specific consent','competent review','affected-party governance'],executed_external_actions:0};default:throw new a.AdmissionError('UNKNOWN_OPERATION');}}
function run(q){let f;try{f=a.validate(q);if(!operations.includes(q.operation))throw new a.AdmissionError('UNKNOWN_OPERATION');}catch(e){if(e instanceof a.AdmissionError)return{ok:false,error:e.code,original_completion_credit:0};throw e;}return{ok:true,contract_id:q.contract_id,input_sha256:a.hash(q),core_outcome:q.operation==='pg_trinity_map'?'represented':q.operation==='pg_observation_gap'?'open_gap':q.operation==='pg_authority_gate'?'exact_gate':'completed',value:value(q.operation,f),boundary:'Finite declared graph only; no live reliability or authority claim.'};}
module.exports={value,run,operations};
