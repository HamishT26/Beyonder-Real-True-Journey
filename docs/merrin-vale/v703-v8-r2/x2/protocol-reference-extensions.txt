'use strict';
// Reference uses explicit simple paths and path-hitting subsets, not production deletion search.
const r=require('../x1/protocol-reference.txt');
const ns=f=>Array.from({length:f.nodes.length},(_,i)=>i),edgeSort=(a,b)=>a[0]-b[0]||a[1]-b[1],cutSort=(a,b)=>a.length-b.length||JSON.stringify(a).localeCompare(JSON.stringify(b));
function paths(f){const found=[];function visit(at,vertices,edges){if(f.targets.includes(at))found.push({vertices:vertices.slice(),edges:edges.slice()});for(const [a,b]of f.edges)if(a===at&&!vertices.includes(b))visit(b,vertices.concat(b),edges.concat([[a,b]]));}visit(f.start,[f.start],[]);return found;}
function allSubsets(items){const result=[[]];for(const x of items)for(const row of result.slice())result.push(row.concat([x]));return result;}
function hitting(items,sets,key=x=>JSON.stringify(x)){const hits=allSubsets(items).filter(c=>sets.every(s=>s.some(x=>c.some(y=>key(x)===key(y)))));return hits.filter(c=>!hits.some(d=>d.length<c.length&&d.every(x=>c.some(y=>key(x)===key(y))))).sort(cutSort);}
function expected(op,f){const p=paths(f),d=r.closure(f);switch(op){case'pg_dominators':return{completion_reachable:p.length>0,mandatory_states:p.length?ns(f).filter(n=>p.every(x=>x.vertices.includes(n))):[]};case'pg_edge_cuts':{const c=hitting(f.edges.slice().sort(edgeSort),p.map(x=>x.edges));return{already_unreachable:p.length===0,minimum_size:c.length?c[0].length:null,minimal_cuts:c};}case'pg_vertex_cuts':{const items=ns(f).filter(n=>n!==f.start&&!f.targets.includes(n)),c=hitting(items,p.map(x=>x.vertices));return{already_unreachable:p.length===0,minimum_size:c.length?c[0].length:null,minimal_cuts:c};}case'pg_longest_dag':{const cyc=ns(f).some(n=>f.edges.some(e=>e[0]===n&&e[1]===n)||ns(f).some(m=>m!==n&&isFinite(d[n][m])&&isFinite(d[m][n])));return{acyclic:!cyc,longest_distance:cyc||!p.length?null:Math.max(...p.map(x=>x.edges.length))};}case'pg_trap_components':return r.components(f,d).filter(g=>g.some(n=>isFinite(d[f.start][n]))&&g.every(n=>!f.targets.includes(n))&&f.edges.every(([x,y])=>!g.includes(x)||g.includes(y)));case'pg_failure_frontier':return f.edges.filter(e=>isFinite(d[f.start][e[0]])&&!f.forbidden.includes(e[0])&&f.forbidden.includes(e[1])).slice().sort(edgeSort);case'pg_reverse_distance':return ns(f).map(n=>{const x=Math.min(...f.targets.map(t=>d[n][t]));return{node:n,distance:isFinite(x)?x:null};});case'pg_trinity_map':return{representation:'declared-state-transition-analogy',mind:'finite reachability and obstruction',body:'phase admission and evidence capture',heart:'authority and failure retention',real_system_observed:false};case'pg_observation_gap':return{observed_live_traces:0,required:['real task-service traces','provider availability observations','independent reproduction'],gap_open:true};case'pg_authority_gate':return{granted_authority:false,required:['scope-specific consent','competent review','affected-party governance'],executed_external_actions:0};default:throw Error('Reference operation');}}
module.exports={paths,expected};
