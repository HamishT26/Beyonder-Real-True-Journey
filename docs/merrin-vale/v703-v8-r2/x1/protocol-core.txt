'use strict';
const crypto=require('crypto');
const stable=x=>x===null||typeof x!=='object'?JSON.stringify(x):Array.isArray(x)?'['+x.map(stable).join(',')+']':'{'+Object.keys(x).sort().map(k=>JSON.stringify(k)+':'+stable(x[k])).join(',')+'}';
const hash=x=>crypto.createHash('sha256').update(stable(x)).digest('hex');
class AdmissionError extends Error{constructor(code){super(code);this.code=code;}}
const need=(v,c)=>{if(!v)throw new AdmissionError(c);},keys=(o,expected)=>o!==null&&typeof o==='object'&&!Array.isArray(o)&&Object.keys(o).sort().join('|')===expected.slice().sort().join('|');
function validate(q){need(keys(q,['contract_id','operation','fixture']),'FIELD_SET');need(typeof q.contract_id==='string'&&/^[A-Za-z0-9_-]{1,100}$/.test(q.contract_id),'CONTRACT_ID');need(typeof q.operation==='string','OPERATION');const f=q.fixture;need(keys(f,['fixture_id','nodes','edges','start','targets','forbidden','horizon','synthetic']),'FIXTURE_FIELDS');need(typeof f.fixture_id==='string'&&f.fixture_id.length>0&&f.fixture_id.length<=100,'FIXTURE_ID');need(f.synthetic===true,'SYNTHETIC_ONLY');need(Array.isArray(f.nodes)&&f.nodes.length>=1&&f.nodes.length<=8&&f.nodes.every(Number.isInteger)&&new Set(f.nodes).size===f.nodes.length&&f.nodes.every(n=>n>=0&&n<f.nodes.length),'NODES');need(Array.isArray(f.edges)&&f.edges.length<=16,'EDGES');need(f.edges.every(e=>Array.isArray(e)&&e.length===2&&e.every(n=>Number.isInteger(n)&&f.nodes.includes(n))),'EDGE_ENDPOINT');need(new Set(f.edges.map(e=>e.join(':'))).size===f.edges.length,'DUPLICATE_EDGE');need(Number.isInteger(f.start)&&f.nodes.includes(f.start),'START');for(const k of ['targets','forbidden'])need(Array.isArray(f[k])&&new Set(f[k]).size===f[k].length&&f[k].every(n=>Number.isInteger(n)&&f.nodes.includes(n)),'STATE_SET');need(f.targets.length>0,'TARGETS');need(Number.isInteger(f.horizon)&&f.horizon>=0&&f.horizon<=8,'HORIZON');return f;}
const sorted=xs=>[...xs].sort((a,b)=>a-b);
function adjacency(f,reverse=false){const a=Object.fromEntries(f.nodes.map(n=>[n,[]]));for(const [x,y]of f.edges)a[reverse?y:x].push(reverse?x:y);for(const k in a)a[k].sort((x,y)=>x-y);return a;}
function reach(f,seeds=[f.start],reverse=false){const a=adjacency(f,reverse),seen=new Set(seeds),queue=[...seeds];for(let i=0;i<queue.length;i++)for(const n of a[queue[i]])if(!seen.has(n)){seen.add(n);queue.push(n);}return sorted(seen);}
function distances(f,seeds=[f.start],reverse=false){const a=adjacency(f,reverse),d=Object.fromEntries(f.nodes.map(n=>[n,null])),q=[...seeds];for(const s of seeds)d[s]=0;for(let i=0;i<q.length;i++)for(const v of a[q[i]])if(d[v]===null){d[v]=d[q[i]]+1;q.push(v);}return sorted(f.nodes).map(n=>({node:n,distance:d[n]}));}
function scc(f){const a=adjacency(f),index={},low={},stack=[],active=new Set(),groups=[];let next=0;function visit(v){index[v]=low[v]=next++;stack.push(v);active.add(v);for(const w of a[v]){if(index[w]===undefined){visit(w);low[v]=Math.min(low[v],low[w]);}else if(active.has(w))low[v]=Math.min(low[v],index[w]);}if(low[v]===index[v]){const g=[];let w;do{w=stack.pop();active.delete(w);g.push(w);}while(w!==v);groups.push(sorted(g));}}for(const n of sorted(f.nodes))if(index[n]===undefined)visit(n);return groups.sort((a,b)=>a[0]-b[0]);}
function value(op,f){switch(op){case'pg_profile':return{nodes:f.nodes.length,edges:f.edges.length,targets:f.targets.length,outdegrees:sorted(f.nodes).map(n=>[n,f.edges.filter(e=>e[0]===n).length])};case'pg_reachable':return reach(f);case'pg_can_finish':return reach(f,f.targets,true);case'pg_shortest_completion':{const ds=distances(f).filter(r=>f.targets.includes(r.node)&&r.distance!==null),d=ds.length?Math.min(...ds.map(r=>r.distance)):null;return{distance:d,targets_at_min:ds.filter(r=>r.distance===d).map(r=>r.node)};}case'pg_unsafe_reach':{const states=reach(f).filter(n=>f.forbidden.includes(n));return{states,unsafe_reachable:states.length>0};}case'pg_deadlocks':return reach(f).filter(n=>!f.targets.includes(n)&&!f.edges.some(e=>e[0]===n));case'pg_scc':return scc(f);case'pg_cycle_states':return sorted(scc(f).flatMap(g=>g.length>1?g:g.filter(n=>f.edges.some(e=>e[0]===n&&e[1]===n))));case'pg_walk_counts':{let v=Array(f.nodes.length).fill(0);v[f.start]=1;const counts=[];for(let k=0;k<=f.horizon;k++){counts.push(f.targets.reduce((s,n)=>s+v[n],0));const next=Array(f.nodes.length).fill(0);for(const [a,b]of f.edges)next[b]+=v[a];v=next;}return{horizon:f.horizon,counts_by_length:counts};}case'pg_transition_closure':return sorted(f.nodes).flatMap(a=>reach(f,[a]).map(b=>[a,b]));default:throw new AdmissionError('UNKNOWN_OPERATION');}}
function run(q){let f;try{f=validate(q);if(!operations.includes(q.operation))throw new AdmissionError('UNKNOWN_OPERATION');}catch(e){if(e instanceof AdmissionError)return{ok:false,error:e.code,original_completion_credit:0};throw e;}return{ok:true,contract_id:q.contract_id,input_sha256:hash(q),core_outcome:'completed',value:value(q.operation,f),boundary:'Finite declared graph only; no live reliability or authority claim.'};}
const operations=['pg_profile','pg_reachable','pg_can_finish','pg_shortest_completion','pg_unsafe_reach','pg_deadlocks','pg_scc','pg_cycle_states','pg_walk_counts','pg_transition_closure'];
module.exports={validate,AdmissionError,stable,hash,sorted,adjacency,reach,distances,scc,value,run,operations};
