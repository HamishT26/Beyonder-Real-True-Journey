'use strict';
// Independent matrix closure and explicit bounded walk enumeration; no production import.
function closure(f){const n=f.nodes.length,d=Array.from({length:n},(_,i)=>Array.from({length:n},(_,j)=>i===j?0:Infinity));for(const [a,b]of f.edges)d[a][b]=Math.min(d[a][b],1);for(let k=0;k<n;k++)for(let i=0;i<n;i++)for(let j=0;j<n;j++)d[i][j]=Math.min(d[i][j],d[i][k]+d[k][j]);return d;}
const ids=f=>Array.from({length:f.nodes.length},(_,i)=>i);
function components(f,d=closure(f)){const remaining=new Set(ids(f)),groups=[];while(remaining.size){const a=Math.min(...remaining),g=ids(f).filter(b=>remaining.has(b)&&isFinite(d[a][b])&&isFinite(d[b][a]));g.forEach(x=>remaining.delete(x));groups.push(g);}return groups;}
function expected(op,f){const d=closure(f),ns=ids(f),reachable=ns.filter(n=>isFinite(d[f.start][n]));switch(op){case'pg_profile':return{nodes:ns.length,edges:f.edges.length,targets:f.targets.length,outdegrees:ns.map(n=>[n,f.edges.reduce((s,e)=>s+Number(e[0]===n),0)])};case'pg_reachable':return reachable;case'pg_can_finish':return ns.filter(n=>f.targets.some(t=>isFinite(d[n][t])));case'pg_shortest_completion':{const min=Math.min(...f.targets.map(t=>d[f.start][t]));return{distance:isFinite(min)?min:null,targets_at_min:isFinite(min)?ns.filter(n=>f.targets.includes(n)&&d[f.start][n]===min):[]};}case'pg_unsafe_reach':{const states=ns.filter(n=>f.forbidden.includes(n)&&isFinite(d[f.start][n]));return{states,unsafe_reachable:states.length>0};}case'pg_deadlocks':return reachable.filter(n=>!f.targets.includes(n)&&f.edges.filter(e=>e[0]===n).length===0);case'pg_scc':return components(f,d);case'pg_cycle_states':return ns.filter(a=>f.edges.some(([b,c])=>a===b&&a===c)||ns.some(b=>b!==a&&isFinite(d[a][b])&&isFinite(d[b][a])));case'pg_walk_counts':{const counts=Array(f.horizon+1).fill(0);function visit(at,depth){if(f.targets.includes(at))counts[depth]++;if(depth<f.horizon)for(const [a,b]of f.edges)if(a===at)visit(b,depth+1);}visit(f.start,0);return{horizon:f.horizon,counts_by_length:counts};}case'pg_transition_closure':return ns.flatMap(a=>ns.filter(b=>isFinite(d[a][b])).map(b=>[a,b]));default:throw Error('Reference operation');}}
module.exports={closure,components,expected};
