'use strict';
// Registered GHC launcher guard. It controls this entrypoint, not arbitrary shell use.
const fs=require('fs'),path=require('path'),cp=require('child_process');
const allowed=Object.freeze(['gpt-5.6-sol','gpt-6-astra']);
class LaunchError extends Error{constructor(code,message){super(message);this.code=code;}}
function parse(argv){const at=argv.indexOf('--'),front=at<0?argv:argv.slice(0,at),child=at<0?[]:argv.slice(at+1),o={child};for(let i=0;i<front.length;i++){const key=front[i];if(key==='--check-only'){if(o.checkOnly)throw new LaunchError('DUPLICATE_OPTION','Specify check-only once.');o.checkOnly=true;continue;}if(!['--model','--entrypoint','--codex','--cwd'].includes(key))throw new LaunchError('UNKNOWN_OPTION','Unknown launcher option: '+key);const k=key.slice(2);if(o[k]!==undefined)throw new LaunchError('DUPLICATE_OPTION','Specify '+key+' once.');const val=front[++i];if(!val||val.startsWith('--'))throw new LaunchError('MISSING_OPTION_VALUE','A value is required for '+key+'.');o[k]=val;}return o;}
function preflight(o){if(!allowed.includes(o.model))throw new LaunchError('MODEL_NOT_ALLOWED','Choose gpt-5.6-sol or gpt-6-astra. No phase process, worktree or baton was created.');if(o.entrypoint&&o.codex)throw new LaunchError('AMBIGUOUS_ENTRYPOINT','Choose one Node phase entrypoint or the installed Codex entrypoint.');if(!o.checkOnly&&!o.entrypoint&&!o.codex)throw new LaunchError('ENTRYPOINT_REQUIRED','Supply --entrypoint with a TXT phase script, --codex with the installed Codex script, or --check-only.');for(let i=0;i<o.child.length;i++){const s=o.child[i];if(/^-[mp]/.test(s)||/^--(?:model|model-provider|profile)(?:=|$)/.test(s)||/^(?:model|model_provider|profile)\s*=/.test(s)||/^(?:-c=?|--config=)(?:model|model_provider|profile)\s*=/.test(s))throw new LaunchError('CHILD_MODEL_OVERRIDE','Model or profile overrides must be selected at the guarded launcher.');if((s==='-c'||s==='--config')&&/^(?:model|model_provider|profile)\s*=/.test(o.child[i+1]||''))throw new LaunchError('CHILD_MODEL_OVERRIDE','Child configuration cannot override the selected model.');}return{ok:true,model:o.model,allowed_models:allowed,guard:'GHC_MODEL_ALLOWLIST_V16',mutation_performed:false,model_request_sent:false,availability_scope:'Current local GHC allowlist; a provider can still reject an otherwise allowed model at execution time.'};}
function main(argv){let o,p;try{o=parse(argv);p=preflight(o);}catch(e){if(!(e instanceof LaunchError))throw e;console.error(JSON.stringify({ok:false,error:e.code,message:e.message,child_started:false,mutation_performed:false}));return2();return;}if(o.checkOnly){console.log(JSON.stringify(p));return;}const file=path.resolve(o.entrypoint||o.codex),ext=path.extname(file).toLowerCase();if(o.entrypoint&&ext!=='.txt'||o.codex&&!['.js','.mjs','.cjs'].includes(ext)){console.error(JSON.stringify({ok:false,error:'ENTRYPOINT_FORMAT',child_started:false}));process.exitCode=2;return;}if(!fs.existsSync(file)||!fs.statSync(file).isFile()){console.error(JSON.stringify({ok:false,error:'ENTRYPOINT_MISSING',child_started:false}));process.exitCode=2;return;}const cwd=o.cwd?path.resolve(o.cwd):process.cwd();if(!fs.existsSync(cwd)||!fs.statSync(cwd).isDirectory()){console.error(JSON.stringify({ok:false,error:'CWD_MISSING',child_started:false}));process.exitCode=2;return;}const args=o.codex?[file,'--model',o.model,...o.child]:[file,...o.child],r=cp.spawnSync(process.execPath,args,{cwd,env:{...process.env,GHC_MODEL:o.model,GHC_MODEL_GUARD:'v16'},stdio:'inherit',shell:false});if(r.error){console.error(JSON.stringify({ok:false,error:'CHILD_START_FAILED',child_started:false}));process.exitCode=1;return;}process.exitCode=Number.isInteger(r.status)?r.status:1;}
function return2(){process.exitCode=2;}
module.exports={allowed,parse,preflight,LaunchError,main};if(require.main===module)main(process.argv.slice(2));
