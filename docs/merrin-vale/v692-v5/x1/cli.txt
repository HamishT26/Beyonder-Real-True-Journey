const fs=require('fs');
function decode(text){const parsed=JSON.parse(text);let i=0;const ws=()=>{while(i<text.length&&/\s/.test(text[i]))i++;};function str(){const start=i++;while(i<text.length){if(text[i]==='\\'){i+=2;continue;}if(text[i++]==='"')return JSON.parse(text.slice(start,i));}throw Error('invalid_json');}function value(){ws();if(text[i]==='{'){i++;ws();const seen=new Set();if(text[i]==='}'){i++;return;}for(;;){ws();const key=str();if(seen.has(key))throw Error('duplicate_key');seen.add(key);ws();i++;value();ws();const c=text[i++];if(c==='}')break;}}else if(text[i]==='['){i++;ws();if(text[i]===']'){i++;return;}for(;;){value();ws();if(text[i++]===']')break;}}else if(text[i]==='"')str();else while(i<text.length&&!/[\s,}\]]/.test(text[i]))i++;}value();return parsed;}
function run(api,allowed){let result;try{const file=process.argv[2];let b;if(file){if(fs.statSync(file).size>65536)throw Error('request_too_large');b=fs.readFileSync(file);}else{b=Buffer.alloc(65537);let size=0,n;while(size<b.length&&(n=fs.readSync(0,b,size,b.length-size,null))>0)size+=n;b=b.subarray(0,size);}if(b.length>65536)throw Error('request_too_large');const x=decode(b.toString('utf8'));result=allowed.includes(x?.op)?api.evaluate(x):{ok:false,value:null,error:'unknown_operation',external_credit:false};}catch(e){result={ok:false,value:null,error:['duplicate_key','request_too_large'].includes(e.message)?e.message:'invalid_json',external_credit:false};}process.stdout.write(JSON.stringify(result)+'\n');}
