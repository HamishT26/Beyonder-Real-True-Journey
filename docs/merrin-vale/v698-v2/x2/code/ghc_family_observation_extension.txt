'use strict';
module.exports=function(core){
const ops=['or_information_loss','or_refinement_gain','or_quotient_edges','or_path_lift','or_observation_language','or_spurious_words','or_minimum_sensor_sets','or_minimum_sensor_cost','or_observation_gap','or_authority_gate'];
const claims={real_sensor_identification:['measurement_protocol','calibration','uncertainty','independent_observation'],empirical_gmut:['covariant_action','observable_map','dataset','likelihood'],production_thos:['real_operational_trials','failure_model','independent_safety_review'],production_identity:['real_keys_and_proofs','lifecycle','trust_governance','privacy_security_review'],affected_accessibility:['assistive_technology_trials','affected_user_review','accessible_remedy']};
const roles={deployment:['competent_operational_review','real_system_evidence'],maori_wording:['maori_authority','affected_governance'],personhood:['not_established_by_finite_software'],rights_allocation:['affected_party_authority','legal_governance'],stage20:['all_protected_evidence_and_authority_gates']};
function lex(a,b){for(let i=0;i<Math.min(a.length,b.length);i++)if(a[i]!==b[i])return a[i]-b[i];return a.length-b.length;}
const unique=words=>[...new Map(words.map(w=>[JSON.stringify(w),w])).values()].sort(lex);
function quotient(x,p){const sets=core.cells(p).map(()=>new Set());for(let s=0;s<p.length;s++)for(const t of x.transitions[s])sets[p[s]].add(p[t]);return sets.map(s=>[...s].sort((a,b)=>a-b));}
function language(edges,labels,horizon){let frontier=new Map(edges.map((_,s)=>[s+'|'+labels[s],{state:s,word:[labels[s]]}]));for(let k=0;k<horizon;k++){const next=new Map();for(const {state,word} of frontier.values())for(const t of edges[state]){const w=[...word,labels[t]];next.set(t+'|'+w.join(','),{state:t,word:w});}frontier=next;}return unique([...frontier.values()].map(r=>r.word));}
function lift(x,p){let paths=new Map(p.flatMap((v,s)=>v===x.path[0]?[[s,[s]]]:[])),frontiers=[[...paths.keys()].sort((a,b)=>a-b)];for(let k=1;k<x.path.length;k++){const next=new Map();for(const [s,word] of paths)for(const t of x.transitions[s])if(p[t]===x.path[k]){const w=[...word,t],old=next.get(t);if(!old||lex(w,old)<0)next.set(t,w);}paths=next;frontiers.push([...paths.keys()].sort((a,b)=>a-b));}const q=quotient(x,p),empty=frontiers.findIndex(f=>f.length===0);return{abstract_edges_exist:x.path.slice(1).every((v,i)=>q[x.path[i]].includes(v)),realizable:paths.size>0,witness:[...paths.values()].sort(lex)[0]||null,frontiers,first_empty_frontier:empty<0?null:empty};}
function cover(x,weights){const target=(1<<x.goals.length)-1,coverage=x.features.map(f=>x.goals.reduce((m,[a,b],i)=>f[a]!==f[b]?m|(1<<i):m,0));let dp=new Map([[0,{cost:0,sets:[[]]}]]);for(let i=0;i<coverage.length;i++){const next=new Map(dp);for(const [m,value] of dp){const merged=m|coverage[i],cost=value.cost+weights[i],sets=value.sets.map(s=>[...s,i]),old=next.get(merged);if(!old||cost<old.cost)next.set(merged,{cost,sets});else if(cost===old.cost)next.set(merged,{cost,sets:unique([...old.sets,...sets])});}dp=next;}return dp.get(target)||null;}
function calculate(op,x){const p=core.normalized(x.left),q=core.normalized(x.right),n=p.length;switch(op){
case'or_information_loss':{const lost=[],gained=[];for(let i=0;i<n;i++)for(let j=i+1;j<n;j++){if(p[i]!==p[j]&&q[i]===q[j])lost.push([i,j]);if(p[i]===p[j]&&q[i]!==q[j])gained.push([i,j]);}return{left_refines_right:core.splitting(p,q)===null,lost,gained,loss_count:lost.length,gain_count:gained.length,real_privacy_verified:false};}
case'or_refinement_gain':{const labels=core.commonRefinement(p,q),old=core.rough(p,x.target),refined=core.rough(labels,x.target);return{refined_labels:labels,added_certain:refined.lower.filter(i=>!old.lower.includes(i)),removed_possible:old.upper.filter(i=>!refined.upper.includes(i)),lower_monotone:old.lower.every(i=>refined.lower.includes(i)),upper_monotone:refined.upper.every(i=>old.upper.includes(i))};}
case'or_quotient_edges':{const q=quotient(x,p);return{cells:core.cells(p),edges:q.flatMap((a,i)=>a.map(j=>[i,j])),edge_semantics:'existential_concrete_pair'};}
case'or_path_lift':return lift(x,p);
case'or_observation_language':{const words=language(x.transitions,p,x.horizon);return{horizon:x.horizon,words,count:words.length,unbounded_claim:false};}
case'or_spurious_words':{const concrete=language(x.transitions,p,x.horizon),adj=quotient(x,p),abstract=language(adj,adj.map((_,i)=>i),x.horizon),keys=new Set(concrete.map(w=>JSON.stringify(w))),spurious=abstract.filter(w=>!keys.has(JSON.stringify(w)));return{horizon:x.horizon,concrete_count:concrete.length,abstract_count:abstract.length,spurious_count:spurious.length,first_spurious_word:spurious[0]||null,real_system_counterexample:false};}
case'or_minimum_sensor_sets':{const best=cover(x,x.features.map(()=>1));return{feasible:best!==null,minimum_size:best?.cost??null,solutions:best?best.sets.sort(lex):[],goal_pairs:x.goals.length};}
case'or_minimum_sensor_cost':{const best=cover(x,x.costs);return{feasible:best!==null,minimum_cost:best?.cost??null,solutions:best?best.sets.sort(lex):[],within_declared_budget:best!==null&&best.cost<=x.budget,cost_units:'declared_integer_model_units'};}
case'or_observation_gap':return{claim:x.claim,missing:claims[x.claim].filter(r=>!x.provided.includes(r)),externally_verified:false,finite_model_is_measurement:false};
case'or_authority_gate':return{request:x.request,decision:'exact_gate',missing:[...roles[x.request]],authorized:false};
default:core.fail('unknown_operation');}}
return{operations:ops,evaluate:r=>core.route(r,ops,calculate,op=>op==='or_information_loss'?'represented':op==='or_observation_gap'?'open_gap':op==='or_authority_gate'?'exact_gate':'completed')};
};
