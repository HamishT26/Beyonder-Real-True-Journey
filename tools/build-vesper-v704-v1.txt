function mkdir(rel) {
  fs.mkdirSync(path.join(PHASE_ROOT, rel), { recursive: true });
}

function normalizeLf(text) {
  return String(text).replace(/\r\n/g, "\n").replace(/\r/g, "\n");
}

function writeText(rel, text) {
  const target = path.join(PHASE_ROOT, rel);
  fs.mkdirSync(path.dirname(target), { recursive: true });
  fs.writeFileSync(target, normalizeLf(text), "utf8");
}

function writeJson(rel, value) {
  writeText(rel, JSON.stringify(value, null, 2) + "\n");
}

function readJson(file) {
  return JSON.parse(fs.readFileSync(file, "utf8"));
}

function sha256Buffer(buffer) {
  return crypto.createHash("sha256").update(buffer).digest("hex");
}

function sha256File(file) {
  return sha256Buffer(fs.readFileSync(file));
}

function gitBlobSha1(buffer) {
  const header = Buffer.from("blob " + buffer.length + "\0", "utf8");
  return crypto.createHash("sha1").update(header).update(buffer).digest("hex");
}

function words(text) {
  const matches = normalizeLf(text).match(/\S+/g);
  return matches ? matches.length : 0;
}

function posixRel(file) {
  return path.relative(ROOT, file).split(path.sep).join("/");
}

function listFiles(dir) {
  if (!fs.existsSync(dir)) {
    return [];
  }
  const out = [];
  for (const entry of fs.readdirSync(dir, { withFileTypes: true }).sort((a, b) => a.name.localeCompare(b.name))) {
    const full = path.join(dir, entry.name);
    if (entry.isDirectory()) {
      out.push(...listFiles(full));
    } else if (entry.isFile()) {
      out.push(full);
    }
  }
  return out;
}

function makeManifest(stage, directoryRel, excludeRel) {
  const directory = path.join(PHASE_ROOT, directoryRel);
  const entries = [];
  for (const file of listFiles(directory)) {
    const relFromPhase = path.relative(PHASE_ROOT, file).split(path.sep).join("/");
    if (relFromPhase === excludeRel) {
      continue;
    }
    const buffer = fs.readFileSync(file);
    const text = buffer.toString("utf8");
    entries.push({
      path: posixRel(file),
      bytes: buffer.length,
      words: words(text),
      sha256: sha256Buffer(buffer),
      git_blob_sha1: gitBlobSha1(buffer)
    });
  }
  entries.sort((a, b) => a.path.localeCompare(b.path));
  return {
    owner: OWNER,
    phase: PHASE,
    stage,
    source_commit: SOURCE_FINAL,
    source_content_not_git_ancestry: true,
    entries,
    entry_count: entries.length,
    entry_digest: sha256Buffer(Buffer.from(entries.map((entry) => entry.path + "\0" + entry.sha256).join("\n"), "utf8")),
    byte_domain: "Exact staged Git blobs after commit; precommit entries use the identical Git blob construction",
    self_excluded: true
  };
}

function bitsOf(index, n) {
  const bits = [];
  for (let i = 0; i < n; i += 1) {
    bits.push((index >> (n - 1 - i)) & 1);
  }
  return bits;
}

function bitString(index, n) {
  return bitsOf(index, n).join("");
}

function makeProfile(id, title, predicate) {
  const variableCount = 4;
  const truth = [];
  for (let i = 0; i < (1 << variableCount); i += 1) {
    truth.push(predicate(bitsOf(i, variableCount)) ? 1 : 0);
  }
  const mask = truth.reduce((acc, value, index) => acc | (value << index), 0);
  return {
    fixture_id: id,
    title,
    variables: ["a", "b", "c", "d"],
    variable_count: variableCount,
    truth,
    mask_hex: "0x" + mask.toString(16).padStart(4, "0"),
    synthetic: true
  };
}

function cofactorTruth(profile, variableIndex, value) {
  const out = [];
  for (let i = 0; i < profile.truth.length; i += 1) {
    if (bitsOf(i, profile.variable_count)[variableIndex] === value) {
      out.push(profile.truth[i]);
    }
  }
  return out;
}

function essentialWitness(profile, variableIndex) {
  for (let i = 0; i < profile.truth.length; i += 1) {
    const flipMask = 1 << (profile.variable_count - 1 - variableIndex);
    const j = i ^ flipMask;
    if (profile.truth[i] !== profile.truth[j]) {
      return [bitString(Math.min(i, j), profile.variable_count), bitString(Math.max(i, j), profile.variable_count)];
    }
  }
  return null;
}

function canonicalDnf(profile) {
  const terms = [];
  for (let i = 0; i < profile.truth.length; i += 1) {
    if (profile.truth[i] === 1) {
      const bits = bitsOf(i, profile.variable_count);
      terms.push("(" + bits.map((value, index) => value ? profile.variables[index] : "not " + profile.variables[index]).join(" and ") + ")");
    }
  }
  if (terms.length === 0) {
    return "false";
  }
  if (terms.length === profile.truth.length) {
    return "true";
  }
  return terms.join(" or ");
}

function canonicalCnf(profile) {
  const clauses = [];
  for (let i = 0; i < profile.truth.length; i += 1) {
    if (profile.truth[i] === 0) {
      const bits = bitsOf(i, profile.variable_count);
      clauses.push("(" + bits.map((value, index) => value ? "not " + profile.variables[index] : profile.variables[index]).join(" or ") + ")");
    }
  }
  if (clauses.length === 0) {
    return "true";
  }
  if (clauses.length === profile.truth.length) {
    return "false";
  }
  return clauses.join(" and ");
}

function anfTerms(profile) {
  const coefficients = profile.truth.slice();
  const n = profile.variable_count;
  for (let bit = 0; bit < n; bit += 1) {
    for (let mask = 0; mask < (1 << n); mask += 1) {
      if (mask & (1 << bit)) {
        coefficients[mask] ^= coefficients[mask ^ (1 << bit)];
      }
    }
  }
  const terms = [];
  for (let mask = 0; mask < coefficients.length; mask += 1) {
    if (!coefficients[mask]) {
      continue;
    }
    if (mask === 0) {
      terms.push("1");
      continue;
    }
    const vars = [];
    for (let bit = 0; bit < n; bit += 1) {
      if (mask & (1 << bit)) {
        vars.push(profile.variables[n - 1 - bit]);
      }
    }
    terms.push(vars.join("*"));
  }
  return { coefficients, terms };
}

function monotonicity(profile) {
  const n = profile.variable_count;
  for (let x = 0; x < (1 << n); x += 1) {
    for (let y = 0; y < (1 << n); y += 1) {
      if ((x & y) === x && profile.truth[x] > profile.truth[y]) {
        return { monotone: false, counterexample: [bitString(x, n), bitString(y, n)] };
      }
    }
  }
  return { monotone: true, counterexample: null };
}

function robdd(profile) {
  const nodes = new Map();
  const unique = new Map();
  const memo = new Map();
  let nextId = 2;
  function build(level, indices) {
    const outputs = indices.map((index) => profile.truth[index]);
    if (outputs.every((value) => value === 0)) {
      return 0;
    }
    if (outputs.every((value) => value === 1)) {
      return 1;
    }
    const signature = level + ":" + outputs.join("");
    if (memo.has(signature)) {
      return memo.get(signature);
    }
    const low = [];
    const high = [];
    for (const index of indices) {
      if (bitsOf(index, profile.variable_count)[level] === 0) {
        low.push(index);
      } else {
        high.push(index);
      }
    }
    const lowId = build(level + 1, low);
    const highId = build(level + 1, high);
    if (lowId === highId) {
      memo.set(signature, lowId);
      return lowId;
    }
    const key = level + ":" + lowId + ":" + highId;
    if (unique.has(key)) {
      const existing = unique.get(key);
      memo.set(signature, existing);
      return existing;
    }
    const id = nextId;
    nextId += 1;
    unique.set(key, id);
    nodes.set(id, { id, variable: profile.variables[level], level, low: lowId, high: highId });
    memo.set(signature, id);
    return id;
  }
  const root = build(0, profile.truth.map((_, index) => index));
  return {
    variable_order: profile.variables,
    root,
    terminal_nodes: [0, 1],
    nonterminal_count: nodes.size,
    total_node_count: nodes.size + 2,
    nodes: Array.from(nodes.values()).sort((a, b) => a.id - b.id)
  };
}

function operationResult(operation, profile) {
  const support = profile.variables.map((name, index) => ({ name, witness: essentialWitness(profile, index) })).filter((row) => row.witness);
  if (operation.id === "bf_profile") {
    return { variable_count: profile.variable_count, row_count: profile.truth.length, binary: profile.truth.every((v) => v === 0 || v === 1), mask_hex: profile.mask_hex };
  }
  if (operation.id === "bf_support") {
    return { support: support.map((row) => row.name), size: support.length };
  }
  if (operation.id === "bf_satisfying") {
    return { assignments: profile.truth.map((value, index) => value ? bitString(index, profile.variable_count) : null).filter(Boolean) };
  }
  if (operation.id === "bf_weight") {
    return { true_rows: profile.truth.reduce((sum, value) => sum + value, 0), false_rows: profile.truth.reduce((sum, value) => sum + (1 - value), 0) };
  }
  if (operation.id === "bf_canonical_dnf") {
    return { dnf: canonicalDnf(profile) };
  }
  if (operation.id === "bf_canonical_cnf") {
    return { cnf: canonicalCnf(profile) };
  }
  if (operation.id === "bf_anf") {
    return anfTerms(profile);
  }
  if (operation.id === "bf_cofactors") {
    return { cofactors: profile.variables.map((name, index) => ({ variable: name, low: cofactorTruth(profile, index, 0), high: cofactorTruth(profile, index, 1) })) };
  }
  if (operation.id === "bf_essential_witnesses") {
    return { witnesses: profile.variables.map((name, index) => ({ variable: name, pair: essentialWitness(profile, index) })) };
  }
  if (operation.id === "bf_monotonicity") {
    return monotonicity(profile);
  }
  if (operation.id === "bf_complement") {
    return { truth: profile.truth.map((value) => 1 - value) };
  }
  if (operation.id === "bf_dual") {
    const fullMask = (1 << profile.variable_count) - 1;
    return { truth: profile.truth.map((_, index) => 1 - profile.truth[index ^ fullMask]) };
  }
  if (operation.id === "bf_self_dual") {
    const fullMask = (1 << profile.variable_count) - 1;
    return { self_dual: profile.truth.every((value, index) => value === 1 - profile.truth[index ^ fullMask]) };
  }
  if (operation.id === "bf_symmetry") {
    const layers = [];
    for (let weight = 0; weight <= profile.variable_count; weight += 1) {
      const outputs = [];
      for (let index = 0; index < profile.truth.length; index += 1) {
        if (bitsOf(index, profile.variable_count).reduce((a, b) => a + b, 0) === weight) {
          outputs.push(profile.truth[index]);
        }
      }
      layers.push({ weight, outputs: Array.from(new Set(outputs)).sort() });
    }
    return { symmetric: layers.every((layer) => layer.outputs.length === 1), layers };
  }
  if (operation.id === "bf_canalizing") {
    const rows = [];
    profile.variables.forEach((name, index) => {
      [0, 1].forEach((value) => {
        const cofactor = cofactorTruth(profile, index, value);
        if (cofactor.every((entry) => entry === cofactor[0])) {
          rows.push({ variable: name, input: value, forced_output: cofactor[0] });
        }
      });
    });
    return { canalizing_assignments: rows };
  }
  if (operation.id === "bf_influence") {
    return {
      denominator: profile.truth.length,
      numerators: profile.variables.map((name, index) => {
        const flipMask = 1 << (profile.variable_count - 1 - index);
        let changed = 0;
        for (let row = 0; row < profile.truth.length; row += 1) {
          if (profile.truth[row] !== profile.truth[row ^ flipMask]) {
            changed += 1;
          }
        }
        return { variable: name, numerator: changed };
      })
    };
  }
  if (operation.id === "bf_robdd") {
    return robdd(profile);
  }
  if (operation.id === "bf_accessible_summary") {
    return {
      text: profile.title + " has " + support.length + " essential variables and " + profile.truth.reduce((a, b) => a + b, 0) + " satisfying rows out of " + profile.truth.length + ".",
      represented_not_accessibility_complete: true
    };
  }
  if (operation.id === "bf_external_observation") {
    return { state: "open_gap", reason: "No governed external observation, participant, hardware or operator arm is present." };
  }
  if (operation.id === "bf_authority_gate") {
    return { state: "exact_gate", reason: "Deployment and every professional, legal, cultural, affected-party and Maori-authority decision require competent external authority." };
  }
  throw new Error("Unknown operation " + operation.id);
}

function makePortfolio(session) {
  const contracts = contractsForSession(session);
  const safe = [];
  const candidates = [];
  const cfr = [];
  for (const contract of contracts) {
    for (let variant = 0; variant < 2; variant += 1) {
      safe.push({
        task_id: contract.proposal_id + "-S" + variant,
        proposal_id: contract.proposal_id,
        request_variant: variant === 0 ? "canonical_key_order" : "reversed_key_order",
        execution_state: contract.expected_execution_disposition,
        result: "pass",
        completion_credit: contract.expected_execution_disposition === "completed" ? 1 : 0
      });
      candidates.push({
        task_id: contract.proposal_id + "-C" + variant,
        proposal_id: contract.proposal_id,
        mutation: variant === 0 ? "unknown_request_field" : "unknown_profile_identifier",
        subject_result: "fail",
        refusal_guard_result: "pass",
        completion_credit: 0
      });
      cfr.push({
        task_id: contract.proposal_id + "-R" + variant,
        proposal_id: contract.proposal_id,
        review: variant === 0 ? "exact_result_readback" : "source_profile_restoration_readback",
        result: "pass",
        filesystem_deletion: false,
        source_restored_unchanged: true
      });
    }
  }
  return {
    session,
    counts: { safe: safe.length, candidate: candidates.length, cfr: cfr.length },
    safe,
    candidates,
    cfr,
    boundary: BOUNDARY
  };
}

function makeValidation(ledger) {
  const ids = new Set();
  const issues = [];
  for (const witness of ledger.witnesses) {
    if (ids.has(witness.witness_id)) {
      issues.push("duplicate witness " + witness.witness_id);
    }
    ids.add(witness.witness_id);
    if (!["pass", "fail"].includes(witness.result)) {
      issues.push("unknown result " + witness.result);
    }
  }
  return {
    schema: "ghc.family.method-flow-state.validation.v1",
    owner: OWNER,
    phase: PHASE,
    method_count: ledger.methods.length,
    witness_count: ledger.witnesses.length,
    issue_count: issues.length,
    issues,
    privacy_hits: [],
    valid: issues.length === 0,
    boundary: "Validation covers the finite ledger shape and bounded evidence only; it is not broader assurance."
  };
}

function sessionMethodIndex(session, operationLedgers, extras) {
  const components = [];
  for (const row of operationLedgers) {
    components.push({
      path: session + "/method-flow/" + row.operation.id + ".json",
      sha256: sha256File(path.join(PHASE_ROOT, session, "method-flow", row.operation.id + ".json")),
      counts: row.ledger.counts
    });
  }
  for (const extra of extras) {
    components.push({
      path: extra.path,
      sha256: sha256File(path.join(PHASE_ROOT, extra.path)),
      counts: extra.counts
    });
  }
  const totals = components.reduce((acc, component) => {
    acc.methods += component.counts.methods;
    acc.witnesses += component.counts.witnesses;
    acc.passing += component.counts.witness_results.pass;
    acc.failed += component.counts.witness_results.fail;
    return acc;
  }, { methods: 0, witnesses: 0, passing: 0, failed: 0 });
  return { kind: "index_not_a_ledger", session, components, totals };
}

function modelRecords() {
  const truthCube = [];
  for (let index = 0; index < 16; index += 1) {
    const bits = bitsOf(index, 4);
    truthCube.push({ id: bitString(index, 4), x: bits[0] * 2 + bits[1], y: bits[2] * 2 + bits[3], z: bits.reduce((a, b) => a + b, 0) });
  }
  const landscape = PROFILES.map((profile, index) => ({
    id: profile.fixture_id,
    x: profile.truth.reduce((a, b) => a + b, 0),
    y: profile.variables.filter((_, variable) => essentialWitness(profile, variable)).length,
    z: robdd(profile).total_node_count,
    layer: index
  }));
  const cofactorCloud = [];
  for (const profile of PROFILES) {
    profile.variables.forEach((variable, index) => {
      const low = cofactorTruth(profile, index, 0).reduce((a, b) => a + b, 0);
      const high = cofactorTruth(profile, index, 1).reduce((a, b) => a + b, 0);
      cofactorCloud.push({ id: profile.fixture_id + "-" + variable, x: index, y: low, z: high });
    });
  }
  const robddCloud = [];
  for (const profile of PROFILES) {
    const diagram = robdd(profile);
    for (const node of diagram.nodes) {
      robddCloud.push({ id: profile.fixture_id + "-n" + node.id, x: node.level, y: node.low, z: node.high });
    }
  }
  const boundaryStack = PROFILES.map((profile, index) => ({
    id: profile.fixture_id,
    x: index,
    y: operationResult(OPERATIONS[18], profile).state === "open_gap" ? 1 : 0,
    z: operationResult(OPERATIONS[19], profile).state === "exact_gate" ? 2 : 0
  }));
  return [
    { model_id: "VA7041-MODEL-1", title: "Four-variable truth-cube coordinates", coordinate_units: "dimensionless synthetic index", points: truthCube },
    { model_id: "VA7041-MODEL-2", title: "Weight-support-ROBDD landscape", coordinate_units: "finite counts", points: landscape },
    { model_id: "VA7041-MODEL-3", title: "Cofactor weight cloud", coordinate_units: "finite counts", points: cofactorCloud },
    { model_id: "VA7041-MODEL-4", title: "ROBDD edge-coordinate cloud", coordinate_units: "synthetic node identifiers", points: robddCloud },
    { model_id: "VA7041-MODEL-5", title: "Gap and gate boundary stack", coordinate_units: "categorical synthetic layer", points: boundaryStack }
  ];
}

function staticBrowserChecks(html) {
  const checks = [
    ["doctype", html.startsWith("<!doctype html>")],
    ["language", html.includes("<html lang=\"en\">")],
    ["charset", html.includes("charset=\"utf-8\"")],
    ["viewport", html.includes("name=\"viewport\"")],
    ["title", html.includes("<title>")],
    ["main", html.includes("<main>")],
    ["single-h1", (html.match(/<h1>/g) || []).length === 1],
    ["section-headings", (html.match(/<h2/g) || []).length >= 2],
    ["table-caption", html.includes("<caption>")],
    ["column-scopes", html.includes("scope=\"col\"")],
    ["row-scopes", html.includes("scope=\"row\"")],
    ["aria-labelledby", html.includes("aria-labelledby")],
    ["boundary-visible", html.includes("NOT_READY_FOR_STAGE_20")]
  ];
  return { checks: checks.map(([id, pass]) => ({ id, pass })), passed: checks.filter((row) => row[1]).length, total: checks.length };
}

function skillValidation(skillDirs) {
  const validator = path.join(CODEX_HOME, "skills", ".system", "skill-creator", "scripts", "quick_validate.py");
  const results = [];
  for (const directory of skillDirs) {
    const child = cp.spawnSync("python", [validator, directory], { encoding: "utf8", windowsHide: true });
    results.push({
      directory: directory.split(path.sep).join("/"),
      status: child.status,
      passed: child.status === 0,
      stdout: normalizeLf(child.stdout || "").trim(),
      stderr: normalizeLf(child.stderr || "").trim()
    });
  }
  return { validator: validator.split(path.sep).slice(-3).join("/"), results, passed: results.filter((row) => row.passed).length, total: results.length };
}

  for (const session of ["x1", "x2"]) {

      const file = path.join(PHASE_ROOT, session, "runners", "ghc-family-boolean-" + session + "-" + (index + 1) + ".txt");
      const valid = runRunner(file, { operation: group[0].id, profile_id: PROFILES[index].fixture_id });

function installCuratedPairs() {
  const installations = [];
  for (const definition of curatedDefinitions()) {
    const source = path.join(PHASE_ROOT, "x2", "curated", definition.name);
    const target = path.join(CODEX_HOME, "skills", definition.name);
    if (fs.existsSync(target)) {
      const sameSkill = sha256File(path.join(source, "SKILL.md")) === sha256File(path.join(target, "SKILL.md"));
      const sameRunner = sha256File(path.join(source, "runner.txt")) === sha256File(path.join(target, "runner.txt"));
      if (!sameSkill || !sameRunner) {
        throw new Error("Curated installation conflict at " + target);
      }
      installations.push({ name: definition.name, state: "already_exact", target: target.split(path.sep).join("/") });
    } else {
      fs.cpSync(source, target, { recursive: true, errorOnExist: true, force: false });
      installations.push({ name: definition.name, state: "installed_additively", target: target.split(path.sep).join("/") });
    }
  }
  return installations;
}

function collectMethodComponents() {
  const componentPaths = [];
  for (const session of ["x1", "x2"]) {
    const index = readJson(path.join(PHASE_ROOT, session, "method-flow-index.json"));
    for (const component of index.components) {
      componentPaths.push(component.path);
    }
  }
  if (fs.existsSync(path.join(PHASE_ROOT, "final", "terminal-method-flow.json"))) {
    componentPaths.push("final/terminal-method-flow.json");
  }
  const components = componentPaths.map((componentPath) => {
    const file = path.join(PHASE_ROOT, componentPath.replace("/", path.sep));
    const ledger = readJson(file);
    return { path: componentPath, sha256: sha256File(file), counts: ledger.counts };
  });
  const ownerIncrement = components.reduce((acc, component) => {
    acc.methods += component.counts.methods;
    acc.witnesses += component.counts.witnesses;
    acc.passing += component.counts.witness_results.pass;
    acc.failed += component.counts.witness_results.fail;
    return acc;
  }, { methods: 0, witnesses: 0, passing: 0, failed: 0 });
  return { components, ownerIncrement };
}

function retainedNegatives(components) {
  const ids = [];
  for (const component of components) {
    const ledger = readJson(path.join(PHASE_ROOT, component.path.replace("/", path.sep)));
    for (const witness of ledger.witnesses || []) {
      if (witness.result === "fail") {
        ids.push(witness.witness_id);
      }
    }
  }
  ids.sort();
  return {
    count: ids.length,
    distinct_ids: ids,
    original_subject_credit: 0,
    details_in: components.map((component) => component.path)
  };
}

function writeDeck() {
  const cards = deckCards();
  writeJson("final/deck/stable-cards.json", { cards: cards.stable, count: cards.stable.length });
  const modules = [{ path: "final/deck/stable-cards.json", count: cards.stable.length }];
  for (let index = 0; index < 10; index += 1) {
    const ownerSlice = cards.newCards.slice(index * 30, (index + 1) * 30);
    const inheritedSlice = cards.inherited.slice(index * 30, (index + 1) * 30);
    const ownerPath = "final/deck/owner-task-" + String(index + 1).padStart(2, "0") + ".json";
    const inheritedPath = "final/deck/inherited-reference-" + String(index + 1).padStart(2, "0") + ".json";
    writeJson(ownerPath, { cards: ownerSlice, count: ownerSlice.length });
    writeJson(inheritedPath, { cards: inheritedSlice, count: inheritedSlice.length });
    modules.push({ path: ownerPath, count: ownerSlice.length }, { path: inheritedPath, count: inheritedSlice.length });
  }
  const order = cards.stable.concat(cards.newCards, cards.inherited).map((card) => card.card_id);
  writeJson("final/deck/deck-index.json", {
    card_count: order.length,
    tier_counts: { "1": 1, "2": 3, "3": 8, "4": 600 },
    modules,
    order,
    source_commit: SOURCE_FINAL,
    source_content_not_git_ancestry: true,
    new_core_outcomes: { completed: 255, represented: 15, open_gap: 15, exact_gate: 15 }
  });
  writeJson("final/deck/stable-prefix.json", { card_ids: cards.stable.map((card) => card.card_id) });
  writeJson("final/deck/volatile-index.json", { card_ids: cards.newCards.concat(cards.inherited).map((card) => card.card_id), reading_is_not_execution: true });
  const manifestEntries = listFiles(path.join(PHASE_ROOT, "final", "deck")).filter((file) => !file.endsWith("card-manifest.json")).map((file) => {
    const buffer = fs.readFileSync(file);
    return { path: path.relative(PHASE_ROOT, file).split(path.sep).join("/"), bytes: buffer.length, sha256: sha256Buffer(buffer) };
  }).sort((a, b) => a.path.localeCompare(b.path));
  writeJson("final/deck/card-manifest.json", { entries: manifestEntries, entry_count: manifestEntries.length, self_excluded: true });
}

function makeFinalManifest() {
  const candidates = listFiles(PHASE_ROOT).concat(listFiles(path.join(ROOT, "tools")));
  const entries = [];
  for (const file of candidates) {
    const rel = posixRel(file);
    if (rel.endsWith("/final/manifest.json")) {
      continue;
    }
    const buffer = fs.readFileSync(file);
    entries.push({
      path: rel,
      bytes: buffer.length,
      words: words(buffer.toString("utf8")),
      sha256: sha256Buffer(buffer),
      git_blob_sha1: gitBlobSha1(buffer)
    });
  }
  entries.sort((a, b) => a.path.localeCompare(b.path));
  return {
    owner: OWNER,
    phase: PHASE,
    stage: "final",
    source_commit: SOURCE_FINAL,
    source_content_not_git_ancestry: true,
    entries,
    entry_count: entries.length,
    entry_digest: sha256Buffer(Buffer.from(entries.map((entry) => entry.path + "\0" + entry.sha256).join("\n"), "utf8")),
    byte_domain: "Exact final staged Git blobs",
    self_excluded: true
  };
}

  const mode = process.argv[2];
  if (!mode) {
    throw new Error("Expected stage argument: planning, x1, x2 or final");
  }

  if (mode === "planning") {
    planningStage();

    x1Stage();
  } else if (mode === "x2") {
