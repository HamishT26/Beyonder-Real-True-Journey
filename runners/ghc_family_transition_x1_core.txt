'use strict';
const p=require('./ghc_family_transition_protocol.txt');
const FIELDS={lts_enabled:[],lts_reachable:[],lts_bad_trace:['target'],lts_deadlocks:[],lts_ef:['target'],lts_af:['target'],lts_eg:['allowed'],lts_eu:['target','allowed'],lts_scc:[],lts_conflicts:[]};
const outgoing=(m,s)=>m.edges.filter(e=>e[0]===s).sort(p.compare);
const succ=(m,s)=>{const es=outgoing(m,s);return es.length?[...new Set(es.map(e=>e[2]))]:[s];};
function reach(m,start=m.start){const seen=new Set([start]),queue=[start];for(let i=0;i<queue.length;i++)for(const e of outgoing(m,queue[i]))if(!seen.has(e[2])){seen.add(e[2]);queue.push(e[2]);}return m.states.filter(v=>seen.has(v));}
function badTrace(m,targets){const queue=[{states:[m.start],edges:[]}],seen=new Set([m.start]);for(let i=0;i<queue.length;i++){const x=queue[i],v=x.states.at(-1);if(targets.includes(v))return x;for(const e of outgoing(m,v))if(!seen.has(e[2])){seen.add(e[2]);queue.push({states:[...x.states,e[2]],edges:[...x.edges,e]});}}return null;}
function least(m,seed,predicate){const win=new Set(seed);let changed;do{changed=false;for(const s of m.states)if(!win.has(s)&&predicate(s,win)){win.add(s);changed=true;}}while(changed);return m.states.filter(s=>win.has(s));}
function eg(m,allowed){const keep=new Set(allowed);let changed;do{changed=false;for(const s of [...keep])if(!succ(m,s).some(t=>keep.has(t))){keep.delete(s);changed=true;}}while(changed);return m.states.filter(s=>keep.has(s));}
function scc(m){let index=0;const indices=new Map(),low=new Map(),stack=[],on=new Set(),groups=[];function visit(v){indices.set(v,index);low.set(v,index++);stack.push(v);on.add(v);for(const e of outgoing(m,v)){const w=e[2];if(!indices.has(w)){visit(w);low.set(v,Math.min(low.get(v),low.get(w)));}else if(on.has(w))low.set(v,Math.min(low.get(v),indices.get(w)));}if(low.get(v)===indices.get(v)){const g=[];let w;do{w=stack.pop();on.delete(w);g.push(w);}while(w!==v);g.sort((a,b)=>m.states.indexOf(a)-m.states.indexOf(b));groups.push(g);}}m.states.forEach(v=>{if(!indices.has(v))visit(v);});groups.sort((a,b)=>m.states.indexOf(a[0])-m.states.indexOf(b[0]));return groups.map(g=>({states:g,cyclic:g.length>1||outgoing(m,g[0]).some(e=>e[2]===g[0])}));}
function value(q){if(!q||!Object.hasOwn(FIELDS,q.op))p.fail('unknown_operation');p.object(q,['op','model',...FIELDS[q.op]]);const m=p.model(q.model);if(q.target)p.stateSet(q.target,m);if(q.allowed)p.stateSet(q.allowed,m);
switch(q.op){case'lts_enabled':return [...new Set(outgoing(m,m.start).map(e=>e[1]))].sort();case'lts_reachable':return reach(m);case'lts_bad_trace':return badTrace(m,q.target);case'lts_deadlocks':return reach(m).filter(s=>!outgoing(m,s).length);case'lts_ef':return least(m,q.target,(s,w)=>outgoing(m,s).some(e=>w.has(e[2])));case'lts_af':return least(m,q.target,(s,w)=>succ(m,s).every(t=>w.has(t)));case'lts_eg':return eg(m,q.allowed);case'lts_eu':return least(m,q.target,(s,w)=>q.allowed.includes(s)&&outgoing(m,s).some(e=>w.has(e[2])));case'lts_scc':return scc(m);case'lts_conflicts':return m.states.flatMap(s=>[...new Set(outgoing(m,s).map(e=>e[1]))].sort().map(a=>({state:s,action:a,successors:[...new Set(outgoing(m,s).filter(e=>e[1]===a).map(e=>e[2]))].sort()})).filter(x=>x.successors.length>1));}}
const run=q=>p.envelope(value,q);module.exports={run,value,FIELDS,outgoing,succ,reach,badTrace,least,eg,scc};if(require.main===module)p.cli(run);
