'use strict';
const fs=require('node:fs'),path=require('node:path'),assert=require('node:assert/strict'),ROOT=path.resolve(__dirname,'..'),BASE=path.join(ROOT,'docs','sable-rook','v690-v5'),core=require(path.join(BASE,'x2','core.txt'));
const proposals=JSON.parse(fs.readFileSync(path.join(BASE,'plan','new-proposals.json'),'utf8')).proposals.filter(r=>r.session==='x2'),checks=[],add=(id,f)=>{try{f();checks.push({id,result:'pass'});}catch(e){checks.push({id,result:'fail',error:e.message});}};
for(const p of proposals){const before=JSON.stringify(p.request);add(p.proposal_id+'-positive',()=>assert.deepEqual(core.evaluate(p.request),p.expected_output));add(p.proposal_id+'-candidate',()=>assert.deepEqual(core.evaluate(p.candidate_subject),p.candidate_expected));add(p.proposal_id+'-nonmutation',()=>assert.equal(JSON.stringify(p.request),before));}
for(const [op,allowed] of Object.entries(core.allowed)){add('family-'+op+'-case-token',()=>assert.ok(allowed.includes('case_token')));add('family-'+op+'-closed-envelope',()=>assert.deepEqual(Object.keys(core.evaluate(proposals.find(p=>p.operation===op).request)).sort(),['error','ok','value']));}
const failed=checks.filter(r=>r.result==='fail'),receipt={schema:'ghc.family.x2-tests.v5',owner:'Sable Rook',phase:'v690-v5',session:'x2',checks,passed:checks.length-failed.length,total:checks.length,failed,positive_cases:100,rejecting_candidates:100,nonmutation_reviews:100,same_owner_only:true,independent_reproduction:false,terminal_verdict:'NOT_READY_FOR_STAGE_20'};const target=path.resolve(ROOT,process.argv[2]);fs.mkdirSync(path.dirname(target),{recursive:true});fs.writeFileSync(target,JSON.stringify(receipt,null,2)+'\n');process.stdout.write(JSON.stringify({passed:receipt.passed,total:receipt.total,failed:failed.length})+'\n');if(failed.length)process.exitCode=1;
