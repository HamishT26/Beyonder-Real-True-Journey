'use strict';
const u=require('../scripts/ghc_family_avelin_evidence.txt'),core=require('../runners/ghc_family_transition_x1_core.txt'),protocol=require('../runners/ghc_family_transition_protocol.txt');
const checks=[],add=(id,ok,detail=null)=>checks.push({id,result:ok?'pass':'fail',detail:ok?null:detail});
const definitions=u.read('plan/new-proposals.json').proposals.filter(p=>p.session==='x1');
for(const d of definitions){const before=u.canon(d.request),actual=core.run(d.request),negative=core.run(d.candidate_subject);add(d.proposal_id+'-positive',u.canon(actual)===u.canon(d.expected_output),{actual,expected:d.expected_output});add(d.proposal_id+'-refusal',u.canon(negative)===u.canon(d.candidate_expected),{negative,expected:d.candidate_expected});add(d.proposal_id+'-nonmutation',before===u.canon(d.request));}
const models=u.read('plan/operation-families.json').models;
for(let k=0;k<models.length;k++){const model=models[k],target=model.states.filter((_,i)=>i%2===0),allowed=model.states.filter(s=>!target.includes(s));const af=core.run({op:'lts_af',model,target}).value,eg=core.run({op:'lts_eg',model,allowed}).value;add('AF-EG-duality-'+k,model.states.every(s=>af.includes(s)!==eg.includes(s)));const gs=core.scc(model);add('SCC-partition-'+k,gs.flatMap(g=>g.states).length===model.states.length&&new Set(gs.flatMap(g=>g.states)).size===model.states.length&&gs.every(g=>g.states.every(a=>g.states.every(b=>core.reach(model,a).includes(b)))));}
const dead=models[0];add('deadlock-visible',u.canon(core.run({op:'lts_deadlocks',model:dead}).value)==='["s0"]');add('deadlock-not-eventually-empty',core.run({op:'lts_af',model:dead,target:[]}).value.length===0);add('deadlock-stutters-for-EG',core.run({op:'lts_eg',model:dead,allowed:['s0']}).value.length===1);add('zero-step-counterexample',core.run({op:'lts_bad_trace',model:dead,target:['s0']}).value.edges.length===0);
for(const [id,text,expected] of [['duplicate-json','{"op":"x","op":"y"}','duplicate_key'],['nonfinite-json','1e999','nonfinite_number'],['trailing-json','{} false','json_syntax']]){let actual;try{protocol.parse(text);actual='accepted';}catch(e){actual=e.message;}add(id,actual===expected,{actual,expected});}
const result={schema:'ghc.family.owner-tests.v5',owner:u.OWNER,phase:u.PHASE,session:'x1',checks,passed:checks.filter(c=>c.result==='pass').length,total:checks.length,failed:checks.filter(c=>c.result==='fail'),same_owner_only:true,independent_reproduction:false,boundary:u.boundary};
if(process.argv[2])u.fs.writeFileSync(process.argv[2],JSON.stringify(result,null,2)+'\n');console.log(JSON.stringify({passed:result.passed,total:result.total,failed:result.failed}));if(result.failed.length)process.exitCode=1;
