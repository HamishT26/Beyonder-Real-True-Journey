def dump(path: Path, value: object) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, indent=2, ensure_ascii=False, sort_keys=False) + "\n", encoding="utf-8", newline="\n")


def write(path: Path, text: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(text.rstrip() + "\n", encoding="utf-8", newline="\n")


def sha_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def stable_digest(value: object) -> str:
    return sha_bytes(json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":")).encode("utf-8"))


def load_seren_proposals() -> list[dict]:

    for name in ("proposals-x1.json", "proposals-x2.json"):

        rows.extend(payload["proposals"])
    if len(rows) != 300:
        raise RuntimeError(f"expected 300 Seren proposals, found {len(rows)}")
    return rows
