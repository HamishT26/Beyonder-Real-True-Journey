function sha(value) { return crypto.createHash("sha256").update(value).digest("hex"); }
function git(args, buffer = false) { return cp.execFileSync("git", args, { cwd: ROOT, encoding: buffer ? null : "utf8", maxBuffer: 256 * 1024 * 1024 }); }
function readJson(rel) { return JSON.parse(fs.readFileSync(path.join(ROOT, rel), "utf8")); }
function blob(commit, rel) { return git(["show", commit + ":" + rel], true); }
function verifyManifest(commit, rel) {
  const manifest = JSON.parse(blob(commit, rel).toString("utf8"));
  const errors = [];
  for (const entry of manifest.entries) {
    const data = blob(commit, entry.path);
    if (data.length !== entry.bytes || sha(data) !== entry.sha256) errors.push(entry.path);
  }
  return { entries: manifest.entries.length, errors };
}
function check(name, passed, detail) { return { name, passed: Boolean(passed), detail }; }
