function textBytes(value) {
  return Buffer.from(String(value).replace(/\r\n/g, "\n"), "utf8");
}
function sha(value) {
  return crypto.createHash("sha256").update(Buffer.isBuffer(value) ? value : textBytes(value)).digest("hex");
}
function writeText(rel, value) {
  const abs = path.join(ROOT, rel);
  fs.mkdirSync(path.dirname(abs), { recursive: true });
  fs.writeFileSync(abs, textBytes(value));
}
function writeJson(rel, value) {
  writeText(rel, JSON.stringify(value, null, 2) + "\n");
}
function readJson(abs) {
  return JSON.parse(fs.readFileSync(abs, "utf8"));
}
function relFromRoot(abs) {
  return path.relative(ROOT, abs).replace(/\\/g, "/");
}
function git(args, cwd) {

}
function canonicalTokens(s) {
  return String(s).toLowerCase().replace(/[^a-z0-9]+/g, " ").trim().split(/\s+/).filter(Boolean);
}
function jaccard(a, b) {
  const A = new Set(canonicalTokens(a));
  const B = new Set(canonicalTokens(b));
  const I = [...A].filter(x => B.has(x)).length;
  const U = new Set([...A, ...B]).size;
  return U ? I / U : 1;
}
