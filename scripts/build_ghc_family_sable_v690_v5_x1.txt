function ordered(value) {
  if (Array.isArray(value)) return value.map(ordered);
  if (value && typeof value === 'object') return Object.fromEntries(Object.keys(value).sort().map((key) => [key, ordered(value[key])]));
  return value;
}
function stable(value) { return JSON.stringify(ordered(value), null, 2) + '\n'; }
function compact(value) { return JSON.stringify(ordered(value)); }
function sha(value) { return crypto.createHash('sha256').update(Buffer.isBuffer(value) ? value : Buffer.from(String(value))).digest('hex'); }
function writeJson(relative, value) { const target = path.join(X1, relative); fs.mkdirSync(path.dirname(target), {recursive: true}); fs.writeFileSync(target, stable(value), 'utf8'); }
function writeText(relative, value) { const target = path.join(X1, relative); fs.mkdirSync(path.dirname(target), {recursive: true}); fs.writeFileSync(target, value.trimEnd() + '\n', 'utf8'); }
