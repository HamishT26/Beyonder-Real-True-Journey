'use strict';
const fs=require('node:fs'),path=require('node:path'),crypto=require('node:crypto'),cp=require('node:child_process');
const ROOT=path.resolve(__dirname,'..');
const command=process.argv[2],manifestRel=process.argv[3];
if(!command||!manifestRel)throw Error('usage: node owner_manifest.txt --build|--verify manifest-relative-path');
const git=(args,encoding='utf8')=>cp.execFileSync('git',args,{cwd:ROOT,encoding});
const digest=bytes=>crypto.createHash('sha256').update(bytes).digest('hex');
const entryFor=rel=>{const bytes=git(['show',`:${rel}`],null);return{path:rel,bytes:bytes.length,sha256:digest(bytes)};};
if(command==='--build-pair'){
 const ownerRel=process.argv[4];if(!ownerRel)throw Error('--build-pair delta-manifest owner-manifest');const exclusions=[manifestRel,ownerRel].sort(),staged=git(['diff','--cached','--name-only','--diff-filter=ACMR']).trim().split(/\r?\n/).filter(Boolean).sort(),files=git(['ls-files']).trim().split(/\r?\n/).filter(Boolean).sort();
 const deltaEntries=staged.filter(rel=>!exclusions.includes(rel)).map(entryFor),ownerEntries=files.filter(rel=>!exclusions.includes(rel)).map(entryFor);
 const delta={schema:'ghc.family.raw-git-manifest.v5',owner:'Sable Rook',phase:'v690-v5',lifecycle_state:'FINAL_DELTA',domain:'raw_git_index_blob',entries:deltaEntries,entry_count:deltaEntries.length,self_exclusions:exclusions};
 const owner={schema:'ghc.family.raw-git-owner-manifest.v5',owner:'Sable Rook',phase:'v690-v5',domain:'raw_git_index_blob',entries:ownerEntries,entry_count:ownerEntries.length,self_exclusions:exclusions,owner_file_count:ownerEntries.length+exclusions.length,file_ceiling:2000};
 for(const [rel,payload] of [[manifestRel,delta],[ownerRel,owner]]){const target=path.join(ROOT,rel);fs.mkdirSync(path.dirname(target),{recursive:true});fs.writeFileSync(target,JSON.stringify(payload,null,2)+'\n','utf8');}
 process.stdout.write(JSON.stringify({built:true,delta_entries:deltaEntries.length,owner_entries:ownerEntries.length,exclusions:2,owner_files:owner.owner_file_count})+'\n');
}else if(command==='--verify-pair'){
 const ownerRel=process.argv[4];if(!ownerRel)throw Error('--verify-pair delta-manifest owner-manifest');const verify=(rel,universe)=>{const m=JSON.parse(fs.readFileSync(path.join(ROOT,rel),'utf8')),expected=[...m.entries.map(r=>r.path),...m.self_exclusions].sort(),missing=expected.filter(r=>!universe.includes(r)),extra=universe.filter(r=>!expected.includes(r)),mismatches=[];for(const e of m.entries){const bytes=git(['show',`:${e.path}`],null);if(bytes.length!==e.bytes||digest(bytes)!==e.sha256)mismatches.push(e.path);}return{valid:!missing.length&&!extra.length&&!mismatches.length,entries:m.entries.length,exclusions:m.self_exclusions.length,missing,extra,mismatches};};
 const staged=git(['diff','--cached','--name-only','--diff-filter=ACMR']).trim().split(/\r?\n/).filter(Boolean).sort(),files=git(['ls-files']).trim().split(/\r?\n/).filter(Boolean).sort(),delta=verify(manifestRel,staged),owner=verify(ownerRel,files),valid=delta.valid&&owner.valid&&files.length<2000;process.stdout.write(JSON.stringify({valid,delta,owner,owner_files:files.length})+'\n');if(!valid)process.exitCode=1;
}else if(command==='--build'){
 const files=git(['ls-files']).trim().split(/\r?\n/).filter(Boolean).sort(),entries=[];
 for(const rel of files){if(rel===manifestRel)continue;const bytes=git(['show',`:${rel}`],null);entries.push({path:rel,bytes:bytes.length,sha256:digest(bytes)});}
 const payload={schema:'ghc.family.raw-git-owner-manifest.v5',owner:'Sable Rook',phase:'v690-v5',domain:'raw_git_index_blob',entries,entry_count:entries.length,self_exclusions:[manifestRel],owner_file_count:entries.length+1,file_ceiling:2000};
 const target=path.join(ROOT,manifestRel);fs.mkdirSync(path.dirname(target),{recursive:true});fs.writeFileSync(target,JSON.stringify(payload,null,2)+'\n','utf8');process.stdout.write(JSON.stringify({built:true,entries:entries.length,exclusions:1,owner_files:entries.length+1})+'\n');
}else if(command==='--verify'){
 const m=JSON.parse(fs.readFileSync(path.join(ROOT,manifestRel),'utf8')),files=git(['ls-files']).trim().split(/\r?\n/).filter(Boolean).sort(),expected=[...m.entries.map(r=>r.path),...m.self_exclusions].sort(),missing=expected.filter(r=>!files.includes(r)),extra=files.filter(r=>!expected.includes(r)),mismatches=[];
 for(const e of m.entries){const bytes=git(['show',`:${e.path}`],null);if(bytes.length!==e.bytes||digest(bytes)!==e.sha256)mismatches.push(e.path);}
 const valid=!missing.length&&!extra.length&&!mismatches.length&&m.owner_file_count===files.length&&files.length<2000;process.stdout.write(JSON.stringify({valid,entries:m.entries.length,exclusions:m.self_exclusions.length,owner_files:files.length,missing,extra,mismatches})+'\n');if(!valid)process.exitCode=1;
}else throw Error('unknown command');
