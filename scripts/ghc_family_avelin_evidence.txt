'use strict';
const fs=require('node:fs'),path=require('node:path'),cp=require('node:child_process'),crypto=require('node:crypto');
const BASE='docs/avelin-reed/v690-v6',OWNER='Avelin Reed',PHASE='v690-v6';
const boundary='Bounded same-owner synthetic evidence only; no empirical, participant, identity, professional, production, legal, cultural, affected-party, Maori-authority, complete privacy/accessibility, exhaustive-security, independent-reproduction, consciousness/personhood, AGI/ASI, Theory-of-Everything or Stage 20 claim. NOT_READY_FOR_STAGE_20.';
const hash=b=>crypto.createHash('sha256').update(b).digest('hex');
const stable=v=>v===null||typeof v!=='object'?v:Array.isArray(v)?v.map(stable):Object.fromEntries(Object.keys(v).sort().map(k=>[k,stable(v[k])]));
const canon=v=>JSON.stringify(stable(v));
const git=(...a)=>cp.execFileSync('git',a,{encoding:'utf8'}).trim();
const read=r=>JSON.parse(fs.readFileSync(path.join(BASE,r),'utf8'));
function put(r,v,exclusive=true){const p=path.join(BASE,r);if(exclusive&&fs.existsSync(p))throw Error('refuse overwrite '+r);fs.mkdirSync(path.dirname(p),{recursive:true});fs.writeFileSync(p,typeof v==='string'?v:JSON.stringify(v,null,2)+'\n');}
function paths(dir='.') {const out=[];for(const e of fs.readdirSync(dir,{withFileTypes:true})){if(e.name==='.git')continue;const p=path.join(dir,e.name);if(e.isDirectory())out.push(...paths(p));else out.push(p.split(path.sep).join('/').replace(/^\.\//,''));}return out.sort();}
function manifest(out,selected,exclusions=[]){if(!exclusions.includes(out))exclusions.push(out);const entries=selected.filter(p=>!exclusions.includes(p)).sort().map(p=>{const b=fs.readFileSync(p);if(b.includes(13))throw Error('CR found '+p);return{path:p,bytes:b.length,sha256:hash(b)};});const value={schema:'ghc.family.raw-manifest.v5',owner:OWNER,phase:PHASE,byte_domain:'exact UTF-8 normalized LF working bytes equal Git blobs',entries,self_exclusions:exclusions,selected_count:selected.length,boundary};fs.writeFileSync(out,JSON.stringify(value,null,2)+'\n');return value;}
function ledger(stage){return{schema:'ghc.family.method-flow-state.v1',owner:OWNER,phase:PHASE,stage,identity_boundary:boundary,execution_authority:'owner_self_scoped_delta',methods:[],witnesses:[],state_events:[],recommendations:[],counts:{},boundary};}
function method(l,id,title,negativeIds,workaround){l.methods.push({method_id:id,title,failure_signature:title,trigger_preconditions:['Declared owner scope and frozen input','Matching stage'],privacy_class:'sanitized_public',approval_class:'safe_now',candidate_workaround:workaround,validation_witness_ids:[],recurrence_guard:'Full typed equality and retained failures; no successful canonical replay.',rollback:'Select prior compatible caller and preserve all evidence.',recommendation_state:'candidate',supersedes:[],protected_gates:read('plan/source-provenance.json').source_overlay?['All scientific, identity, professional, production, legal, cultural, affected-party, Maori-authority and Stage 20 boundaries']:['Stage 20'],retained_negative_ids:negativeIds,scope_boundary:boundary});l.state_events.push({method_id:id,from:'observed',to:'candidate',reason:'Declared bounded workaround'});}
function witness(l,id,mid,result,procedure,expected,observed,negativeIds=[]){l.witnesses.push({witness_id:id,method_id:mid,procedure,scope:'Avelin '+PHASE+' '+l.stage,expected,observed,result,same_owner_only:true,independent_reproduction:false,retained_negative_ids:negativeIds,original_subject_success_credit:result==='fail'?0:null,boundary});if(result==='pass'){const m=l.methods.find(m=>m.method_id===mid);m.validation_witness_ids.push(id);if(m.recommendation_state==='candidate'){m.recommendation_state='validated';l.state_events.push({method_id:mid,from:'candidate',to:'validated',witness_id:id});}}}
function finalize(l){l.counts={methods:l.methods.length,direct_witnesses:l.witnesses.length,failed_witnesses:l.witnesses.filter(w=>w.result==='fail').length,passing_witnesses:l.witnesses.filter(w=>w.result==='pass').length};return l;}
function validLedger(l){const ids=new Set(l.witnesses.map(w=>w.witness_id));return ids.size===l.witnesses.length&&l.witnesses.every(w=>l.methods.some(m=>m.method_id===w.method_id)&&['pass','fail'].includes(w.result)&&w.same_owner_only&&!w.independent_reproduction&&(w.result!=='fail'||w.original_subject_success_credit===0))&&l.methods.every(m=>m.retained_negative_ids.length&&m.validation_witness_ids.every(w=>ids.has(w))&&m.validation_witness_ids.some(id=>l.witnesses.some(w=>w.witness_id===id&&w.result==='pass')));}
module.exports={fs,path,cp,BASE,OWNER,PHASE,boundary,hash,canon,git,read,put,paths,manifest,ledger,method,witness,finalize,validLedger};
