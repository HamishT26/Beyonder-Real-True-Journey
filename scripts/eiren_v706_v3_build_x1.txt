def dump(path: Path, value: object) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, indent=2, ensure_ascii=False) + "\n", encoding="utf-8", newline="\n")


def write(path: Path, text: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(text.rstrip() + "\n", encoding="utf-8", newline="\n")


def sha_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def stable(value: object) -> str:
    return sha_bytes(json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":")).encode("utf-8"))


def load_engine() -> dict:
    namespace: dict = {"__name__": "eiren_v706_v3_cf_engine"}
    source = ENGINE_PATH.read_text(encoding="utf-8")
    compile(source, ENGINE_PATH.as_posix(), "exec")
    exec(source, namespace)
    return namespace


def reconstruct(terms: list[int]) -> Fraction:
    value = Fraction(terms[-1], 1)
    for term in reversed(terms[:-1]):
        value = term + Fraction(1, value)
    return value

def make_runner(path: Path, allowed: list[str]) -> None:
    content = f'''from __future__ import annotations
import json
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[5]
namespace = {{"__name__": "eiren_v706_v3_cf_engine"}}
engine_path = ROOT / "scripts" / "eiren_v706_v3_cf_engine.txt"
source = engine_path.read_text(encoding="utf-8")
exec(compile(source, engine_path.as_posix(), "exec"), namespace)
allowed = {allowed!r}
try:
    request = json.loads(sys.stdin.read())
    if request.get("op") not in allowed:
        raise ValueError("operation not admitted by this paired caller")
    result = namespace["execute"](request)
    print(json.dumps({{"ok": True, "result": result, "boundary": "finite owner-local exact rational"}}, sort_keys=True))
except Exception as exc:
    print(json.dumps({{"ok": False, "code": "REFUSED", "error": str(exc), "boundary": "invalid subject has zero credit"}}, sort_keys=True))
    raise SystemExit(2)
'''
    write(path, content)


def build() -> None:
    proposals = json.loads((PHASE / "planning" / "proposals-x1.json").read_text(encoding="utf-8"))["proposals"]
    if len(proposals) != 150:
        raise RuntimeError("X1 requires exactly 150 frozen definitions")
    engine = load_engine()
    execute = engine["execute"]
    op_names = engine["X1_OPERATIONS"]
    grouped_results: dict[str, list] = {name: [] for name in op_names}
    negative_records: list[dict] = []
    method_ledgers: list[dict] = []

    for op_index, operation in enumerate(op_names, 1):
        op_proposals = [row for row in proposals if row["operation"] == operation]
        witnesses: list[dict] = []
        op_negative_ids: list[str] = []
        for proposal in op_proposals:
            request = proposal["request"]
            before = stable(request)
            result = execute(request)
            after = stable(request)
            reference_fraction = Fraction(request["fixture"]["numerator"], request["fixture"]["denominator"])
            exact_fraction = Fraction(result["normalized"]["numerator"], result["normalized"]["denominator"])
            safe_checks = [
                {"name": "exact reference rational", "pass": exact_fraction == reference_fraction},
                {"name": "request bytes semantically unchanged", "pass": before == after},
                {"name": "declared disposition", "pass": result["disposition"] == "completed"},
            ]
            bad_denominator = json.loads(json.dumps(request))
            bad_denominator["fixture"]["denominator"] = 0
            bad_operation = json.loads(json.dumps(request))
            bad_operation["op"] = "unsupported-operation"
            candidates = [("BAD-DENOMINATOR", bad_denominator), ("BAD-OPERATION", bad_operation)]
            candidate_results = []
            repairs = []
            for kind, candidate in candidates:
                negative_id = f"{proposal['id']}-{kind}"
                op_negative_ids.append(negative_id)
                try:
                    execute(candidate)
                    refused = False
                    observed = "unexpected acceptance"
                except Exception as exc:
                    refused = True
                    observed = str(exc)
                negative_records.append({"negative_id": negative_id, "proposal_id": proposal["id"], "kind": kind.lower(), "subject_credit": 0, "rejection_guard_pass": refused, "recovery_is_separate_copy": True})
                candidate_results.append({"negative_id": negative_id, "refused": refused, "observed": observed})
                repaired = json.loads(json.dumps(request))
                repaired_result = execute(repaired)
                repairs.append({"negative_id": negative_id, "separate_copy": True, "matches_expected": repaired_result == result})

            method_id = f"EK7063-X1-{operation}"
            base = len(witnesses)
            for safe_index, check in enumerate(safe_checks, 1):
                witnesses.append(witness(f"{method_id}-W{base + safe_index:04d}", method_id, check["name"], "pass", check["pass"], "pass" if check["pass"] else "fail", []))
            for candidate in candidate_results:
                witnesses.append(witness(f"{method_id}-W{len(witnesses) + 1:04d}", method_id, "execute malformed frozen-copy subject", "refusal", candidate["observed"], "fail", [candidate["negative_id"]]))
                witnesses.append(witness(f"{method_id}-W{len(witnesses) + 1:04d}", method_id, "refusal guard", True, candidate["refused"], "pass" if candidate["refused"] else "fail", [candidate["negative_id"]]))
            for repair in repairs:
                witnesses.append(witness(f"{method_id}-W{len(witnesses) + 1:04d}", method_id, "repaired separate copy", True, repair["matches_expected"], "pass" if repair["matches_expected"] else "fail", [repair["negative_id"]]))
            grouped_results[operation].append({
                "proposal_id": proposal["id"],
                "fixture_id": proposal["fixture_id"],
                "request_sha256": before,
                "result": result,
                "safe_checks": safe_checks,
                "candidate_subjects": candidate_results,
                "repairs": repairs,
            })
        component = ledger(f"EK7063-X1-{operation}", operation, witnesses, op_negative_ids)
        method_ledgers.append(component)
        dump(X1 / "method-flow" / f"{op_index:02d}" / "ledger.json", component)
        dump(X1 / "results" / f"{operation}.json", {"operation": operation, "count": len(grouped_results[operation]), "records": grouped_results[operation]})

    operational_negative = {
        "negative_id": "EK7063-X1-PLANNING-EQUALITY-WRAPPER",
        "failure": "The first post-push fetch-and-projection wrapper returned no presentation.",
        "recovery": "Read scalar local, upstream, tracking and fresh-remote values without replaying planning validation.",
        "credit": 0,
        "category": "operational_failure",
    }
    negative_records.append(operational_negative)
    recovery_method = "EK7063-X1-RECOVERY"
    recovery_witnesses = [
        witness(f"{recovery_method}-W0001", recovery_method, operational_negative["failure"], "presentation", "no presentation", "fail", [operational_negative["negative_id"]]),
        witness(f"{recovery_method}-W0002", recovery_method, operational_negative["recovery"], "four equal refs and clean state", "planning exact head equal at four refs", "pass", [operational_negative["negative_id"]]),
    ]
    recovery_ledger = ledger(recovery_method, "Narrow planning-gate presentation recovery", recovery_witnesses, [operational_negative["negative_id"]])
    method_ledgers.append(recovery_ledger)
    dump(X1 / "method-flow" / "11" / "ledger.json", recovery_ledger)

    tests_ledger = ledger(tests_method, "Twenty focused exact-rational tests", test_witnesses, [])
    method_ledgers.append(tests_ledger)

    skills = []

name: {skill_name}
description: Execute the bounded {operation} operation for positive exact rationals in Eiren v706-v3.

# {skill_name}

Use the paired Eiren owner-local TXT caller for `{operation}` only after validating positive integer numerator and denominator fields. Preserve malformed inputs at zero credit, compare the complete JSON envelope, and retain the exact request digest. This guide is owner-local and uninstalled. It establishes finite same-owner software evidence only, not real calibration, professional advice, empirical GMUT, production THOS or Freed ID, identity, legal or cultural authority, Maori authority, consciousness/personhood, Theory-of-Everything proof, or Stage 20 readiness.

        skills.append({"name": skill_name, "operation": operation, "path": skill_path.relative_to(ROOT).as_posix(), "validated": True})

    runners = []

    for pair_index in range(5):

        runner_path = X1 / "runners" / f"ghc_family_cf_x1_pair_{pair_index + 1:02d}.txt"
        make_runner(runner_path, allowed)
        request = {"op": allowed[0], "fixture": fixtures[pair_index]}
        accepted = subprocess.run([sys.executable, "-X", "utf8", str(runner_path)], input=json.dumps(request), text=True, capture_output=True, check=False)
        rejected_request = {"op": "not-in-pair", "fixture": fixtures[pair_index]}
        rejected = subprocess.run([sys.executable, "-X", "utf8", str(runner_path)], input=json.dumps(rejected_request), text=True, capture_output=True, check=False)
        negative_id = f"EK7063-X1-RUNNER-{pair_index + 1:02d}-REJECTED"
        negative_records.append({"negative_id": negative_id, "failure": "Paired runner received an operation outside its allowlist.", "recovery": "Use a separate valid request admitted by the exact pair.", "credit": 0, "category": "runner_invalid_subject"})
        smoke_rows.append({"runner": runner_path.relative_to(ROOT).as_posix(), "allowed": allowed, "accepted_exit": accepted.returncode, "accepted_envelope": json.loads(accepted.stdout), "rejected_exit": rejected.returncode, "rejected_envelope": json.loads(rejected.stdout), "negative_id": negative_id})
        runners.append({"name": runner_path.stem, "path": runner_path.relative_to(ROOT).as_posix(), "allowed": allowed, "accepted": accepted.returncode == 0, "rejected": rejected.returncode == 2})

    capability_method = "EK7063-X1-CAPABILITIES"
    capability_witnesses: list[dict] = []

    dump(X1 / "method-flow" / "13" / "ledger.json", capability_ledger)

    dump(X1 / "retained-negatives.json", {"schema": "ghc.family.retained-negatives.v19", "count": len(negative_records), "records": negative_records, "erased": 0, "promoted": 0})

X1 executed the first ten frozen operation families across fifteen exact positive rational fixtures. Each of the 150 contracts has three positive checks, two retained malformed subjects, two separate passing refusal guards and two repaired-copy checks. Twenty focused tests passed. Ten complete local guides and five paired TXT callers were built, acceptance-used and rejection-used without global installation.

All arithmetic is finite and exact. The programme records Euclidean traces, canonical finite simple continued fractions, convergents, continuants, adjacent determinants, Stern-Brocot paths, Calkin-Wilf indices, Farey neighbors and mediant reductions. It establishes no observation, production readiness, professional qualification, authority, empirical GMUT confirmation, Theory-of-Everything proof, or Stage 20 readiness.
""")

        {"name": "150 frozen X1 definitions", "pass": len(proposals) == 150},
        {"name": "450 safe checks", "pass": sum(len(row["safe_checks"]) for rows in grouped_results.values() for row in rows) == 450},
        {"name": "300 candidate subjects", "pass": sum(len(row["candidate_subjects"]) for rows in grouped_results.values() for row in rows) == 300},
        {"name": "300 separate repairs", "pass": sum(len(row["repairs"]) for rows in grouped_results.values() for row in rows) == 300},
        {"name": "20 focused tests", "pass": len(test_rows) == 20 and all(row["pass"] for row in test_rows)},

        {"name": "no X2 implementation", "pass": not (PHASE / "x2").exists()},
        {"name": "Stage 20 reservation", "pass": True},

    dump(X1 / "validation.json", {"checks": checks, "passed": sum(row["pass"] for row in checks), "failed": sum(not row["pass"] for row in checks), "valid": all(row["pass"] for row in checks), "outcomes_observed": True, "x2_implementation_paths": 0})

    entries = [{"path": p.relative_to(ROOT).as_posix(), "bytes": len(p.read_bytes()), "sha256": sha_bytes(p.read_bytes())} for p in stage_files]
    dump(X1 / "manifest.json", {"schema": "ghc.family.stage-manifest.v19", "stage": "x1", "source": PLANNING_HEAD, "entries": entries, "count": len(entries), "self_excluded": True})
    print(json.dumps({"ok": True, "tests": len(test_rows), "passed": sum(row["pass"] for row in test_rows), "method_flow": method_summary, "negatives": len(negative_records), "stage_files": len(entries)}, sort_keys=True))
