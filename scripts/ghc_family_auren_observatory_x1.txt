class ContractError extends Error { constructor(code) { super(code); this.code = code; } }
const fail = (code) => { throw new ContractError(code); };
function fields(value, names) { if (!value || typeof value !== 'object' || Array.isArray(value)) fail('invalid_object'); const actual = Object.keys(value); for (const name of names) if (!Object.hasOwn(value, name)) fail('missing_field'); for (const name of actual) if (!names.includes(name)) fail('unknown_field'); }
function integer(value, minimum = 0) { if (!Number.isSafeInteger(value) || value < minimum) fail('invalid_integer'); return value; }
function array(value, minimum = 0, maximum = 30) { if (!Array.isArray(value) || value.length < minimum || value.length > maximum) fail('invalid_array'); return value; }
function text(value, maximum = 200) { if (typeof value !== 'string' || !value.length || value.length > maximum) fail('invalid_text'); return value; }
function boolean(value) { if (typeof value !== 'boolean') fail('invalid_boolean'); return value; }
function gcd(a, b) { a = a < 0n ? -a : a; b = b < 0n ? -b : b; while (b) { const t = a % b; a = b; b = t; } return a || 1n; }
function frac(n, d = 1n) { n = BigInt(n); d = BigInt(d); if (d === 0n) fail('zero_denominator'); if (d < 0n) { n = -n; d = -d; } const g = gcd(n, d); return [n / g, d / g]; }
function add(a, b) { return frac(a[0] * b[1] + b[0] * a[1], a[1] * b[1]); }
function mul(a, b) { return frac(a[0] * b[0], a[1] * b[1]); }
function out(a) { return { num: a[0].toString(), den: a[1].toString() }; }
function binomial(n, k) { n = BigInt(n); k = BigInt(k); if (k < 0n || k > n) return 0n; if (k > n - k) k = n - k; let r = 1n; for (let i = 1n; i <= k; i++) r = r * (n - k + i) / i; return r; }
function cloned(value) { return JSON.parse(JSON.stringify(value)); }

function evaluate(request, table=operations) { const before=JSON.stringify(request); try { fields(request,['op','payload']); if(typeof request.op!=='string')fail('invalid_operation'); if(!Object.hasOwn(table,request.op))fail('unknown_operation'); const result={ok:true,error:null,value:table[request.op](request.payload)}; if(JSON.stringify(request)!==before)throw new Error('input_mutation'); return result; } catch(e) { if(e instanceof ContractError)return{ok:false,error:e.code,value:null}; throw e; } }
function cli(table=operations) { let s='';process.stdin.setEncoding('utf8');process.stdin.on('data',(b)=>{s+=b;if(s.length>262144){process.stderr.write('input_limit\n');process.exit(2);}});process.stdin.on('end',()=>{let request;try{request=JSON.parse(s);}catch{process.stdout.write(JSON.stringify({ok:false,error:'invalid_json',value:null})+'\n');return;}process.stdout.write(JSON.stringify(evaluate(request,table))+'\n');}); }
