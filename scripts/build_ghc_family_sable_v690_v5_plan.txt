function ordered(value) {
  if (Array.isArray(value)) return value.map(ordered);
  if (value && typeof value === 'object') {
    return Object.fromEntries(Object.keys(value).sort().map((key) => [key, ordered(value[key])]));
  }
  return value;
}

function stable(value) {
  return JSON.stringify(ordered(value), null, 2) + '\n';
}

function compact(value) {
  return JSON.stringify(ordered(value));
}

function sha(value) {
  const data = Buffer.isBuffer(value) ? value : Buffer.from(String(value));
  return crypto.createHash('sha256').update(data).digest('hex');
}

function writeJson(relative, value) {
  const target = path.join(BASE, relative);
  fs.mkdirSync(path.dirname(target), {recursive: true});
  fs.writeFileSync(target, stable(value), 'utf8');
}

function writeText(relative, value) {
  const target = path.join(BASE, relative);
  fs.mkdirSync(path.dirname(target), {recursive: true});
  fs.writeFileSync(target, value.trimEnd() + '\n', 'utf8');
}

function gcd(a, b) {
  a = Math.abs(a); b = Math.abs(b);
  while (b) [a, b] = [b, a % b];
  return a || 1;
}

function fraction(n, d) {
  const g = gcd(n, d);
  return {numerator: n / g, denominator: d / g};
}

function topological(payload) {
  const incoming = Object.fromEntries(payload.nodes.map((node) => [node, 0]));
  const outgoing = Object.fromEntries(payload.nodes.map((node) => [node, []]));
  for (const [from, to] of payload.edges) { incoming[to] += 1; outgoing[from].push(to); }
  const ready = payload.nodes.filter((node) => incoming[node] === 0).sort();
  const order = [];
  while (ready.length) {
    const node = ready.shift(); order.push(node);
    for (const next of outgoing[node].sort()) {
      incoming[next] -= 1;
      if (incoming[next] === 0) { ready.push(next); ready.sort(); }
    }
  }
  return {acyclic: order.length === payload.nodes.length, order};
}

function oracle(operation, payload) {
  if (operation === 'dag_topological_order') return topological(payload);
  if (operation === 'backdoor_path_screen') {
    const open = payload.paths.filter((row) => row.starts_backdoor && (!row.collider_blocked || row.collider_conditioned)).map((row) => row.id).sort();
    return {open_backdoor_paths: open, count: open.length};
  }
  if (operation === 'adjustment_set_filter') {
    const admissible = payload.candidates.filter((set) => payload.required.every((x) => set.includes(x)) && set.every((x) => !payload.forbidden.includes(x))).map((set) => [...set].sort());
    return {admissible};
  }
  if (operation === 'frontdoor_condition_matrix') {
    const conditions = ['intercepts_treatment_effect', 'treatment_outcome_backdoor_blocked', 'mediator_outcome_backdoor_blocked'];
    const missing = conditions.filter((key) => payload[key] !== true);
    return {identifiable: missing.length === 0, missing};
  }
  if (operation === 'counterfactual_consistency') {
    const violations = payload.rows.filter((row) => row.observed_treatment === row.intervention && row.observed_outcome !== row.counterfactual_outcome).map((row) => row.id).sort();
    return {consistent: violations.length === 0, violations};
  }
  if (operation === 'positivity_overlap_bins') {
    const supported = payload.strata.filter((row) => row.treated > 0 && row.control > 0).map((row) => row.id);
    const unsupported = payload.strata.filter((row) => row.treated === 0 || row.control === 0).map((row) => row.id);
    return {supported, unsupported, complete_overlap: unsupported.length === 0};
  }
  if (operation === 'instrument_assumption_ledger') {
    const required = ['relevance', 'exclusion', 'independence'];
    const missing = required.filter((key) => payload.assumptions[key] !== 'declared');
    return {all_declared: missing.length === 0, missing};
  }
  if (operation === 'negative_control_pairing') {
    const pairs = [];
    for (const exposure of payload.exposures) for (const outcome of payload.outcomes) if (exposure.domain === outcome.domain) pairs.push(`${exposure.id}:${outcome.id}`);
    return {pairs: pairs.sort(), pair_count: pairs.length};
  }
  if (operation === 'estimand_population_contract') {
    return {estimand: payload.estimand, target_population: payload.target_population, denominator: payload.included, excluded: payload.excluded, empty: payload.included === 0};
  }
  if (operation === 'sensitivity_interval_projection') {
    return {lower: payload.estimate - payload.bias_bound, upper: payload.estimate + payload.bias_bound, assumption: payload.assumption};
  }
  if (operation === 'audit_filter_projection') {
    return {visible_ids: payload.rows.filter((row) => payload.statuses.includes(row.status) && payload.pillars.includes(row.pillar)).map((row) => row.id)};
  }
  if (operation === 'uncertainty_narration') {
    return {text: `${payload.label}: ${payload.value} ${payload.unit}; bounded interval ${payload.lower} to ${payload.upper}; source ${payload.source_status}.`, machine: {value: payload.value, lower: payload.lower, upper: payload.upper, unit: payload.unit}};
  }
  if (operation === 'evidence_expiry_state') {
    const state = payload.revoked ? 'revoked' : payload.observed_at >= payload.expires_at ? 'expired' : 'active';
    return {state, usable: state === 'active'};
  }
  if (operation === 'focus_order_trace') {
    const active = payload.elements.filter((row) => !row.hidden && !row.disabled).sort((a, b) => a.order - b.order || a.id.localeCompare(b.id));
    const orders = active.map((row) => row.order);
    return {focus_order: active.map((row) => row.id), duplicate_order: orders.some((value, index) => index > 0 && orders[index - 1] === value)};
  }
  if (operation === 'table_sort_stability') {
    const rows = payload.rows.map((row, index) => ({...row, _index: index})).sort((a, b) => (a.value - b.value) * (payload.direction === 'desc' ? -1 : 1) || a._index - b._index);
    return {row_ids: rows.map((row) => row.id), direction: payload.direction};
  }
  if (operation === 'error_summary_links') {
    const linked = payload.errors.filter((row) => payload.targets.includes(row.target)).map((row) => row.field).sort();
    const missing = payload.errors.filter((row) => !payload.targets.includes(row.target)).map((row) => row.field).sort();
    return {linked, missing_targets: missing};
  }
  if (operation === 'live_region_queue') {
    const sorted = [...payload.events].sort((a, b) => a.time - b.time || a.id.localeCompare(b.id));
    return {assertive: sorted.filter((row) => row.urgency === 'assertive').map((row) => row.id), polite: sorted.filter((row) => row.urgency === 'polite').map((row) => row.id), evaluation: 'structural_proxy'};
  }
  if (operation === 'locale_number_readback') {
    const value = payload.numerator / payload.denominator;
    return {canonical_decimal: value.toFixed(3), requested_locale: payload.locale, spoken_review: 'reserved', evaluation: 'structural_proxy'};
  }
  if (operation === 'real_user_evaluation_gate') {
    return {state: payload.participants > 0 && payload.assistive_technologies.length > 0 ? 'evidence_present_requires_review' : 'OPEN_GAP', participant_rows: payload.participants, manual_review: 'reserved'};
  }
  if (operation === 'authority_release_gate') {
    const vacant = payload.required_roles.filter((row) => row.status !== 'authorized').map((row) => row.role).sort();
    return {state: vacant.length ? 'EXACT_GATE' : 'requires_live_authority_verification', vacant_roles: vacant, software_authority: false};
  }
  throw new Error(`unknown operation ${operation}`);
}

  const n = caseIndex;
  if (operation === 'dag_topological_order') return {nodes: [`A${n}`, `B${n}`, `C${n}`, `D${n}`], edges: n % 2 ? [[`A${n}`, `B${n}`], [`A${n}`, `C${n}`], [`B${n}`, `D${n}`]] : [[`A${n}`, `C${n}`], [`B${n}`, `C${n}`], [`C${n}`, `D${n}`]]};
  if (operation === 'backdoor_path_screen') return {paths: [{id: `p${n}a`, starts_backdoor: true, collider_blocked: n % 2 === 0, collider_conditioned: false}, {id: `p${n}b`, starts_backdoor: true, collider_blocked: true, collider_conditioned: n % 3 === 0}, {id: `p${n}c`, starts_backdoor: false, collider_blocked: false, collider_conditioned: false}]};
  if (operation === 'adjustment_set_filter') return {required: [`baseline${n}`], forbidden: [`mediator${n}`, `instrument${n}`], candidates: [[`baseline${n}`], [`baseline${n}`, `age${n}`], [`baseline${n}`, `mediator${n}`], [`instrument${n}`]]};
  if (operation === 'frontdoor_condition_matrix') return {intercepts_treatment_effect: true, treatment_outcome_backdoor_blocked: n % 4 !== 0, mediator_outcome_backdoor_blocked: n % 3 !== 0};
  if (operation === 'counterfactual_consistency') return {rows: [{id: `r${n}a`, observed_treatment: 1, intervention: 1, observed_outcome: n % 2, counterfactual_outcome: n % 2}, {id: `r${n}b`, observed_treatment: 0, intervention: 1, observed_outcome: 0, counterfactual_outcome: 1}, {id: `r${n}c`, observed_treatment: 0, intervention: 0, observed_outcome: 1, counterfactual_outcome: n % 5 === 0 ? 0 : 1}]};
  if (operation === 'positivity_overlap_bins') return {strata: [{id: `s${n}a`, treated: n + 1, control: n + 2}, {id: `s${n}b`, treated: n % 3 === 0 ? 0 : n, control: n + 1}, {id: `s${n}c`, treated: n + 2, control: n % 4 === 0 ? 0 : n}]};
