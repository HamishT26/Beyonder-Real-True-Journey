'use strict';
const u=require('./ghc_family_avelin_evidence.txt'),{fs,path,cp,read,put}=u;
const session=process.argv[2],python=process.argv[3],validator=process.argv[4];if(!['x1','x2'].includes(session)||!python||!validator)throw Error('session, existing Python, installed validator required');
const defs=read('plan/new-proposals.json').proposals.filter(d=>d.session===session),plan=read('plan/skills-runners-promotions.json'),guides=plan.local_skills.filter(x=>x.session===session),checks=[];
for(const g of guides){const rel=u.BASE+'/'+session+'/skills/'+g.name,run=cp.spawnSync(python,['-X','utf8','-B',validator,rel],{encoding:'utf8',env:{...process.env,PYTHONDONTWRITEBYTECODE:'1'}});checks.push({kind:'system_skill_validator',path:rel,exit:run.status,passed:run.status===0,output:(run.stdout+run.stderr).trim()});}
const l=u.ledger(session+'-capabilities');
for(const [i,name] of plan.runners[session].entries()){const positive=defs[i*20],rel='runners/'+name,id='AR6906-'+session.toUpperCase()+'-RUNNER-'+(i+1),negative=id+'-OUTSIDE';u.method(l,id,'Paired transition runner group '+(i+1),[negative],'Retain the paired operation boundary and reject other families.');const pos=cp.spawnSync(process.execPath,[rel],{encoding:'utf8',input:JSON.stringify(positive.request)});const neg=cp.spawnSync(process.execPath,[rel],{encoding:'utf8',input:JSON.stringify({op:'unselected_operation'})});let a,b;try{a=JSON.parse(pos.stdout);b=JSON.parse(neg.stdout);}catch(e){a=pos.stdout;b=neg.stdout;}const ok=pos.status===0&&u.canon(a)===u.canon(positive.expected_output),refused=neg.status===2&&u.canon(b)===u.canon({ok:false,value:null,error:'outside_group'});checks.push({kind:'runner_positive',path:rel,passed:ok});checks.push({kind:'runner_outside_refusal',path:rel,passed:refused});u.witness(l,id+'-POSITIVE',id,ok?'pass':'fail','Invoke paired runner via Node stdin',positive.expected_output,a);u.witness(l,negative,id,'fail','Outside-group subject','An in-group operation',b,[negative]);u.witness(l,id+'-REFUSAL',id,refused?'pass':'fail','Outside-group refusal guard',{ok:false,value:null,error:'outside_group'},b,[negative]);}
u.finalize(l);put(session+'/capability-validation.json',{checks,passed:checks.filter(c=>c.passed).length,total:checks.length,tool_versions:{node:process.version,skill_validator:'installed system Skill Creator quick_validate.py'},same_owner_only:true,boundary:u.boundary});put(session+'/method-flow/capability-ledger.json',l);console.log(JSON.stringify({session,checks:checks.filter(c=>c.passed).length+'/'+checks.length,counts:l.counts}));if(checks.some(c=>!c.passed))process.exitCode=1;
