'use strict';
const u=require('./ghc_family_avelin_evidence.txt');
// Five bounded pattern classes; definitions are not proof of exhaustive privacy.
const patterns={private_route:/\b[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}\b|"(?:threadId|providerTabId|clientThreadId)"\s*:/gi,private_path:/[A-Z]:[\\/](?:Users|GHC-Archives)[\\/]|\/(?:Users|home)\/[^\s"']+/g,credential:/\b(?:sk-|ghp_)[A-Za-z0-9_-]{20,}|-----BEGIN [A-Z ]*PRIVATE KEY-----/g,session_stream:/"(?:session_meta|response_item|event_msg)"\s*:/g,protected_contact:/\b[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}\b/gi};
function scan(paths){const candidates=[];for(const file of paths){const text=u.fs.readFileSync(file,'utf8');for(const [kind,rx] of Object.entries(patterns)){rx.lastIndex=0;for(const m of text.matchAll(rx))candidates.push({path:file,line:text.slice(0,m.index).split('\n').length,kind,matched_bytes:Buffer.byteLength(m[0]),classification:'review_required'});}}return candidates;}
if(require.main===module){const [stage,selection]=process.argv.slice(2);if(!stage)throw Error('stage required');const files=selection?JSON.parse(u.fs.readFileSync(selection,'utf8')):u.paths().filter(p=>/\.(json|md|txt|html)$/.test(p));const candidates=scan(files);const result={schema:'ghc.family.five-class-privacy.v5',owner:u.OWNER,phase:u.PHASE,stage,scanned_files:files.length,scope:'Exact owner files only',classes:Object.keys(patterns),candidate_count:candidates.length,candidates,confirmed_hit_count:candidates.length,classification_rule:'No candidate exemption is applied automatically. Stop for explicit review of any match.',complete_privacy:false,boundary:u.boundary};u.put(stage+'/privacy-scan.json',result);console.log(JSON.stringify({files:files.length,candidates:candidates.length}));if(candidates.length)process.exitCode=1;}
module.exports={scan,patterns};
