'use strict';
const fs=require('node:fs'),vm=require('node:vm'),path=require('node:path');
const files=process.argv.slice(2),results=[];
if(!files.length)throw Error('supply one or more JavaScript text paths');
for(const file of files){try{new vm.Script(fs.readFileSync(file,'utf8'),{filename:path.basename(file)});results.push({file:path.basename(file),valid:true});}catch(error){results.push({file:path.basename(file),valid:false,error:error.message});}}
const valid=results.every(row=>row.valid);process.stdout.write(JSON.stringify({valid,files:results.length,results})+'\n');if(!valid)process.exitCode=1;
